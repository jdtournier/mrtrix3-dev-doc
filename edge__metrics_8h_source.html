124 class Metric_invnodevolume : public Metric_base {
125 
126  public:
127  Metric_invnodevolume (Image<node_t>& node_image) :
128  Metric_base (false)
129  {
130  for (auto l = Loop() (node_image); l; ++l) {
131  const node_t node_index = node_image.value();
132  if (node_index >= node_volumes.size())
133  node_volumes.resize (node_index + 1, 0);
134  node_volumes[node_index]++;
135  }
136  }
137 
138  virtual double operator() (const Streamline<>& tck, const NodePair& nodes) const override
139  {
140  const size_t volume_sum = node_volumes[nodes.first] + node_volumes[nodes.second];
141  if (!volume_sum) return 0.0;
142  return (2.0 / double(volume_sum));
143  }
144 
145  virtual double operator() (const Streamline<>& tck, const std::vector<node_t>& nodes) const override
146  {
147  size_t volume_sum = 0;
148  for (std::vector<node_t>::const_iterator n = nodes.begin(); n != nodes.end(); ++n)
149  volume_sum += node_volumes[*n];
150  if (!volume_sum) return 0.0;
151  return (nodes.size() / double(volume_sum));
152  }
