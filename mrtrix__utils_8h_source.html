69  void read_mrtrix_header (Header& H, SourceType& kv)
70  {
71 
72  std::string dtype, layout;
73  std::vector<int> dim;
74  std::vector<float> transform, dw_scheme, vox, scaling;
75  std::vector<std::string> units, labels;
76 
77  std::string key, value;
78  while (next_keyvalue (kv, key, value)) {
79  key = lowercase (key);
80  if (key == "dim") dim = parse_ints (value);
81  else if (key == "vox") vox = parse_floats (value);
82  else if (key == "layout") layout = value;
83  else if (key == "datatype") dtype = value;
84  else if (key == "scaling") scaling = parse_floats (value);
85  else if (key == "comments") H.comments().push_back (value);
86  else if (key == "units") units = split (value, "\\");
87  else if (key == "labels") labels = split (value, "\\");
88  else if (key == "transform") {
89  std::vector<float> V (parse_floats (value));
90  transform.insert (transform.end(), V.begin(), V.end());
91  }
92  else if (key == "dw_scheme") {
93  std::vector<float> V (parse_floats (value));
94  dw_scheme.insert (dw_scheme.end(), V.begin(), V.end());
95  }
96  else if (key.size() && value.size()) {
97  if (H[key].size())
98  H[key] += '\n';
99  H[key] += value;
100  }
101  }
102 
103  if (dim.empty())
104  throw Exception ("missing \"dim\" specification for MRtrix image \"" + H.name() + "\"");
105  H.set_ndim (dim.size());
106  for (size_t n = 0; n < dim.size(); n++) {
107  if (dim[n] < 1)
108  throw Exception ("invalid dimensions for MRtrix image \"" + H.name() + "\"");
109  H.dim(n) = dim[n];
110  }
111 
112  if (vox.empty())
113  throw Exception ("missing \"vox\" specification for MRtrix image \"" + H.name() + "\"");
114  if (vox.size() < 3)
115  throw Exception ("too few entries in \"vox\" specification for MRtrix image \"" + H.name() + "\"");
116  for (size_t n = 0; n < std::min (vox.size(), H.ndim()); n++) {
117  if (vox[n] < 0.0)
118  throw Exception ("invalid voxel size for MRtrix image \"" + H.name() + "\"");
119  H.vox(n) = vox[n];
120  }
121 
122 
123  if (dtype.empty())
124  throw Exception ("missing \"datatype\" specification for MRtrix image \"" + H.name() + "\"");
125  H.datatype() = DataType::parse (dtype);
126 
127 
128  if (layout.empty())
129  throw Exception ("missing \"layout\" specification for MRtrix image \"" + H.name() + "\"");
130  std::vector<ssize_t> ax = Axis::parse (H.ndim(), layout);
131  for (size_t i = 0; i < ax.size(); ++i)
132  H.stride(i) = ax[i];
133 
134  if (transform.size()) {
135 
136  if (transform.size() < 9)
137  throw Exception ("invalid \"transform\" specification for MRtrix image \"" + H.name() + "\"");
138 
139  H.transform().allocate (4,4);
140  int count = 0;
141  for (int row = 0; row < 3; ++row)
142  for (int col = 0; col < 4; ++col)
143  H.transform() (row,col) = transform[count++];
144  H.transform()(3,0) = H.transform()(3,1) = H.transform()(3,2) = 0.0;
145  H.transform()(3,3) = 1.0;
146  }
147 
148 
149  if (dw_scheme.size()) {
150  if (dw_scheme.size() % 4) {
151  INFO ("invalid \"dw_scheme\" specification for MRtrix image \"" + H.name() + "\" - ignored");
152  }
153  else {
154  H.DW_scheme().allocate (dw_scheme.size() /4, 4);
155  int count = 0;
156  for (size_t row = 0; row < H.DW_scheme().rows(); ++row)
157  for (size_t col = 0; col < 4; ++col)
158  H.DW_scheme()(row,col) = dw_scheme[count++];
159  }
160  }
161 
162 
163  if (scaling.size()) {
164  if (scaling.size() != 2)
165  throw Exception ("invalid \"scaling\" specification for MRtrix image \"" + H.name() + "\"");
166  H.intensity_offset() = scaling[0];
167  H.intensity_scale() = scaling[1];
168  }
