179  std::atomic<uint8_t>* at = reinterpret_cast<std::atomic<uint8_t>*> (as<uint8_t>(data) + (i/8));
180  uint8_t prev = *at, new_value;
181  do {
182  if (value) new_value = prev | (BITMASK >> i%8);
183  else new_value = prev & ~(BITMASK >> i%8);
184  } while (!at->compare_exchange_weak (prev, new_value));
