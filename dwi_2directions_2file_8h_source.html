34  template <typename ValueType>
35  inline Math::Matrix<ValueType> load_cartesian (const std::string& filename)
36  {
37  Math::Matrix<ValueType> directions (filename);
38  if (directions.columns() == 2)
39  directions = Math::SH::S2C (directions);
40  else {
41  if (directions.columns() != 3)
42  throw Exception ("unexpected number of columns for directions file \"" + filename + "\"");
43  for (size_t n = 0; n < directions.rows(); ++n) {
44  ValueType norm = Math::norm (directions.row(n));
45  if (std::abs(ValueType(1.0) - norm) > 1e-4)
46  WARN ("directions file \"" + filename + "\" contains non-unit direction vectors");
47  directions.row(n) *= norm ? ValueType(1.0)/norm : ValueType(0.0);
48  }
49  }
50  return directions;
51  }
