40  template <class Container>
41  value_type operator() (const Container& vec)
42  {
43  diff.resize (vec.size());
44  med_diff.resize (vec.size());
45  for (size_t j = 0; j < vec.size(); ++j) {
46  for (size_t i = 0; i < vec.size(); ++i)
47  diff[i] = std::abs (vec[i] - vec[j]);
48  med_diff[j] = Math::median (diff);
49  }
