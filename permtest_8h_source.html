106  class Processor { MEMALIGN (Processor<StatsType>)
107  public:
108  Processor (const StatsType& stats_calculator,
109  const std::shared_ptr<EnhancerBase> enhancer,
110  const vector_type& empirical_enhanced_statistics,
111  const vector_type& default_enhanced_statistics,
112  const std::shared_ptr<vector_type> default_enhanced_statistics_neg,
113  vector_type& perm_dist_pos,
114  std::shared_ptr<vector_type> perm_dist_neg,
115  vector<size_t>& global_uncorrected_pvalue_counter,
116  std::shared_ptr< vector<size_t> > global_uncorrected_pvalue_counter_neg) :
117  stats_calculator (stats_calculator),
118  enhancer (enhancer), empirical_enhanced_statistics (empirical_enhanced_statistics),
119  default_enhanced_statistics (default_enhanced_statistics), default_enhanced_statistics_neg (default_enhanced_statistics_neg),
120  statistics (stats_calculator.num_elements()), enhanced_statistics (stats_calculator.num_elements()),
121  uncorrected_pvalue_counter (stats_calculator.num_elements(), 0),
122  perm_dist_pos (perm_dist_pos), perm_dist_neg (perm_dist_neg),
123  global_uncorrected_pvalue_counter (global_uncorrected_pvalue_counter),
124  global_uncorrected_pvalue_counter_neg (global_uncorrected_pvalue_counter_neg),
125  mutex (new std::mutex())
126  {
127  if (global_uncorrected_pvalue_counter_neg)
128  uncorrected_pvalue_counter_neg.reset (new vector<size_t>(stats_calculator.num_elements(), 0));
129  }

132  ~Processor () {
133  std::lock_guard<std::mutex> lock (*mutex);
134  for (size_t i = 0; i < stats_calculator.num_elements(); ++i) {
135  global_uncorrected_pvalue_counter[i] += uncorrected_pvalue_counter[i];
136  if (global_uncorrected_pvalue_counter_neg)
137  (*global_uncorrected_pvalue_counter_neg)[i] = (*uncorrected_pvalue_counter_neg)[i];
138  }
139  }

141 
142  bool operator() (const Permutation& permutation)
143  {
144  stats_calculator (permutation.data, statistics);
145  if (enhancer) {
146  perm_dist_pos[permutation.index] = (*enhancer) (statistics, enhanced_statistics);
147  } else {
148  enhanced_statistics = statistics;
149  perm_dist_pos[permutation.index] = enhanced_statistics.maxCoeff();
150  }
151 
152  if (empirical_enhanced_statistics.size()) {
153  perm_dist_pos[permutation.index] = 0.0;
154  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
155  enhanced_statistics[i] /= empirical_enhanced_statistics[i];
156  perm_dist_pos[permutation.index] = std::max(perm_dist_pos[permutation.index], enhanced_statistics[i]);
157  }
158  }
159 
160  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
161  if (default_enhanced_statistics[i] > enhanced_statistics[i])
162  uncorrected_pvalue_counter[i]++;
163  }
164 
165  // Compute the opposite contrast
166  if (perm_dist_neg) {
167  statistics = -statistics;
168 
169  (*perm_dist_neg)[permutation.index] = (*enhancer) (statistics, enhanced_statistics);
170 
171  if (empirical_enhanced_statistics.size()) {
172  (*perm_dist_neg)[permutation.index] = 0.0;
173  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
174  enhanced_statistics[i] /= empirical_enhanced_statistics[i];
175  (*perm_dist_neg)[permutation.index] = std::max ((*perm_dist_neg)[permutation.index], enhanced_statistics[i]);
176  }
177  }
178 
179  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
180  if ((*default_enhanced_statistics_neg)[i] > enhanced_statistics[i])
181  (*uncorrected_pvalue_counter_neg)[i]++;
182  }
183  }
184  return true;
185  }
186 
187  protected:
188  StatsType stats_calculator;
189  std::shared_ptr<EnhancerBase> enhancer;
190  const vector_type& empirical_enhanced_statistics;
191  const vector_type& default_enhanced_statistics;
192  const std::shared_ptr<vector_type> default_enhanced_statistics_neg;

207  template <class StatsType>
208  void precompute_empirical_stat (const StatsType& stats_calculator, const std::shared_ptr<EnhancerBase> enhancer,
209  PermutationStack& perm_stack, vector_type& empirical_statistic)
210  {
211  vector<size_t> global_enhanced_count (empirical_statistic.size(), 0);
212  {
213  PreProcessor<StatsType> preprocessor (stats_calculator, enhancer, empirical_statistic, global_enhanced_count);
214  Thread::run_queue (perm_stack, Permutation(), Thread::multi (preprocessor));
215  }
216  for (ssize_t i = 0; i < empirical_statistic.size(); ++i) {
217  if (global_enhanced_count[i] > 0)
218  empirical_statistic[i] /= static_cast<default_type> (global_enhanced_count[i]);
219  }
220  }

224  // Precompute the default statistic image and enhanced statistic. We need to precompute this for calculating the uncorrected p-values.
225  template <class StatsType>
226  void precompute_default_permutation (const StatsType& stats_calculator,
227  const std::shared_ptr<EnhancerBase> enhancer,
228  const vector_type& empirical_enhanced_statistic,
229  vector_type& default_enhanced_statistics,
230  std::shared_ptr<vector_type> default_enhanced_statistics_neg,
231  vector_type& default_statistics)
232  {
233  vector<size_t> default_labelling (stats_calculator.num_subjects());
234  for (size_t i = 0; i < default_labelling.size(); ++i)
235  default_labelling[i] = i;
236  stats_calculator (default_labelling, default_statistics);
237  (*enhancer) (default_statistics, default_enhanced_statistics);
238 
239  if (empirical_enhanced_statistic.size())
240  default_enhanced_statistics /= empirical_enhanced_statistic;
241 
242  // Compute the opposite contrast
243  if (default_enhanced_statistics_neg) {
244  default_statistics = -default_statistics;
245 
246  (*enhancer) (default_statistics, *default_enhanced_statistics_neg);
247 
248  if (empirical_enhanced_statistic.size())
249  (*default_enhanced_statistics_neg) /= empirical_enhanced_statistic;
250 
251  // revert default_statistics to positive contrast for output
252  default_statistics = -default_statistics;
253  }

257  inline void run_permutations (PermutationStack& perm_stack,
258  const StatsType& stats_calculator,
259  const std::shared_ptr<EnhancerBase> enhancer,
260  const vector_type& empirical_enhanced_statistic,
261  const vector_type& default_enhanced_statistics,
262  const std::shared_ptr<vector_type> default_enhanced_statistics_neg,
263  vector_type& perm_dist_pos,
264  std::shared_ptr<vector_type> perm_dist_neg,
265  vector_type& uncorrected_pvalues,
266  std::shared_ptr<vector_type> uncorrected_pvalues_neg)
267  {
268  vector<size_t> global_uncorrected_pvalue_count (stats_calculator.num_elements(), 0);
269  std::shared_ptr< vector<size_t> > global_uncorrected_pvalue_count_neg;
270  if (perm_dist_neg)
271  global_uncorrected_pvalue_count_neg.reset (new vector<size_t> (stats_calculator.num_elements(), 0));
272 
273  {
274  Processor<StatsType> processor (stats_calculator, enhancer,
275  empirical_enhanced_statistic,
276  default_enhanced_statistics, default_enhanced_statistics_neg,
277  perm_dist_pos, perm_dist_neg,
278  global_uncorrected_pvalue_count, global_uncorrected_pvalue_count_neg);
279  Thread::run_queue (perm_stack, Permutation(), Thread::multi (processor));
280  }
281 
282  for (size_t i = 0; i < stats_calculator.num_elements(); ++i) {
283  uncorrected_pvalues[i] = global_uncorrected_pvalue_count[i] / default_type(perm_stack.num_permutations);
284  if (perm_dist_neg)
285  (*uncorrected_pvalues_neg)[i] = (*global_uncorrected_pvalue_count_neg)[i] / default_type(perm_stack.num_permutations);
286  }
