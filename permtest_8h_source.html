231  size_t num_permutations, std::vector<double>& empirical_statistic)

251  template <class StatsType, class EnhancementType>
252  inline void precompute_default_permutation (const StatsType& stats_calculator, const EnhancementType& enhancer,
253  const std::shared_ptr<std::vector<double> >& empirical_enhanced_statistic,
254  std::vector<value_type>& default_enhanced_statistics, std::shared_ptr<std::vector<value_type> >& default_enhanced_statistics_neg,
255  std::vector<value_type>& default_statistics)
256  {
257  std::vector<size_t> default_labelling (stats_calculator.num_subjects());
258  for (size_t i = 0; i < default_labelling.size(); ++i)
259  default_labelling[i] = i;
260  value_type max_stat = 0.0, min_stat = 0.0;
261  stats_calculator (default_labelling, default_statistics, max_stat, min_stat);
262  max_stat = enhancer (max_stat, default_statistics, default_enhanced_statistics);
263 
264  if (empirical_enhanced_statistic) {
265  for (size_t i = 0; i < default_statistics.size(); ++i)
266  default_enhanced_statistics[i] /= (*empirical_enhanced_statistic)[i];
267  }
268 
269  // Compute the opposite contrast
270  if (default_enhanced_statistics_neg) {
271  for (size_t i = 0; i < default_statistics.size(); ++i)
272  default_statistics[i] = -default_statistics[i];
273 
274  max_stat = enhancer (-min_stat, default_statistics, *default_enhanced_statistics_neg);
275 
276  if (empirical_enhanced_statistic) {
277  for (size_t i = 0; i < default_statistics.size(); ++i)
278  (*default_enhanced_statistics_neg)[i] /= (*empirical_enhanced_statistic)[i];
279  }
280  // revert default_statistics to positive contrast for output
281  for (size_t i = 0; i < default_statistics.size(); ++i)
282  default_statistics[i] = -default_statistics[i];
283  }
