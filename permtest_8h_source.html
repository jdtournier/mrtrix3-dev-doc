53  template <class StatsType>
54  class PreProcessor { MEMALIGN (PreProcessor<StatsType>)
55  public:
56  PreProcessor (const StatsType& stats_calculator,
57  const std::shared_ptr<EnhancerBase> enhancer,
58  vector_type& global_enhanced_sum,
59  vector<size_t>& global_enhanced_count) :
60  stats_calculator (stats_calculator),
61  enhancer (enhancer), global_enhanced_sum (global_enhanced_sum),
62  global_enhanced_count (global_enhanced_count), enhanced_sum (vector_type::Zero (global_enhanced_sum.size())),
63  enhanced_count (global_enhanced_sum.size(), 0.0), stats (global_enhanced_sum.size()),
64  enhanced_stats (global_enhanced_sum.size()), mutex (new std::mutex()) {}
65 
66  ~PreProcessor ()
67  {
68  std::lock_guard<std::mutex> lock (*mutex);
69  for (ssize_t i = 0; i < global_enhanced_sum.size(); ++i) {
70  global_enhanced_sum[i] += enhanced_sum[i];
71  global_enhanced_count[i] += enhanced_count[i];
72  }
73  }
74 
75  bool operator() (const Permutation& permutation)
76  {
77  stats_calculator (permutation.data, stats);
78  (*enhancer) (stats, enhanced_stats);
79  for (ssize_t i = 0; i < enhanced_stats.size(); ++i) {
80  if (enhanced_stats[i] > 0.0) {
81  enhanced_sum[i] += enhanced_stats[i];
82  enhanced_count[i]++;
83  }
84  }
85  return true;
86  }
87 
88  protected:
89  StatsType stats_calculator;
90  std::shared_ptr<EnhancerBase> enhancer;
91  vector_type& global_enhanced_sum;
92  vector<size_t>& global_enhanced_count;
93  vector_type enhanced_sum;
94  vector<size_t> enhanced_count;
95  vector_type stats;
96  vector_type enhanced_stats;
97  std::shared_ptr<std::mutex> mutex;
98  };

104  template <class StatsType>
105  class Processor { MEMALIGN (Processor<StatsType>)
106  public:
107  Processor (const StatsType& stats_calculator,
108  const std::shared_ptr<EnhancerBase> enhancer,
109  const vector_type& empirical_enhanced_statistics,
110  const vector_type& default_enhanced_statistics,
111  const std::shared_ptr<vector_type> default_enhanced_statistics_neg,
112  vector_type& perm_dist_pos,
113  std::shared_ptr<vector_type> perm_dist_neg,
114  vector<size_t>& global_uncorrected_pvalue_counter,
115  std::shared_ptr< vector<size_t> > global_uncorrected_pvalue_counter_neg) :
116  stats_calculator (stats_calculator),
117  enhancer (enhancer), empirical_enhanced_statistics (empirical_enhanced_statistics),
118  default_enhanced_statistics (default_enhanced_statistics), default_enhanced_statistics_neg (default_enhanced_statistics_neg),
119  statistics (stats_calculator.num_elements()), enhanced_statistics (stats_calculator.num_elements()),
120  uncorrected_pvalue_counter (stats_calculator.num_elements(), 0),
121  perm_dist_pos (perm_dist_pos), perm_dist_neg (perm_dist_neg),
122  global_uncorrected_pvalue_counter (global_uncorrected_pvalue_counter),
123  global_uncorrected_pvalue_counter_neg (global_uncorrected_pvalue_counter_neg),
124  mutex (new std::mutex())
125  {
126  if (global_uncorrected_pvalue_counter_neg)
127  uncorrected_pvalue_counter_neg.reset (new vector<size_t>(stats_calculator.num_elements(), 0));
128  }
129 
130 
131  ~Processor () {
132  std::lock_guard<std::mutex> lock (*mutex);
133  for (size_t i = 0; i < stats_calculator.num_elements(); ++i) {
134  global_uncorrected_pvalue_counter[i] += uncorrected_pvalue_counter[i];
135  if (global_uncorrected_pvalue_counter_neg)
136  (*global_uncorrected_pvalue_counter_neg)[i] = (*uncorrected_pvalue_counter_neg)[i];
137  }
138  }
139 
140 
141  bool operator() (const Permutation& permutation)
142  {
143  stats_calculator (permutation.data, statistics);
144  if (enhancer) {
145  perm_dist_pos[permutation.index] = (*enhancer) (statistics, enhanced_statistics);
146  } else {
147  enhanced_statistics = statistics;
148  perm_dist_pos[permutation.index] = enhanced_statistics.maxCoeff();
149  }
150 
151  if (empirical_enhanced_statistics.size()) {
152  perm_dist_pos[permutation.index] = 0.0;
153  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
154  enhanced_statistics[i] /= empirical_enhanced_statistics[i];
155  perm_dist_pos[permutation.index] = std::max(perm_dist_pos[permutation.index], enhanced_statistics[i]);
156  }
157  }
158 
159  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
160  if (default_enhanced_statistics[i] > enhanced_statistics[i])
161  uncorrected_pvalue_counter[i]++;
162  }
163 
164  // Compute the opposite contrast
165  if (perm_dist_neg) {
166  statistics = -statistics;
167 
168  (*perm_dist_neg)[permutation.index] = (*enhancer) (statistics, enhanced_statistics);
169 
170  if (empirical_enhanced_statistics.size()) {
171  (*perm_dist_neg)[permutation.index] = 0.0;
172  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
173  enhanced_statistics[i] /= empirical_enhanced_statistics[i];
174  (*perm_dist_neg)[permutation.index] = std::max ((*perm_dist_neg)[permutation.index], enhanced_statistics[i]);
175  }
176  }
177 
178  for (ssize_t i = 0; i < enhanced_statistics.size(); ++i) {
179  if ((*default_enhanced_statistics_neg)[i] > enhanced_statistics[i])
180  (*uncorrected_pvalue_counter_neg)[i]++;
181  }
182  }
183  return true;
184  }
185 
186  protected:
187  StatsType stats_calculator;
188  std::shared_ptr<EnhancerBase> enhancer;
189  const vector_type& empirical_enhanced_statistics;
190  const vector_type& default_enhanced_statistics;
191  const std::shared_ptr<vector_type> default_enhanced_statistics_neg;
192  vector_type statistics;
193  vector_type enhanced_statistics;
194  vector<size_t> uncorrected_pvalue_counter;
195  std::shared_ptr<vector<size_t> > uncorrected_pvalue_counter_neg;
196  vector_type& perm_dist_pos;
197  std::shared_ptr<vector_type> perm_dist_neg;
198 
199  vector<size_t>& global_uncorrected_pvalue_counter;
200  std::shared_ptr<vector<size_t> > global_uncorrected_pvalue_counter_neg;
201  std::shared_ptr<std::mutex> mutex;
202  };

206  template <class StatsType>
207  void precompute_empirical_stat (const StatsType& stats_calculator, const std::shared_ptr<EnhancerBase> enhancer,
208  PermutationStack& perm_stack, vector_type& empirical_statistic)
209  {
210  vector<size_t> global_enhanced_count (empirical_statistic.size(), 0);
211  {
212  PreProcessor<StatsType> preprocessor (stats_calculator, enhancer, empirical_statistic, global_enhanced_count);
213  Thread::run_queue (perm_stack, Permutation(), Thread::multi (preprocessor));
214  }
215  for (ssize_t i = 0; i < empirical_statistic.size(); ++i) {
216  if (global_enhanced_count[i] > 0)
217  empirical_statistic[i] /= static_cast<default_type> (global_enhanced_count[i]);
218  }
219  }

223  // Precompute the default statistic image and enhanced statistic. We need to precompute this for calculating the uncorrected p-values.
224  template <class StatsType>
225  void precompute_default_permutation (const StatsType& stats_calculator,
226  const std::shared_ptr<EnhancerBase> enhancer,
227  const vector_type& empirical_enhanced_statistic,
228  vector_type& default_enhanced_statistics,
229  std::shared_ptr<vector_type> default_enhanced_statistics_neg,
230  vector_type& default_statistics)
231  {
232  vector<size_t> default_labelling (stats_calculator.num_subjects());
233  for (size_t i = 0; i < default_labelling.size(); ++i)
234  default_labelling[i] = i;
235  stats_calculator (default_labelling, default_statistics);
236  (*enhancer) (default_statistics, default_enhanced_statistics);
237 
238  if (empirical_enhanced_statistic.size())
239  default_enhanced_statistics /= empirical_enhanced_statistic;
240 
241  // Compute the opposite contrast
242  if (default_enhanced_statistics_neg) {
243  default_statistics = -default_statistics;
244 
245  (*enhancer) (default_statistics, *default_enhanced_statistics_neg);
246 
247  if (empirical_enhanced_statistic.size())
248  (*default_enhanced_statistics_neg) /= empirical_enhanced_statistic;
249 
250  // revert default_statistics to positive contrast for output
251  default_statistics = -default_statistics;
252  }

255  template <class StatsType>
256  inline void run_permutations (PermutationStack& perm_stack,
257  const StatsType& stats_calculator,
258  const std::shared_ptr<EnhancerBase> enhancer,
259  const vector_type& empirical_enhanced_statistic,
260  const vector_type& default_enhanced_statistics,
261  const std::shared_ptr<vector_type> default_enhanced_statistics_neg,
262  vector_type& perm_dist_pos,
263  std::shared_ptr<vector_type> perm_dist_neg,
264  vector_type& uncorrected_pvalues,
265  std::shared_ptr<vector_type> uncorrected_pvalues_neg)
266  {
267  vector<size_t> global_uncorrected_pvalue_count (stats_calculator.num_elements(), 0);
268  std::shared_ptr< vector<size_t> > global_uncorrected_pvalue_count_neg;
269  if (perm_dist_neg)
270  global_uncorrected_pvalue_count_neg.reset (new vector<size_t> (stats_calculator.num_elements(), 0));
271 
272  {
273  Processor<StatsType> processor (stats_calculator, enhancer,
274  empirical_enhanced_statistic,
275  default_enhanced_statistics, default_enhanced_statistics_neg,
276  perm_dist_pos, perm_dist_neg,
277  global_uncorrected_pvalue_count, global_uncorrected_pvalue_count_neg);
278  Thread::run_queue (perm_stack, Permutation(), Thread::multi (processor));
279  }
280 
281  for (size_t i = 0; i < stats_calculator.num_elements(); ++i) {
282  uncorrected_pvalues[i] = global_uncorrected_pvalue_count[i] / default_type(perm_stack.num_permutations);
283  if (perm_dist_neg)
284  (*uncorrected_pvalues_neg)[i] = (*global_uncorrected_pvalue_count_neg)[i] / default_type(perm_stack.num_permutations);
285  }
