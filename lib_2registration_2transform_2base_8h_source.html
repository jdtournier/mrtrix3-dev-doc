235  void compute_halfspace_transformations() {
236  Eigen::Matrix<ParameterType, 4, 4> tmp;
237  tmp.setIdentity();
238  tmp.template block<3,4>(0,0) = trafo.matrix();
239  assert ((tmp.template block<3,3>(0,0).isApprox (trafo.linear())));
240  assert (tmp.determinant() > 0);
241  tmp = tmp.sqrt().eval();
242  trafo_half.matrix() = tmp.template block<3,4>(0,0);
243  trafo_half_inverse.matrix() = trafo_half.inverse().matrix();
244  assert (trafo.matrix().isApprox ((trafo_half*trafo_half).matrix()));
245  assert (trafo.inverse().matrix().isApprox ((trafo_half_inverse*trafo_half_inverse).matrix()));
