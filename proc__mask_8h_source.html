99 
100  // Test to see if the image has already been re-gridded to match the fixel image
101  // If it has, can do a direct import
102  if (Image::dimensions_match (v_5tt, in_anat, 0, 3)) {
103  INFO ("5TT image dimensions match fixel image - importing directly");
104  auto v_anat = in_anat.voxel();
105  Image::copy (v_anat, v_5tt);
106  } else {
107  Image::ThreadedLoop threaded_loop ("resampling ACT 5TT image to fixel image space...", in_dwi, 0, 3);
108  ResampleFunctor<Set> functor (in_dwi, in_anat, *act_5tt);
109  threaded_loop.run (functor);
110  }
111 
112  // Once all of the 5TT data has been read in, use it to derive the processing mask
113  Image::LoopInOrder loop (v_5tt, 0, 3);
114  v_5tt[3] = 2; // Access the WM fraction
115  for (auto l = loop (v_5tt, mask); l; ++l)
116  mask.value() = Math::pow2<float> (v_5tt.value()); // Processing mask value is the square of the WM fraction
117 
118  } else {
119 
120  auto dwi = in_dwi.voxel();
121  auto f = [] (decltype(dwi)& dwi, decltype(mask)& mask) {
122  typedef typename std::remove_reference<decltype(in_dwi)>::type::value_type value_type;
123  mask.value() = (dwi.value() && std::isfinite (static_cast<value_type> (dwi.value()))) ? 1.0 : 0.0;
124  };
125  Image::ThreadedLoop ("Creating homogeneous processing mask...", dwi, 0, 3).run (f, dwi, mask);
