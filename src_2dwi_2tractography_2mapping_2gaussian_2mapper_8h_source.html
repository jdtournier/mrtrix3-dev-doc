119  void TrackMapper::voxelise (const Streamline<>& tck, Cont& output) const
120  {
121 
122  size_t prev = 0;
123  const size_t last = tck.size() - 1;
124 
125  Eigen::Vector3i vox;
126  for (size_t i = 0; i != last; ++i) {
127  vox = round (scanner2voxel * tck[i]);
128  if (check (vox, info)) {
129  const Eigen::Vector3f dir ((tck[i+1] - tck[prev]).normalized());
130  const float factor = tck_index_to_factor (i);
131  add_to_set (output, vox, dir, 1.0f, factor);
132  }
133  prev = i;
134  }
135 
136  vox = round (scanner2voxel * tck[last]);
137  if (check (vox, info)) {
138  const Eigen::Vector3f dir ((tck[last] - tck[prev]).normalized());
139  const float factor = tck_index_to_factor (last);
140  add_to_set (output, vox, dir, 1.0f, factor);

153  template <class Cont>
154  void TrackMapper::voxelise_precise (const Streamline<>& tck, Cont& out) const
155  {
156  typedef Eigen::Vector3f PointF;
157 
158  static const float accuracy = Math::pow2 (0.005 * std::min (info.spacing (0), std::min (info.spacing (1), info.spacing (2))));
159 
160  if (tck.size() < 2)
161  return;
162 
163  Math::Hermite<float> hermite (0.1);
164 
165  const PointF tck_proj_front = (tck[ 0 ] * 2.0) - tck[ 1 ];
166  const PointF tck_proj_back = (tck[tck.size()-1] * 2.0) - tck[tck.size()-2];
167 
168  unsigned int p = 0;
169  PointF p_voxel_exit = tck.front();
170  float mu = 0.0;
171  bool end_track = false;
172  Eigen::Vector3i next_voxel (round (scanner2voxel * tck.front()));
173 
174  do {
175 
176  const PointF p_voxel_entry (p_voxel_exit);
177  PointF p_prev (p_voxel_entry);
178  float length = 0.0;
179  const float index_voxel_entry = float(p) + mu;
180  const Eigen::Vector3i this_voxel = next_voxel;
181 
182  while ((p != tck.size()) && ((next_voxel = round (scanner2voxel * tck[p])) == this_voxel)) {
183  length += (p_prev - tck[p]).norm();
184  p_prev = tck[p];
185  ++p;
186  mu = 0.0;
187  }
188 
189  if (p == tck.size()) {
190  p_voxel_exit = tck.back();
191  end_track = true;
192  } else {
193 
194  float mu_min = mu;
195  float mu_max = 1.0;
196 
197  const PointF* p_one = (p == 1) ? &tck_proj_front : &tck[p - 2];
198  const PointF* p_four = (p == tck.size() - 1) ? &tck_proj_back : &tck[p + 1];
199 
200  PointF p_min = p_prev;
201  PointF p_max = tck[p];
202 
203  while ((p_min - p_max).squaredNorm() > accuracy) {
204 
205  mu = 0.5 * (mu_min + mu_max);
206  hermite.set (mu);
207  const PointF p_mu = hermite.value (*p_one, tck[p - 1], tck[p], *p_four);
208  const Eigen::Vector3i mu_voxel = round (scanner2voxel * p_mu);
209 
210  if (mu_voxel == this_voxel) {
211  mu_min = mu;
212  p_min = p_mu;
213  } else {
214  mu_max = mu;
215  p_max = p_mu;
216  next_voxel = mu_voxel;
217  }
218 
219  }
220  p_voxel_exit = p_max;
221 
222  }
223 
224  length += (p_prev - p_voxel_exit).norm();
225  PointF traversal_vector = (p_voxel_exit - p_voxel_entry).normalized();
226  if (traversal_vector.allFinite() && check (this_voxel, info)) {
227  const float index_voxel_exit = float(p) + mu;
228  const size_t mean_tck_index = std::round (0.5 * (index_voxel_entry + index_voxel_exit));
229  const float factor = tck_index_to_factor (mean_tck_index);
230  add_to_set (out, this_voxel, traversal_vector, length, factor);
231  }
232 
233  } while (!end_track);
234 
235  }

240  void TrackMapper::voxelise_ends (const Streamline<>& tck, Cont& out) const
241  {
242  for (size_t end = 0; end != 2; ++end) {
243  const Eigen::Vector3i vox = round (scanner2voxel * (end ? tck.back() : tck.front()));
244  if (check (vox, info)) {
245  const Eigen::Vector3f dir = (end ? (tck[tck.size()-1] - tck[tck.size()-2]) : (tck[0] - tck[1])).normalized();
246  const float factor = (end ? factors.back() : factors.front());
247  add_to_set (out, vox, dir, 1.0f, factor);
248  }

280  inline float TrackMapper::tck_index_to_factor (const size_t i) const
281  {
282  const float ideal_index = float(i) / float(upsampler.get_ratio());
283  const size_t lower_index = std::max (size_t(std::floor (ideal_index)), size_t(0));
284  const size_t upper_index = std::min (size_t(std::ceil (ideal_index)), size_t(factors.size() - 1));
285  const float mu = ideal_index - float(lower_index);
286  return ((mu * factors[upper_index]) + ((1.0f-mu) * factors[lower_index]));
287  }
