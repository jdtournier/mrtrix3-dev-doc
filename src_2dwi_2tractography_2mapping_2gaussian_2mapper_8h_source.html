54  void set_gaussian_FWHM (const default_type FWHM)
55  {
56  if (track_statistic != GAUSSIAN)
57  throw Exception ("Cannot set Gaussian FWHM unless the track statistic is Gaussian");
58  const default_type theta = FWHM / (2.0 * std::sqrt (2.0 * std::log (2.0)));
59  gaussian_denominator = 2.0 * Math::pow2 (theta);
60  }

63  // Have to re-define the functor here, so that the alternative voxelise() methods can be called
64  // (Can't define these virtual because they're templated)
65  // Also means that the call to run_queue() must cast to Gaussian::TrackMapper rather than using base class
66  template <class Cont>
67  bool operator() (Streamline<>& in, Cont& out) const
68  {
69  out.clear();
70  out.index = in.index;
71  out.weight = in.weight;
72  if (in.empty())
73  return true;
74  if (preprocess (in, out) || map_zero) {
75  Streamline<> temp;
76  upsampler (in, temp);
77  if (precise)
78  voxelise_precise (temp, out);
79  else if (ends_only)
80  voxelise_ends (temp, out);
81  else
82  voxelise (temp, out);
83  postprocess (temp, out);
84  }
85  return true;
86  }

119  template <class Cont>
120  void TrackMapper::voxelise (const Streamline<>& tck, Cont& output) const
121  {
122 
123  size_t prev = 0;
124  const size_t last = tck.size() - 1;
125 
126  Eigen::Vector3i vox;
127  for (size_t i = 0; i != last; ++i) {
128  vox = round (scanner2voxel * tck[i]);
129  if (check (vox, info)) {
130  const Eigen::Vector3 dir ((tck[i+1] - tck[prev]).cast<default_type>().normalized());
131  const default_type factor = tck_index_to_factor (i);
132  add_to_set (output, vox, dir, 1.0, factor);
133  }
134  prev = i;
135  }
136 
137  vox = round (scanner2voxel * tck[last]);
138  if (check (vox, info)) {
139  const Eigen::Vector3 dir ((tck[last] - tck[prev]).cast<default_type>().normalized());
140  const default_type factor = tck_index_to_factor (last);
141  add_to_set (output, vox, dir, 1.0f, factor);
142  }
143 
144  for (auto& i : output)
145  i.normalize();
146 
147  }

154  template <class Cont>
155  void TrackMapper::voxelise_precise (const Streamline<>& tck, Cont& out) const
156  {
157 
158  using point_type = Streamline<>::point_type;
159  using value_type = Streamline<>::value_type;
160 
161 
162  static const default_type accuracy = Math::pow2 (0.005 * std::min (info.spacing (0), std::min (info.spacing (1), info.spacing (2))));
163 
164  if (tck.size() < 2)
165  return;
166 
167  Math::Hermite<value_type> hermite (0.1);
168 
169  const point_type tck_proj_front = (tck[ 0 ] * 2.0) - tck[ 1 ];
170  const point_type tck_proj_back = (tck[tck.size()-1] * 2.0) - tck[tck.size()-2];
171 
172  unsigned int p = 0;
173  point_type p_voxel_exit = tck.front();
174  default_type mu = 0.0;
175  bool end_track = false;
176  Eigen::Vector3i next_voxel (round (scanner2voxel * tck.front()));
177 
178  do {
179 
180  const point_type p_voxel_entry (p_voxel_exit);
181  point_type p_prev (p_voxel_entry);
182  default_type length = 0.0;
183  const default_type index_voxel_entry = default_type(p) + mu;
184  const Eigen::Vector3i this_voxel = next_voxel;
185 
186  while ((p != tck.size()) && ((next_voxel = round (scanner2voxel * tck[p])) == this_voxel)) {
187  length += (p_prev - tck[p]).norm();
188  p_prev = tck[p];
189  ++p;
190  mu = 0.0;
191  }
192 
193  if (p == tck.size()) {
194  p_voxel_exit = tck.back();
195  end_track = true;
196  } else {
197 
198  default_type mu_min = mu;
199  default_type mu_max = 1.0;
200 
201  const point_type* p_one = (p == 1) ? &tck_proj_front : &tck[p - 2];
202  const point_type* p_four = (p == tck.size() - 1) ? &tck_proj_back : &tck[p + 1];
203 
204  point_type p_min = p_prev;
205  point_type p_max = tck[p];
206 
207  while ((p_min - p_max).squaredNorm() > accuracy) {
208 
209  mu = 0.5 * (mu_min + mu_max);
210  hermite.set (mu);
211  const point_type p_mu = hermite.value (*p_one, tck[p - 1], tck[p], *p_four);
212  const Eigen::Vector3i mu_voxel = round (scanner2voxel * p_mu);
213 
214  if (mu_voxel == this_voxel) {
215  mu_min = mu;
216  p_min = p_mu;
217  } else {
218  mu_max = mu;
219  p_max = p_mu;
220  next_voxel = mu_voxel;
221  }
222 
223  }
224  p_voxel_exit = p_max;
225 
226  }
227 
228  length += (p_prev - p_voxel_exit).norm();
229  Eigen::Vector3 traversal_vector = (p_voxel_exit - p_voxel_entry).cast<default_type>().normalized();
230  if (traversal_vector.allFinite() && check (this_voxel, info)) {
231  const default_type index_voxel_exit = default_type(p) + mu;
232  const size_t mean_tck_index = std::round (0.5 * (index_voxel_entry + index_voxel_exit));
233  const default_type factor = tck_index_to_factor (mean_tck_index);
234  add_to_set (out, this_voxel, traversal_vector, length, factor);
235  }
236 
237  } while (!end_track);
238 
239  }

243  template <class Cont>
244  void TrackMapper::voxelise_ends (const Streamline<>& tck, Cont& out) const
245  {
246  for (size_t end = 0; end != 2; ++end) {
247  const Eigen::Vector3i vox = round (scanner2voxel * (end ? tck.back() : tck.front()));
248  if (check (vox, info)) {
249  const Eigen::Vector3 dir = (end ? (tck[tck.size()-1] - tck[tck.size()-2]) : (tck[0] - tck[1])).cast<default_type>().normalized();
250  const default_type factor = (end ? factors.back() : factors.front());
251  add_to_set (out, vox, dir, 1.0, factor);
252  }
253  }
254  }

284  inline default_type TrackMapper::tck_index_to_factor (const size_t i) const
285  {
286  const default_type ideal_index = default_type(i) / default_type(upsampler.get_ratio());
287  const size_t lower_index = std::max (size_t(std::floor (ideal_index)), size_t(0));
288  const size_t upper_index = std::min (size_t(std::ceil (ideal_index)), size_t(factors.size() - 1));
289  const default_type mu = ideal_index - default_type(lower_index);
290  return ((mu * factors[upper_index]) + ((1.0-mu) * factors[lower_index]));
291  }
