129 void TrackMapper::voxelise (const Streamline<>& tck, Cont& output) const
130 {
131 
132  size_t prev = 0;
133  const size_t last = tck.size() - 1;
134 
135  Point<int> vox;
136  for (size_t i = 0; i != last; ++i) {
137  vox = round (transform.scanner2voxel (tck[i]));
138  if (check (vox, info)) {
139  const Point<float> dir ((tck[i+1] - tck[prev]).normalise());
140  const float factor = tck_index_to_factor (i);
141  add_to_set (output, vox, dir, 1.0f, factor);

143  prev = i;
144  }
145 
146  vox = round (transform.scanner2voxel (tck[last]));
147  if (check (vox, info)) {

149  const float factor = tck_index_to_factor (last);
150  add_to_set (output, vox, dir, 1.0f, factor);

163 template <class Cont>
164 void TrackMapper::voxelise_precise (const Streamline<>& tck, Cont& out) const
165 {
166  typedef Point<float> PointF;
167 
168  static const float accuracy = Math::pow2 (0.005 * minvalue (info.vox (0), info.vox (1), info.vox (2)));
169 
170  if (tck.size() < 2)
171  return;
172 
173  Math::Hermite<float> hermite (0.1);
174 
175  const PointF tck_proj_front = (tck[ 0 ] * 2.0) - tck[ 1 ];
176  const PointF tck_proj_back = (tck[tck.size()-1] * 2.0) - tck[tck.size()-2];
177 
178  unsigned int p = 0;
179  PointF p_voxel_exit = tck.front();
180  float mu = 0.0;
181  bool end_track = false;
182  Point<int> next_voxel (round (transform.scanner2voxel (tck.front())));
183 
184  do {
185 
186  const PointF p_voxel_entry (p_voxel_exit);
187  PointF p_prev (p_voxel_entry);
188  float length = 0.0;
189  const float index_voxel_entry = float(p) + mu;
190  const Point<int> this_voxel = next_voxel;
191 
192  while ((p != tck.size()) && ((next_voxel = round (transform.scanner2voxel (tck[p]))) == this_voxel)) {
193  length += dist (p_prev, tck[p]);
194  p_prev = tck[p];
195  ++p;
196  mu = 0.0;
197  }
198 
199  if (p == tck.size()) {
200  p_voxel_exit = tck.back();
201  end_track = true;
202  } else {
203 
204  float mu_min = mu;
205  float mu_max = 1.0;
206 
207  const PointF* p_one = (p == 1) ? &tck_proj_front : &tck[p - 2];
208  const PointF* p_four = (p == tck.size() - 1) ? &tck_proj_back : &tck[p + 1];
209 
210  PointF p_min = p_prev;
211  PointF p_max = tck[p];
212 
213  while (dist2 (p_min, p_max) > accuracy) {
214 
215  mu = 0.5 * (mu_min + mu_max);
216  hermite.set (mu);
217  const PointF p_mu = hermite.value (*p_one, tck[p - 1], tck[p], *p_four);
218  const Point<int> mu_voxel = round (transform.scanner2voxel (p_mu));
219 
220  if (mu_voxel == this_voxel) {
221  mu_min = mu;
222  p_min = p_mu;
223  } else {
224  mu_max = mu;
225  p_max = p_mu;
226  next_voxel = mu_voxel;
227  }
228 
229  }
230  p_voxel_exit = p_max;
231 
232  }
233 
234  length += dist (p_prev, p_voxel_exit);
235  PointF traversal_vector ((p_voxel_exit - p_voxel_entry).normalise());
236  if (traversal_vector.valid() && check (this_voxel, info)) {
237  const float index_voxel_exit = float(p) + mu;
238  const size_t mean_tck_index = std::round (0.5 * (index_voxel_entry + index_voxel_exit));
239  const float factor = tck_index_to_factor (mean_tck_index);
240  add_to_set (out, this_voxel, traversal_vector, length, factor);
241  }
242 
243  } while (!end_track);
244 
245 }

250 void TrackMapper::voxelise_ends (const Streamline<>& tck, Cont& out) const
251 {
252  for (size_t end = 0; end != 2; ++end) {
253  const Point<int> vox = round (transform.scanner2voxel (end ? tck.back() : tck.front()));
254  if (check (vox, info)) {
255  const Point<float> dir = (end ? (tck[tck.size()-1] - tck[tck.size()-2]) : (tck[0] - tck[1])).normalise();
256  const float factor = (end ? factors.back() : factors.front());
257  add_to_set (out, vox, dir, 1.0f, factor);
258  }
259  }

290 inline float TrackMapper::tck_index_to_factor (const size_t i) const
291 {
292  const float ideal_index = float(i) / float(upsampler.get_ratio());
293  const size_t lower_index = std::max (size_t(std::floor (ideal_index)), size_t(0));
294  const size_t upper_index = std::min (size_t(std::ceil (ideal_index)), size_t(factors.size() - 1));
295  const float mu = ideal_index - float(lower_index);
296  return ((mu * factors[upper_index]) + ((1.0f-mu) * factors[lower_index]));
297 }
