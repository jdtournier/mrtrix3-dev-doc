72  void set_voxel_size (const std::vector<default_type>& voxel_size)
73  {
74  if (voxel_size.size() != 3)
75  throw Exception ("the voxel size must be defined using a value for all three dimensions.");
76 
77  Eigen::Vector3 original_extent;
78  for (size_t j = 0; j < 3; ++j) {
79  if (voxel_size[j] <= 0.0)
80  throw Exception ("the voxel size must be larger than zero");
81  original_extent[j] = axes_[j].size * axes_[j].spacing;
82  axes_[j].size = std::round (axes_[j].size * axes_[j].spacing / voxel_size[j] - 0.0001); // round down at .5
83  // Here we adjust the translation to ensure the image extent is centered wrt the original extent.
84  // This is important when the new voxel size is not an exact multiple of the original extent
85  for (size_t i = 0; i < 3; ++i)
86  transform_(i,3) += 0.5 * ((voxel_size[j] - axes_[j].spacing) + (original_extent[j] - (axes_[j].size * voxel_size[j]))) * transform_(i,j);
87  axes_[j].spacing = voxel_size[j];
88  }
