56  transform (inv_warp),

62  template <class InvDFVoxelType>
63  void operator() (InvDFVoxelType& inv_warp) {
64 
65  Point<float> truth = transform.voxel2scanner (inv_warp);
66  Point<float> current;
67  for (inv_warp[3] = 0; inv_warp[3] < 3; ++inv_warp[3])
68  current[(size_t)inv_warp[3]] = inv_warp.value();
69 
70  size_t iter = 0;
71  value_type error = std::numeric_limits<value_type>::max();
72  while (iter < max_iter && error > error_tolerance) {
73  error = update (current, truth);
74  ++iter;
75  }
76 
77  for (inv_warp[3] = 0; inv_warp[3] < 3; ++inv_warp[3])
78  inv_warp.value() = current[(size_t)inv_warp[3]];
79  }
80 
81  private:
82 
83  value_type update (Point<float>& current, Point<float>& truth) {
84 
85  warped_moving_positions.scanner (current);
86  Point<float> discrepancy;
87  value_type error = 0.0;
88  for (warped_moving_positions[3] = 0; warped_moving_positions[0] < 3; ++warped_moving_positions[3]) {
89  size_t dim = warped_moving_positions[3];
90  discrepancy[dim] = truth[dim] - warped_moving_positions.value();
91  error += Math::pow2 (discrepancy[dim]);
92  }
93  current += discrepancy;
94  return error;
95  }

149  check_dimensions (warp, inv_warp);
150 
151  if (!is_initialised)
152  displacement2deformation (inv_warp, inv_warp);
153 
154  BufferScratch<float> positions (*this);
155  BufferScratch<float>::voxel_type positions_vox (positions);
156  Image::Registration::displacement2deformation (positions_vox, positions_vox);
157  Interp::Cubic<BufferScratch<float>::voxel_type> interp (positions_vox);
158 
159  BufferScratch<float> warped_positions (*this);
160  BufferScratch<float>::voxel_type warped_positions_vox (warped_positions);
161 
162  LoopInOrder loop (warp, 0, 3);
163  for (loop.start (warp, warped_positions_vox); loop.ok(); loop.next (warp, warped_positions_vox)) {
164  Point<float> moving_pos;
165  for (warp[3] = 0; warp[3] < 3; ++warp[3])
166  moving_pos[(size_t)warp[3]] = warp.value();
167  interp.scanner (moving_pos);
168  for (interp[3] = 0, warped_positions_vox[3] = 0; interp[3] < 3; ++interp[3], ++warped_positions_vox[3])
169  warped_positions_vox.value() = interp.value();
170  }
171 
172  ThreadKernel<BufferScratch<float>::voxel_type>
173  warp_updator (warped_positions_vox, inv_warp, max_iter, error_tolerance);
174  ThreadedLoop ("inverting warp field...", inv_warp, 1, 0, 3)
175  .run (warp_updator, inv_warp);
176  }
