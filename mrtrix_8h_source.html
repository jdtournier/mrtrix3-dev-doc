206  {

218  template <class T> inline T to (const std::string& string)
219  {
220  const std::string stripped (strip (string));
221  std::istringstream stream (stripped);
222  T value;
223  stream >> value;
224  if (stream.fail()) {
225  if (std::is_floating_point<T>::value) {
226  const std::string lstring = lowercase (stripped);
227  if (lstring == "nan")
228  return std::numeric_limits<T>::quiet_NaN();
229  else if (lstring == "-nan")
230  return -std::numeric_limits<T>::quiet_NaN();
231  else if (lstring == "inf")
232  return std::numeric_limits<T>::infinity();
233  else if (lstring == "-inf")
234  return -std::numeric_limits<T>::infinity();
235  }
236  throw Exception ("error converting string \"" + string + "\" to type \"" + typeid(T).name() + "\"");
237  } else if (!stream.eof()) {
238  throw Exception ("incomplete use of string \"" + string + "\" in conversion to type \"" + typeid(T).name() + "\"");
239  }
240  return value;
241  }

266  template <> inline cfloat to<cfloat> (const std::string& string)
267  {
268  if (!string.size())
269  throw Exception ("cannot convert empty string to complex float");
270 
271  const std::string stripped = strip (string);
272  vector<cfloat> candidates;
273  for (ssize_t i = -1; i <= ssize_t(stripped.size()); ++i) {
274  std::string first, second;
275  if (i == -1) {
276  first = "0";
277  second = stripped;
278  } else if (i == ssize_t(stripped.size())) {
279  first = stripped;
280  second = "0i";
281  } else {
282  if (!(stripped[i] == '+' || stripped[i] == '-'))
283  continue;
284  first = stripped.substr(0, i);
285  second = stripped.substr(stripped[i] == '-' ? i : i+1);
286  }
287  if (!(second.back() == 'i' || second.back() == 'j'))
288  continue;
289  second.pop_back();
290  if (second.empty() || second == "-" || second == "+")
291  second.push_back ('1');
292  try {
293  candidates.push_back (cfloat (to<float> (first), to<float> (second)));
294  } catch (Exception&) { }
295  }
296 
297  if (candidates.empty())
298  throw Exception ("error converting string \"" + string + "\" to complex float (no valid conversion)");
299 
300  for (size_t i = 1; i != candidates.size(); ++i) {
301  if (!(candidates[i].real() == candidates[0].real() ||
302  (std::isnan (candidates[i].real()) && std::isnan (candidates[0].real()))))
303  throw Exception ("error converting string \"" + string + "\" to complex float (ambiguity in real component)");
304  if (!(candidates[i].imag() == candidates[0].imag() ||
305  (std::isnan (candidates[i].imag()) && std::isnan (candidates[0].imag()))))
306  throw Exception ("error converting string \"" + string + "\" to complex float (ambiguity in imaginary component)");
307  }
308  return candidates[0];
309  }
