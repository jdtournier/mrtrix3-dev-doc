30  class RigidLinearNonSymmetricUpdate {
31  public:
32  template <typename ValueType>
33  inline bool operator() (Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& newx,
34  const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& x,
35  const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& g,
36  ValueType step_size) {
37  assert (newx.size() == x.size());
38  assert (g.size() == x.size());
39  newx = x - step_size * g;
40  Eigen::Matrix<ValueType, 4, 4> X;
41  Registration::Transform::param_vec2mat(newx, X);
42  Eigen::Matrix<ValueType, 3, 3> L(X.template block<3,3>(0,0));
43  Eigen::Matrix<ValueType, 3, 3> R;
44  project_linear2rotation(L, R);
45  X.template block<3,3>(0,0) = R;
46  Registration::Transform::param_mat2vec(X, newx);
47  return !(newx.isApprox(x));
48  }

68  template<class ValueType>
69  void project_linear2rotation(const Eigen::Matrix<ValueType, 3, 3>& linear, Eigen::Matrix<ValueType, 3, 3>& rotation) const
70  {
71  Eigen::JacobiSVD<Eigen::Matrix<ValueType, 3, 3>> svd(linear, Eigen::ComputeFullU | Eigen::ComputeFullV);
72 
73  ValueType x = (svd.matrixU() * svd.matrixV().adjoint()).determinant(); // so x has absolute value 1
74  Eigen::Matrix<ValueType, 3, 1> sv(svd.singularValues());
75  sv.coeffRef(0) *= x;
76  Eigen::Matrix<ValueType, 3, 3> m(svd.matrixU());
77  m.col(0) /= x;
78  rotation = m * svd.matrixV().adjoint();
79  assert((rotation * rotation.transpose().eval()).isApprox(Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>::Identity(3,3)) && "orthonormal?");
80  }
