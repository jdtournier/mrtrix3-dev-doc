45  class VersorUpdate {
46  public:
47  template <typename ValueType>
48  inline bool operator() (Math::Vector<ValueType>& newx,
49  const Math::Vector<ValueType>& x,
50  const Math::Vector<ValueType>& g,
51  ValueType step_size) {
52 
53  Vector<ValueType> axis (3);
54  axis[0] = g[0];
55  axis[1] = g[1];
56  axis[2] = g[2];
57  Versor<ValueType> gradient_rotation;
58  gradient_rotation.set (axis, -step_size * Math::norm (axis));
59 
60  Vector<ValueType> right_part (3);
61  right_part[0] = x[0];
62  right_part[1] = x[1];
63  right_part[2] = x[2];
64 
65  Versor<ValueType> current_rotation;
66  current_rotation.set (right_part);
67 
68  Versor<ValueType> new_rotation = current_rotation * gradient_rotation;
69 
70  newx[0] = new_rotation[1];
71  newx[1] = new_rotation[2];
72  newx[2] = new_rotation[3];
73  newx[3] = x[3] - step_size * g[3];
74  newx[4] = x[4] - step_size * g[4];
75  newx[5] = x[5] - step_size * g[5];
76 
77  bool changed = false;
78  for (size_t n = 0; n < x.size(); ++n) {
79  if (newx[n] != x[n])
80  changed = true;
81  }
82  return changed;
83  }
84  };

92  inline bool operator() (Math::Vector<ValueType>& newx,
93  const Math::Vector<ValueType>& x,
94  Math::Vector<ValueType>& g,
95  ValueType step_size) {
96  Math::Vector<ValueType> vx (x.sub(0,4));
97  Math::Vector<ValueType> vg (g.sub(0,4));
98 
99  ValueType dp = Math::dot (vx, vg);

104  return false;
105 
106  Math::Vector<ValueType> v (newx.sub(0,4));
107  Math::normalise (v);
108  return newx != x;
109  }

143  void get_jacobian_wrt_params (const PointType& p, Matrix<ValueType>& jacobian) const {
144 
145  const ValueType vw = versor_[0];
146  const ValueType vx = versor_[1];
147  const ValueType vy = versor_[2];
148  const ValueType vz = versor_[3];
149 
150  jacobian.resize(3, 6);
151  jacobian.zero();
152 
153  const double px = p[0] - this->centre[0];
154  const double py = p[1] - this->centre[1];
155  const double pz = p[2] - this->centre[2];
156 
157  const double vxx = vx * vx;
158  const double vyy = vy * vy;
159  const double vzz = vz * vz;
160  const double vww = vw * vw;
161  const double vxy = vx * vy;
162  const double vxz = vx * vz;
163  const double vxw = vx * vw;
164  const double vyz = vy * vz;
165  const double vyw = vy * vw;
166  const double vzw = vz * vw;
167 
168  jacobian(0,0) = 2.0 * ( ( vyw + vxz ) * py + ( vzw - vxy ) * pz ) / vw;
169  jacobian(1,0) = 2.0 * ( ( vyw - vxz ) * px - 2 * vxw * py + ( vxx - vww ) * pz ) / vw;
170  jacobian(2,0) = 2.0 * ( ( vzw + vxy ) * px + ( vww - vxx ) * py - 2 * vxw * pz ) / vw;
171  jacobian(0,1) = 2.0 * ( -2 * vyw * px + ( vxw + vyz ) * py + ( vww - vyy ) * pz ) / vw;
172  jacobian(1,1) = 2.0 * ( ( vxw - vyz ) * px + ( vzw + vxy ) * pz ) / vw;
173  jacobian(2,1) = 2.0 * ( ( vyy - vww ) * px + ( vzw - vxy ) * py - 2 * vyw * pz ) / vw;
174  jacobian(0,2) = 2.0 * ( -2 * vzw * px + ( vzz - vww ) * py + ( vxw - vyz ) * pz ) / vw;
175  jacobian(1,2) = 2.0 * ( ( vww - vzz ) * px - 2 * vzw * py + ( vyw + vxz ) * pz ) / vw;
176  jacobian(2,2) = 2.0 * ( ( vxw + vyz ) * px + ( vyw - vxz ) * py ) / vw;
177  jacobian(0,3) = 1.0;
178  jacobian(1,4) = 1.0;
179  jacobian(2,5) = 1.0;

280  jacobian(1,2) = jacobian(0,1);
