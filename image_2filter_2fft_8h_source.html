109 
110  Image::BufferScratch<cdouble> temp_data (info());
111  auto temp_voxel = temp_data.voxel();
112  Image::copy (input, temp_voxel);
113  if (progress) ++(*progress);
114 
115  for (std::vector<size_t>::const_iterator axis = axes_to_process.begin(); axis != axes_to_process.end(); ++axis) {
116  std::vector<size_t> axes = Stride::order (temp_voxel);
117  for (size_t n = 0; n < axes.size(); ++n) {
118  if (axes[n] == *axis) {
119  axes.erase (axes.begin() + n);
120  break;
121  }
122  }
123  FFTKernel<decltype(temp_voxel)> kernel (temp_voxel, *axis, inverse);
124  ThreadedLoop (temp_voxel, axes, 1).run (kernel);
125  if (progress) ++(*progress);
126  }
127 
128  if (centre_zero_) {
129  Image::LoopInOrder loop (output);
130  for (auto l = loop (output); l; ++l) {
131  Image::Nav::set_pos (temp_voxel, output);

133  temp_voxel[*flip_axis] = (temp_voxel[*flip_axis] >= (temp_voxel.dim (*flip_axis) / 2)) ?
134  (temp_voxel[*flip_axis] - (temp_voxel.dim (*flip_axis) / 2)) :
