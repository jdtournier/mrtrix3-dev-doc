101  template <class InputComplexVoxelType, class OutputComplexVoxelType>
102  void operator() (InputComplexVoxelType& input, OutputComplexVoxelType& output)
103  {
104 
105  Ptr<ProgressBar> progress;
106  if (message.size())
107  progress = new ProgressBar (message, axes_to_process.size() + 2);
108 
109  Image::BufferScratch<cdouble> temp_data (info());
110  Image::BufferScratch<cdouble>::voxel_type temp_voxel (temp_data);
111  Image::copy (input, temp_voxel);
112  if (progress) ++(*progress);
113 
114  for (std::vector<size_t>::const_iterator axis = axes_to_process.begin(); axis != axes_to_process.end(); ++axis) {
115  std::vector<size_t> axes = Stride::order (temp_voxel);
116  for (size_t n = 0; n < axes.size(); ++n) {
117  if (axes[n] == *axis) {
118  axes.erase (axes.begin() + n);
119  break;
120  }
121  }
122  FFTKernel< Image::BufferScratch<cdouble>::voxel_type > kernel (temp_voxel, *axis, inverse);
123  ThreadedLoop (temp_voxel, axes, 1).run (kernel);
124  if (progress) ++(*progress);
125  }
126 
127  if (centre_zero_) {
128  Image::LoopInOrder loop (output);
129  for (loop.start (output); loop.ok(); loop.next (output)) {
130  Image::Nav::set_pos (temp_voxel, output);
131  for (std::vector<size_t>::const_iterator flip_axis = axes_to_process.begin(); flip_axis != axes_to_process.end(); ++flip_axis)
132  temp_voxel[*flip_axis] = (temp_voxel[*flip_axis] >= (temp_voxel.dim (*flip_axis) / 2)) ?
133  (temp_voxel[*flip_axis] - (temp_voxel.dim (*flip_axis) / 2)) :
134  (temp_voxel[*flip_axis] + (temp_voxel.dim (*flip_axis) / 2));
135  output.value() = temp_voxel.value();
136  }
137  } else {
138  Image::copy (temp_voxel, output);
139  }
140  }
