117  void set_directions (Math::Matrix<float>& dirs_az_el, float angular_threshold) {
118  angular_threshold = angular_threshold * M_PI / 180.0;
119  Math::Matrix<float> vert (dirs_az_el.rows(), 3);
120  for (size_t d = 0; d < dirs_az_el.rows(); d++) {
121  vert(d,0) = sin(dirs_az_el(d,1)) * cos(dirs_az_el(d,0));
122  vert(d,1) = sin(dirs_az_el(d,1)) * sin(dirs_az_el(d,0));
123  vert(d,2) = cos(dirs_az_el(d,1));
124  }
125  dir_adjacency_matrix_.resize(dirs_az_el.rows(), dirs_az_el.rows(), 0.0);
126  for (size_t m = 0; m < dirs_az_el.rows(); m++) {
127  for (size_t n = m + 1; n < dirs_az_el.rows(); n++) {
128  float angle = Math::acos(Math::dot(vert.row(m), vert.row(n)));
129  if (angle > M_PI_2)
130  angle = M_PI - angle;
131  if (angle < angular_threshold) {
132  dir_adjacency_matrix_ (m,n) = 1;
133  dir_adjacency_matrix_ (n,m) = 1;
134  } else {
135  dir_adjacency_matrix_ (m,n) = 0;
136  dir_adjacency_matrix_ (n,m) = 0;
137  }
138  }
139  }

147  template <class MaskVoxelType>
148  const std::vector<std::vector<int> >& precompute_adjacency (MaskVoxelType& mask) {
149 
150  ProgressBar progress ("Precomputing voxel adjacency from mask...");
151  if (!dim_to_ignore_.size())
152  dim_to_ignore_.resize (mask.ndim(), false);
153  Image::BufferScratch<uint32_t> index_data (mask);
154  Image::BufferScratch<uint32_t>::voxel_type index_image (index_data);
155  // 1st pass, store mask image indices and their index in the array
156  Image::LoopInOrder loop (mask);
157  for (loop.start (mask, index_image); loop.ok(); loop.next (mask, index_image)) {
158  progress++;
159  if (mask.value() >= 0.5) {
160  // For each voxel, store the index within mask_indices for 2nd pass
161  index_image.value() = mask_indices_.size();
162  std::vector<int> index (mask.ndim());
163  for (size_t dim = 0; dim < mask.ndim(); dim++)
164  index[dim] = mask[dim];
165  mask_indices_.push_back (index);
166  } else {
167  index_image.value() = 0;
168  }
169  }
170  // Here we pre-compute the offsets for our neighbours in 3D space
171  std::vector< Point<int> > neighbour_offsets;
172  Point<int> offset;
173  for (offset[0] = -1; offset[0] <= 1; offset[0]++) {
174  for (offset[1] = -1; offset[1] <= 1; offset[1]++) {
175  for (offset[2] = -1; offset[2] <= 1; offset[2]++) {
176  if (offset.norm2() && (do_26_connectivity_ || offset.norm() == 1)) {
177  for (size_t dim = 0; dim != 3; ++dim)
178  if (Math::abs (offset[dim]) && dim_to_ignore_[dim])
179  continue;
180  neighbour_offsets.push_back (offset);
181  }
182  }
183  }
184  }
185  // 2nd pass, define adjacency
186  MaskVoxelType mask_neigh (mask);
187  for (std::vector<std::vector<int> >::const_iterator it = mask_indices_.begin(); it != mask_indices_.end(); ++it) {
188  progress++;
189  std::vector<uint32_t> neighbour_indices;
190  if (mask.ndim() == 4)
191  mask_neigh[3] = (*it)[3];
192  for (std::vector< Point<int> >::const_iterator offset = neighbour_offsets.begin(); offset != neighbour_offsets.end(); ++offset) {
193  for (size_t dim = 0; dim < 3; dim++)
194  mask_neigh[dim] = (*it)[dim] + (*offset)[dim];
195  if (Image::Nav::within_bounds (mask_neigh)) {
196  if (mask_neigh.value() >= 0.5)
197  neighbour_indices.push_back (Image::Nav::get_value_at_pos (index_image, mask_neigh));
198  }
199  }
200  // here we handle the 4th dimension
201  if (mask.ndim() == 4 && !dim_to_ignore_[3]) {
202  Image::Nav::set_pos (mask_neigh, *it);
203  if (dir_adjacency_matrix_.is_set()) {
204  for (int i = 0; i < mask.dim(3); i++) {
205  if (dir_adjacency_matrix_((*it)[3], i)) {
206  mask_neigh[3] = i;
207  if (mask_neigh.value() > 0.5)
208  neighbour_indices.push_back (Image::Nav::get_value_at_pos (index_image, mask_neigh));
209  }
210  }
211  } else {
212  if ((*it)[3] > 0) { // boundary check
213  mask_neigh[3] = (*it)[3] - 1;
214  if (mask_neigh.value() > 0.5)
215  neighbour_indices.push_back (Image::Nav::get_value_at_pos (index_image, mask_neigh));
216  }
217  if ((*it)[3] + 1 < mask.dim (3)) { // boundary check
218  mask_neigh[3] = (*it)[3] + 1;
219  if (mask_neigh.value() > 0.5)
220  neighbour_indices.push_back (Image::Nav::get_value_at_pos (index_image, mask_neigh));
221  }
222  }
223  }
224  adjacent_indices_.push_back(neighbour_indices);
225  }
226  return mask_indices_;
227  }

255  // use a non-recursive depth first search to agglomerate adjacent voxels
256  void depth_first_search (uint32_t root,
257  cluster& cluster,
258  std::vector<uint32_t>& labels) const {
259  uint32_t node = root;
260  std::stack<uint32_t> stack;
261  while (true) {
262  labels[node] = cluster.label;
263  stack.push (node);
264  cluster.size++;
265  if (next_neighbour (node, labels)) {
266  continue;
267  } else {
268  do {
269  if (stack.top() == root)
270  return;
271  stack.pop();
272  node = stack.top();
273  } while (!next_neighbour (node, labels));
274  }
275  }
276  }

365  template <class InputVoxelType, class OutputVoxelType>
366  void operator() (InputVoxelType& in, OutputVoxelType& out) {
367  std::vector<cluster> clusters;
368  std::vector<uint32_t> labels;
369 
370  Ptr<ProgressBar> progress;
371  if (message.size())
372  progress = new ProgressBar (message);
373 
374  Connector connector (do_26_connectivity_);
375  if (dim_to_ignore_.size())
376  connector.set_dim_to_ignore(dim_to_ignore_);
377  if (directions_.is_set())
378  connector.set_directions (directions_, angular_threshold_);
379  {
380  LogLevelLatch level(0);
381  connector.precompute_adjacency(in);
382  }
383  if (progress) ++(*progress);
384  std::vector<std::vector<int> > mask_indices = connector.run (clusters, labels);
385  if (progress) ++(*progress);
386  std::sort (clusters.begin(), clusters.end(), compare_clusters);
387  if (progress) ++(*progress);
388 
389  std::vector<int> label_lookup (clusters.size(), 0);
390  for (uint32_t c = 0; c < clusters.size(); c++)
391  label_lookup[clusters[c].label - 1] = c + 1;
392 
393  Image::LoopInOrder loop(out);
394  for (loop.start(out); loop.ok(); loop.next(out))
395  out.value() = 0;
396 
397  for (uint32_t i = 0; i < mask_indices.size(); i++) {
398  for (size_t dim = 0; dim < out.ndim(); dim++)
399  out[dim] = mask_indices[i][dim];
400  if (largest_only_) {
401  if (label_lookup[labels[i] - 1] == 1) {
402  out.value() = 1;
403  }
404  } else {
405  out.value() = label_lookup[labels[i] - 1];
406  }
407  }
408  }
