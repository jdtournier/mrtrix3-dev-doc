125  template <class MaskVoxelType>
126  const std::vector<std::vector<int> >& precompute_adjacency (MaskVoxelType& mask) {
127 
128  Image::BufferScratch<uint32_t> index_data (mask);
129  auto index_image = index_data.voxel();
130 
131  // 1st pass, store mask image indices and their index in the array
132  Image::LoopInOrder loop (mask);
133  for (auto l = loop (mask, index_image); l; ++l) {
134  if (mask.value() >= 0.5) {
135  // For each voxel, store the index within mask_indices for 2nd pass
136  index_image.value() = mask_indices.size();
137  std::vector<int> index (mask.ndim());
138  for (size_t dim = 0; dim < mask.ndim(); dim++)
139  index[dim] = mask[dim];
140  mask_indices.push_back (index);
141  } else {
142  index_image.value() = 0;
143  }
144  }
145  // Here we pre-compute the offsets for our neighbours in 4D space
146  std::vector< std::vector<int> > neighbour_offsets;
147  std::vector<int> offset (4);
148  for (offset[0] = -1; offset[0] <= 1; offset[0]++) {
149  for (offset[1] = -1; offset[1] <= 1; offset[1]++) {
150  for (offset[2] = -1; offset[2] <= 1; offset[2]++) {
151  for (offset[3] = -1; offset[3] <= 1; offset[3]++) {
152  if (!do_26_connectivity && ((abs(offset[0]) + abs(offset[1]) + abs(offset[2]) + abs(offset[3])) > 1))
153  continue;
154  if ((abs(offset[0]) && dim_to_ignore[0]) || (abs(offset[1]) && dim_to_ignore[1]) ||
155  (abs(offset[2]) && dim_to_ignore[2]) || (abs(offset[3]) && dim_to_ignore[3]))
156  continue;
157  neighbour_offsets.push_back (offset);
158  }
159  }
160  }
161  }
162  // 2nd pass, define adjacency
163  MaskVoxelType mask_neigh (mask);
164  for (std::vector<std::vector<int> >::const_iterator it = mask_indices.begin(); it != mask_indices.end(); ++it) {
165  std::vector<uint32_t> neighbour_indices;
166  for (std::vector< std::vector<int> >::const_iterator offset = neighbour_offsets.begin(); offset != neighbour_offsets.end(); ++offset) {
167  for (size_t dim = 0; dim < mask.ndim(); dim++)
168  mask_neigh[dim] = (*it)[dim] + (*offset)[dim];
169  if (Image::Nav::within_bounds (mask_neigh)) {
170  if (mask_neigh.value() >= 0.5)
171  neighbour_indices.push_back (Image::Nav::get_value_at_pos (index_image, mask_neigh));
172  }
173  }
174  adjacent_indices.push_back (neighbour_indices);
175  }
176 
177  return mask_indices;
178  }

206  // use a non-recursive depth first search to agglomerate adjacent voxels
207  void depth_first_search (uint32_t root,
208  cluster& cluster,
209  std::vector<uint32_t>& labels) const {
210  uint32_t node = root;
211  std::stack<uint32_t> stack;
212  while (true) {
213  labels[node] = cluster.label;
214  stack.push (node);
215  cluster.size++;
216  if (next_neighbour (node, labels)) {
217  continue;
218  } else {
219  do {
220  if (stack.top() == root)
221  return;
222  stack.pop();
223  node = stack.top();
224  } while (!next_neighbour (node, labels));
225  }
226  }
227  }
