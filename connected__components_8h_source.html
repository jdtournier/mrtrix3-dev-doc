112  template <class MaskImageType>
113  const std::vector<std::vector<int> >& precompute_adjacency (MaskImageType& mask) {
114 
115  auto index_image = Image<uint32_t>::scratch (mask);
116 
117  // 1st pass, store mask image indices and their index in the array
118  for (auto l = Loop (mask) (mask, index_image); l; ++l) {
119  if (mask.value() >= 0.5) {
120  // For each voxel, store the index within mask_indices for 2nd pass
121  index_image.value() = mask_indices.size();
122  std::vector<int> index (mask.ndim());
123  for (size_t dim = 0; dim < mask.ndim(); dim++)
124  index[dim] = mask.index(dim);
125  mask_indices.push_back (index);
126  } else {
127  index_image.value() = 0;
128  }
129  }
130  // Here we pre-compute the offsets for our neighbours in 4D space
131  std::vector< std::vector<int> > neighbour_offsets;
132  std::vector<int> offset (4);
133  for (offset[0] = -1; offset[0] <= 1; offset[0]++) {
134  for (offset[1] = -1; offset[1] <= 1; offset[1]++) {
135  for (offset[2] = -1; offset[2] <= 1; offset[2]++) {
136  for (offset[3] = -1; offset[3] <= 1; offset[3]++) {
137  if (!do_26_connectivity && ((abs(offset[0]) + abs(offset[1]) + abs(offset[2]) + abs(offset[3])) > 1))
138  continue;
139  if ((abs(offset[0]) && dim_to_ignore[0]) || (abs(offset[1]) && dim_to_ignore[1]) ||
140  (abs(offset[2]) && dim_to_ignore[2]) || (abs(offset[3]) && dim_to_ignore[3]))
141  continue;
142  neighbour_offsets.push_back (offset);
143  }
144  }
145  }
146  }
147  // 2nd pass, define adjacency
148  MaskImageType mask_neigh (mask);
149  for (std::vector<std::vector<int> >::const_iterator it = mask_indices.begin(); it != mask_indices.end(); ++it) {
150  std::vector<uint32_t> neighbour_indices;
151  for (std::vector< std::vector<int> >::const_iterator offset = neighbour_offsets.begin(); offset != neighbour_offsets.end(); ++offset) {
152  for (size_t dim = 0; dim < mask.ndim(); dim++)
153  mask_neigh.index(dim) = (*it)[dim] + (*offset)[dim];
154  if (!is_out_of_bounds (mask_neigh)) {
155  if (mask_neigh.value() >= 0.5) {
156  assign_pos_of (mask_neigh).to (index_image);
157  neighbour_indices.push_back (index_image.value());
158  }
159  }
160  }
161  adjacent_indices.push_back (neighbour_indices);
162  }
163 
164  return mask_indices;
165  }

193  // use a non-recursive depth first search to agglomerate adjacent voxels
194  void depth_first_search (uint32_t root,
195  cluster& cluster,
196  std::vector<uint32_t>& labels) const {
197  uint32_t node = root;
198  std::stack<uint32_t> stack;
199  while (true) {
200  labels[node] = cluster.label;
201  stack.push (node);
202  cluster.size++;
203  if (next_neighbour (node, labels)) {
204  continue;
205  } else {
206  do {
207  if (stack.top() == root)
208  return;
209  stack.pop();
210  node = stack.top();
211  } while (!next_neighbour (node, labels));
212  }
213  }
214  }

311

315  if (progress)
316  ++(*progress);
317 
318  std::vector<int> label_lookup (clusters.size(), 0);
319  for (uint32_t c = 0; c < clusters.size(); c++)
320  label_lookup[clusters[c].label - 1] = c + 1;
321 
322  for (auto l = Loop (out) (out); l; ++l)
323  out.value() = 0;
324

326  {
327  assign_pos_of (mask_indices[i]).to (out);
328  if (largest_only) {
329  if (label_lookup[labels[i] - 1] == 1)
330  out.value() = 1;
331  } else {
