59  // Perform connected components on the mask.
60  const vector<vector<int> >& run (vector<cluster>& clusters,
61  vector<uint32_t>& labels) const {
62  labels.resize (adjacent_indices.size(), 0);
63  uint32_t current_label = 1;
64  for (uint32_t i = 0; i < labels.size(); i++) {
65  // this node has not been already clustered
66  if (labels[i] == 0) {
67  cluster cluster;
68  cluster.label = current_label;
69  cluster.size = 0;
70  depth_first_search (i, cluster, labels);
71  clusters.push_back (cluster);
72  current_label++;
73  }
74  }
75  if (clusters.size() > std::numeric_limits<uint32_t>::max())
76  throw Exception ("The number of clusters is larger than can be labelled with an unsigned 32bit integer.");
77  return mask_indices;
78  }

81  // Perform connected components on data with the defined threshold. Assumes adjacency is the same as the mask.
82  template <class VectorType>
83  void run (vector<cluster>& clusters,
84  vector<uint32_t>& labels,
85  const VectorType& data,
86  const float threshold) const {
87  labels.resize (adjacent_indices.size(), 0);
88  uint32_t current_label = 1;
89  for (uint32_t i = 0; i < labels.size(); i++) {
90  // this node has not been already clustered and is above threshold
91  if (labels[i] == 0 && data[i] > threshold) {
92  cluster cluster;
93  cluster.label = current_label;
94  cluster.size = 0;
95  depth_first_search (i, cluster, labels, data, threshold);
96  clusters.push_back (cluster);
97  current_label++;
98  }
99  }
100  if (clusters.size() > std::numeric_limits<uint32_t>::max())
101  throw Exception ("The number of clusters is larger than can be labelled with an unsigned 32bit integer.");
102  }

112  template <class MaskImageType>
113  const vector<vector<int> >& precompute_adjacency (MaskImageType& mask) {
114 
115  auto index_image = Image<uint32_t>::scratch (mask);
116 
117  // 1st pass, store mask image indices and their index in the array
118  for (auto l = Loop (mask) (mask, index_image); l; ++l) {
119  if (mask.value() >= 0.5) {
120  // For each voxel, store the index within mask_indices for 2nd pass
121  index_image.value() = mask_indices.size();
122  vector<int> index (mask.ndim());
123  for (size_t dim = 0; dim < mask.ndim(); dim++)
124  index[dim] = mask.index(dim);
125  mask_indices.push_back (index);
126  } else {
127  index_image.value() = 0;
128  }
129  }
130  // Here we pre-compute the offsets for our neighbours in 4D space
131  vector< vector<int> > neighbour_offsets;
132  vector<int> offset (4);
133  for (offset[0] = -1; offset[0] <= 1; offset[0]++) {
134  for (offset[1] = -1; offset[1] <= 1; offset[1]++) {
135  for (offset[2] = -1; offset[2] <= 1; offset[2]++) {
136  for (offset[3] = -1; offset[3] <= 1; offset[3]++) {
137  if (!do_26_connectivity && ((abs(offset[0]) + abs(offset[1]) + abs(offset[2]) + abs(offset[3])) > 1))
138  continue;
139  if ((abs(offset[0]) && dim_to_ignore[0]) || (abs(offset[1]) && dim_to_ignore[1]) ||
140  (abs(offset[2]) && dim_to_ignore[2]) || (abs(offset[3]) && dim_to_ignore[3]))
141  continue;
142  neighbour_offsets.push_back (offset);
143  }
144  }
145  }
146  }
147  // 2nd pass, define adjacency
148  MaskImageType mask_neigh (mask);
149  for (vector<vector<int> >::const_iterator it = mask_indices.begin(); it != mask_indices.end(); ++it) {
150  vector<uint32_t> neighbour_indices;
151  for (vector< vector<int> >::const_iterator offset = neighbour_offsets.begin(); offset != neighbour_offsets.end(); ++offset) {
152  for (size_t dim = 0; dim < mask.ndim(); dim++)
153  mask_neigh.index(dim) = (*it)[dim] + (*offset)[dim];
154  if (!is_out_of_bounds (mask_neigh)) {
155  if (mask_neigh.value() >= 0.5) {
156  assign_pos_of (mask_neigh).to (index_image);
157  neighbour_indices.push_back (index_image.value());
158  }
159  }
160  }
161  adjacent_indices.push_back (neighbour_indices);
162  }
163 
164  return mask_indices;
165  }

194  // use a non-recursive depth first search to agglomerate adjacent voxels
195  void depth_first_search (uint32_t root,
196  cluster& cluster,
197  vector<uint32_t>& labels) const {
198  uint32_t node = root;
199  std::stack<uint32_t> stack;
200  while (true) {
201  labels[node] = cluster.label;
202  stack.push (node);
203  cluster.size++;
204  if (next_neighbour (node, labels)) {
205  continue;
206  } else {
207  do {
208  if (stack.top() == root)
209  return;
210  stack.pop();
211  node = stack.top();
212  } while (!next_neighbour (node, labels));
213  }
214  }
215  }

293  template <class InputVoxelType, class OutputVoxelType>
294  void operator() (InputVoxelType& in, OutputVoxelType& out)
295  {
296  Connector connector (do_26_connectivity);
297 
298  if (dim_to_ignore.size())
299  connector.set_dim_to_ignore (dim_to_ignore);
300 
301  connector.precompute_adjacency (in);
302 
303  std::unique_ptr<ProgressBar> progress;
304  if (message.size()) {
305  progress.reset (new ProgressBar (message));
306  ++(*progress);
307  }
308 
309  vector<cluster> clusters;
310  vector<uint32_t> labels;
311  vector<vector<int> > mask_indices = connector.run (clusters, labels);
312 
313  if (progress)
314  ++(*progress);
315  std::sort (clusters.begin(), clusters.end(), compare_clusters);
316  if (progress)
317  ++(*progress);
318 
319  vector<int> label_lookup (clusters.size(), 0);
320  for (uint32_t c = 0; c < clusters.size(); c++)
321  label_lookup[clusters[c].label - 1] = c + 1;
322 
323  for (auto l = Loop (out) (out); l; ++l)
324  out.value() = 0;
325 
326  for (uint32_t i = 0; i < mask_indices.size(); i++)
327  {
328  assign_pos_of (mask_indices[i]).to (out);
329  if (largest_only) {
330  if (label_lookup[labels[i] - 1] == 1)
331  out.value() = 1;
332  } else {
333  out.value() = label_lookup[labels[i] - 1];
334  }
335  }
336  }
