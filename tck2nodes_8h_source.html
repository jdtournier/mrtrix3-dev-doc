41 // Provides a common interface for assigning a streamline to the relevant parcellation node pair
42 // Note that this class is NOT copy-constructed, so derivative classes must be thread-safe
43 class Tck2nodes_base {

45  public:
46  Tck2nodes_base (const Image<node_t>& nodes_data, const bool pair) :
47  nodes (nodes_data),
48  transform (new Transform (nodes)),
49  pair (pair) { }

52 
53  virtual ~Tck2nodes_base() { }
54 
55  NodePair operator() (const Tractography::Streamline<>& tck) const
56  {
57  assert (pair);
58  Image<node_t> v (nodes);
59  const node_t node_one = select_node (tck, v, false);
60  const node_t node_two = select_node (tck, v, true);
61  return std::make_pair (node_one, node_two);
62  }
63 
64  bool operator() (const Streamline<>& tck, std::vector<node_t>& out) const
65  {

70  }
