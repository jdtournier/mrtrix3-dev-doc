57 // Provides a common interface for assigning a streamline to the relevant parcellation node pair
58 // Note that this class is NOT copy-constructed, so derivative classes must be thread-safe
59 class Tck2nodes_base {
60 
61  public:
62  Tck2nodes_base (Image::Buffer<node_t>& nodes_data) :
63  nodes (nodes_data),
64  transform (nodes) { }
65 
66  Tck2nodes_base (const Tck2nodes_base& that) :
67  nodes (that.nodes),
68  transform (that.transform) { }
69 
70  virtual ~Tck2nodes_base() { }
71 
72  virtual NodePair operator() (const Streamline<>& tck) const {
73  VoxelType voxel (nodes);
74  const node_t node_one = select_node (tck, voxel, false);
75  const node_t node_two = select_node (tck, voxel, true);
76  if (node_two < node_one)
77  return std::make_pair (node_two, node_one);
78  else
79  return std::make_pair (node_one, node_two);
80  }
81 
82  protected:
83  typedef Image::Buffer<node_t>::voxel_type VoxelType;
84 
85  Image::Buffer<node_t>& nodes;
86  Image::Transform transform;
87 
88  virtual node_t select_node (const Streamline<>& tck, VoxelType& voxel, bool end) const {
89  throw Exception ("Calling empty virtual function Tck2nodes_base::select_node()");
90  }
91 
92 };
