39 // These constants set how many times a tracking algorithm should attempt to propagate
40 // from a given seed point, based on the mechanism used to provide the seed point
41 //
42 // Mechanisms that provide random seed locations
43 #define MAX_TRACKING_SEED_ATTEMPTS_RANDOM 1
44 //
45 // Dynamic seeding also provides the mean direction of the fixel, so only a small number of
46 // attempts should be required to find a direction above the FOD amplitude threshold;
47 // this will however depend on this threshold as well as the angular threshold
48 #define MAX_TRACKING_SEED_ATTEMPTS_DYNAMIC 50
49 //
50 // GM-WM interface seeding incurs a decent overhead when generating the seed points;
51 // therefore want to make maximal use of each seed point generated, bearing in mind that
52 // the FOD amplitudes may be small there.
53 #define MAX_TRACKING_SEED_ATTEMPTS_GMWMI 300
54 //
55 // Mechanisms that provide a fixed number of seed points; hence the maximum effort should
56 // be made to find an appropriate tracking direction from every seed point provided
57 #define MAX_TRACKING_SEED_ATTEMPTS_FIXED 5000
