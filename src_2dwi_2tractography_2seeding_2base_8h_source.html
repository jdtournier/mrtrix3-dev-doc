37 // These constants set how many times a tracking algorithm should attempt to propagate
38 // from a given seed point, based on the mechanism used to provide the seed point
39 //
40 // Mechanisms that provide random seed locations
41 #define MAX_TRACKING_SEED_ATTEMPTS_RANDOM 1
42 //
43 // Dynamic seeding also provides the mean direction of the fixel, so only a small number of
44 // attempts should be required to find a direction above the FOD amplitude threshold;
45 // this will however depend on this threshold as well as the angular threshold
46 #define MAX_TRACKING_SEED_ATTEMPTS_DYNAMIC 50
47 //
48 // GM-WM interface seeding incurs a decent overhead when generating the seed points;
49 // therefore want to make maximal use of each seed point generated, bearing in mind that
50 // the FOD amplitudes may be small there.
51 #define MAX_TRACKING_SEED_ATTEMPTS_GMWMI 300
52 //
53 // Mechanisms that provide a fixed number of seed points; hence the maximum effort should
54 // be made to find an appropriate tracking direction from every seed point provided
55 #define MAX_TRACKING_SEED_ATTEMPTS_FIXED 5000
