32  vector<Eigen::Vector3f> direction_grid (float max_angle, float spacing)
33  {
34  const float maxR = Math::pow2 (max_angle / spacing);
35  vector<Eigen::Vector3f> list;
36  ssize_t extent = std::ceil (max_angle / spacing);
37 
38  for (ssize_t i = -extent; i <= extent; ++i) {
39  for (ssize_t j = -extent; j <= extent; ++j) {
40  float x = i + 0.5*j, y = SQRT_3_OVER_2*j;
41  float n = Math::pow2(x) + Math::pow2(y);
42  if (n > maxR)
43  continue;
44  n = spacing * std::sqrt (n);
45  float z = std::cos (n);
46  if (n) n = spacing * std::sin (n) / n;
47  list.push_back ({ n*x, n*y, z });
48  }
49  }
50 
51  return (list);
52  }

63  void calibrate (Method& method)
64  {
65  typename Method::Calibrate calibrate_func (method);
66  const float sqrt3 = std::sqrt (3.0);
67 
68  vector<Pair> amps;
69  for (float el = 0.0; el < method.S.max_angle; el += 0.001) {
70  amps.push_back (Pair (el, calibrate_func (el)));
71  if (!std::isfinite (amps.back().amp) || amps.back().amp <= 0.0) break;
72  }
73  float zero = amps.back().el;
74 
75  float N_min = Inf, theta_min = NaN, ratio = NaN;
76  for (size_t i = 1; i < amps.size(); ++i) {
77  float N = Math::pow2 (method.S.max_angle);
78  float Ns = N * (1.0+amps[0].amp/amps[i].amp)/(2.0*Math::pow2(zero));
79  auto dirs = direction_grid (method.S.max_angle + amps[i].el, sqrt3*amps[i].el);
80  N = Ns+dirs.size();
81  //std::cout << amps[i].el << " " << amps[i].amp << " " << Ns << " " << dirs.size() << " " << Ns+dirs.size() << "\n";
82  if (N > 0.0 && N < N_min) {
83  N_min = N;
84  theta_min = amps[i].el;
85  ratio = amps[0].amp / amps[i].amp;
86  }
87  }
88

90  method.calibrate_ratio = ratio;

92  INFO ("rejection sampling will use " + str (method.calibrate_list.size())
