41  template <typename value_type>
42  std::vector<Point<value_type> > direction_grid (value_type max_angle, value_type spacing)
43  {
44  const value_type maxR = Math::pow2 (max_angle / spacing);
45  std::vector<Point<value_type> > list;
46  ssize_t extent = std::ceil (max_angle / spacing);
47 
48  for (ssize_t i = -extent; i <= extent; ++i) {
49  for (ssize_t j = -extent; j <= extent; ++j) {
50  value_type x = i + 0.5*j, y = SQRT_3_OVER_2*j;
51  value_type n = Math::pow2(x) + Math::pow2(y);
52  if (n > maxR)
53  continue;
54  n = spacing * std::sqrt (n);
55  value_type z = std::cos (n);
56  if (n) n = spacing * std::sin (n) / n;
57  list.push_back (Point<value_type> (n*x, n*y, z));
58  }
59  }
60 
61  return (list);
62  }
