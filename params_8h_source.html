103  void update_control_points () {
104  const Eigen::Vector3 centre = transformation.get_centre();
105  control_points.resize(4, 4);
106  // tetrahedron centred at centre of midspace scaled by control_point_exent
107  control_points << 1.0, -1.0, -1.0, 1.0,
108  1.0, 1.0, -1.0, -1.0,
109  1.0, -1.0, 1.0, -1.0,
110  1.0, 1.0, 1.0, 1.0;
111  for (size_t i = 0; i < 3; ++i)
112  control_points.row(i) *= control_point_exent[i];
113  control_points.block<3,4>(0,0).colwise() += centre;
114  }

133  void make_diagnostics_image (const std::basic_string<char>& image_path, bool masked = true) {
134  Header header (midway_image);
135  header.datatype() = DataType::Float64;
136  header.ndim() = 4;
137  header.size(3) = 3;
138  // auto check = Image<float>::scratch (header);
139  auto check = Image<default_type>::create (image_path, header);
140 
141  auto trafo1 = transformation.get_transform_half();
142  auto trafo2 = transformation.get_transform_half_inverse();
143 
144  Adapter::Reslice<Interp::Linear, Im1ImageType > im1_reslicer (
145  im1_image, midway_image, trafo1, Adapter::AutoOverSample, NAN);
146  Adapter::Reslice<Interp::Linear, Im2ImageType > im2_reslicer (
147  im2_image, midway_image, trafo2, Adapter::AutoOverSample, NAN);
148 
149  auto T = MR::Transform(midway_image).voxel2scanner;
150  Eigen::Vector3 midway_point, voxel_pos, im1_point, im2_point;
151 
152  for (auto i = Loop (midway_image) (check, im1_reslicer, im2_reslicer); i; ++i) {
153  voxel_pos = Eigen::Vector3 ((default_type)check.index(0), (default_type)check.index(1), (default_type)check.index(2));
154  midway_point = T * voxel_pos;
155 
156  check.index(3) = 0;
157  check.value() = im1_reslicer.value();
158  if (masked and im1_mask_interp) {
159  transformation.transform_half (im1_point, midway_point);
160  im1_mask_interp->scanner (im1_point);
161  if (im1_mask_interp->value() <= 0.5)
162  check.value() = NAN;
163  }
164 
165  check.index(3) = 1;
166  check.value() = im2_reslicer.value();
167  if (masked and im2_mask_interp) {
168  transformation.transform_half_inverse (im2_point, midway_point);
169  im2_mask_interp->scanner (im1_point);
170  if (im2_mask_interp->value() <= 0.5)
171  check.value() = NAN;
172  }
173  check.index(3) = 0;
174  }
175  INFO("diagnostics image written");
176  // display (check);
177  }
