104  void update_control_points () {
105  const Eigen::Vector3 centre = transformation.get_centre();
106  control_points.resize(4, 4);
107  // tetrahedron centred at centre of midspace scaled by control_point_exent
108  control_points << 1.0, -1.0, -1.0, 1.0,
109  1.0, 1.0, -1.0, -1.0,
110  1.0, -1.0, 1.0, -1.0,
111  1.0, 1.0, 1.0, 1.0;
112  for (size_t i = 0; i < 3; ++i)
113  control_points.row(i) *= control_point_exent[i];
114  control_points.block<3,4>(0,0).colwise() += centre;
115  }

134  void make_diagnostics_image (const std::basic_string<char>& image_path, bool masked = true) {
135  Header header (midway_image);
136  header.datatype() = DataType::Float64;
137  header.ndim() = 4;
138  header.size(3) = 3;
139  // auto check = Image<float>::scratch (header);
140  auto check = Image<default_type>::create (image_path, header);
141 
142  auto trafo1 = transformation.get_transform_half();
143  auto trafo2 = transformation.get_transform_half_inverse();
144 
145  Adapter::Reslice<Interp::Linear, Im1ImageType > im1_reslicer (
146  im1_image, midway_image, trafo1, Adapter::AutoOverSample, NAN);
147  Adapter::Reslice<Interp::Linear, Im2ImageType > im2_reslicer (
148  im2_image, midway_image, trafo2, Adapter::AutoOverSample, NAN);
149 
150  auto T = MR::Transform(midway_image).voxel2scanner;
151  Eigen::Vector3 midway_point, voxel_pos, im1_point, im2_point;
152 
153  for (auto i = Loop (midway_image) (check, im1_reslicer, im2_reslicer); i; ++i) {
154  voxel_pos = Eigen::Vector3 ((default_type)check.index(0), (default_type)check.index(1), (default_type)check.index(2));
155  midway_point = T * voxel_pos;
156 
157  check.index(3) = 0;
158  check.value() = im1_reslicer.value();
159  if (masked and im1_mask_interp) {
160  transformation.transform_half (im1_point, midway_point);
161  im1_mask_interp->scanner (im1_point);
162  if (im1_mask_interp->value() <= 0.5)
163  check.value() = NAN;
164  }
165 
166  check.index(3) = 1;
167  check.value() = im2_reslicer.value();
168  if (masked and im2_mask_interp) {
169  transformation.transform_half_inverse (im2_point, midway_point);
170  im2_mask_interp->scanner (im1_point);
171  if (im2_mask_interp->value() <= 0.5)
172  check.value() = NAN;
173  }
174  check.index(3) = 0;
175  }
176  INFO("diagnostics image written");
177  // display (check);
178  }
