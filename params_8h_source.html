99  void update_control_points () {
100  const Eigen::Vector3 centre = transformation.get_centre();
101  control_points.resize(4, 4);
102  // tetrahedron centred at centre of midspace scaled by control_point_exent
103  control_points << 1.0, -1.0, -1.0, 1.0,
104  1.0, 1.0, -1.0, -1.0,
105  1.0, -1.0, 1.0, -1.0,
106  1.0, 1.0, 1.0, 1.0;
107  for (size_t i = 0; i < 3; ++i)
108  control_points.row(i) *= control_point_exent[i];
109  control_points.block<3,4>(0,0).colwise() += centre;
110  }
