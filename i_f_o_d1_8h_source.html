167  term_t next ()
168  {
169  if (!get_data (source))
170  return EXIT_IMAGE;
171 
172  float max_val = 0.0;
173  for (size_t i = 0; i < calibrate_list.size(); ++i) {
174  float val = FOD (rotate_direction (dir, calibrate_list[i]));
175  if (std::isnan (val))
176  return EXIT_IMAGE;
177  else if (val > max_val)
178  max_val = val;
179  }
180 
181  if (max_val <= 0.0)
182  return CALIBRATE_FAIL;

185 
186  num_sample_runs++;
187 
188  for (size_t n = 0; n < S.max_trials; n++) {
189  Eigen::Vector3f new_dir = rand_dir (dir);
190  float val = FOD (new_dir);
191 
192  if (val > S.threshold) {
193 
194  if (val > max_val) {
195  DEBUG ("max_val exceeded!!! (val = " + str(val) + ", max_val = " + str (max_val) + ")");
196  ++num_truncations;
197  if (val/max_val > max_truncation)
198  max_truncation = val/max_val;
199  }
200 
201  if (uniform(*rng) < val/max_val) {
202  dir = new_dir;
203  dir.normalize();
204  pos += S.step_size * dir;
205  mean_sample_num += n;
