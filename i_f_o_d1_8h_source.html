166  term_t next ()
167  {
168  if (!get_data (source))
169  return EXIT_IMAGE;
170 
171  float max_val = 0.0;
172  for (size_t i = 0; i < calibrate_list.size(); ++i) {
173  float val = FOD (rotate_direction (dir, calibrate_list[i]));
174  if (std::isnan (val))
175  return EXIT_IMAGE;
176  else if (val > max_val)
177  max_val = val;
178  }
179 
180  if (max_val <= 0.0)
181  return CALIBRATOR;
182 
183  max_val *= calibrate_ratio;
184 
185  num_sample_runs++;
186 
187  for (size_t n = 0; n < S.max_trials; n++) {
188  Eigen::Vector3f new_dir = rand_dir (dir);
189  float val = FOD (new_dir);
190 
191  if (val > S.threshold) {
192 
193  if (val > max_val) {
194  DEBUG ("max_val exceeded!!! (val = " + str(val) + ", max_val = " + str (max_val) + ")");
195  ++num_truncations;
196  if (val/max_val > max_truncation)
197  max_truncation = val/max_val;
198  }
199 
200  if (uniform(*rng) < val/max_val) {
201  dir = new_dir;
202  dir.normalize();
203  pos += S.step_size * dir;
204  mean_sample_num += n;
205  return CONTINUE;
206  }
207 
208  }
209  }
210 
211  return BAD_SIGNAL;
