127  template <class IndexHeaderType>
128  FORCE_INLINE index_type get_number_of_fixels (IndexHeaderType& index_header) {
129  check_index_image (index_header);
130  if (index_header.keyval().count (n_fixels_key)) {
131  return std::stoul (index_header.keyval().at(n_fixels_key));
132  } else {
133  auto index_image = Image<index_type>::open (index_header.name());
134  index_image.index(3) = 1;
135  index_type num_fixels = 0;
136  index_type max_offset = 0;
137  for (auto i = MR::Loop (index_image, 0, 3) (index_image); i; ++i) {
138  if (index_image.value() > max_offset) {
139  max_offset = index_image.value();
140  index_image.index(3) = 0;
141  num_fixels = index_image.value();
142  index_image.index(3) = 1;
143  }
144  }
145  return (max_offset + num_fixels);
146  }
147  }

150  template <class IndexHeaderType, class DataHeaderType>
151  FORCE_INLINE bool fixels_match (const IndexHeaderType& index_header, const DataHeaderType& data_header)
152  {
153  bool fixels_match (false);
154 
155  if (is_index_image (index_header)) {
156  if (index_header.keyval().count (n_fixels_key)) {
157  fixels_match = std::stoul (index_header.keyval().at(n_fixels_key)) == (index_type)data_header.size(0);
158  } else {
159  auto index_image = Image<index_type>::open (index_header.name());
160  index_image.index(3) = 1;
161  index_type num_fixels = 0;
162  index_type max_offset = 0;
163  for (auto i = MR::Loop (index_image, 0, 3) (index_image); i; ++i) {
164  if (index_image.value() > max_offset) {
165  max_offset = index_image.value();
166  index_image.index(3) = 0;
167  num_fixels = index_image.value();
168  index_image.index(3) = 1;
169  }
170  }
171  fixels_match = (max_offset + num_fixels) == (index_type)data_header.size(0);
172  }
173  }
174 
175  return fixels_match;
176  }

189  FORCE_INLINE void check_fixel_directory (const std::string &path, bool create_if_missing = false, bool check_if_empty = false)
190  {
191  std::string path_temp = path;
192  // handle the use case when a fixel command is run from inside a fixel directory
193  if (path.empty())
194  path_temp = Path::cwd();
195 
196  bool exists (true);
197 
198  if (!(exists = Path::exists (path_temp))) {
199  if (create_if_missing) File::mkdir (path_temp);
200  else throw Exception ("Fixel directory (" + str(path_temp) + ") does not exist");
201  }
202  else if (!Path::is_dir (path_temp))
203  throw Exception (str(path_temp) + " is not a directory");
204 
205  if (check_if_empty && Path::Dir (path_temp).read_name ().size () != 0)
206  throw Exception ("Expected fixel directory " + path_temp + " to be empty.");

210  FORCE_INLINE Header find_index_header (const std::string &fixel_directory_path)
211  {
212  Header header;
213  check_fixel_directory (fixel_directory_path);
214 
215  for (std::initializer_list<const std::string>::iterator it = supported_sparse_formats.begin();
216  it !=supported_sparse_formats.end(); ++it) {
217  std::string full_path = Path::join (fixel_directory_path, "index" + *it);
218  if (Path::exists(full_path)) {
219  if (header.valid())
220  throw InvalidFixelDirectoryException ("Multiple index images found in directory " + fixel_directory_path);
221  header = Header::open (full_path);
222  }
223  }
224  if (!header.valid())
225  throw InvalidFixelDirectoryException ("Could not find index image in directory " + fixel_directory_path);
226 
227  check_index_image (header);
228  return header;
229  }

232  FORCE_INLINE vector<Header> find_data_headers (const std::string &fixel_directory_path, const Header &index_header, const bool include_directions = false)
233  {
234  check_index_image (index_header);
235  auto dir_walker = Path::Dir (fixel_directory_path);
236  vector<std::string> file_names;
237  {
238  std::string temp;
239  while ((temp = dir_walker.read_name()).size())
240  file_names.push_back (temp);
241  }
242  std::sort (file_names.begin(), file_names.end());
243 
244  vector<Header> data_headers;
245  for (auto fname : file_names) {
246  if (Path::has_suffix (fname, supported_sparse_formats)) {
247  try {
248  auto H = Header::open (Path::join (fixel_directory_path, fname));
249  if (is_data_file (H)) {
250  if (fixels_match (index_header, H)) {
251  if (!is_directions_file (H) || include_directions)
252  data_headers.emplace_back (std::move (H));
253  } else {
254  WARN ("fixel data file (" + fname + ") does not contain the same number of elements as fixels in the index file");
255  }
256  }
257  } catch (...) {
258  WARN ("unable to open file \"" + fname + "\" as potential fixel data file");
259  }
260  }
261  }
262 
263  return data_headers;
264  }

268  FORCE_INLINE Header find_directions_header (const std::string fixel_directory_path)
269  {
270  bool directions_found (false);
271  Header header;
272  check_fixel_directory (fixel_directory_path);
273  Header index_header = Fixel::find_index_header (fixel_directory_path);
274 
275  auto dir_walker = Path::Dir (fixel_directory_path);
276  std::string fname;
277  while ((fname = dir_walker.read_name()).size()) {
278  if (is_directions_filename (fname)) {
279  Header tmp_header = Header::open (Path::join (fixel_directory_path, fname));
280  if (is_directions_file (tmp_header)) {
281  if (fixels_match (index_header, tmp_header)) {
282  if (directions_found == true)
283  throw Exception ("multiple directions files found in fixel image directory: " + fixel_directory_path);
284  directions_found = true;
285  header = std::move (tmp_header);
286  } else {
287  WARN ("fixel directions file (" + fname + ") does not contain the same number of elements as fixels in the index file" );
288  }
289  }
290  }
291  }
292 
293  if (!directions_found)
294  throw InvalidFixelDirectoryException ("Could not find directions image in directory " + fixel_directory_path);
295 
296  return header;
297  }

335  FORCE_INLINE void copy_index_file (const std::string& input_directory, const std::string& output_directory) {
336  Header input_header = Fixel::find_index_header (input_directory);
337  check_fixel_directory (output_directory, true);
338 
339  std::string output_path = Path::join (output_directory, Path::basename (input_header.name()));
340 
341  // If the index file already exists check it is the same as the input index file
342  if (Path::exists (output_path) && !App::overwrite_files) {
343  auto input_image = input_header.get_image<index_type>();
344  auto output_image = Image<index_type>::open (output_path);
345 
346  if (!images_match_abs (input_image, output_image))
347  throw Exception ("output sparse image directory (" + output_directory + ") already contains index file, "
348  "which is not the same as the expected output. Use -force to override if desired");
349 
350  } else {
351  auto output_image = Image<index_type>::create (Path::join (output_directory, Path::basename (input_header.name())), input_header);
352  auto input_image = input_header.get_image<index_type>();
353  threaded_copy (input_image, output_image);
