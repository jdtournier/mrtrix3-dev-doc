299  void truncate_track (std::vector< Point<value_type> >& tck, const int revert_step)
300  {
301  // Need to be able to get an estimate of the tangent at the new endpoint
302  // Removing start of current arc (counts as 1 if it exists) plus 1 arclength for each remaining revert_step
303  const int points_to_remove = (sample_idx ? sample_idx : S.num_samples) + ((revert_step - 1) * S.num_samples);
304  const int new_end_idx = (int)tck.size() - 1 - points_to_remove;
305  if (new_end_idx <= 1) {
306  tck.clear();
307  pos.invalidate();
308  dir.invalidate();
309  return;
310  }
311  dir = (tck[new_end_idx + 1] - tck[new_end_idx - 1]).normalise();
312 
313  // Erase the track up to the correct point
314  tck.erase (tck.begin() + new_end_idx + 1, tck.end());
315 
316  // Need to get the path probability contribution from the FOD at this point
317  pos = tck.back();
318  get_data (source);
319  half_log_prob0 = 0.5 * std::log (FOD (dir));
320 
321  // Make sure that arc is re-calculated when next() is called
322  sample_idx = S.num_samples;
323 
324  // Need to update sgm_depth appropriately, remembering that it is tracked by exec
325  act().sgm_depth = MAX (0, act().sgm_depth - points_to_remove);
326  }

374  value_type path_prob (std::vector< Point<value_type> >& positions, std::vector< Point<value_type> >& tangents)
375  {
376 
377  // Early exit for ACT when path is not sensible
378  if (S.is_act()) {
379  if (!act().fetch_tissue_data (positions[S.num_samples - 1]))
380  return (NAN);
381  if (act().tissues().get_csf() >= 0.5)
382  return 0.0;
383  }
384 
385  value_type log_prob = half_log_prob0;
386  for (size_t i = 0; i < S.num_samples; ++i) {
387 
388  value_type fod_amp = FOD (positions[i], tangents[i]);
389  if (std::isnan (fod_amp))
390  return (NAN);
391  if (fod_amp < S.threshold)
392  return 0.0;
393  fod_amp = std::log (fod_amp);
394  if (i < S.num_samples-1) {
395  log_prob += fod_amp;
396  } else {
397  last_half_log_probN = 0.5*fod_amp;
398  log_prob += last_half_log_probN;
399  }
400  }
401 
402  return std::exp (S.fod_power * log_prob);
403  }

407  void get_path (std::vector< Point<value_type> >& positions, std::vector< Point<value_type> >& tangents, const Point<value_type>& end_dir) const
408  {
409  value_type cos_theta = end_dir.dot (dir);
410  cos_theta = std::min (cos_theta, value_type(1.0));
411  value_type theta = std::acos (cos_theta);
412 
413  if (theta) {
414 
415  Point<value_type> curv = end_dir - cos_theta * dir;
416  curv.normalise();
417  value_type R = S.step_size / theta;
418 
419  for (size_t i = 0; i < S.num_samples-1; ++i) {
420  value_type a = (theta * (i+1)) / S.num_samples;
421  value_type cos_a = std::cos (a);
422  value_type sin_a = std::sin (a);
423  positions[i] = pos + R * (sin_a * dir + (value_type(1.0) - cos_a) * curv);
424  tangents[i] = cos_a * dir + sin_a * curv;
425  }
426  positions[S.num_samples-1] = pos + R * (std::sin (theta) * dir + (value_type(1.0)-cos_theta) * curv);
427  tangents[S.num_samples-1] = end_dir;
428 
429  } else { // straight on:
430 
431  for (size_t i = 0; i < S.num_samples; ++i) {
432  value_type f = (i+1) * (S.step_size / S.num_samples);
433  positions[i] = pos + f * dir;
434  tangents[i] = dir;
435  }
436 
437  }
438  }
