185  template <typename ValueType, typename CoefType>
186  inline ValueType value (const CoefType& coefs, ValueType cos_elevation, ValueType cos_azimuth, ValueType sin_azimuth, int lmax)
187  {
188  ValueType amplitude = 0.0;
189  VLA_MAX (AL, ValueType, lmax+1, 64);
190  Legendre::Plm_sph (AL, lmax, 0, ValueType (cos_elevation));
191  for (int l = 0; l <= lmax; l+=2)
192  amplitude += AL[l] * coefs[index (l,0)];
193  ValueType c0 (1.0), s0 (0.0);
194  for (int m = 1; m <= lmax; m++) {
195  Legendre::Plm_sph (AL, lmax, m, ValueType (cos_elevation));
196  ValueType c = c0 * cos_azimuth - s0 * sin_azimuth; // std::cos(m*azimuth)
197  ValueType s = s0 * cos_azimuth + c0 * sin_azimuth; // std::sin(m*azimuth)
198  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
199 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
200  amplitude += AL[l] * Math::sqrt2 * (c * coefs[index (l,m)] + s * coefs[index (l,-m)]);
201 #else
202  amplitude += AL[l] * (c * coefs[index (l,m)] + s * coefs[index (l,-m)]);
203 #endif
204  }
205  c0 = c;
206  s0 = s;
207  }
208  return amplitude;
209  }

236  template <typename ValueType>
237  inline Vector<ValueType>& delta (Vector<ValueType>& delta_vec, const Point<ValueType>& unit_dir, int lmax)
238  {
239  delta_vec.allocate (NforL (lmax));
240  ValueType rxy = std::sqrt ( pow2(unit_dir[1]) + pow2(unit_dir[0]) );
241  ValueType cp = (rxy) ? unit_dir[0]/rxy : 1.0;
242  ValueType sp = (rxy) ? unit_dir[1]/rxy : 0.0;
243  VLA_MAX (AL, ValueType, lmax+1, 64);
244  Legendre::Plm_sph (AL, lmax, 0, ValueType (unit_dir[2]));
245  for (int l = 0; l <= lmax; l+=2)
246  delta_vec[index (l,0)] = AL[l];
247  ValueType c0 (1.0), s0 (0.0);
248  for (int m = 1; m <= lmax; m++) {
249  Legendre::Plm_sph (AL, lmax, m, ValueType (unit_dir[2]));
250  ValueType c = c0 * cp - s0 * sp;
251  ValueType s = s0 * cp + c0 * sp;
252  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
253 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
254  delta_vec[index (l,m)] = AL[l] * Math::sqrt2 * c;
255  delta_vec[index (l,-m)] = AL[l] * Math::sqrt2 * s;
256 #else
257  delta_vec[index (l,m)] = AL[l] * 2.0 * c;
258  delta_vec[index (l,-m)] = AL[l] * 2.0 * s;
259 #endif
260  }
261  c0 = c;
262  s0 = s;
263  }
264  return delta_vec;
265  }

588  c0 = c;
589  s0 = s;
590  }
591  return v;

606  inline ValueType get_peak (const ValueType* sh, int lmax, Point<ValueType>& unit_init_dir, PrecomputedAL<ValueType>* precomputer = NULL)
607  {
608  assert (unit_init_dir.valid());
609  for (int i = 0; i < 50; i++) {
610  ValueType az = atan2 (unit_init_dir[1], unit_init_dir[0]);
611  ValueType el = acos (unit_init_dir[2]);
612  ValueType amplitude, dSH_del, dSH_daz, d2SH_del2, d2SH_deldaz, d2SH_daz2;
613  derivatives (sh, lmax, el, az, amplitude, dSH_del, dSH_daz, d2SH_del2, d2SH_deldaz, d2SH_daz2, precomputer);
614 
615  ValueType del = sqrt (dSH_del*dSH_del + dSH_daz*dSH_daz);
616  ValueType daz = dSH_daz/del;
617  del = dSH_del/del;
618 
619  ValueType dSH_dt = daz*dSH_daz + del*dSH_del;
620  ValueType d2SH_dt2 = daz*daz*d2SH_daz2 + 2.0*daz*del*d2SH_deldaz + del*del*d2SH_del2;
621  ValueType dt = d2SH_dt2 ? (-dSH_dt / d2SH_dt2) : 0.0;
622 
623  if (dt < 0.0) dt = -dt;
624  if (dt > MAX_DIR_CHANGE) dt = MAX_DIR_CHANGE;
625 
626  del *= dt;
627  daz *= dt;
628 
629  unit_init_dir += Point<ValueType> (del*std::cos (az) *std::cos (el) - daz*std::sin (az), del*std::sin (az) *std::cos (el) + daz*std::cos (az), -del*std::sin (el));
630  unit_init_dir.normalise();
631 
632  if (dt < ANGLE_TOLERANCE) return amplitude;
633  }
634 
635  unit_init_dir.invalidate();
636  DEBUG ("failed to find SH peak!");
637  return NAN;
638  }

646  template <typename ValueType>
647  inline void derivatives (
648  const ValueType* sh, const int lmax, const ValueType elevation, const ValueType azimuth, ValueType& amplitude,
649  ValueType& dSH_del, ValueType& dSH_daz, ValueType& d2SH_del2, ValueType& d2SH_deldaz,
650  ValueType& d2SH_daz2, PrecomputedAL<ValueType>* precomputer)
651  {
652  ValueType sel = std::sin (elevation);
653  ValueType cel = std::cos (elevation);
654  bool atpole = sel < 1e-4;
655 
656  dSH_del = dSH_daz = d2SH_del2 = d2SH_deldaz = d2SH_daz2 = 0.0;
657  VLA_MAX (AL, ValueType, NforL_mpos (lmax), 64);
658 
659  if (precomputer) {
660  PrecomputedFraction<ValueType> f;
661  precomputer->set (f, elevation);
662  precomputer->get (AL, f);
663  }
664  else {
665  VLA_MAX (buf, ValueType, lmax+1, 64);
666  for (int m = 0; m <= lmax; m++) {
667  Legendre::Plm_sph (buf, lmax, m, cel);
668  for (int l = ( (m&1) ?m+1:m); l <= lmax; l+=2)
669  AL[index_mpos (l,m)] = buf[l];
670  }
671  }
672 
673  amplitude = sh[0] * AL[0];
674  for (int l = 2; l <= (int) lmax; l+=2) {
675  const ValueType& v (sh[index (l,0)]);
676  amplitude += v * AL[index_mpos (l,0)];
677  dSH_del += v * sqrt (ValueType (l* (l+1))) * AL[index_mpos (l,1)];
678  d2SH_del2 += v * (sqrt (ValueType (l* (l+1) * (l-1) * (l+2))) * AL[index_mpos (l,2)] - l* (l+1) * AL[index_mpos (l,0)]) /2.0;
679  }
680 
681  for (int m = 1; m <= lmax; m++) {
682 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
683  ValueType caz = Math::sqrt2 * std::cos (m*azimuth);
684  ValueType saz = Math::sqrt2 * std::sin (m*azimuth);
685 #else
686  ValueType caz = std::cos (m*azimuth);
687  ValueType saz = std::sin (m*azimuth);
688 #endif
689  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
690  const ValueType& vp (sh[index (l,m)]);
691  const ValueType& vm (sh[index (l,-m)]);
692  amplitude += (vp*caz + vm*saz) * AL[index_mpos (l,m)];
693 
694  ValueType tmp = sqrt (ValueType ( (l+m) * (l-m+1))) * AL[index_mpos (l,m-1)];
695  if (l > m) tmp -= sqrt (ValueType ( (l-m) * (l+m+1))) * AL[index_mpos (l,m+1)];
696  tmp /= -2.0;
697  dSH_del += (vp*caz + vm*saz) * tmp;
698 
699  ValueType tmp2 = - ( (l+m) * (l-m+1) + (l-m) * (l+m+1)) * AL[index_mpos (l,m)];
700  if (m == 1) tmp2 -= sqrt (ValueType ( (l+m) * (l-m+1) * (l+m-1) * (l-m+2))) * AL[index_mpos (l,1)];
701  else tmp2 += sqrt (ValueType ( (l+m) * (l-m+1) * (l+m-1) * (l-m+2))) * AL[index_mpos (l,m-2)];
702  if (l > m+1) tmp2 += sqrt (ValueType ( (l-m) * (l+m+1) * (l-m-1) * (l+m+2))) * AL[index_mpos (l,m+2)];
703  tmp2 /= 4.0;
704  d2SH_del2 += (vp*caz + vm*saz) * tmp2;
705 
706  if (atpole) dSH_daz += (vm*caz - vp*saz) * tmp;
707  else {
708  d2SH_deldaz += m * (vm*caz - vp*saz) * tmp;
709  dSH_daz += m * (vm*caz - vp*saz) * AL[index_mpos (l,m)];
710  d2SH_daz2 -= (vp*caz + vm*saz) * m*m * AL[index_mpos (l,m)];
711  }
712 
713  }
714  }
715 
716  if (!atpole) {
717  dSH_daz /= sel;
718  d2SH_deldaz /= sel;
719  d2SH_daz2 /= sel*sel;
720  }
721  }
