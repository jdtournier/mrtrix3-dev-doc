87  tck.clear();
88 
89  if (!in.is_open())
90  return false;
91 
92  do {
93  auto p = get_next_point();
94  if (std::isinf (p[0])) {
95  in.close();
96  check_excess_weights();
97  return false;
98  }
99  if (in.eof()) {
100  in.close();
101  check_excess_weights();
102  return false;
103  }
104 
105  if (std::isnan (p[0])) {
106  tck.index = current_index++;
107 
108  if (weights_file) {
109 
110  (*weights_file) >> tck.weight;
111  if (weights_file->fail()) {
112  WARN ("Streamline weights file contains less entries than .tck file; only read " + str(current_index-1) + " streamlines");
113  in.close();
114  tck.clear();
115  return false;
116  }
117 
118  } else {
119  tck.weight = 1.0;
120  }
121 
122  return true;
123  }
124 
125  tck.push_back (p);
126  } while (in.good());

128  in.close();

211  template <class ValueType = float>
212  class WriterUnbuffered : public __WriterBase__<ValueType>, public WriterInterface<ValueType>
213  { NOMEMALIGN
214  public:
215  using __WriterBase__<ValueType>::count;
216  using __WriterBase__<ValueType>::total_count;
217  using __WriterBase__<ValueType>::name;
218  using __WriterBase__<ValueType>::dtype;
219  using __WriterBase__<ValueType>::create;
220  using __WriterBase__<ValueType>::verify_stream;
221  using __WriterBase__<ValueType>::update_counts;
222  using __WriterBase__<ValueType>::open_success;
223 
224  using vector_type = Eigen::Matrix<ValueType,3,1>;
225 
227  WriterUnbuffered (const std::string& file, const Properties& properties) :
228  __WriterBase__<ValueType> (file) {
229 
230  if (!Path::has_suffix (name, ".tck"))
231  throw Exception ("output track files must use the .tck suffix");
232 
233  File::OFStream out;
234  try {
235  out.open (name, std::ios::out | std::ios::binary | std::ios::trunc);
236  } catch (Exception& e) {
237  throw Exception (e, "Unable to create output track file");
238  }
239 
240  const_cast<Properties&> (properties).set_timestamp();
241  const_cast<Properties&> (properties).set_version_info();
242 
243  create (out, properties, "tracks");
244  barrier_addr = out.tellp();
245 
246  vector_type x;
247  format_point (barrier(), x);
248  out.write (reinterpret_cast<char*> (&x[0]), sizeof (x));
249  if (!out.good())
250  throw Exception ("error writing tracks file \"" + name + "\": " + strerror (errno));
251  open_success = true;
252 
253  auto opt = App::get_options ("tck_weights_out");
254  if (opt.size())
255  set_weights_path (opt[0][0]);
256  }
257 
259  bool operator() (const Streamline<ValueType>& tck) {
260  if (tck.size()) {
261  // allocate buffer on the stack for performance:
262  NON_POD_VLA (buffer, vector_type, tck.size()+2);
263  for (size_t n = 0; n < tck.size(); ++n)
264  format_point (tck[n], buffer[n]);
265  format_point (delimiter(), buffer[tck.size()]);
266 
267  commit (buffer, tck.size()+1);
268 
269  if (weights_name.size())
270  write_weights (str(tck.weight) + "\n");
271 
272  ++count;
273  }
274  ++total_count;
275  return true;
276  }

298  void format_point (const vector_type& src, vector_type& dest) {
299  using namespace ByteOrder;
300  if (dtype.is_little_endian())
301  dest = { LE(src[0]), LE(src[1]), LE(src[2]) };
302  else
303  dest = { BE(src[0]), BE(src[1]), BE(src[2]) };
304  }

308  File::OFStream out (weights_name, std::ios::in | std::ios::out | std::ios::binary | std::ios::ate);

318  void commit (vector_type* data, size_t num_points) {
319  if (num_points == 0 || !open_success)
320  return;
321 
322  int64_t prev_barrier_addr = barrier_addr;
323 
324  format_point (barrier(), data[num_points]);
325  File::OFStream out (name, std::ios::in | std::ios::out | std::ios::binary | std::ios::ate);
326  out.write (reinterpret_cast<const char* const> (data+1), sizeof (vector_type) * num_points);
327  verify_stream (out);
328  barrier_addr = int64_t (out.tellp()) - sizeof(vector_type);
329  out.seekp (prev_barrier_addr, out.beg);
330  out.write (reinterpret_cast<const char* const> (data), sizeof(vector_type));
331  verify_stream (out);
332  update_counts (out);
333  }

375  //CONF option: TrackWriterBufferSize
376  //CONF default: 16777216
377  //CONF The size of the write-back buffer (in bytes) to use when
378  //CONF writing track files. MRtrix will store the output tracks in a
379  //CONF relatively large buffer to limit the number of write() calls,
380  //CONF avoid associated issues such as file fragmentation.
381  Writer (const std::string& file, const Properties& properties, size_t default_buffer_capacity = 16777216) :
382  WriterUnbuffered<ValueType> (file, properties),
383  buffer_capacity (File::Config::get_int ("TrackWriterBufferSize", default_buffer_capacity) / sizeof (vector_type)),
384  buffer (new vector_type [buffer_capacity]),

395  bool operator() (const Streamline<ValueType>& tck) {
396  if (tck.size()) {
397  if (buffer_size + tck.size() + 2 > buffer_capacity)
398  commit ();
399 
400  for (const auto& i : tck)
401  add_point (i);
402  add_point (delimiter());
403 
404  if (weights_name.size())
405  weights_buffer += str (tck.weight) + ' ';
406 
407  ++count;
408  }
409  ++total_count;
410  return true;
411  }
