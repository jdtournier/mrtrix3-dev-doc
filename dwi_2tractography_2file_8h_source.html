246  if (tck.size()) {
247  // allocate buffer on the stack for performance:

304  void commit (vector_type* data, size_t num_points) {
305  if (num_points == 0)
306  return;
307 
308  int64_t prev_barrier_addr = barrier_addr;
309 
310  format_point (barrier(), data[num_points]);
311  File::OFStream out (name, std::ios::in | std::ios::out | std::ios::binary | std::ios::ate);
312  out.write (reinterpret_cast<const char* const> (data+1), sizeof (vector_type) * num_points);
313  verify_stream (out);
314  barrier_addr = int64_t (out.tellp()) - sizeof(vector_type);
315  out.seekp (prev_barrier_addr, out.beg);
316  out.write (reinterpret_cast<const char* const> (data), sizeof(vector_type));
317  verify_stream (out);
318  update_counts (out);
319  }
