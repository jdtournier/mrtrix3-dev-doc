53  double stageShift(const Particle* par, const Point_t& pos, const Point_t& dir)
54  {
55  dEint = 0.0;
56  if (par->hasPredecessor()) {
57  int a = (par->getPredecessor()->getPredecessor() == par) ? -1 : 1;
58  dEint -= calcEnergy(par, -1, par->getPredecessor(), a);
59  Point_t ep (pos);
60  ep -= Particle::L * dir;
61  dEint += calcEnergy(pos, ep, par->getPredecessor()->getPosition(), par->getPredecessor()->getEndPoint(a));
62  }
63  if (par->hasSuccessor()) {
64  int a = (par->getSuccessor()->getPredecessor() == par) ? -1 : 1;
65  dEint -= calcEnergy(par, 1, par->getSuccessor(), a);
66  Point_t ep (pos);
67  ep += Particle::L * dir;
68  dEint += calcEnergy(pos, ep, par->getSuccessor()->getPosition(), par->getSuccessor()->getEndPoint(a));
69  }
70  return dEint / stats.getTint();
71  }
