48  // Note that this function does not take into account grouping of subjects and therefore generated

50  // If the number of subjects is large then the likelihood of generating duplicates is low.
51  inline void generate_permutations (const size_t num_perms,
52  const size_t num_subjects,
53  std::vector<std::vector<size_t> >& permutations,
54  bool include_default)
55  {
56  permutations.clear();
57  std::vector<size_t> default_labelling (num_subjects);
58  for (size_t i = 0; i < num_subjects; ++i)
59  default_labelling[i] = i;
60  size_t p = 0;
61  if (include_default) {
62  permutations.push_back (default_labelling);
63  ++p;
64  }
65  for (;p < num_perms; ++p) {
66  std::vector<size_t> permuted_labelling (default_labelling);
67  do {
68  std::random_shuffle (permuted_labelling.begin(), permuted_labelling.end());
69  } while (is_duplicate_permutation (permuted_labelling, permutations));
70  permutations.push_back (permuted_labelling);
71  }

76  const std::vector<value_type>& stats,

78  {

80  for (ssize_t i = 0; i < perm_dist.size(); i++)
81  permutations[i] = perm_dist[i];
82  std::sort (permutations.begin(), permutations.end());
83  pvalues.resize (stats.size());
84  for (size_t i = 0; i < stats.size(); ++i) {
85  if (stats[i] > 0.0) {
86  value_type pvalue = 1.0;
87  for (size_t j = 0; j < permutations.size(); ++j) {
88  if (stats[i] < permutations[j]) {

90  break;
91  }
