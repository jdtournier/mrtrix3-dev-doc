331  template <typename ValueType>
332  void* Image<ValueType>::Buffer::get_data_pointer ()
333  {
334  if (data_buffer) // already allocated via with_direct_io()
335  return data_buffer.get();
336 
337  assert (io && "data pointer will only be set for valid Images");
338  if (!io->is_file_backed()) // this is a scratch image
339  return io->segment(0);
340 
341  // check whether we can still do direct IO
342  // if so, return address where mapped
343  if (io->nsegments() == 1 && datatype() == DataType::from<ValueType>() && intensity_offset() == 0.0 && intensity_scale() == 1.0)
344  return io->segment(0);
345 
346  // can't do direct IO
347  return nullptr;

407  template <typename ValueType>
408  Image<ValueType> Image<ValueType>::with_direct_io (Stride::List with_strides)
409  {
410  if (buffer->data_buffer)
411  throw Exception ("FIXME: don't invoke 'with_direct_io()' on images already using direct IO!");
412  if (!buffer->get_io())
413  throw Exception ("FIXME: don't invoke 'with_direct_io()' on non-validated images!");
414  if (!buffer.unique())
415  throw Exception ("FIXME: don't invoke 'with_direct_io()' on images if other copies exist!");
416 
417  bool preload = ( buffer->datatype() != DataType::from<ValueType>() ) || ( buffer->get_io()->files.size() > 1 );
418  if (with_strides.size()) {
419  auto new_strides = Stride::get_actual (Stride::get_nearest_match (*this, with_strides), *this);
420  preload |= ( new_strides != Stride::get (*this) );
421  with_strides = new_strides;
422  }
423  else
424  with_strides = Stride::get (*this);
425 
426  if (!preload)
427  return std::move (*this);
428 
429  // do the preload:
430 
431  // the buffer into which to copy the data:
432  const auto buffer_size = footprint<ValueType> (voxel_count (*this));
433  buffer->data_buffer = std::unique_ptr<uint8_t[]> (new uint8_t [buffer_size]);
434 
435  if (buffer->get_io()->is_image_new()) {
436  // no need to preload if data is zero anyway:
437  memset (buffer->data_buffer.get(), 0, buffer_size);
438  }
439  else {
440  auto src (*this);
441  TmpImage<ValueType> dest = { *buffer, buffer->data_buffer.get(), std::vector<ssize_t> (ndim(), 0), with_strides, Stride::offset (with_strides, *this) };
442  threaded_copy_with_progress_message ("preloading data for \"" + name() + "\"", src, dest);
443  }
444 
445  return Image (buffer, with_strides);
446  }

452  template <typename ValueType>
453  std::string Image<ValueType>::dump_to_mrtrix_file (std::string filename, bool) const
454  {
455  if (!data_pointer || ( !Path::has_suffix (filename, ".mih") && !Path::has_suffix (filename, ".mif") ))
456  throw Exception ("FIXME: image not suitable for use with 'Image::dump_to_mrtrix_file()'");
457 
458  // try to dump file to mrtrix format if possible (direct IO)
459  if (filename == "-")
460  filename = File::create_tempfile (0, "mif");
461 
462  DEBUG ("dumping image \"" + name() + "\" to file \"" + filename + "\"...");
463 
464  File::OFStream out (filename, std::ios::out | std::ios::binary);
465  out << "mrtrix image\n";
466  Formats::write_mrtrix_header (*buffer, out);
467 
468  const bool single_file = Path::has_suffix (filename, ".mif");
469  std::string data_filename = filename;
470 
471  int64_t offset = 0;
472  out << "file: ";
473  if (single_file) {
474  offset = out.tellp() + int64_t(18);
475  offset += ((4 - (offset % 4)) % 4);
476  out << ". " << offset << "\nEND\n";
477  }
478  else {
479  data_filename = filename.substr (0, filename.size()-4) + ".dat";
480  out << Path::basename (data_filename) << "\n";
481  out.close();
482  out.open (data_filename, std::ios::out | std::ios::binary);
483  }
484 
485  const int64_t data_size = footprint (*buffer);
486  out.seekp (offset, out.beg);
487  out.write ((const char*) data_pointer, data_size);
488  if (!out.good())
489  throw Exception ("error writing back contents of file \"" + data_filename + "\": " + strerror(errno));
490  out.close();
491 
492  // If data_size exceeds some threshold, ostream artificially increases the file size beyond that required at close()
493  // TODO check whether this is still needed...?
494  File::resize (data_filename, offset + data_size);
495 
496  return filename;
497  }
