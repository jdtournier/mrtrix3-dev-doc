86  void check()
87  {
88  if (!handler_)
89  throw Exception ("cannot create sparse image buffer for image with no handler");
90  if (typeid (*handler_) != typeid (Handler::Sparse))
91  throw Exception ("cannot create sparse image buffer for accessing non-sparse image");
92  // Use the header information rather than trying to access this from the handler
93  std::map<std::string, std::string>::const_iterator name_it = find(Image::Sparse::name_key);
94  if (name_it == end())
95  throw Exception ("cannot create sparse image buffer without knowledge of underlying class type in the image header");
96  const std::string& class_name = name_it->second;
97  if (str(typeid(SparseDataType).name()) != class_name)
98  throw Exception ("class type of sparse image buffer does not match that in image header");
99  std::map<std::string, std::string>::const_iterator size_it = find(Image::Sparse::size_key);
100  if (size_it == end())
101  throw Exception ("cannot create sparse image buffer without knowledge of underlying class size in the image header");
102  const size_t class_size = to<size_t>(size_it->second);
103  if (sizeof(SparseDataType) != class_size)
104  throw Exception ("class size of sparse image buffer does not match that in image header");
105  DEBUG ("Sparse image buffer verified for accessing image " + name() + " using type " + str(typeid(SparseDataType).name()));
106  }
