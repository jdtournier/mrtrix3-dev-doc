37 // Base class to handle case where the factor contributed by the streamline varies along its length
38 // (currently only occurs when the track-wise statistic is Gaussian)
39 class VoxelAddon
40 {
41  public:
42  VoxelAddon() : sum_factors (0.0) { }
43  VoxelAddon (const float v) : sum_factors (v) { }
44  float get_factor() const { return sum_factors; }
45  protected:
46  void operator+= (const float f) const { sum_factors += f; }
47  void operator= (const float f) { sum_factors = f; }
48  void operator= (const VoxelAddon& that) { sum_factors = that.sum_factors; }
49  void normalise (const float l) const { sum_factors /= l; }
50  private:
51  mutable float sum_factors;
52 };
