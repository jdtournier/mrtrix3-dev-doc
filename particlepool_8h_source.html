47  Particle* create(const Point_t& pos, const Point_t& dir)
48  {
49  std::lock_guard<std::mutex> lock (mutex);
50  if (avail.empty()) {
51  pool.emplace_back(pos, dir);
52  return &pool.back();
53  } else {

80  Particle* random() {
81  std::lock_guard<std::mutex> lock (mutex);
82  if (pool.size() > avail.size())
83  {
84  std::uniform_int_distribution<size_t> dist(0, pool.size()-1);
85  for (int k = 0; k != 5; ++k) {
