82  Particle* random() {
83  std::lock_guard<std::mutex> lock (mutex);
84  if (pool.size() > avail.size())
85  {
86  std::uniform_int_distribution<size_t> dist(0, pool.size()-1);
87  for (int k = 0; k != 5; ++k) {
88  Particle* p = &pool[dist(rng)];
