81  Particle* random() {
82  std::lock_guard<std::mutex> lock (mutex);
83  if (pool.size() > avail.size())
84  {
85  std::uniform_int_distribution<size_t> dist(0, pool.size()-1);
86  for (int k = 0; k != 5; ++k) {
87  Particle* p = &pool[dist(rng)];
