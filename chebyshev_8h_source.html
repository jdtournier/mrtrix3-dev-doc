43  template <typename T> inline T eval (const double* coef, const int order, const T lower, const T upper, const T x)
44  {
45  T y = (2.0*x - lower - upper) / (upper - lower);
46  T d = 0.0, dd = 0.0;
47  for (int i = order; i >= 1; i--) {
48  T temp = d;
49  d = 2.0*y*d - dd + coef[i];
50  dd = temp;
51  }
52  return (y*d - dd + 0.5 * coef[0]);
53  }
