99  inline double calcShiftProb(const Particle* par, const Point_t& pos, const Point_t& dir) const
100  {
101  Point_t Dpos = par->getPosition() - pos;
102  Point_t Ddir = par->getDirection() - dir;
103  return gaussian_pdf(Dpos, sigpos) * gaussian_pdf(Ddir, sigdir);
104  }
105 
106  inline double gaussian_pdf(const Point_t& x, double sigma) const {
107  return std::exp( -x.squaredNorm() / (2*sigma) ) / std::sqrt( 2*Math::pi * sigma*sigma);
108  }
