98  inline double calcShiftProb(const Particle* par, const Point_t& pos, const Point_t& dir) const
99  {
100  Point_t Dpos = par->getPosition() - pos;
101  Point_t Ddir = par->getDirection() - dir;
102  return gaussian_pdf(Dpos, sigpos) * gaussian_pdf(Ddir, sigdir);
103  }

105  inline double gaussian_pdf(const Point_t& x, double sigma) const {
106  return std::exp( -x.squaredNorm() / (2*sigma) ) / std::sqrt( 2*Math::pi * sigma*sigma);
