593  FORCE_INLINE bool push (T*& item) {
594  std::unique_lock<std::mutex> lock (mutex);
595  more_space.wait (lock, [this]{ return !(full() && reader_count); });
596  if (!reader_count) return false;
597  *back = item;
598  back = inc (back);
599  if (item_stack.empty()) {
600  item = new T;
601  items.push_back (std::unique_ptr<T> (item));
602  }
603  else {
604  item = item_stack.top();
605  item_stack.pop();
606  }
607  more_data.notify_one();

609  }

611  FORCE_INLINE bool pop (T*& item) {
612  std::unique_lock<std::mutex> lock (mutex);
613  if (item)
614  item_stack.push (item);
