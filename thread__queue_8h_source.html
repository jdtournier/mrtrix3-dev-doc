615  bool push (T*& item) {
616  {
617  std::unique_lock<std::mutex> lock (mutex);
618  more_space.wait (lock, [this]{ return !(full() && reader_count); });
619  if (!reader_count) return false;
620  *back = item;
621  back = inc (back);
622  if (item_stack.empty()) {
623  item = new T;
624  items.push_back (item);
625  }
626  else {
627  item = item_stack.top();
628  item_stack.pop();
629  }
630  }
631  more_data.notify_one();
632  return true;
633  }

635  bool pop (T*& item) {
636  {
637  std::unique_lock<std::mutex> lock (mutex);
638  if (item)
639  item_stack.push (item);
640  item = nullptr;
641  more_data.wait (lock, [this]{ return !(empty() && writer_count); });
