586  FORCE_INLINE bool push (T*& item) {
587  std::unique_lock<std::mutex> lock (mutex);
588  more_space.wait (lock, [this]{ return !(full() && reader_count); });
589  if (!reader_count) return false;
590  *back = item;
591  back = inc (back);
592  if (item_stack.empty()) {
593  item = new T;
594  items.push_back (std::unique_ptr<T> (item));
595  }
596  else {
597  item = item_stack.top();
598  item_stack.pop();
599  }
600  more_data.notify_one();
601  return true;
602  }
603 
604  FORCE_INLINE bool pop (T*& item) {
605  std::unique_lock<std::mutex> lock (mutex);
606  if (item)
607  item_stack.push (item);
608  item = nullptr;
609  more_data.wait (lock, [this]{ return !(empty() && writer_count); });
610  if (empty() && !writer_count)
611  return false;
612  item = *front;
613  front = inc (front);
614  more_space.notify_one();
615  return true;
616  }
