350  template <> inline void put<bool> (const bool value, void* data, size_t i, bool)
351  {
352  std::atomic<uint8_t>* at = reinterpret_cast<std::atomic<uint8_t>*> (((uint8_t*) data) + (i/8));
353  uint8_t prev = *at, new_value;
354  do {
355  if (value) new_value = prev | (BITMASK >> i%8);
356  else new_value = prev & ~(BITMASK >> i%8);
357  } while (!at->compare_exchange_weak (prev, new_value));
