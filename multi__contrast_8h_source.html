36 FORCE_INLINE void check_image_output (const std::string& image_name, const Header& reference) {
37 vector<std::string> V;
38 if (!image_name.size()) throw Exception ("image output path is empty");
39 if (Path::exists (image_name) && !App::overwrite_files)
40 throw Exception ("output image \"" + image_name + "\" already exists (use -force option to force overwrite)");
41
42 Header H = reference;
43 File::NameParser parser;
44 parser.parse (image_name);
45 vector<int> Pdim (parser.ndim());
46
47 H.name() = image_name;
48
49 const Formats::Base** format_handler = Formats::handlers;
50 for (; *format_handler; format_handler++)
51 if ((*format_handler)->check (H, H.ndim() - Pdim.size()))
52 break;
53
54 if (!*format_handler) {
55 const std::string basename = Path::basename (image_name);
56 const size_t extension_index = basename.find_last_of (".");
57 if (extension_index == std::string::npos)
58 throw Exception ("unknown format for image \"" + image_name + "\" (no file extension specified)");
59 else
60 throw Exception ("unknown format for image \"" + image_name + "\" (unsupported file extension: " + basename.substr (extension_index) + ")");
61 }
62 }

64 struct MultiContrastSetting { NOMEMALIGN
65 size_t start; // index to volume in image holding all tissue contrasts
66 size_t nvols; // number of volumes preloaded into image holding all tissue contrasts
67 ssize_t lmax; // maximum requested lmax
68 bool do_reorientation; // registration treats this image as (possibly 4D) scalar image
69 size_t image_nvols; // number of volumes in original image
70 ssize_t image_lmax; // lmax available in image (preloaded, might differ from original image), 0 if not an FOD image
71 default_type weight;
72
73 MultiContrastSetting (): start (0), nvols (0), do_reorientation (false), weight (1.0) {}
74 MultiContrastSetting (size_t image_nvols, bool do_reorientation = false, ssize_t limit_lmax = std::numeric_limits<ssize_t>::max()) :
75 start (0), nvols (image_nvols), do_reorientation (do_reorientation), weight (1.0) {
76 if (do_reorientation) {
77 image_lmax = Math::SH::LforN (image_nvols);
78 lmax = image_lmax;
79 } else {
80 image_lmax = 0;
81 lmax = 0;
82 }
83 lower_lmax (limit_lmax);
84 }
85
86 FORCE_INLINE void lower_lmax (ssize_t new_lmax) {
87 assert (new_lmax >= 0);
88 if (new_lmax < lmax) {
89 lmax = new_lmax;
90 if (new_lmax > 0) {
91 nvols = Math::SH::NforL (new_lmax);
92 } else {
93 nvols = 1;
94 }
95 }
96 }
97 };
