149  // set_matrix updates the 3x3 matrix and also updates the translation
150  void set_matrix (const Eigen::Matrix<ParameterType, 3, 3>& mat) {
151  transform_type Tc2, To, R0;
152  Tc2.setIdentity();
153  To.setIdentity();
154  R0.setIdentity();
155  To.translation() = offset;
156  Tc2.translation() = centre - 0.5 * offset;
157  R0.linear() = mat;
158  trafo = Tc2 * To * R0 * Tc2.inverse();
159  compute_halfspace_transformations();
160  }

247  void compute_halfspace_transformations() {
248  Eigen::Matrix<ParameterType, 4, 4> tmp;
249  tmp.setIdentity();
250  tmp.template block<3,4>(0,0) = trafo.matrix();
251  assert ((tmp.template block<3,3>(0,0).isApprox (trafo.linear())));
252  assert (tmp.determinant() > 0);
253  tmp = tmp.sqrt().eval();
254  trafo_half.matrix() = tmp.template block<3,4>(0,0);
255  trafo_half_inverse.matrix() = trafo_half.inverse().matrix();
256  assert (trafo.matrix().isApprox ((trafo_half*trafo_half).matrix()));
257  assert (trafo.inverse().matrix().isApprox ((trafo_half_inverse*trafo_half_inverse).matrix()));
258  // debug();
259  }
