85  void calc_distance ()
86  {
87  if (!std::isfinite (orientation_z[0]))
88  Math::cross (orientation_z, orientation_x, orientation_y);
89  else {
90  float normal[3];
91  Math::cross (normal, orientation_x, orientation_y);
92  if (Math::dot (normal, orientation_z) < 0.0) {
93  orientation_z[0] = -normal[0];
94  orientation_z[1] = -normal[1];
95  orientation_z[2] = -normal[2];
96  }
97  else {
98  orientation_z[0] = normal[0];
99  orientation_z[1] = normal[1];
100  orientation_z[2] = normal[2];
101  }
102  }
103 
104  Math::normalise (orientation_z);
105  distance = Math::dot (orientation_z, position_vector);
106  }
