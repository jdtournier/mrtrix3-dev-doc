169  template <class Fixel>
170  void Model<Fixel>::map_streamlines (const std::string& path)
171  {
172  Tractography::Properties properties;
173  Tractography::Reader<> file (path, properties);
174 
175  const track_t count = (properties.find ("count") == properties.end()) ? 0 : to<track_t>(properties["count"]);
176  if (!count)
177  throw Exception ("Cannot map streamlines: track file " + Path::basename(path) + " is empty");
178 
179  contributions.assign (count, nullptr);
180 
181  {
182  Mapping::TrackLoader loader (file, count);
183  TrackMappingWorker worker (*this, Mapping::determine_upsample_ratio (Fixel_map<Fixel>::header(), properties, 0.1));
184  Thread::run_queue (loader,
185  Thread::batch (Tractography::Streamline<>()),
186  Thread::multi (worker));
187  }
188 
189  if (!contributions.back()) {
190  track_t num_tracks = 0, max_index = 0;
191  for (track_t i = 0; i != contributions.size(); ++i) {
192  if (contributions[i]) {
193  ++num_tracks;
194  max_index = std::max (max_index, i);
195  }
196  WARN ("Only " + str (num_tracks) + " tracks read from input track file; expected " + str (contributions.size()));
197  contributions.resize (max_index + 1);
198  }
199  }
200 
201  tck_file_path = path;
202 
203  INFO ("Proportionality coefficient after streamline mapping is " + str (mu()));
204  }

210  template <class Fixel>
211  void Model<Fixel>::remove_excluded_fixels ()
212  {
213 
214  const bool remove_untracked_fixels = App::get_options ("remove_untracked").size();
215  auto opt = App::get_options ("fd_thresh");
216  const float min_fibre_density = opt.size() ? float(opt[0][0]) : 0.0;
217 
218  if (!remove_untracked_fixels && !min_fibre_density)
219  return;
220 
221  vector<size_t> fixel_index_mapping (fixels.size(), 0);
222  VoxelAccessor v (accessor());
223 
224  vector<Fixel> new_fixels;
225  new_fixels.push_back (Fixel());
226  FOD_sum = 0.0;
227 
228  for (auto l = Loop (v) (v); l; ++l) {
229  if (v.value()) {
230 
231  size_t new_start_index = new_fixels.size();
232 
233  for (typename Fixel_map<Fixel>::Iterator i = begin(v); i; ++i) {
234  if ((!remove_untracked_fixels || i().get_TD()) && (i().get_FOD() > min_fibre_density)) {
235  fixel_index_mapping [size_t (i)] = new_fixels.size();
236  new_fixels.push_back (i());
237  FOD_sum += i().get_weight() * i().get_FOD();
238  }
239  }
240 
241  delete v.value();
242 
243  if (new_fixels.size() == new_start_index)
244  v.value() = nullptr;
245  else
246  v.value() = new MapVoxel (new_start_index, new_fixels.size() - new_start_index);
247 
248  }
249  }
250 
251  INFO (str (fixels.size() - new_fixels.size()) + " out of " + str(fixels.size()) + " fixels removed from reconstruction (" + str(new_fixels.size()) + ") remaining)");
252 
253  fixels.swap (new_fixels);
254 
255  TrackIndexRangeWriter writer (SIFT_TRACK_INDEX_BUFFER_SIZE, num_tracks(), "Removing excluded fixels");
256  FixelRemapper remapper (*this, fixel_index_mapping);
257  Thread::run_queue (writer, TrackIndexRange(), Thread::multi (remapper));
258 
259  TD_sum = 0.0;
260  for (typename vector<Fixel>::const_iterator i = fixels.begin(); i != fixels.end(); ++i)
261  TD_sum += i->get_weight() * i->get_TD();
262 
263  INFO ("After fixel exclusion, the proportionality coefficient is " + str(mu()));
264 
265  }

273  template <class Fixel>
274  void Model<Fixel>::check_TD()
275  {
276  VAR (TD_sum);
277  double sum_from_fixels = 0.0, sum_from_fixels_weighted = 0.0;
278  for (typename vector<Fixel>::const_iterator i = fixels.begin(); i != fixels.end(); ++i) {
279  sum_from_fixels += i->get_TD();
280  sum_from_fixels_weighted += i->get_TD() * i->get_weight();
281  }
282  VAR (sum_from_fixels);
283  VAR (sum_from_fixels_weighted);
284  double sum_from_tracks = 0.0;
285  for (vector<TrackContribution*>::const_iterator i = contributions.begin(); i != contributions.end(); ++i) {
286  if (*i)
287  sum_from_tracks += (*i)->get_total_contribution();
288  }
289  VAR (sum_from_tracks);
290  }

296  template <class Fixel>
297  void Model<Fixel>::output_non_contributing_streamlines (const std::string& output_path) const
298  {
299  Tractography::Properties p;
300  Tractography::Reader<float> reader (tck_file_path, p);
301  Tractography::Writer<float> writer (output_path, p);
302  Tractography::Streamline<> tck;
303  ProgressBar progress ("Writing non-contributing streamlines output file", contributions.size());
304  track_t tck_counter = 0;
305  while (reader (tck) && tck_counter < contributions.size()) {
306  if (contributions[tck_counter] && !contributions[tck_counter++]->get_total_contribution())
307  writer (tck);
308  else
309  writer.skip();
310  ++progress;
311  }
312  reader.close();
313  }

333  template <class Fixel>
334  bool Model<Fixel>::TrackMappingWorker::operator() (const Tractography::Streamline<>& in)
335  {
336  assert (in.get_index() < master.contributions.size());
337  assert (!master.contributions[in.get_index()]);
338 
339  try {
340 
341  Mapping::SetDixel dixels;
342  mapper (in, dixels);
343 
344  vector<Track_fixel_contribution> masked_contributions;
345  default_type total_contribution = 0.0, total_length = 0.0;
346 
347  for (Mapping::SetDixel::const_iterator i = dixels.begin(); i != dixels.end(); ++i) {
348  total_length += i->get_length();
349  const size_t fixel_index = master.dixel2fixel (*i);
350  if (fixel_index && (i->get_length() > Track_fixel_contribution::min())) {
351  total_contribution += i->get_length() * master.fixels[fixel_index].get_weight();
352  bool incremented = false;
353  for (vector<Track_fixel_contribution>::iterator c = masked_contributions.begin(); !incremented && c != masked_contributions.end(); ++c) {
354  if ((c->get_fixel_index() == fixel_index) && c->add (i->get_length()))
355  incremented = true;
356  }
357  if (!incremented)
358  masked_contributions.push_back (Track_fixel_contribution (fixel_index, i->get_length()));
359  }
360  }
361 
362  master.contributions[in.get_index()] = new TrackContribution (masked_contributions, total_contribution, total_length);
363 
364  TD_sum += total_contribution;
365  for (vector<Track_fixel_contribution>::const_iterator i = masked_contributions.begin(); i != masked_contributions.end(); ++i)
366  fixel_TDs [i->get_fixel_index()] += i->get_length();
367 
368  return true;
369 
370  } catch (...) {
371  throw Exception ("Error allocating memory for streamline visitations");
372  return false;
373  }
374  }

380  template <class Fixel>
381  bool Model<Fixel>::FixelRemapper::operator() (const TrackIndexRange& in)
382  {
383  for (track_t track_index = in.first; track_index != in.second; ++track_index) {
384  if (master.contributions[track_index]) {
385  TrackContribution& this_cont (*master.contributions[track_index]);
386  vector<Track_fixel_contribution> new_cont;
387  double total_contribution = 0.0;
388  for (size_t i = 0; i != this_cont.dim(); ++i) {
389  const size_t new_index = remapper[this_cont[i].get_fixel_index()];
390  if (new_index) {
391  new_cont.push_back (Track_fixel_contribution (new_index, this_cont[i].get_length()));
392  total_contribution += this_cont[i].get_length() * master[new_index].get_weight();
393  }
394  }
395  TrackContribution* new_contribution = new TrackContribution (new_cont, total_contribution, this_cont.get_total_length());
396  delete master.contributions[track_index];
397  master.contributions[track_index] = new_contribution;
398  }
399  }
400  return true;
401  }
