48  class Track_fixel_contribution
49  {
50  public:
51  Track_fixel_contribution (const uint32_t fixel_index, const float length) {
52  const uint32_t length_as_int = std::min (uint32_t(255), uint32_t(Math::round (scale_to_storage * length)));
53  data = (fixel_index & 0x00FFFFFF) | (length_as_int << 24);
54  }
55 
56  Track_fixel_contribution() :
57  data (0) { }
58 
59  uint32_t get_fixel_index() const { return (data & 0x00FFFFFF); }
60  float get_length() const { return (uint32_t((data & 0xFF000000) >> 24) * scale_from_storage); }
61 
62 
63  bool add (const float length)
64  {
65  // Allow summing of multiple contributions to a fixel, UNLESS it would cause truncation, in which
66  // case keep them separate
67  const uint32_t increment = Math::round (scale_to_storage * length);
68  const uint32_t existing = (data & 0xFF000000) >> 24;
69  if (existing + increment > 255)
70  return false;
71  data = (data & 0x00FFFFFF) | ((existing + increment) << 24);
72  return true;
73  }

78  const float max_length = Math::sqrt (Math::pow2 (in.vox(0)) + Math::pow2 (in.vox(1)) + Math::pow2 (in.vox(2)));
79  // TODO Newer mapping performs chordal approximation of length
