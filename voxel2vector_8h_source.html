34  // A class to achieve a mapping from a voxel position in an image
35  // with any number of axes, to an index within a 1D vector of data.
36  class Voxel2Vector
37  { MEMALIGN(Voxel2Vector)
38  public:

40  typedef uint32_t index_t;

42  static const index_t invalid = std::numeric_limits<index_t>::max();

51  size_t size() const { return reverse.size(); }

53  const vector<index_t>& operator[] (const size_t index) const {
54  assert (index < reverse.size());

58  template <class PosType>
59  index_t operator() (const PosType& pos) const {
60  Image<index_t> temp (forward); // For thread-safety
61  assign_pos_of (pos).to (temp);
62  if (is_out_of_bounds (temp))
63  return invalid;
64  return temp.value();
65  }
66

69  vector< vector<index_t> > reverse;
70  };

74  template <class MaskType>
75  Voxel2Vector::Voxel2Vector (MaskType& mask, const Header& data) :
76  forward (Image<index_t>::scratch (data, "Voxel to vector index conversion scratch image"))
77  {
78  if (!dimensions_match (mask, data, 0, std::min (mask.ndim(), data.ndim())))
79  throw Exception ("Dimension mismatch between image data and processing mask");
80  // E.g. Mask may be 3D but data are 4D; for any voxel where the mask is
81  // true, want to include data from all volumes
82  Adapter::Replicate<MaskType> r_mask (mask, data);
83  // Loop in axis order so that those voxels contiguous in memory are still
84  // contiguous in the vectorised data
85  index_t counter = 0;
86  for (auto l = Loop(data) (r_mask, forward); l; ++l) {
87  if (r_mask.value()) {
88  forward.value() = counter++;
89  vector<index_t> pos;
90  for (size_t index = 0; index != data.ndim(); ++index)
91  pos.push_back (forward.index(index));
92  reverse.push_back (pos);
93  } else {
94  forward.value() = invalid;
95  }
96  }
97  DEBUG ("Voxel2Vector class has " + str(reverse.size()) + " non-zero entries");
98  }
