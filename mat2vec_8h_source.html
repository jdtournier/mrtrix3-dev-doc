73 Cont& Mat2Vec::operator() (const matrix_type& in, Cont& out) const
74 {
75  assert (in.rows() == in.cols());
76  assert (in.rows() == size);
77  out.resize (vec_size());
78  for (size_t index = 0; index != vec_size(); ++index) {
79  const std::pair<node_t, node_t> row_column = (*this) (index);
80  out[index] = in (row_column.first, row_column.second);
81  }
82  return out;
