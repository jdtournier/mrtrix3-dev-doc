41  class LinearUpdate {
42  public:
43  template <typename ValueType>
44  inline bool operator() (Math::Vector<ValueType>& newx, const Math::Vector<ValueType>& x,
45  const Math::Vector<ValueType>& g, ValueType step_size) {
46  bool changed = false;
47  for (size_t n = 0; n < x.size(); ++n) {
48  newx[n] = x[n] - step_size * g[n];
49  if (newx[n] != x[n])
50  changed = true;
51  }
52  return changed;
53  }

108  dt = func.init (x);
109  nfeval = 0;

111  compute_normg_and_step_unscaled ();

122  bool iterate (bool verbose = false) {
123  assert (normg != 0.0);
124 
125  while (true) {
126  if (!update_func (x2, x, g, dt))
127  return false;
128 
129  value_type f2 = evaluate_func (x2, g2, verbose);
130 
131  // quadratic minimum:
132  value_type step_length = step_unscaled*dt;
133  value_type denom = 2.0 * (normg*step_length + f2 - f);
134  value_type quadratic_minimum = denom > 0.0 ? normg * step_length / denom : step_up;
135 
136  if (quadratic_minimum < step_down) quadratic_minimum = step_down;
137  if (quadratic_minimum > step_up) quadratic_minimum = step_up;
138 
139  if (f2 < f) {
140  dt *= quadratic_minimum;
141  f = f2;
142  x.swap (x2);
143  g.swap (g2);
144  compute_normg_and_step_unscaled ();
145  return true;
146  }
147 
148  if (quadratic_minimum >= 1.0)
149  quadratic_minimum = 0.5;
150  dt *= quadratic_minimum;
151  }
152  }

173  void compute_normg_and_step_unscaled () {
174  normg = step_unscaled = norm (g);
175  if (preconditioner_weights.size()) {
176  value_type g_projected = 0.0;
177  step_unscaled = 0.0;
178  for (size_t n = 0; n < g.size(); ++n) {
179  step_unscaled += Math::pow2 (g[n]);
180  g_projected += preconditioner_weights[n] * Math::pow2 (g[n]);
181  g[n] *= preconditioner_weights[n];
182  }
183  normg = g_projected / normg;
184  step_unscaled = Math::sqrt (step_unscaled);
185  }
