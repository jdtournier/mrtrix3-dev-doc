31  class LinearUpdate { NOMEMALIGN
32  public:
33  template <typename ValueType>
34  inline bool operator() (Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& newx, const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& x,
35  const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& g, ValueType step_size) {
36  bool changed = false;
37  for (ssize_t n = 0; n < x.size(); ++n) {
38  newx[n] = x[n] - step_size * g[n];
39  if (newx[n] != x[n])
40  changed = true;
41  }
42  return changed;
43  }
44  };

155  bool iterate (std::ostream& log_os) {
156  // assert (normg != 0.0);
157  assert (std::isfinite (normg));
158 
159 
160  while (normg != 0.0) {
161  if (!update_func (x2, x, g, dt))
162  return false;
163 
164  value_type f2 = evaluate_func (x2, g2, verbose);
165 
166  // quadratic minimum:
167  value_type step_length = step_unscaled*dt;
168  value_type denom = 2.0 * (normg*step_length + f2 - f);
169  value_type quadratic_minimum = denom > 0.0 ? normg * step_length / denom : step_up;
170 
171  if (quadratic_minimum < step_down) quadratic_minimum = step_down;
172  if (quadratic_minimum > step_up) quadratic_minimum = step_up;
173 
174  if (f2 < f) {
175  ++niter;
176  dt *= quadratic_minimum;
177  f = f2;
178  x.swap (x2);
179  g.swap (g2);
180  if (log_os) {
181  log_os << niter << delim << nfeval << delim << str(f) << delim << str(dt);
182  for (ssize_t i=0; i< x.size(); ++i){ log_os << delim << str(x(i)); }
183  for (ssize_t i=0; i< x.size(); ++i){ log_os << delim << str(g(i)); }
184  log_os << std::endl;
185  }
186  compute_normg_and_step_unscaled ();
187  return true;
188  }
189 
190  if (quadratic_minimum >= 1.0)
191  quadratic_minimum = 0.5;
192  dt *= quadratic_minimum;
193 
194  if (dt <= 0.0)
195  return false;
196  }
197  return false;
198  }

225  normg = step_unscaled = g.norm();

227  if (normg > 0.0){

233  g.array() *= preconditioner_weights.array();
234  normg = g_projected / normg;
