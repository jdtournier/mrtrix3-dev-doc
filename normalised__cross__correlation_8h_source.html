98  using value_type = typename ImageType3::value_type;
99  in1.index(0) = mask.index(0);
100  in1.index(1) = mask.index(1);
101  in1.index(2) = mask.index(2);
102  value_type value_in1 = in1.value();
103  if (value_in1 != value_in1){ // nan in image 1, update mask
104  DEBUG("nan in image 1.");
105  mask.value() = false;
106  out.row(3) = 0.0;
107  return;
108  }
109  in2.index(0) = mask.index(0);
110  in2.index(1) = mask.index(1);
111  in2.index(2) = mask.index(2);
112  value_type value_in2 = in2.value();
113  if (value_in2 != value_in2){ // nan in image 2, update mask
114  DEBUG("nan in image 2.");
115  mask.value() = false;
116  out.row(3) = 0.0;
117  return;
118  }
119  auto niter = NeighbourhoodIterator(mask, extent);
120  value_type v1 = 0, v1_2 = 0, v2 = 0, v2_2 = 0 , v1_v2 = 0, mean1, mean2;
121  size_t cnt = 0;
122  // DEBUG(str(niter));
123  while(niter.loop()){
124  mask.index(0) = niter.index(0);
125  mask.index(1) = niter.index(1);
126  mask.index(2) = niter.index(2);
127  // DEBUG(" "+str(mask));
128  if (!mask.value())
129  continue;
130  in1.index(0) = niter.index(0);
131  in1.index(1) = niter.index(1);
132  in1.index(2) = niter.index(2);
133  value_type val1 = in1.value();
134  if (val1 != val1){
135  DEBUG("nan in image 1");
136  continue;
137  }
138  in2.index(0) = niter.index(0);
139  in2.index(1) = niter.index(1);
140  in2.index(2) = niter.index(2);
141  value_type val2 = in2.value();
142  if (val2 != val2){
143  DEBUG("nan in image 2");
144  continue;
145  }
146  v1 += val1;
147  v1_2 += val1 * val1;
148  v2 += val2;
149  v2_2 += val2 * val2;
150  v1_v2 += val1 * val2;
151  cnt += 1;
152  }
153  // reset the mask index
154  mask.index(0) = out.index(0);
155  mask.index(1) = out.index(1);
156  mask.index(2) = out.index(2);
157 
158  if (cnt <= 0)
159  throw Exception ("FIXME: neighbourhood does not contain centre");
160 
161  // subtract mean
162  mean1 = v1 / cnt;
163  mean2 = v2 / cnt;
164  v1_2 -= (v1 * mean1);
165  v2_2 -= (v2 * mean2);
166  v1_v2 -= (v2 * mean1);
167 
168  assert(mean1 == mean1);
169  assert(mean2 == mean2);
170  assert(v1_2 == v1_2);
171  assert(v2_2 == v2_2);
172  assert(v1_v2 == v1_v2);

217  default_type precompute(ParamType& parameters) {
218  INFO("precomputing cross correlation data...");

221  using Im1Type = decltype(parameters.im1_image);
222  using Im2Type = decltype(parameters.im2_image);
223  using Im1MaskType = decltype(parameters.im1_mask);
224  using Im2MaskType = decltype(parameters.im2_mask);
225  using ProcessedImageValueType = typename ParamType::ProcessedValueType;
226  using ProcessedMaskType = typename ParamType::ProcessedMaskType;

230  Header midway_header (parameters.midway_image);
231  midway_v2s = MR::Transform (midway_header).voxel2scanner;
232 
233  // store precomputed values in cc_image:
234  // volumes 0 and 1: normalised intensities of both images (Im1 and Im2)
235  // vlumes 2 to 4: neighbourhood dot products Im1.dot(Im2), Im1.dot(Im1), Im2.dot(Im2)
236  auto cc_image_header = Header::scratch (midway_header);
237  cc_image_header.ndim() = 4;
238  cc_image_header.size(3) = 5;
239  ProcessedMaskType cc_mask;
240  auto cc_mask_header = Header::scratch (parameters.midway_image);
241 
242  auto cc_image = cc_image_header.template get_image <ProcessedImageValueType>().with_direct_io(Stride::contiguous_along_axis(3));
243  {
244  LogLevelLatch log_level (0);
245  if (parameters.im1_mask.valid() or parameters.im2_mask.valid())
246  cc_mask = cc_mask_header.template get_image<bool>();
247  if (parameters.im1_mask.valid() and !parameters.im2_mask.valid())
248  Filter::reslice <Interp::Nearest> (parameters.im1_mask, cc_mask, parameters.transformation.get_transform_half());
249  else if (!parameters.im1_mask.valid() and parameters.im2_mask.valid())
250  Filter::reslice <Interp::Nearest> (parameters.im2_mask, cc_mask, parameters.transformation.get_transform_half_inverse(), Adapter::AutoOverSample);
251  else if (parameters.im1_mask.valid() and parameters.im2_mask.valid()){
252  Adapter::Reslice<Interp::Linear, Im1MaskType> mask_reslicer1 (parameters.im1_mask, cc_mask_header, parameters.transformation.get_transform_half());
253  Adapter::Reslice<Interp::Linear, Im2MaskType> mask_reslicer2 (parameters.im2_mask, cc_mask_header, parameters.transformation.get_transform_half_inverse());
254  // TODO should be faster to just loop over m1:
255  // if (m1.value())
256  // assign_pos_of(m1).to(m2); cc_mask.value() = m2.value()
257  auto both = [](decltype(cc_mask)& cc_mask, decltype(mask_reslicer1)& m1, decltype(mask_reslicer2)& m2) {
258  cc_mask.value() = ((m1.value() + m2.value()) / 2.0) > 0.5 ? true : false;
259  };
260  ThreadedLoop (cc_mask).run (both, cc_mask, mask_reslicer1, mask_reslicer2);
261  }
262  }
263 
264  Adapter::Reslice<Interp::Cubic, Im1Type> interp1 (parameters.im1_image, cc_image_header, parameters.transformation.get_transform_half(), Adapter::AutoOverSample, std::numeric_limits<typename Im1Type::value_type>::quiet_NaN());
265 
266  Adapter::Reslice<Interp::Cubic, Im2Type> interp2 (parameters.im2_image, cc_image_header, parameters.transformation.get_transform_half_inverse(), Adapter::AutoOverSample, std::numeric_limits<typename Im2Type::value_type>::quiet_NaN());
267 
268  const auto extent = parameters.get_extent();
269 
270  // TODO unmasked CCPrecomputeFunctor. create a mask (all voxels true) if none given.

272  if (!cc_mask.valid()){
273  cc_mask = cc_mask_header.template get_image<bool>();

279  loop.run (NCCPrecomputeFunctorMasked_Naive<decltype(interp1), decltype(interp2)>(extent, interp1, interp2), parameters.processed_mask, cc_image);
280  parameters.processed_image = cc_image;
281  parameters.processed_image_interp.reset (new CCInterpType (parameters.processed_image));
282  // display<Image<float>>(parameters.processed_image);
283  return 0;

287  default_type operator() (Params& params,
288  const Iterator& iter,
289  Eigen::Matrix<default_type, Eigen::Dynamic, 1>& gradient) {
290  assert (params.processed_image.index(0) == iter.index(0)); // iterates over processed image rather than midway image
291  assert (params.processed_image.index(1) == iter.index(1));
292  assert (params.processed_image.index(2) == iter.index(2));
293 
294  if (params.processed_mask.valid()) {
295  params.processed_mask.index(0) = iter.index(0);
296  params.processed_mask.index(1) = iter.index(1);

298  if (!params.processed_mask.value())
299  return 0.0;
300  }

302  const Eigen::Vector3 pos = Eigen::Vector3(default_type(iter.index(0)), default_type(iter.index(0)), default_type(iter.index(0)));
303 
304  params.processed_image.index(3) = 2;
305  default_type A = params.processed_image.value();
306  params.processed_image.index(3) = 3;
307  default_type B = params.processed_image.value();
308  params.processed_image.index(3) = 4;
309  default_type C = params.processed_image.value();
310  default_type A_BC = A / (B * C);
311 
312  if (A_BC != A_BC){
313  DEBUG("A_BC is NAN");
314  return 0.0;
315  }
316 
317  params.processed_image_interp->voxel(pos);
318  params.processed_image_interp->index (3) = 0;
319  typename Params::Im1ValueType val1;
320  typename Params::Im2ValueType val2;
321  Eigen::Matrix<typename Params::Im1ValueType, 1, 3> grad1;
322  Eigen::Matrix<typename Params::Im2ValueType, 1, 3> grad2;
323 
324  // gradient calculation

326  params.processed_image_interp->value_and_gradient_wrt_scanner(val1, grad1);
327  if (val1 != val1){
328  // this should not happen as the precompute should have changed the mask
329  WARN("FIXME: val1 is nan");
330  return 0.0;
331  }
332  params.processed_image_interp->index(3) = 1;
333  params.processed_image_interp->value_and_gradient_wrt_scanner(val2, grad2);
334  if (val2 != val2){
335  // this should not happen as the precompute should have changed the mask
336  WARN("FIXME: val2 is nan");
337  return 0.0;
338  }

357  // std::cerr << "val2: " << params.processed_image.value() << ", " << val2_scanner << ", " << val2 << std::endl;
358  // }
359  const Eigen::Vector3 midway_point = midway_v2s * pos;
360  Eigen::MatrixXd jacobian = params.transformation.get_jacobian_wrt_params (midway_point);
361  for (ssize_t par = 0; par < gradient.size(); par++) {
362  default_type sum = 0.0;
363  for ( size_t dim = 0; dim < 3; dim++){
364  sum -= A_BC * jacobian (dim, par) * (val2 - A/B * val1) * grad1[dim];

366  }

371  }
