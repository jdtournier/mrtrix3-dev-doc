85  float depth_of (const Eigen::Vector3f& x) const {
86  float d = MVP(2,0)*x[0] + MVP(2,1)*x[1] + MVP(2,2)*x[2] + MVP(2,3);
87  if (MVP(3,2)) d /= MVP(3,0)*x[0] + MVP(3,1)*x[1] + MVP(3,2)*x[2] + MVP(3,3);
88  return d;
89  }

91  Eigen::Vector3f model_to_screen (const Eigen::Vector3f& x) const {
92  Eigen::Vector3f S (
93  MVP(0,0)*x[0] + MVP(0,1)*x[1] + MVP(0,2)*x[2] + MVP(0,3),
94  MVP(1,0)*x[0] + MVP(1,1)*x[1] + MVP(1,2)*x[2] + MVP(1,3),
95  MVP(2,0)*x[0] + MVP(2,1)*x[1] + MVP(2,2)*x[2] + MVP(2,3));
96  if (MVP(3,2))
97  S /= MVP(3,0)*x[0] + MVP(3,1)*x[1] + MVP(3,2)*x[2] + MVP(3,3);
98  S[0] = viewport[0] + 0.5f*viewport[2]*(1.0f+S[0]);
99  S[1] = viewport[1] + 0.5f*viewport[3]*(1.0f+S[1]);
100  return S;
101  }

103  Eigen::Vector3f model_to_screen_direction (const Eigen::Vector3f& dir) const {
104  Eigen::Vector3f S (
105  MVP(0,0)*dir[0] + MVP(0,1)*dir[1] + MVP(0,2)*dir[2],
106  MVP(1,0)*dir[0] + MVP(1,1)*dir[1] + MVP(1,2)*dir[2],
107  MVP(2,0)*dir[0] + MVP(2,1)*dir[1] + MVP(2,2)*dir[2]);
108  S[0] *= 0.5f*viewport[2];
109  S[1] *= 0.5f*viewport[3];

113  Eigen::Vector3f screen_to_model (float x, float y, float depth) const {
114  x = 2.0f*(x-viewport[0])/viewport[2] - 1.0f;
115  y = 2.0f*(y-viewport[1])/viewport[3] - 1.0f;
116  Eigen::Vector3f S (
117  iMVP(0,0)*x + iMVP(0,1)*y + iMVP(0,2)*depth + iMVP(0,3),
118  iMVP(1,0)*x + iMVP(1,1)*y + iMVP(1,2)*depth + iMVP(1,3),
119  iMVP(2,0)*x + iMVP(2,1)*y + iMVP(2,2)*depth + iMVP(2,3));
120  if (MVP(3,2))
121  S /= iMVP(3,0)*x + iMVP(3,1)*y + iMVP(3,2)*depth + iMVP(3,3);
122  return S;
123  }

212  void render_text (const std::string& text, int position, int line = 0) const {
213  QString s (text.c_str());
214  int x, y;
215 
216  if (position & RightEdge) x = width() - font.metric.height() / 2 - font.metric.width (s);
217  else if (position & LeftEdge) x = font.metric.height() / 2;
218  else x = (width() - font.metric.width (s)) / 2;
219 
220  if (position & TopEdge) y = height() - 1.5 * font.metric.height() - line * font.metric.lineSpacing();
221  else if (position & BottomEdge) y = font.metric.height() / 2 + line * font.metric.lineSpacing();
222  else y = (height() - font.metric.height()) / 2 - line * font.metric.lineSpacing();
223 
224  render_text (x, y, text);
225  }
