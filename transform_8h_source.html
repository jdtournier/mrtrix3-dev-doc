40  template <class InfoType>
41  Transform (const InfoType& info) :
42  out_of_bounds (true)
43  {
44  bounds[0] = info.dim(0) - 0.5;
45  bounds[1] = info.dim(1) - 0.5;
46  bounds[2] = info.dim(2) - 0.5;
47  voxelsize[0] = info.vox(0);
48  voxelsize[1] = info.vox(1);
49  voxelsize[2] = info.vox(2);
50 
51  set_matrix (V2S, info.transform());
52  for (size_t i = 0; i < 3; i++) {
53  for (size_t j = 0; j < 3; j++) {
54  V2S[i][j] *= info.vox (j);
55  }
56  }
57 
58  int signum;
59  Math::Permutation p (4);
60  Math::Matrix<float> D (4,4);
61  Math::Matrix<float> M (4,4);
62  for (size_t i = 0; i < 3; i++)
63  for (size_t j = 0; j < 4; j++)
64  D(i,j) = V2S[i][j];
65  D (3,0) = D (3,1) = D (3,2) = 0.0;
66  D (3,3) = 1.0;
67  Math::LU::decomp (D, p, signum);
68  Math::LU::inv (M, D, p);
69  M (3,0) = M (3,1) = M (3,2) = 0.0;
70  M (3,3) = 1.0;
71  set_matrix (S2V, M);
72 
73  set_matrix (I2S, info.transform());
74 
75  D = info.transform();
76  Math::LU::decomp (D, p, signum);
77  Math::LU::inv (M, D, p);
78  M (3,0) = M (3,1) = M (3,2) = 0.0;
79  M (3,3) = 1.0;
80  set_matrix (S2I, M);
81  }

215  static inline void transform_position (P1& y, const Math::Matrix<T>& M, const P2& x)
216  {
217  y[0] = M (0,0) *x[0] + M (0,1) *x[1] + M (0,2) *x[2] + M (0,3);
218  y[1] = M (1,0) *x[0] + M (1,1) *x[1] + M (1,2) *x[2] + M (1,3);
219  y[2] = M (2,0) *x[0] + M (2,1) *x[1] + M (2,2) *x[2] + M (2,3);
