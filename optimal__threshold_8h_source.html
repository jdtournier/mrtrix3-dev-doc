132  template <class InputVoxelType, class MaskVoxelType>
133  class ImageCorrelationCostFunction {
134 
135  public:
136  typedef typename InputVoxelType::value_type value_type;
137  typedef typename MaskVoxelType::value_type mask_value_type;
138 
139  ImageCorrelationCostFunction (InputVoxelType& input, MaskVoxelType* mask = NULL) :
140  input (input),
141  mask (mask) {
142  double sum_sqr = 0.0, sum = 0.0;
143  count = 0;
144 
145  Image::ThreadedLoop loop (input);
146  if (mask) {
147  Adapter::Replicate<MaskVoxelType> replicated_mask (*mask, input);
148  loop.run (MeanStdFunctor (sum, sum_sqr, count), input, replicated_mask);
149  }
150  else {
151  loop.run (MeanStdFunctor (sum, sum_sqr, count), input);
152  }
153 
154  input_image_mean = sum / count;
155  input_image_stdev = sqrt ((sum_sqr - sum * input_image_mean) / count);
156  }
157 
158  value_type operator() (value_type threshold) const {
159  double sum = 0;
160  double mean_xy = 0.0;
161 
162  Image::ThreadedLoop loop (input);
163  if (mask) {
164  Adapter::Replicate<MaskVoxelType> replicated_mask (*mask, input);
165  loop.run (CorrelationFunctor (threshold, sum, mean_xy), input, replicated_mask);
166  }
167  else
168  loop.run (CorrelationFunctor (threshold, sum, mean_xy), input);
169 
170  mean_xy /= count;
171  double covariance = mean_xy - (sum / count) * input_image_mean;
172  double mask_stdev = sqrt ((sum - double (sum * sum) / count) / count);
173 
174  return -covariance / (input_image_stdev * mask_stdev);
175  }

178  InputVoxelType& input;
179  MaskVoxelType* mask;
180  size_t count;
181  double input_image_mean;
182  double input_image_stdev;
183  };

186  template <class InputVoxelType, class MaskVoxelType>
187  typename InputVoxelType::value_type estimate_optimal_threshold (InputVoxelType& input, MaskVoxelType* mask)
188  {
189  typedef typename InputVoxelType::value_type input_value_type;
190 
191  input_value_type min, max;
192  Image::min_max (input, min, max);
193 
194  input_value_type optimal_threshold = 0.0;
195  {
196  ImageCorrelationCostFunction<InputVoxelType, MaskVoxelType> cost_function (input, mask);
197  optimal_threshold = Math::golden_section_search (cost_function, "optimising threshold...",
198  min + 0.001*(max-min), (min+max)/2.0 , max-0.001*(max-min));
199  }
200 
201  return optimal_threshold;
202  }
