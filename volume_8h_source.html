131  void set_vertices_for_slice_render (const Projection& projection, float depth)
132  {
133  vertices[0] = projection.screen_to_model (projection.x_position(), projection.y_position()+projection.height(), depth);
134  vertices[2] = projection.screen_to_model (projection.x_position(), projection.y_position(), depth);
135  vertices[4] = projection.screen_to_model (projection.x_position()+projection.width(), projection.y_position(), depth);
136  vertices[6] = projection.screen_to_model (projection.x_position()+projection.width(), projection.y_position()+projection.height(), depth);
137 
138  const Eigen::Vector3f sizes (_header.size (0), _header.size (1), _header.size (2));
139  vertices[1] = div ((_transform.scanner2voxel.cast<float>() * vertices[0]) + Eigen::Vector3f { 0.5, 0.5, 0.5 }, sizes);
140  vertices[3] = div ((_transform.scanner2voxel.cast<float>() * vertices[2]) + Eigen::Vector3f { 0.5, 0.5, 0.5 }, sizes);
141  vertices[5] = div ((_transform.scanner2voxel.cast<float>() * vertices[4]) + Eigen::Vector3f { 0.5, 0.5, 0.5 }, sizes);
142  vertices[7] = div ((_transform.scanner2voxel.cast<float>() * vertices[6]) + Eigen::Vector3f { 0.5, 0.5, 0.5 }, sizes);
143  }
