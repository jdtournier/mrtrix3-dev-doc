38 template <class VectorType>
39 Regrid (const ImageType& original, const VectorType& from, const VectorType& size, const value_type fill = 0) :
40 base_type (original),
41 from_ (container_cast<decltype(from_)>(from)),
42 size_ (container_cast<decltype(size_)>(size)),
43 index_invalid_lower_upper ([&]{ vector<vector<ssize_t>> v; for (size_t d = 0; d < from_.size(); ++d) {
44 v.push_back (vector<ssize_t> {from_[d] < 0 ? -(ssize_t) from_[d] - 1 : -1, original.size(d) - from_[d]}); } return v; }()),
45 index_requires_bound_check ([&]{ vector<bool> v; for (size_t d = 0; d < from_.size(); ++d) {
46 v.push_back (from_[d] < 0 || size_[d] > original.size(d) - from_[d]); } return v; }()),
47 fill_ (fill),
48 transform_ (original.transform()),
49 index_ (ndim(), 0) {
50 assert(from_.size() == size_.size());
51 assert(from_.size() == ndim());
52
53 for (size_t n = 0; n < ndim(); ++n)
54 if (size_[n] < 0)
55 throw Exception("FIXME: negative size in Regrid adapter");
56
57 // adjust location of origin
58 for (size_t j = 0; j < 3; ++j)
59 for (size_t i = 0; i < 3; ++i)
60 transform_(i,3) += from[j] * spacing(j) * transform_(i,j);
61
62 }
63
64 void reset () {
65 for (size_t n = 0; n < ndim(); ++n)
66 set_pos (n, 0);
67 }
68
69 size_t ndim () const { return size_.size(); }
70 ssize_t size (size_t axis) const { return size_ [axis]; }
71 const transform_type& transform() const { return transform_; }
72
73 ssize_t get_index (size_t axis) const {
74 return index_requires_bound_check[axis] ? index_[axis] : parent().index(axis) - from_[axis];
75 }
76
77 void move_index (size_t axis, ssize_t increment) {
78 if (index_requires_bound_check[axis]) {
79 index_[axis] += increment;
80 if (increment > 0) {
81 if (index_[axis] < index_invalid_lower_upper[axis][1])
82 parent().index(axis) = index_[axis] + from_[axis];
83 } else
84 if (index_[axis] > index_invalid_lower_upper[axis][0])
85 parent().index(axis) = index_[axis] + from_[axis];
86 } else
87 parent().index(axis) += increment;
88 }
89
90 value_type value () {
91 for (size_t axis = 0; axis < index_.size(); ++axis)
92 if (index_requires_bound_check[axis] &&
93 (index_[axis] >= index_invalid_lower_upper[axis][1] || index_[axis] <= index_invalid_lower_upper[axis][0]))
94 return fill_;
95 return parent().value();
96 }
