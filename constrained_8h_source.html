119  void init ()
120  {
121  assert (response.size());
122  if (lmax_data > lmax) lmax_data = lmax;
123  INFO ("calculating even spherical harmonic components up to order " + str (lmax_data) + " for initialisation");
124 
125  if (init_filter.size() < size_t (lmax_data/2) +1)
126  throw Exception ("not enough initial filter coefficients supplied for lmax = " + str (lmax_data));
127 
128  Math::Vector<value_type> RH;
129  Math::SH::SH2RH (RH, response);
130  RH.resize (1+lmax/2,0);
131 
132  // inverse sdeconv for initialisation:
133  Math::Matrix<value_type> fconv;
134  Math::SH::init_transform (fconv, DW_dirs, lmax_data);
135  rconv.allocate (fconv.columns(), fconv.rows());
136  fconv.diagonal() += 1.0e-2;
137  //fconv.save ("fconv.txt");
138  Math::pinv (rconv, fconv);
139  //rconv.save ("rconv.txt");
140  size_t l = 0, nl = 1;
141  for (size_t row = 0; row < rconv.rows(); ++row) {
142  if (row >= nl) {
143  l++;
144  nl = Math::SH::NforL (2*l);
145  }
146  for (size_t col = 0; col < rconv.columns(); col++)
147  rconv (row, col) *= init_filter[l] / RH[l];
148  }
149 
150  // forward sconv for iteration, using all response function
151  // coefficients up to the requested lmax:
152  Math::SH::init_transform (fconv, DW_dirs, lmax);
153  l = 0;
154  nl = 1;
155  for (size_t col = 0; col < fconv.columns(); ++col) {
156  if (col >= nl) {
157  l++;
158  nl = Math::SH::NforL (2*l);
159  }
160  for (size_t row = 0; row < fconv.rows(); ++row)
161  fconv (row,col) *= RH[l];
162  }
163 
164  // high-res sampling to apply constraint:
165  Math::SH::init_transform (HR_trans, HR_dirs, lmax);
166  value_type constraint_multiplier = neg_lambda * 50.0 * response[0] / value_type (HR_trans.rows());
167  HR_trans *= constraint_multiplier;
168 
169  // adjust threshold accordingly:
170  threshold *= constraint_multiplier;
171 
172  // precompute as much as possible ahead of Cholesky decomp:
173  assert (fconv.columns() <= HR_trans.columns());
174  M.allocate (DW_dirs.rows(), HR_trans.columns());
175  M.sub (0,M.rows(),0,fconv.columns()) = fconv;
176  M.sub (0,M.rows(),fconv.columns(),M.columns()) = 0.0;
177  Mt_M.allocate (M.columns(), M.columns());
178  rankN_update (Mt_M, M, CblasTrans, CblasLower, value_type (1.0), value_type (0.0));
179 
180  INFO ("constrained spherical deconvolution initiated successfully");
181  }

236  bool iterate() {
237  neg.clear();
238  Math::mult (HR_amps, shared.HR_trans, F);
239  for (size_t n = 0; n < HR_amps.size(); n++)
240  if (HR_amps[n] < shared.threshold)
241  neg.push_back (n);
242 
243  if (computed_once && old_neg == neg)
244  if (old_neg == neg)
245  return true;
246 
247  for (size_t i = 0; i < work.rows(); i++)
248  for (size_t j = 0; j <= i; j++)
249  work (i,j) = shared.Mt_M (i,j);
250 
251  // min-norm constraint:
252  if (norm_lambda) {
253 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
254  work.diagonal() += norm_lambda;
255 #else
256  int l = 0;
257  for (size_t i = 0; i < work.rows(); ++i) {
258  if (Math::SH::index (l,0) == i) {
259  work(i,i) += norm_lambda;
260  l+=2;
261  }
262  else
263  work(i,i) += 0.5 * norm_lambda;
264  }
265 #endif
266  }
267 
268  if (neg.size()) {
269  HR_T.allocate (neg.size(), shared.HR_trans.columns());
270  for (size_t i = 0; i < neg.size(); i++)
271  HR_T.row (i) = shared.HR_trans.row (neg[i]);
272  rankN_update (work, HR_T, CblasTrans, CblasLower, value_type (1.0), value_type (1.0));
273  }
274 
275  F = Mt_b;
276 
277  Math::Cholesky::decomp (work);
278  Math::Cholesky::solve (F, work);
279  computed_once = true;
280  old_neg = neg;
281 
282  return false;
283  }
