109  void init ()
110  {
111  using namespace Math::SH;
112 
113  assert (response.size());
114  lmax_response = std::min (lmax_response, std::min (lmax_data, lmax));
115  INFO ("calculating even spherical harmonic components up to order " + str (lmax_response) + " for initialisation");
116 
117  if (!init_filter.size())
118  init_filter = Eigen::VectorXd::Ones(3);
119  init_filter.conservativeResize (size_t (lmax_response/2)+1);
120 
121  auto RH = SH2RH (response);
122  if (RH.size() < 1+lmax/2) {
123  RH.conservativeResize (1+lmax/2);
124  RH.tail (RH.size() - response.size()).setZero();
125  }
126 
127  // inverse sdeconv for initialisation:
128  auto fconv = init_transform (DW_dirs, lmax_response);
129  rconv.resize (fconv.cols(), fconv.rows());
130  fconv.diagonal().array() += 1.0e-2;
131  //fconv.save ("fconv.txt");
132  rconv = Math::pinv (fconv);
133  //rconv.save ("rconv.txt");
134  ssize_t l = 0, nl = 1;
135  for (ssize_t row = 0; row < rconv.rows(); ++row) {
136  if (row >= nl) {
137  l++;
138  nl = NforL (2*l);
139  }
140  rconv.row (row).array() *= init_filter[l] / RH[l];
141  }
142 
143  // forward sconv for iteration, using all response function
144  // coefficients up to the requested lmax:
145  INFO ("calculating even spherical harmonic components up to order " + str (lmax) + " for output");
146  fconv = init_transform (DW_dirs, lmax);
147  l = 0;
148  nl = 1;
149  for (ssize_t col = 0; col < fconv.cols(); ++col) {
150  if (col >= nl) {
151  l++;
152  nl = NforL (2*l);
153  }
154  fconv.col (col).array() *= RH[l];
155  }
156 
157  // high-res sampling to apply constraint:
158  HR_trans = init_transform (HR_dirs, lmax);
159  default_type constraint_multiplier = neg_lambda * 50.0 * response[0] / default_type (HR_trans.rows());
160  HR_trans.array() *= constraint_multiplier;
161 
162  // adjust threshold accordingly:
163  threshold *= constraint_multiplier;
164 
165  // precompute as much as possible ahead of Cholesky decomp:
166  assert (fconv.cols() <= HR_trans.cols());
167  M.resize (DW_dirs.rows(), HR_trans.cols());
168  M.leftCols (fconv.cols()) = fconv;
169  M.rightCols (M.cols() - fconv.cols()).setZero();
170  Mt_M.resize (M.cols(), M.cols());
171  Mt_M.triangularView<Eigen::Lower>() = M.transpose() * M;
172 
173 
174  // min-norm constraint:
175  if (norm_lambda) {
176  norm_lambda *= NORM_LAMBDA_MULTIPLIER * Mt_M (0,0);
177 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
178  Mt_M.diagonal().array() += norm_lambda;
179 #else
180  int l = 0;
181  for (size_t i = 0; i < Mt_M.rows(); ++i) {
182  if (Math::SH::index (l,0) == i) {
183  Mt_M(i,i) += norm_lambda;
184  l+=2;
185  }
186  else
187  Mt_M(i,i) += 0.5 * norm_lambda;
188  }
189 #endif
190  }
191 
192  INFO ("constrained spherical deconvolution initialised successfully");
193  }

255  bool iterate() {
256  neg.clear();
257  HR_amps = shared.HR_trans * F;
258  for (ssize_t n = 0; n < HR_amps.size(); n++)
259  if (HR_amps[n] < shared.threshold)
260  neg.push_back (n);
261 
262  if (computed_once && old_neg == neg)
263  if (old_neg == neg)
264  return true;
265 
266  work.triangularView<Eigen::Lower>() = shared.Mt_M.triangularView<Eigen::Lower>();
267 
268  if (neg.size()) {
269  for (size_t i = 0; i < neg.size(); i++)
270  HR_T.row (i) = shared.HR_trans.row (neg[i]);
271  auto HR_T_view = HR_T.topRows (neg.size());
272  work.triangularView<Eigen::Lower>() += HR_T_view.transpose() * HR_T_view;
273  }
274 
275  F.noalias() = llt.compute (work.triangularView<Eigen::Lower>()).solve (Mt_b);
276 
277  computed_once = true;
278  old_neg = neg;
279 
280  return false;
281  }
