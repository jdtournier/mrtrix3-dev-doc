230 
231  template <class... ImageType>
232  struct Run { NOMEMALIGN
233  const size_t from, to;
234  const std::tuple<ImageType&...> vox;
235  const ssize_t size0;
236  bool ok;
237  FORCE_INLINE Run (const size_t axis_from, const size_t axis_to, const std::tuple<ImageType&...>& vox) :
238  from (axis_from), to (axis_to ? axis_to : std::get<0>(vox).ndim()), vox (vox), size0 (std::get<0>(vox).size(from)), ok (true) {
239  for (size_t n = from; n < to; ++n)
240  apply (set_pos (n, 0), vox);
241  }
242  FORCE_INLINE operator bool() const { return ok; }
243  FORCE_INLINE void operator++() {
244  apply (inc_pos (from), vox);
245  if (std::get<0>(vox).index(from) < size0)
246  return;
247 
248  apply (set_pos (from, 0), vox);
249  size_t axis = from+1;
250  while (axis < to) {
251  apply (inc_pos (axis), vox);
252  if (std::get<0>(vox).index(axis) < std::get<0>(vox).size(axis))
253  return;
254  apply (set_pos (axis, 0), vox);
255  ++axis;
256  }
257  ok = false;
258  }
259  FORCE_INLINE void operator++(int) { operator++(); }
260  };
261 
262  template <class... ImageType>
263  FORCE_INLINE Run<ImageType...> operator() (ImageType&... images) const { return { from, to, std::tie (images...) }; }
