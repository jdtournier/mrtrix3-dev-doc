29  transform_type linear;
30  const auto it = input_warps.keyval().find (name);
31  if (it != input_warps.keyval().end()) {
32  const auto lines = split_lines (it->second);
33  if (lines.size() != 3)
34  throw Exception ("linear transform in initialisation syn warps image header does not contain 3 rows");
35  for (size_t row = 0; row < 3; ++row) {
36  const auto values = split (lines[row], " ", true);
37  if (values.size() != 4)
38  throw Exception ("linear transform in initialisation syn warps image header does not contain 4 columns");
39  for (size_t col = 0; col < 4; ++col)
40  linear (row, col) = std::stod (values[col]);
41  }
42  } else {
43  throw Exception ("no linear transform found in initialisation syn warps image header");
44  }
