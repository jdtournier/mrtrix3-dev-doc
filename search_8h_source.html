131  void run ( bool debug = false ) {
132 
133  std::string what = global_search? "global" : "local";
134  size_t iterations = global_search? global_search_iterations : (rot_angles.size() * local_search_directions);
135  ProgressBar progress ("performing " + what + " search for best rotation", iterations);
136  overlap_it.resize (iterations);
137  cost_it.resize (iterations);
138  trafo_it.reserve (iterations);
139 
140  if (!global_search) {
141  gen_uniform_rotation_axes (local_search_directions, 180.0); // full sphere
142  az_el_to_cartesian();
143  }
144 
145  size_t iteration (0);
146  ssize_t cnt (0);
147  Eigen::Matrix<default_type, Eigen::Dynamic, 1> gradient (local_trafo.size());
148  Eigen::VectorXd cost = Eigen::VectorXd::Zero(1,1);
149  transform_type T;
150  {
151  ParamType parameters = get_parameters ();
152  // parameters.make_diagnostics_image ("/tmp/debugme"+str(iteration)+".mif", true); // REMOVEME
153  Metric::ThreadKernel<MetricType, ParamType> kernel (metric, parameters, cost, gradient, &cnt);
154  ThreadedLoop (parameters.midway_image, 0, 3).run (kernel);
155  assert (cnt > 0);
156  overlap_it[0] = cnt;
157  cost_it[0] = cost(0) / static_cast<default_type>(cnt);
158  T = parameters.transformation.get_transform();
159  trafo_it.push_back (T);
160  }
161 
162  transform_type Tc2, To, R0;
163  Tc2.setIdentity();
164  To.setIdentity();
165  R0.setIdentity();
166  To.translation() = offset;
167  Tc2.translation() = centre - 0.5 * offset;
168 
169 
170  while ( ++iteration < iterations ) {
171  ++progress;
172  if (global_search)
173  gen_random_quaternion ();
174  else
175  gen_local_quaternion ();
176 
177  R0.linear() = quat.normalized().toRotationMatrix();
178  transform_type T = Tc2 * To * R0 * Tc2.inverse();
179 
180  local_trafo.set_transform<decltype(T)>(T);
181  ParamType parameters = get_parameters ();
182  // parameters.make_diagnostics_image ("/tmp/debugme"+str(iteration)+".mif", true); // REMOVEME
183  cost.fill(0);
184  cnt = 0;
185  Metric::ThreadKernel<MetricType, ParamType> kernel (metric, parameters, cost, gradient, &cnt);
186  ThreadedLoop (parameters.midway_image, 0, 3).run (kernel);
187  DEBUG ("rotation search: iteration " + str(iteration) + " cost: " + str(cost) + " cnt: " + str(cnt));
188  // write_images ( "im1_" + str(iteration) + ".mif", "im2_" + str(iteration) + ".mif");
189  if (cnt == 0)
190  WARN ("rotation search: overlap count is zero");
191  overlap_it[iteration] = cnt;
192  cost_it[iteration] = cost(0) / static_cast<default_type>(cnt);
193  trafo_it.push_back (T);
194  }
195  // if (debug) {
196  // save_matrix(cost_it, "/tmp/cost_before.txt");
197  // save_matrix(overlap_it, "/tmp/overlap.txt");
198  // }
199  // best trafo := lowest cost per voxel with at least mean overlap
200  {
201  auto max_ = Eigen::MatrixXd::Constant(cost_it.rows(), 1, std::numeric_limits<default_type>::max());
202  default_type mean_overlap = static_cast<default_type>(overlap_it.sum()) / static_cast<default_type>(iterations);
203  // reject solutions with less than mean overlap by setting cost to max
204  cost_it = (overlap_it.array() > mean_overlap).select(cost_it, max_);
205  std::ptrdiff_t i;
206  min_cost = cost_it.minCoeff(&i);
207  T = trafo_it[i];
208  best_trafo = T;
209  }
210  // if (debug) {
211  // save_matrix(cost_it, "/tmp/cost_after.txt");
212  // Eigen::VectorXd t(2);
213  // t(0) = cost_it(0);
214  // t(1) = min_cost;
215  // save_matrix(t, "/tmp/cost_mass_chosen.txt");
216  // save_matrix(centre, "/tmp/centre.txt");
217  // parameters.transformation.set_transform (best_trafo);
218  // write_images ( "/tmp/im1_best.mif", "/tmp/im2_best.mif");
219  // }
220  input_trafo.set_transform<transform_type> (best_trafo);
221

250  // gen_random_quaternion generates random element of SO(3)
251  FORCE_INLINE void gen_random_quaternion () {
252  // Eigen 3.3.0: quat = Eigen::Quaternion<default_type,Eigen::autoalign>::UnitRandom ();
253  // http://planning.cs.uiuc.edu/node198.html
254  const default_type u1 = rnd ();
255  const default_type u2 = rnd () * 2.0 * Math::pi;
256  const default_type u3 = rnd () * 2.0 * Math::pi;
257  assert (u1 < 1.0 && u1 >= 0.0);
258  assert (u2 < 2.0 * Math::pi && u2 >= 0.0);
259  assert (u3 < 2.0 * Math::pi && u3 >= 0.0);
260  const default_type a = std::sqrt(1.0 - u1);
261  const default_type b = std::sqrt(u1);
262  quat = Eigen::Quaternion<default_type> (a * std::sin(u2), a * std::cos(u2), b * std::sin(u3), b * std::cos(u3));
263  }

265  // gen_uniform_rotation_axes generates roughly uniformly distributed points on sphere
266  // starting on z-axis up to -z-axis (max_cone_angle_deg=180). points are stored as matrix
267  // of azimuth and elevation
268  FORCE_INLINE void gen_uniform_rotation_axes ( const size_t& n_dir, const default_type& max_cone_angle_deg ) {
269  assert (n_dir > 1);
270  assert (max_cone_angle_deg > 0.0);
271  assert (max_cone_angle_deg <= 180.0);
272 
273  const default_type golden_ratio ((1.0 + std::sqrt (5.0)) / 2.0);
274  const default_type golden_angle (2.0 * Math::pi * (1.0 - 1.0 / golden_ratio));
275 
276  az_el.resize (n_dir,2);
277  Eigen::Matrix<default_type, Eigen::Dynamic, 1> idx (n_dir);
278  for (size_t i = 0; i < n_dir; ++i)
279  idx(i) = i;
280  az_el.col(0) = idx * golden_angle;
281 
282  // el(i) = acos (1-(1-cosd(max_cone_angle_deg))*i/(n_dir-1) )
283  default_type a = (1.0 - std::cos(Math::pi / 180.0 * default_type (max_cone_angle_deg))) / (default_type (n_dir - 1));
284  az_el.col(1).array() = - a * idx.array() + 1.0;
285  for (size_t i = 0; i < n_dir; ++i)
286  az_el(i, 1) = std::acos (az_el(i, 1));
287  }

289  // convert spherical coordinates (az_el) to cartesian coordinates (xyz)
290  FORCE_INLINE void az_el_to_cartesian () {
291  xyz.resize (az_el.rows(), 3);
292  Eigen::VectorXd el_sin = az_el.col(1).array().sin();
293  xyz.col(0).array() = el_sin.array() * az_el.col(0).array().cos();
294  xyz.col(1).array() = el_sin.array() * az_el.col(0).array().sin();
295  xyz.col(2).array() = az_el.col(1).array().cos();
296  }

298  FORCE_INLINE void gen_local_quaternion () {
299  // Eigen::Vector3d normal ( -std::cos (az_el(idx,1)), -std::sin (az_el(idx,0)) * std::cos (az_el(idx,0)), 0.0);
300  // Eigen::AngleAxis<default_type> aa (az_el(idx,1), normal);
301  // Eigen::Vector3d normal ( xyz.row(idx) );
302  // Eigen::AngleAxis<default_type> aa (angle, normal);
303  if (idx_dir == local_search_directions) {
304  idx_dir = 0;
305  ++idx_angle;
306  assert (idx_angle < rot_angles.size());
307  }
308  quat = Eigen::Quaternion<default_type> ( Eigen::AngleAxis<default_type> (rot_angles[idx_angle], xyz.row(idx_dir)) );
309  ++idx_dir;
310  }
