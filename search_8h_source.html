133  void run ( bool debug = false ) {
134 
135  std::string what = global_search? "global" : "local";
136  size_t iterations = global_search? global_search_iterations : (rot_angles.size() * local_search_directions);
137  ProgressBar progress ("performing " + what + " search for best rotation", iterations);
138  overlap_it.resize (iterations);
139  cost_it.resize (iterations);
140  trafo_it.reserve (iterations);
141 
142  if (!global_search) {
143  gen_uniform_rotation_axes (local_search_directions, 180.0); // full sphere
144  az_el_to_cartesian();
145  }
146 
147  size_t iteration (0);
148  ssize_t cnt (0);
149  Eigen::Matrix<default_type, Eigen::Dynamic, 1> gradient (local_trafo.size());
150  Eigen::VectorXd cost = Eigen::VectorXd::Zero(1,1);
151  transform_type T;
152  {
153  ParamType parameters = get_parameters ();
154  Metric::ThreadKernel<MetricType, ParamType> kernel (metric, parameters, cost, gradient, &cnt);
155  ThreadedLoop (parameters.midway_image, 0, 3).run (kernel);
156  assert (cnt > 0);
157  overlap_it[0] = cnt;
158  cost_it[0] = cost(0) / static_cast<default_type>(cnt);
159  T = parameters.transformation.get_transform();
160  trafo_it.push_back (T);
161  }
162 
163  transform_type Tc2, To, R0;
164  Tc2.setIdentity();
165  To.setIdentity();
166  R0.setIdentity();
167  To.translation() = offset;
168  Tc2.translation() = centre - 0.5 * offset;
169 
170 
171  while ( ++iteration < iterations ) {
172  ++progress;
173  if (global_search) {
174  gen_random_quaternion ();
175  }
176  else {
177  gen_local_quaternion ();
178  }
179  R0.linear() = quat.matrix();
180  transform_type T = Tc2 * To * R0 * Tc2.inverse();
181  local_trafo.set_transform<decltype(T)>(T);
182  ParamType parameters = get_parameters ();
183  // parameters.transformation.set_transform<decltype(T)>(T);
184  cost.fill(0);
185  cnt = 0;
186  Metric::ThreadKernel<MetricType, ParamType> kernel (metric, parameters, cost, gradient, &cnt);
187  ThreadedLoop (parameters.midway_image, 0, 3).run (kernel);
188  DEBUG ("rotation search: iteration " + str(iteration) + " cost: " + str(cost) + " cnt: " + str(cnt));
189  // write_images ( "im1_" + str(iteration) + ".mif", "im2_" + str(iteration) + ".mif");
190  overlap_it[iteration] = cnt;
191  cost_it[iteration] = cost(0) / static_cast<default_type>(cnt);
192  trafo_it.push_back (T);
193  }
194  // if (debug) {
195  // save_matrix(cost_it, "/tmp/cost_before.txt");
196  // save_matrix(overlap_it, "/tmp/overlap.txt");
197  // }
198  // best trafo := lowest cost per voxel with at least mean overlap
199  {
200  auto max_ = Eigen::MatrixXd::Constant(cost_it.rows(), 1, std::numeric_limits<default_type>::max());
201  default_type mean_overlap = static_cast<default_type>(overlap_it.sum()) / static_cast<default_type>(iterations);
202  // reject solutions with less than mean overlap by setting cost to max
203  cost_it = (overlap_it.array() > mean_overlap).select(cost_it, max_);
204  std::ptrdiff_t i;
205  min_cost = cost_it.minCoeff(&i);
206  T = trafo_it[i];
207  best_trafo = T;
208  }
209  // if (debug) {
210  // save_matrix(cost_it, "/tmp/cost_after.txt");
211  // Eigen::VectorXd t(2);
212  // t(0) = cost_it(0);
213  // t(1) = min_cost;
214  // save_matrix(t, "/tmp/cost_mass_chosen.txt");
215  // save_matrix(centre, "/tmp/centre.txt");
216  // parameters.transformation.set_transform (best_trafo);
217  // write_images ( "/tmp/im1_best.mif", "/tmp/im2_best.mif");
218  // }
219  input_trafo.set_transform<transform_type> (best_trafo);
220 
221  };

249  // gen_random_quaternion generates random quaternion (rotation around random direction
250  // by random angle)
251  inline void gen_random_quaternion () {
252  Eigen::Matrix<default_type, 4, 1> v(rndn(), rndn(), rndn(), rndn());
253  v.array() /= v.norm();
254  quat = Eigen::Quaternion<default_type> (v);
255  };

257  // gen_uniform_rotation_axes generates roughly uniformly distributed points on sphere
258  // starting on z-axis up to -z-axis (max_cone_angle_deg=180). points are stored as matrix
259  // of azimuth and elevation
260  void gen_uniform_rotation_axes ( const size_t& n_dir, const default_type& max_cone_angle_deg ) {
261  assert (n_dir > 1);
262  assert (max_cone_angle_deg > 0.0);
263  assert (max_cone_angle_deg <= 180.0);
264 
265  const default_type golden_ratio ((1.0 + std::sqrt (5.0)) / 2.0);
266  const default_type golden_angle (2.0 * Math::pi * (1.0 - 1.0 / golden_ratio));
267 
268  az_el.resize (n_dir,2);
269  Eigen::Matrix<default_type, Eigen::Dynamic, 1> idx (n_dir);
270  for (size_t i = 0; i < n_dir; ++i)
271  idx(i) = i;
272  az_el.col(0) = idx * golden_angle;
273 
274  // el(i) = acos (1-(1-cosd(max_cone_angle_deg))*i/(n_dir-1) )
275  default_type a = (1.0 - std::cos(Math::pi / 180.0 * default_type (max_cone_angle_deg))) / (default_type (n_dir - 1));
276  az_el.col(1).array() = - a * idx.array() + 1.0;
277  for (size_t i = 0; i < n_dir; ++i)
278  az_el(i, 1) = std::acos (az_el(i, 1));
279  }

281  // convert spherical coordinates (az_el) to cartesian coordinates (xyz)
282  inline void az_el_to_cartesian () {
283  xyz.resize (az_el.rows(), 3);
284  Eigen::VectorXd el_sin = az_el.col(1).array().sin();
285  xyz.col(0).array() = el_sin.array() * az_el.col(0).array().cos();
286  xyz.col(1).array() = el_sin.array() * az_el.col(0).array().sin();
287  xyz.col(2).array() = az_el.col(1).array().cos();
288  }

290  inline void gen_local_quaternion () {
291  // Eigen::Vector3d normal ( -std::cos (az_el(idx,1)), -std::sin (az_el(idx,0)) * std::cos (az_el(idx,0)), 0.0);
292  // Eigen::AngleAxis<default_type> aa (az_el(idx,1), normal);
293  // Eigen::Vector3d normal ( xyz.row(idx) );
294  // Eigen::AngleAxis<default_type> aa (angle, normal);
295  if (idx_dir == local_search_directions) {
296  idx_dir = 0;
297  ++idx_angle;
298  assert (idx_angle < rot_angles.size());
299  }
300  quat = Eigen::Quaternion<default_type> ( Eigen::AngleAxis<default_type> (rot_angles[idx_angle], xyz.row(idx_dir)) );
301  ++idx_dir;
302  }
