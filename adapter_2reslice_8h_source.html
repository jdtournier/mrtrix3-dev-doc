36 // Partial specialisation for boolean value_type in order to avoid compiler
37 // warning regarding use of multiplication when assigning to a boolean
38 template <typename value_type>
39 typename std::enable_if<std::is_same<value_type, bool>::value, value_type>::type
40 inline normalise (const default_type sum, const default_type norm)
41 {
42 return ((sum*norm) >= 0.5) ? true : false;
43 }
44
45 // Partial specialisation to invoke round-to-nearest when taking an average of integers
46 template <typename value_type>
47 typename std::enable_if<!std::is_same<value_type, bool>::value && std::is_integral<value_type>::value, value_type>::type
48 inline normalise (const default_type sum, const default_type norm)
49 {
50 return value_type(std::round (sum*norm));
51 }

53 template <typename value_type>
54 typename std::enable_if<std::is_floating_point<value_type>::value, value_type>::type
55 inline normalise (const default_type sum, const default_type norm)
56 {

117 template <class HeaderType>
118 Reslice (const ImageType& original,
119 const HeaderType& reference,
120 const transform_type& transform = NoTransform,
121 const vector<int>& oversample = AutoOverSample,
122 const value_type value_when_out_of_bounds = Interp::Base<ImageType>::default_out_of_bounds_value()) :
123 interp (original, value_when_out_of_bounds),
124 x { 0, 0, 0 },
125 dim { reference.size(0), reference.size(1), reference.size(2) },
126 vox { reference.spacing(0), reference.spacing(1), reference.spacing(2) },
127 transform_ (reference.transform()),
128 direct_transform (Transform(original).scanner2voxel * transform * Transform(reference).voxel2scanner) {
129 using namespace Eigen;
130 assert (ndim() >= 3);
131
132 if (oversample.size()) {
133 assert (oversample.size() == 3);
134 if (oversample[0] < 1 || oversample[1] < 1 || oversample[2] < 1)
135 throw Exception ("oversample factors must be greater than zero");
136 OS[0] = oversample[0];
137 OS[1] = oversample[1];
138 OS[2] = oversample[2];
139 }
140 else {
141 Vector3 y = direct_transform * Vector3 (0.0, 0.0, 0.0);
142 Vector3 x0 = direct_transform * Vector3 (1.0, 0.0, 0.0);
143 OS[0] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
144 x0 = direct_transform * Vector3 (0.0, 1.0, 0.0);
145 OS[1] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
146 x0 = direct_transform * Vector3 (0.0, 0.0, 1.0);
147 OS[2] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
148 }
149
150 if (OS[0] * OS[1] * OS[2] > 1) {
151 INFO ("using oversampling factors [ " + str (OS[0]) + " " + str (OS[1]) + " " + str (OS[2]) + " ]");
152 oversampling = true;
153 norm = 1.0;
154 for (size_t i = 0; i < 3; ++i) {
155 inc[i] = 1.0/default_type (OS[i]);
156 from[i] = 0.5* (inc[i]-1.0);
157 norm *= OS[i];
158 }
159 norm = 1.0 / norm;
160 }
161 else oversampling = false;
162 }

176 void reset () {
177 x[0] = x[1] = x[2] = 0;
178 for (size_t n = 3; n < interp.ndim(); ++n)
179 interp.index(n) = 0;
180 }
181
182 value_type value () {
183 using namespace Eigen;
184 if (oversampling) {
185 Vector3 d (x[0]+from[0], x[1]+from[1], x[2]+from[2]);
186 default_type sum (0.0);
187 Vector3 s;
188 for (int z = 0; z < OS[2]; ++z) {
189 s[2] = d[2] + z*inc[2];
190 for (int y = 0; y < OS[1]; ++y) {
191 s[1] = d[1] + y*inc[1];
192 for (int x = 0; x < OS[0]; ++x) {
193 s[0] = d[0] + x*inc[0];
194 if (interp.voxel (direct_transform * s))
195 sum += interp.value();
196 }
197 }
198 }
199 return normalise<value_type> (sum, norm);
200 }
201 interp.voxel (direct_transform * Vector3 (x[0], x[1], x[2]));
202 return interp.value();
203 }
204
205 ssize_t get_index (size_t axis) const { return axis < 3 ? x[axis] : interp.index(axis); }
206 void move_index (size_t axis, ssize_t increment) {
207 if (axis < 3) x[axis] += increment;
208 else interp.index(axis) += increment;
209 }
