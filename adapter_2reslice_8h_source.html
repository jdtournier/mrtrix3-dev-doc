90  direct_transform (Transform(original).scanner2voxel * transform * Transform(reference).voxel2scanner) {
91  using namespace Eigen;
92  assert (ndim() >= 3);
93 
94  if (oversample.size()) {
95  assert (oversample.size() == 3);
96  if (oversample[0] < 1 || oversample[1] < 1 || oversample[2] < 1)
97  throw Exception ("oversample factors must be greater than zero");
98  OS[0] = oversample[0];
99  OS[1] = oversample[1];
100  OS[2] = oversample[2];
101  }
102  else {
103  Vector3 y = direct_transform * Vector3 (0.0, 0.0, 0.0);
104  Vector3 x0 = direct_transform * Vector3 (1.0, 0.0, 0.0);
105  OS[0] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
106  x0 = direct_transform * Vector3 (0.0, 1.0, 0.0);
107  OS[1] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
108  x0 = direct_transform * Vector3 (0.0, 0.0, 1.0);
109  OS[2] = std::ceil ((1.0-std::numeric_limits<default_type>::epsilon()) * (y-x0).norm());
110  }
111 
112  if (OS[0] * OS[1] * OS[2] > 1) {
113  INFO ("using oversampling factors [ " + str (OS[0]) + " " + str (OS[1]) + " " + str (OS[2]) + " ]");
114  oversampling = true;
115  norm = 1.0;
116  for (size_t i = 0; i < 3; ++i) {
117  inc[i] = 1.0/default_type (OS[i]);
118  from[i] = 0.5* (inc[i]-1.0);
119  norm *= OS[i];
120  }
121  norm = 1.0 / norm;
122  }
123  else oversampling = false;
124  }

144  value_type value () {
145  using namespace Eigen;
146  if (oversampling) {
147  Vector3 d (x[0]+from[0], x[1]+from[1], x[2]+from[2]);
148  value_type result = 0.0;
149  Vector3 s;
150  for (int z = 0; z < OS[2]; ++z) {
151  s[2] = d[2] + z*inc[2];
152  for (int y = 0; y < OS[1]; ++y) {
153  s[1] = d[1] + y*inc[1];
154  for (int x = 0; x < OS[0]; ++x) {
155  s[0] = d[0] + x*inc[0];
156  if (interp.voxel (direct_transform * s))
157  result += interp.value();
158  }
159  }
160  }
161  result *= norm;

163  }
