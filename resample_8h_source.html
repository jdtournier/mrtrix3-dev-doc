85 template <typename T>
86 bool Upsampler<T>::operator() (std::vector< Point<T> >& in) const
87 {
88  if (!interp_prepare (in))
89  return false;
90  std::vector< Point<T> > out;
91  for (size_t i = 3; i < in.size(); ++i) {
92  out.push_back (in[i-2]);
93  increment (in[i]);
94  Math::mult (temp, M, data);
95  for (size_t row = 0; row != temp.rows(); ++row)
96  out.push_back (Point<T> (temp.row (row)));
97  }
98  out.push_back (in[in.size() - 2]);
99  out.swap (in);
100  return true;
101 }

105 template <typename T>
106 void Upsampler<T>::set_ratio (const size_t upsample_ratio)
107 {
108  if (upsample_ratio > 1) {
109  const size_t dim = upsample_ratio - 1;
110  // Hermite tension = 0.1;
111  // cubic interpolation (tension = 0.0) looks 'bulgy' between control points
112  Math::Hermite<T> interp (0.1);
113  M.allocate (dim, 4);
114  for (size_t i = 0; i != dim; ++i) {
115  interp.set ((i+1.0) / float(upsample_ratio));
116  for (size_t j = 0; j != 4; ++j)
117  M(i,j) = interp.coef(j);
118  }
119  temp.allocate (dim, 3);
120  } else {
121  M.clear();
122  temp.clear();
123  }
124 }

128 template <typename T>
129 bool Upsampler<T>::interp_prepare (std::vector< Point<T> >& in) const
130 {
131  if (!M.is_set() || in.size() < 2)
132  return false;
133  const size_t s = in.size();
134  // Abandoned curvature-based extrapolation - badly posed when step size is not guaranteed to be consistent,
135  // and probably makes little difference anyways
136  in.insert (in.begin(), in[ 0 ] + (in[1] - in[0]));
137  in.push_back ( in[ s ] + (in[s] - in[s-1]));
138  for (size_t i = 0; i != 3; ++i) {
139  data(0,i) = 0.0;
140  data(1,i) = (in[0])[i];
141  data(2,i) = (in[1])[i];
142  data(3,i) = (in[2])[i];
143  }
144  return true;
145 }

193 template <typename T>
194 bool Downsampler::operator() (std::vector< Point<T> >& tck) const
195 {
196  if (ratio <= 1 || tck.empty())
197  return false;
198  const size_t midpoint = tck.size()/2;
199  size_t index_old = (((midpoint - 1) % ratio) + 1);
200  size_t index_new = 1;
201  while (index_old < tck.size() - 1) {
202  tck[index_new++] = tck[index_old];
203  index_old += ratio;
204  }
205  tck[index_new] = tck.back();
206  tck.resize (index_new + 1);
207  return true;
208 }
