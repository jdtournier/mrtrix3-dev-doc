42 class Voxel : public Point<int>
43 {
44  public:
45  Voxel (const int x, const int y, const int z) { p[0] = x; p[1] = y; p[2] = z; }
46  Voxel (const Point<int>& that) : Point<int> (that) { }
47  Voxel () { memset (p, 0x00, 3 * sizeof(int)); }
48  bool operator< (const Voxel& V) const { return ((p[2] == V.p[2]) ? ((p[1] == V.p[1]) ? (p[0] < V.p[0]) : (p[1] < V.p[1])) : (p[2] < V.p[2])); }
49 };

59 {
60  return (V[0] >= 0 && V[0] < H.dim(0) && V[1] >= 0 && V[1] < H.dim(1) && V[2] >= 0 && V[2] < H.dim(2));
