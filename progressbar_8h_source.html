248  template <class TextFunc>
249  FORCE_INLINE void ProgressBar::update (TextFunc&& text_func, const bool increment)
250  {
251  if (!show)
252  return;
253  double time = timer.elapsed();
254  if (increment && _multiplier) {
255  if (++current_val >= next_percent) {
256  set_text (text_func());
257  _ellipsis.clear();
258  _value = std::round (current_val / _multiplier);
259  next_percent = std::ceil ((_value+1) * _multiplier);
260  next_time = time;
261  display_now();
262  return;
263  }
264  }
265  if (time >= next_time) {
266  set_text (text_func());
267  _ellipsis.clear();
268  if (_multiplier)
269  next_time = time + BUSY_INTERVAL;
270  else {
271  _value = time / BUSY_INTERVAL;
272  do { next_time += BUSY_INTERVAL; }
273  while (next_time <= time);
274  }
275  display_now();
276  }
277  }

309  inline void ProgressBar::run_update_thread (const ThreadType& threads) const
310  {
311  if (!show)
312  return;
313  std::unique_lock<std::mutex> lock (mutex);
314  while (!threads.finished()) {
315  notifier.wait_for (lock, std::chrono::milliseconds(1), []{ return notification_is_genuine; });
316  if (notification_is_genuine) {
317  previous_display_func (*this);
318  notification_is_genuine = false;
