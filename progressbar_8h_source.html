113  template <class TextFunc>
114  FORCE_INLINE void update (TextFunc&& text_func, const bool increment = true) {
115  double time = timer.elapsed();
116  if (increment && multiplier) {
117  if (++current_val >= next_percent) {
118  set_text (text_func());
119  ellipsis.clear();
120  value = std::round (current_val / multiplier);
121  next_percent = std::ceil ((value+1) * multiplier);
122  next_time = time;
123  display_now();
124  return;
125  }
126  }
127  if (time >= next_time) {
128  set_text (text_func());
129  ellipsis.clear();
130  if (multiplier)
131  next_time = time + BUSY_INTERVAL;
132  else {
133  value = time / BUSY_INTERVAL;
134  do { next_time += BUSY_INTERVAL; }
135  while (next_time <= time);
136  }
137  display_now();
138  }
139  }
