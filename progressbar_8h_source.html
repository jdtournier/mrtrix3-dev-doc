115  inline void update (TextFunc&& text_func, const bool increment = true) {
116  double time = timer.elapsed();
117  if (increment && multiplier) {
118  if (++current_val >= next_percent) {
119  set_text (text_func());
120  value = std::round (current_val / multiplier);
121  next_percent = std::ceil ((value+1) * multiplier);
122  next_time = time;
123  display_now();
124  return;
125  }
126  }
127  if (time >= next_time) {
128  set_text (text_func());
129  if (multiplier)
130  next_time = time + BUSY_INTERVAL;
131  else {
132  value = time / BUSY_INTERVAL;
133  do { next_time += BUSY_INTERVAL; }
134  while (next_time <= time);
135  }
136  display_now();
137  }
138  }
