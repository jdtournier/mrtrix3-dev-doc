90  void set_max (size_t target, bool display_immediately = true) {
91  if (target) {
92  multiplier = 0.01 * target;

121  template <class TextFunc>
122  FORCE_INLINE void update (TextFunc&& text_func, const bool increment = true) {
123  double time = timer.elapsed();
124  if (increment && multiplier) {
125  if (++current_val >= next_percent) {
126  set_text (text_func());
127  ellipsis.clear();
128  value = std::round (current_val / multiplier);
129  next_percent = std::ceil ((value+1) * multiplier);
130  next_time = time;
131  display_now();
132  return;
133  }
134  }
135  if (time >= next_time) {
136  set_text (text_func());
137  ellipsis.clear();
138  if (multiplier)
139  next_time = time + BUSY_INTERVAL;
140  else {
141  value = time / BUSY_INTERVAL;
142  do { next_time += BUSY_INTERVAL; }
143  while (next_time <= time);
144  }
145  display_now();
146  }
147  }
