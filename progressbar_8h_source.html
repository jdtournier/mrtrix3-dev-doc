122  template <class TextFunc>
123  FORCE_INLINE void update (TextFunc&& text_func, const bool increment = true) {
124  double time = timer.elapsed();
125  if (increment && multiplier) {
126  if (++current_val >= next_percent) {
127  set_text (text_func());
128  ellipsis.clear();
129  value = std::round (current_val / multiplier);
130  next_percent = std::ceil ((value+1) * multiplier);
131  next_time = time;
132  display_now();
133  return;
134  }
135  }
136  if (time >= next_time) {
137  set_text (text_func());
138  ellipsis.clear();
139  if (multiplier)
140  next_time = time + BUSY_INTERVAL;
141  else {
142  value = time / BUSY_INTERVAL;
143  do { next_time += BUSY_INTERVAL; }
144  while (next_time <= time);
145  }
146  display_now();
147  }
148  }
