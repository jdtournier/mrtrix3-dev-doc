72 template <class InputImageType, class MaskType, class OutputImageType>
73 void operator() (InputImageType& input, MaskType& spatial_prior, OutputImageType& output)
74 {
75 if (output.ndim() > 3)
76 throw Exception ("3D output expected");
77
78 std::unique_ptr<ProgressBar> progress (message.size() ? new ProgressBar (message) : nullptr);
79
80 Image<bool> int_roi = Image<bool>::scratch (Header(spatial_prior), "temporary initial mask");
81 INFO ("creating intensity mask from input mask");
82 Dilate dilation_filter (spatial_prior);
83 dilation_filter.set_npass(2);
84 dilation_filter (spatial_prior, int_roi);
85 ssize_t cnt = 0;
86 for (auto l = Loop (0,3) (int_roi); l; ++l) {
87 cnt += int_roi.value();
88 }
89 ssize_t cnt_lower = std::max<size_t>(10000, std::floor(fov_min * input.size(0) * input.size(1) * input.size(2)));
90 ssize_t cnt_upper = std::floor(fov_max * input.size(0) * input.size(1) * input.size(2));
91 float mad, median, previous_mad, previous_median;
92 calculate_median_mad<Image<float>, Image<bool>> (input, int_roi, cnt, median, mad);
93 INFO ("median: "+str(median));
94 INFO ("mad: "+str(mad));
95 INFO ("lower: " + str(median - zlower * mad) + " upper: " + str(median + zupper * mad));
96
97 INFO ("eroding intensity mask");
98 while (cnt >= cnt_lower) {
99 if (progress)
100 ++(*progress);
101 Erode erosion_filter (int_roi);
102 erosion_filter.set_npass(1);
103 erosion_filter (int_roi, int_roi);
104 cnt = 0;
105 for (auto l = Loop (0,3) (int_roi); l; ++l)
106 cnt += int_roi.value();
107 if (cnt == 0)
108 throw Exception ("mask empty after erosion");
109 previous_median = median;
110 previous_mad = mad;
111 calculate_median_mad<Image<float>, Image<bool>> (input, int_roi, cnt, median, mad);
112 upper = median + zupper * mad;
113 lower = median - zlower * mad;
114 INFO ("median: " + str(median) + ", changed: "+str((median - previous_median) / previous_median));
115 INFO ("mad: " + str(mad) + ", changed: "+str((mad - previous_mad) / previous_mad));
116 INFO ("FOV: " + str(float(cnt) / (input.size(0) * input.size(1) * input.size(2))));
117 INFO ("lower: " + str(lower) + " upper: " + str(upper));
118 INFO ("cnt_upper - cnt: " + str(cnt_upper - cnt));
119 if (lower > 0.0 && ((median + 2.5 * mad) - (previous_median + 2.5 * previous_mad)) < 0.0 && (cnt < cnt_upper))
120 break;
121 }
122
123 if (App::log_level >= 3) {
124 auto masked_image = Image<float>::scratch (input, "robust z score");
125 // output;
126 for (auto l = Loop (0,3) (masked_image, input, int_roi); l; ++l) {
127 if (int_roi.value())
128 masked_image.value() = input.value();
129 }
130 display (masked_image);
131 }
132
133 {
134 INFO ("intensity sample mask");
135 if (progress)
136 ++(*progress);
137
138 Image<float> eroded_zscore_image;
139 if (App::log_level >= 3) {
140 eroded_zscore_image = Image<float>::scratch (input, "robust z score");
141 }
142
143 int maxiter = 5;
144 while (maxiter--) {
145 // refine image mask based on robust Z score
146 cnt = 0;
147 for (auto l = Loop (0,3) (input, int_roi); l; ++l) {
148 if (int_roi.value()) {
149 float z = (input.value() - median) / mad;
150 bool good = (z > -zlower) && (z < zupper);
151 if (App::log_level >= 3) {
152 assign_pos_of(input, 0, 3).to(eroded_zscore_image);
153 eroded_zscore_image.value() = (z > -zlower) && (dont_maskupper || z < zupper) ? z : NaN;
154 }
155 if (good) cnt++;
156 int_roi.value() = good;
157 } else if (App::log_level >= 3) {
158 assign_pos_of(input, 0, 3).to(eroded_zscore_image);
159 eroded_zscore_image.value() = NaN;
160 }
161 }
162 previous_mad = mad;
163 previous_median = median;
164 calculate_median_mad<Image<float>, Image<bool>> (input, int_roi, cnt, median, mad);
165 upper = median + zupper * mad;
166 lower = median - zlower * mad;
167 INFO("median: " + str(median) + ", changed: " + str((median - previous_median)));
168 INFO("mad: " + str(mad) + ", changed: " + str((mad - previous_mad)));
169 INFO("lower: " + str(lower) + " upper: " + str(upper));
170 float change = MR::abs(median - previous_median) / previous_mad;
171 INFO("convergence: "+str(change));
172 if (change < 1e-2)
173 break;
174 }
175 if (App::log_level >= 3)
176 display (eroded_zscore_image);
177 }
178 upper = median + zupper * mad;
179 lower = median - zlower * mad;
180 if (lower < 0.0) {
181 WARN ("likely not converged, setting lower to 0.0");
182 lower = 0.0;
183 }
184
185
186 INFO ("lower: "+str(lower));
187 INFO ("upper: "+str(upper));
188 INFO ("bridge: "+str(bridge));
189
190 mask = Image<bool>::scratch (Header(spatial_prior), "temporary mask");
191 if (progress)
192 ++(*progress);
193
194 for (auto l = Loop (0,3) (input, mask, spatial_prior); l; ++l)
195 mask.value() = spatial_prior.value() && input.value() >= lower && (dont_maskupper || input.value() <= upper);
196
197 if (App::log_level >= 3)
198 display (mask);
199 if (progress)
200 ++(*progress);
201
202 {
203 INFO ("selecting largest ROI");
204 ConnectedComponents connected_filter (mask);
205 connected_filter.set_largest_only (true);
206 connected_filter (mask, mask);
207 if (progress)
208 ++(*progress);
209 }
210
211 for (auto l = Loop (0,3) (mask); l; ++l)
212 mask.value() = !mask.value();
213
214 {
215 INFO ("removing masked out islands");
216 ConnectedComponents connected_filter (mask);
217 connected_filter.set_largest_only (true);
218 connected_filter (mask, mask);
219 if (progress)
220 ++(*progress);
221 }
222
223 if (bridge) {
224 INFO ("bridging");
225 for (auto l = Loop (0,3) (mask); l; ++l)
226 mask.value() = !mask.value();
227 if (progress)
228 ++(*progress);
229 Dilate dilation_filter (mask);
230 dilation_filter.set_npass(bridge);
231 dilation_filter (mask, mask);
232 if (progress)
233 ++(*progress);
234 for (auto l = Loop (0,3) (mask); l; ++l)
235 mask.value() = !mask.value();
236 if (progress)
237 ++(*progress);
238 ConnectedComponents connected_filter (mask);
239 connected_filter.set_largest_only (true);
240 connected_filter (mask, mask);
241 if (progress)
242 ++(*progress);
243 Dilate dilation_filter2 (mask);
244 dilation_filter2.set_npass(bridge);
245 dilation_filter2 (mask, mask);
246 if (progress)
247 ++(*progress);
248 if (App::log_level >= 3)
249 display (mask);
250 }
251
252 for (auto l = Loop (0,3) (mask, spatial_prior); l; ++l)
253 mask.value() = !mask.value() && spatial_prior.value();
254 if (progress)
255 ++(*progress);
256
257 float lo = std::max<float>(median - 2.5 * mad, lower);
258 float hi = std::min<float>(median + 2.5 * mad, upper);
259 for (auto l = Loop (0,3) (input, spatial_prior, mask, output); l; ++l) {
260 if (!spatial_prior.value())
261 continue;
262 float val = input.value();
263 if (mask.value()) {
264 if (val < lo)
265 output.value() = val; // hack
266 else if (val > hi)
267 output.value() = hi;
268 else
269 output.value() = val;
270 continue;
271 } else { // outside refined mask but inside initial mask
272 if (keep_lower && val < lo)
273 output.value() = lo;
274 else if (keep_upper && val > hi)
275 output.value() = hi;
276 }
277 }

279 }

301 template <typename ImageType, typename MaskType>
302 void calculate_median_mad (ImageType& image, MaskType& mask, size_t nvoxels, float& median, float& mad) {
303 MR::vector<float> vals (nvoxels);
304 size_t idx = 0;
305 for (auto l = Loop (0,3) (mask, image); l; ++l) {
306 if (mask.value())
307 vals[idx++] = image.value();
308 assert (idx <= nvoxels);
309 }
310 median = Math::median(vals);
311 for (auto & v : vals)
312 v = MR::abs(v - median);
313 mad = Math::median(vals);
314 }
315
316 };
