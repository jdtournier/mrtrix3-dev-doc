90 class FODSegResult
91 {
92  public:
93  FODSegResult (const DWI::FMLS::FOD_lobes& lobes) :
94  vox (lobes.vox),
95  peak_dir (lobes[0].get_peak_dir()),
96  integral (lobes[0].get_integral()),
97  dispersion (integral / lobes[0].get_peak_value())
98  {
99  assert (lobes.size());
100  float sum_integrals = 0.0;
101  for (size_t i = 1; i != lobes.size(); ++i)
102  sum_integrals += lobes[i].get_integral();
103  volume_ratio = sum_integrals / lobes[0].get_integral();
104  }
105 
106  FODSegResult() : vox (), peak_dir (), integral (NAN), dispersion (NAN), volume_ratio (NAN) { }
