48  NonLinear ():
49  is_initialised (false),
50  max_iter (1, 50),
51  scale_factor (3),
52  update_smoothing (2.0),
53  disp_smoothing (1.0),
54  gradient_step (0.5),
55  do_reorientation (false),
56  fod_lmax (3) {
57  scale_factor[0] = 0.25;
58  scale_factor[1] = 0.5;
59  scale_factor[2] = 1.0;
60  fod_lmax[0] = 0;
61  fod_lmax[1] = 2;
62  fod_lmax[2] = 4;

67  template <class TransformType, class Im1ImageType, class Im2ImageType, class Im1MaskType, class Im2MaskType>
68  void run (TransformType linear_transform,
69  Im1ImageType& im1_image,
70  Im2ImageType& im2_image,
71  Im1MaskType& im1_mask,
72  Im2MaskType& im2_mask) {
73 
74  if (!is_initialised) {
75  im1_to_mid_linear = linear_transform.get_transform_half();
76  im2_to_mid_linear = linear_transform.get_transform_half_inverse();
77 
78  INFO ("Estimating halfway space");
79  std::vector<Eigen::Transform<double, 3, Eigen::Projective> > init_transforms;
80  // define transfomations that will be applied to the image header when the common space is calculated
81  {
82  Eigen::Transform<double, 3, Eigen::Projective> init_trafo_2 = linear_transform.get_transform_half();
83  Eigen::Transform<double, 3, Eigen::Projective> init_trafo_1 = linear_transform.get_transform_half_inverse();
84  init_transforms.push_back (init_trafo_2);
85  init_transforms.push_back (init_trafo_1);
86  }
87 
88  auto padding = Eigen::Matrix<default_type, 4, 1>(0.0, 0.0, 0.0, 0.0);
89  std::vector<Header> headers;
90  headers.push_back (Header (im2_image));
91  headers.push_back (Header (im1_image));
92  midway_image_header = compute_minimum_average_header(headers, 1, padding, init_transforms);
93  } else {
94  // if initialising only perform optimisation at the full resolution level
95  scale_factor.resize (1);
96  scale_factor[0] = 1.0;
97  }
98 
99  if (max_iter.size() == 1)
100  max_iter.resize (scale_factor.size(), max_iter[0]);
101  else if (max_iter.size() != scale_factor.size())
102  throw Exception ("the max number of non-linear iterations needs to be defined for each multi-resolution level (scale_factor)");
103 
104  if (do_reorientation and (fod_lmax.size() != scale_factor.size()))
105  throw Exception ("the lmax needs to be defined for each multi-resolution level (scale factor)");
106  else
107  fod_lmax.resize (scale_factor.size(), 0);
108 
109  for (size_t level = 0; level < scale_factor.size(); level++) {
110  if (is_initialised) {
111  if (do_reorientation) {
112  CONSOLE ("scale factor (init warp resolution), lmax " + str(fod_lmax[level]));
113  } else {
114  CONSOLE ("scale factor (init warp resolution)");
115  }
116  } else {
117  if (do_reorientation) {
118  CONSOLE ("multi-resolution level " + str(level + 1) + ", scale factor " + str(scale_factor[level]) + ", lmax " + str(fod_lmax[level]));
119  } else {
120  CONSOLE ("multi-resolution level " + str(level + 1) + ", scale factor " + str(scale_factor[level]));
121  }
122  }
123 
124  DEBUG ("Resizing midway image based on multi-resolution level");
125 
126  Filter::Resize resize_filter (midway_image_header);
127  resize_filter.set_scale_factor (scale_factor[level]);
128  resize_filter.set_interp_type (1);
129  resize_filter.datatype() = DataType::Float64; // for saving debug output with save()
130 
131  Header midway_image_header_resized = resize_filter;
132  midway_image_header_resized.ndim() = 3;
133 
134  default_type update_smoothing_mm = update_smoothing * ((midway_image_header_resized.spacing(0)
135  + midway_image_header_resized.spacing(1)
136  + midway_image_header_resized.spacing(2)) / 3.0);
137  default_type disp_smoothing_mm = disp_smoothing * ((midway_image_header_resized.spacing(0)
138  + midway_image_header_resized.spacing(1)
139  + midway_image_header_resized.spacing(2)) / 3.0);
140 
141  auto im1_smoothed = Registration::multi_resolution_lmax (im1_image, scale_factor[level], do_reorientation, fod_lmax[level]);
142  auto im2_smoothed = Registration::multi_resolution_lmax (im2_image, scale_factor[level], do_reorientation, fod_lmax[level]);
143 
144  DEBUG ("Initialising scratch images");
145  Header warped_header (midway_image_header_resized);
146  if (im1_image.ndim() == 4) {
147  warped_header.ndim() = 4;
148  warped_header.size(3) = im1_smoothed.size(3);
149  }
150  auto im1_warped = Image<default_type>::scratch (warped_header);
151  auto im2_warped = Image<default_type>::scratch (warped_header);
152 
153  Header field_header (midway_image_header_resized);
154  field_header.ndim() = 4;
155  field_header.size(3) = 3;
156 
157  im1_to_mid_new = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
158  im2_to_mid_new = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
159  im1_update = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
160  im2_update = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
161  im1_update_new = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
162  im2_update_new = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
163 
164  if (!is_initialised) {
165  if (level == 0) {
166  im1_to_mid = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
167  im2_to_mid = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
168  mid_to_im1 = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
169  mid_to_im2 = std::make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
170  } else {
171  DEBUG ("Upsampling fields");
172  {
173  LogLevelLatch level(0);
174  im1_to_mid = reslice (*im1_to_mid, field_header);
175  im2_to_mid = reslice (*im2_to_mid, field_header);
176  mid_to_im1 = reslice (*mid_to_im1, field_header);
177  mid_to_im2 = reslice (*mid_to_im2, field_header);
178  }
179  }
180  }
181 
182  ssize_t iteration = 1;
183  default_type grad_step_altered = gradient_step * (field_header.spacing(0) + field_header.spacing(1) + field_header.spacing(2)) / 3.0;
184  default_type cost = std::numeric_limits<default_type>::max();
185  bool converged = false;
186 
187  while (!converged) {
188  if (iteration > 1) {
189  DEBUG ("smoothing update fields");
190  Filter::Smooth smooth_filter (*im1_update);
191  smooth_filter.set_stdev (update_smoothing_mm);
192  smooth_filter (*im1_update, *im1_update);
193  smooth_filter (*im2_update, *im2_update);
194  }
195 
196  Image<default_type> im1_deform_field = Image<default_type>::scratch (field_header);
197  Image<default_type> im2_deform_field = Image<default_type>::scratch (field_header);
198 
199  if (iteration > 1) {
200  DEBUG ("updating displacement field field");
201  Warp::update_displacement_scaling_and_squaring (*im1_to_mid, *im1_update, *im1_to_mid_new, grad_step_altered);
202  Warp::update_displacement_scaling_and_squaring (*im2_to_mid, *im2_update, *im2_to_mid_new, grad_step_altered);
203 
204  DEBUG ("smoothing displacement field");
205  Filter::Smooth smooth_filter (*im1_to_mid_new);
206  smooth_filter.set_stdev (disp_smoothing_mm);
207  smooth_filter.set_zero_boundary (true);
208  smooth_filter (*im1_to_mid_new, *im1_to_mid_new);
209  smooth_filter (*im2_to_mid_new, *im2_to_mid_new);
210 
211  Registration::Warp::compose_linear_displacement (im1_to_mid_linear, *im1_to_mid_new, im1_deform_field);
212  Registration::Warp::compose_linear_displacement (im2_to_mid_linear, *im2_to_mid_new, im2_deform_field);
213  } else {
214  Registration::Warp::compose_linear_displacement (im1_to_mid_linear, *im1_to_mid, im1_deform_field);
215  Registration::Warp::compose_linear_displacement (im2_to_mid_linear, *im2_to_mid, im2_deform_field);
216  }
217 
218  DEBUG ("warping input images");
219  {
220  LogLevelLatch level (0);
221  Filter::warp<Interp::Linear> (im1_smoothed, im1_warped, im1_deform_field, 0.0);
222  Filter::warp<Interp::Linear> (im2_smoothed, im2_warped, im2_deform_field, 0.0);
223  }
224 
225  if (do_reorientation && fod_lmax[level]) {
226  DEBUG ("Reorienting FODs");
227  Registration::Transform::reorient_warp (im1_warped, im1_deform_field, aPSF_directions);
228  Registration::Transform::reorient_warp (im2_warped, im2_deform_field, aPSF_directions);
229  }
230 
231  DEBUG ("warping mask images");
232  Im1MaskType im1_mask_warped;
233  if (im1_mask.valid()) {
234  im1_mask_warped = Im1MaskType::scratch (midway_image_header_resized);
235  LogLevelLatch level (0);
236  Filter::warp<Interp::Linear> (im1_mask, im1_mask_warped, im1_deform_field, 0.0);
237  }
238  Im1MaskType im2_mask_warped;
239  if (im2_mask.valid()) {
240  im2_mask_warped = Im1MaskType::scratch (midway_image_header_resized);
241  LogLevelLatch level (0);
242  Filter::warp<Interp::Linear> (im2_mask, im2_mask_warped, im2_deform_field, 0.0);
243  }
244 
245  DEBUG ("evaluating metric and computing update field");
246  default_type cost_new = 0.0;
247  size_t voxel_count = 0;
248 
249  if (im1_image.ndim() == 4) {
250  Metric::Demons4D<Im1ImageType, Im2ImageType, Im1MaskType, Im2MaskType> metric (cost_new, voxel_count, im1_warped, im2_warped, im1_mask_warped, im2_mask_warped);
251  ThreadedLoop (im1_warped, 0, 3).run (metric, im1_warped, im2_warped, *im1_update_new, *im2_update_new);
252  } else {
253  Metric::Demons<Im1ImageType, Im2ImageType, Im1MaskType, Im2MaskType> metric (cost_new, voxel_count, im1_warped, im2_warped, im1_mask_warped, im2_mask_warped);
254  ThreadedLoop (im1_warped, 0, 3).run (metric, im1_warped, im2_warped, *im1_update_new, *im2_update_new);
255  }
256 
257  cost_new /= static_cast<default_type>(voxel_count);
258 
259  // If cost is lower then keep new displacement fields and gradients
260  if (cost_new < cost) {
261  cost = cost_new;
262  if (iteration > 1) {
263  std::swap (im1_to_mid_new, im1_to_mid);
264  std::swap (im2_to_mid_new, im2_to_mid);
265  }
266  std::swap (im1_update_new, im1_update);
267  std::swap (im2_update_new, im2_update);
268 
269  DEBUG ("inverting displacement field");
270  {
271  LogLevelLatch level (0);
272  Warp::invert_displacement (*im1_to_mid, *mid_to_im1);
273  Warp::invert_displacement (*im2_to_mid, *mid_to_im2);
274  }
275 
276 
277  } else {
278  converged = true;
279  }
280 
281  if (!converged)
282  INFO (" iteration: " + str(iteration) + " cost: " + str(cost));
283 
284  if (++iteration > max_iter[level])
285  converged = true;
286  }
287  }
288  // Convert all warps to deformation field format for output
289  Registration::Warp::displacement2deformation (*im1_to_mid, *im1_to_mid);
290  Registration::Warp::displacement2deformation (*im2_to_mid, *im2_to_mid);
291  Registration::Warp::displacement2deformation (*mid_to_im1, *mid_to_im1);
292  Registration::Warp::displacement2deformation (*mid_to_im2, *mid_to_im2);
293  if (has_negative_jacobians(*im1_to_mid) || has_negative_jacobians(*im2_to_mid) ||
294  has_negative_jacobians(*mid_to_im1) || has_negative_jacobians(*mid_to_im2))
295  WARN ("final warp computed is not diffeomorphic (negative jacobian determinants detected). Try increasing -nl_disp_smooth or -nl_update_smooth regularisation.");
296  }
