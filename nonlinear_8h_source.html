49  scale_factor (3),
50  update_smoothing (2.0),
51  disp_smoothing (1.0),
52  gradient_step (0.5),
53  do_reorientation (false),
54  fod_lmax (3) {
55  scale_factor[0] = 0.25;
56  scale_factor[1] = 0.5;
57  scale_factor[2] = 1.0;
58  fod_lmax[0] = 0;
59  fod_lmax[1] = 2;

64  template <class TransformType, class Im1ImageType, class Im2ImageType, class Im1MaskType, class Im2MaskType>
65  void run (TransformType linear_transform,
66  Im1ImageType& im1_image,
67  Im2ImageType& im2_image,
68  Im1MaskType& im1_mask,
69  Im2MaskType& im2_mask) {
70 
71  if (!is_initialised) {
72  im1_to_mid_linear = linear_transform.get_transform_half();
73  im2_to_mid_linear = linear_transform.get_transform_half_inverse();
74 
75  INFO ("Estimating halfway space");
76  vector<Eigen::Transform<double, 3, Eigen::Projective>> init_transforms;
77  // define transfomations that will be applied to the image header when the common space is calculated
78  {
79  Eigen::Transform<double, 3, Eigen::Projective> init_trafo_2 = linear_transform.get_transform_half();
80  Eigen::Transform<double, 3, Eigen::Projective> init_trafo_1 = linear_transform.get_transform_half_inverse();
81  init_transforms.push_back (init_trafo_2);
82  init_transforms.push_back (init_trafo_1);
83  }
84 
85  auto padding = Eigen::Matrix<default_type, 4, 1>(0.0, 0.0, 0.0, 0.0);
86  vector<Header> headers;
87  headers.push_back (Header (im2_image));
88  headers.push_back (Header (im1_image));
89  midway_image_header = compute_minimum_average_header(headers, 1, padding, init_transforms);
90  } else {
91  // if initialising only perform optimisation at the full resolution level
92  scale_factor.resize (1);
93  scale_factor[0] = 1.0;
94  }
95 
96  if (max_iter.size() == 1)
97  max_iter.resize (scale_factor.size(), max_iter[0]);
98  else if (max_iter.size() != scale_factor.size())
99  throw Exception ("the max number of non-linear iterations needs to be defined for each multi-resolution level (scale_factor)");
100 
101  if (do_reorientation and (fod_lmax.size() != scale_factor.size()))
102  throw Exception ("the lmax needs to be defined for each multi-resolution level (scale factor)");
103  else
104  fod_lmax.resize (scale_factor.size(), 0);
105 
106  for (size_t level = 0; level < scale_factor.size(); level++) {
107  if (is_initialised) {
108  if (do_reorientation) {
109  CONSOLE ("scale factor (init warp resolution), lmax " + str(fod_lmax[level]));
110  } else {
111  CONSOLE ("scale factor (init warp resolution)");
112  }
113  } else {
114  if (do_reorientation) {
115  CONSOLE ("nonlinear stage " + str(level + 1) + ", scale factor " + str(scale_factor[level]) + ", lmax " + str(fod_lmax[level]));
116  } else {
117  CONSOLE ("nonlinear stage " + str(level + 1) + ", scale factor " + str(scale_factor[level]));
118  }
119  }
120 
121  DEBUG ("Resizing midway image based on multi-resolution level");
122 
123  Filter::Resize resize_filter (midway_image_header);
124  resize_filter.set_scale_factor (scale_factor[level]);
125  resize_filter.set_interp_type (1);
126  resize_filter.datatype() = DataType::Float64; // for saving debug output with save()
127 
128  Header midway_image_header_resized = resize_filter;
129  midway_image_header_resized.ndim() = 3;
130 
131  default_type update_smoothing_mm = update_smoothing * ((midway_image_header_resized.spacing(0)
132  + midway_image_header_resized.spacing(1)
133  + midway_image_header_resized.spacing(2)) / 3.0);
134  default_type disp_smoothing_mm = disp_smoothing * ((midway_image_header_resized.spacing(0)
135  + midway_image_header_resized.spacing(1)
136  + midway_image_header_resized.spacing(2)) / 3.0);
137 
138  auto im1_smoothed = Registration::multi_resolution_lmax (im1_image, scale_factor[level], do_reorientation, fod_lmax[level]);
139  auto im2_smoothed = Registration::multi_resolution_lmax (im2_image, scale_factor[level], do_reorientation, fod_lmax[level]);
140 
141  DEBUG ("Initialising scratch images");
142  Header warped_header (midway_image_header_resized);
143  if (im1_image.ndim() == 4) {
144  warped_header.ndim() = 4;
145  warped_header.size(3) = im1_smoothed.size(3);
146  }
147  auto im1_warped = Image<default_type>::scratch (warped_header);
148  auto im2_warped = Image<default_type>::scratch (warped_header);
149 
150  Header field_header (midway_image_header_resized);
151  field_header.ndim() = 4;
152  field_header.size(3) = 3;
153 
154  im1_to_mid_new = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
155  im2_to_mid_new = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
156  im1_update = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
157  im2_update = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
158  im1_update_new = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
159  im2_update_new = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
160 
161  if (!is_initialised) {
162  if (level == 0) {
163  im1_to_mid = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
164  im2_to_mid = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
165  mid_to_im1 = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
166  mid_to_im2 = make_shared<Image<default_type>>(Image<default_type>::scratch (field_header));
167  } else {
168  DEBUG ("Upsampling fields");
169  {
170  LogLevelLatch level(0);
171  im1_to_mid = reslice (*im1_to_mid, field_header);
172  im2_to_mid = reslice (*im2_to_mid, field_header);
173  mid_to_im1 = reslice (*mid_to_im1, field_header);
174  mid_to_im2 = reslice (*mid_to_im2, field_header);
175  }
176  }
177  }
178 
179  ssize_t iteration = 1;
180  default_type grad_step_altered = gradient_step * (field_header.spacing(0) + field_header.spacing(1) + field_header.spacing(2)) / 3.0;
181  default_type cost = std::numeric_limits<default_type>::max();
182  bool converged = false;
183 
184  while (!converged) {
185  if (iteration > 1) {
186  DEBUG ("smoothing update fields");
187  Filter::Smooth smooth_filter (*im1_update);
188  smooth_filter.set_stdev (update_smoothing_mm);
189  smooth_filter (*im1_update);
190  smooth_filter (*im2_update);
191  }
192 
193  Image<default_type> im1_deform_field = Image<default_type>::scratch (field_header);
194  Image<default_type> im2_deform_field = Image<default_type>::scratch (field_header);
195 
196  if (iteration > 1) {
197  DEBUG ("updating displacement field");
198  Warp::update_displacement_scaling_and_squaring (*im1_to_mid, *im1_update, *im1_to_mid_new, grad_step_altered);
199  Warp::update_displacement_scaling_and_squaring (*im2_to_mid, *im2_update, *im2_to_mid_new, grad_step_altered);
200 
201  DEBUG ("smoothing displacement field");
202  Filter::Smooth smooth_filter (*im1_to_mid_new);
203  smooth_filter.set_stdev (disp_smoothing_mm);
204  smooth_filter.set_zero_boundary (true);
205  smooth_filter (*im1_to_mid_new);
206  smooth_filter (*im2_to_mid_new);
207 
208  Registration::Warp::compose_linear_displacement (im1_to_mid_linear, *im1_to_mid_new, im1_deform_field);
209  Registration::Warp::compose_linear_displacement (im2_to_mid_linear, *im2_to_mid_new, im2_deform_field);
210  } else {
211  Registration::Warp::compose_linear_displacement (im1_to_mid_linear, *im1_to_mid, im1_deform_field);
212  Registration::Warp::compose_linear_displacement (im2_to_mid_linear, *im2_to_mid, im2_deform_field);
213  }
214 
215  DEBUG ("warping input images");
216  {
217  LogLevelLatch level (0);
218  Filter::warp<Interp::Linear> (im1_smoothed, im1_warped, im1_deform_field, 0.0);
219  Filter::warp<Interp::Linear> (im2_smoothed, im2_warped, im2_deform_field, 0.0);
220  }
221 
222  if (do_reorientation && fod_lmax[level]) {
223  DEBUG ("Reorienting FODs");
224  Registration::Transform::reorient_warp (im1_warped, im1_deform_field, aPSF_directions);
225  Registration::Transform::reorient_warp (im2_warped, im2_deform_field, aPSF_directions);
226  }
227 
228  DEBUG ("warping mask images");
229  Im1MaskType im1_mask_warped;
230  if (im1_mask.valid()) {
231  im1_mask_warped = Im1MaskType::scratch (midway_image_header_resized);
232  LogLevelLatch level (0);
233  Filter::warp<Interp::Linear> (im1_mask, im1_mask_warped, im1_deform_field, 0.0);
234  }
235  Im1MaskType im2_mask_warped;
236  if (im2_mask.valid()) {
237  im2_mask_warped = Im1MaskType::scratch (midway_image_header_resized);
238  LogLevelLatch level (0);
239  Filter::warp<Interp::Linear> (im2_mask, im2_mask_warped, im2_deform_field, 0.0);
240  }
241 
242  DEBUG ("evaluating metric and computing update field");
243  default_type cost_new = 0.0;
244  size_t voxel_count = 0;
245 
246  if (im1_image.ndim() == 4) {
247  Metric::Demons4D<Im1ImageType, Im2ImageType, Im1MaskType, Im2MaskType> metric (cost_new, voxel_count, im1_warped, im2_warped, im1_mask_warped, im2_mask_warped);
248  ThreadedLoop (im1_warped, 0, 3).run (metric, im1_warped, im2_warped, *im1_update_new, *im2_update_new);
249  } else {
250  Metric::Demons<Im1ImageType, Im2ImageType, Im1MaskType, Im2MaskType> metric (cost_new, voxel_count, im1_warped, im2_warped, im1_mask_warped, im2_mask_warped);
251  ThreadedLoop (im1_warped, 0, 3).run (metric, im1_warped, im2_warped, *im1_update_new, *im2_update_new);
252  }
253 
254  cost_new /= static_cast<default_type>(voxel_count);
255 
256  // If cost is lower then keep new displacement fields and gradients
257  if (cost_new < cost) {
258  cost = cost_new;
259  if (iteration > 1) {
260  std::swap (im1_to_mid_new, im1_to_mid);
261  std::swap (im2_to_mid_new, im2_to_mid);
262  }
263  std::swap (im1_update_new, im1_update);
264  std::swap (im2_update_new, im2_update);
265 
266  DEBUG ("inverting displacement field");
267  {
268  LogLevelLatch level (0);
269  Warp::invert_displacement (*im1_to_mid, *mid_to_im1);
270  Warp::invert_displacement (*im2_to_mid, *mid_to_im2);
271  }
272 
273 
274  } else {
275  converged = true;
276  }
277 
278  if (!converged)
279  INFO (" iteration: " + str(iteration) + " cost: " + str(cost));
280 
281  if (++iteration > max_iter[level])
282  converged = true;
283  }
284  }
285  // Convert all warps to deformation field format for output
286  Registration::Warp::displacement2deformation (*im1_to_mid, *im1_to_mid);
287  Registration::Warp::displacement2deformation (*im2_to_mid, *im2_to_mid);
288  Registration::Warp::displacement2deformation (*mid_to_im1, *mid_to_im1);
289  Registration::Warp::displacement2deformation (*mid_to_im2, *mid_to_im2);
290  if (has_negative_jacobians(*im1_to_mid) || has_negative_jacobians(*im2_to_mid) ||
291  has_negative_jacobians(*mid_to_im1) || has_negative_jacobians(*mid_to_im2))
292  WARN ("final warp computed is not diffeomorphic (negative jacobian determinants detected). Try increasing -nl_disp_smooth or -nl_update_smooth regularisation.");
293  }
