86  void operator() (InputComplexImageType& input, OutputComplexImageType& output)
87  {
88 
89  std::shared_ptr<ProgressBar> progress (message.size() ? new ProgressBar (message, axes_to_process.size() + 2) : nullptr);
90 
91  auto temp = Image<cdouble>::scratch (*this);
92  copy (input, temp);
93  if (progress)
94  ++(*progress);
95 
96  for (vector<size_t>::const_iterator axis = axes_to_process.begin(); axis != axes_to_process.end(); ++axis) {
97  vector<size_t> axes = Stride::order (temp);
98  for (size_t n = 0; n < axes.size(); ++n) {
99  if (axes[n] == *axis) {
100  axes.erase (axes.begin() + n);
101  break;
102  }
103  }
104  FFTKernel<decltype(temp)> kernel (temp, *axis, inverse);
105  ThreadedLoop (temp, axes, 1).run (kernel);
106  if (progress) ++(*progress);
107  }
108 
109  if (centre_zero_) {
110  for (auto l = Loop (output)(output); l; ++l) {
111  assign_pos_of (output).to (temp);
112  for (vector<size_t>::const_iterator flip_axis = axes_to_process.begin(); flip_axis != axes_to_process.end(); ++flip_axis)
113  temp.index(*flip_axis) = (temp.index(*flip_axis) >= (temp.size (*flip_axis) / 2)) ?
114  (temp.index(*flip_axis) - (temp.size (*flip_axis) / 2)) :
115  (temp.index(*flip_axis) + (temp.size (*flip_axis) / 2));
116  output.value() = temp.value();
