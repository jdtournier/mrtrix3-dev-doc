85 template <class InputComplexImageType, class OutputComplexImageType>
86 void operator() (InputComplexImageType& input, OutputComplexImageType& output)
87 {
88
89 std::shared_ptr<ProgressBar> progress (message.size() ? new ProgressBar (message, axes_to_process.size() + 2) : nullptr);
90
91 auto temp = Image<cdouble>::scratch (*this);
92 copy (input, temp);
93 if (progress)
94 ++(*progress);
95
96 for (vector<size_t>::const_iterator axis = axes_to_process.begin(); axis != axes_to_process.end(); ++axis) {
97 vector<size_t> axes = Stride::order (temp);
98 for (size_t n = 0; n < axes.size(); ++n) {
99 if (axes[n] == *axis) {
100 axes.erase (axes.begin() + n);
101 break;
102 }
103 }
104 FFTKernel<decltype(temp)> kernel (temp, *axis, inverse);
105 ThreadedLoop (temp, axes, 1).run (kernel);
106 if (progress) ++(*progress);
107 }
108
109 if (centre_zero_) {
110 for (auto l = Loop (output)(output); l; ++l) {
111 assign_pos_of (output).to (temp);
112 for (vector<size_t>::const_iterator flip_axis = axes_to_process.begin(); flip_axis != axes_to_process.end(); ++flip_axis)
113 temp.index(*flip_axis) = (temp.index(*flip_axis) >= (temp.size (*flip_axis) / 2)) ?
114 (temp.index(*flip_axis) - (temp.size (*flip_axis) / 2)) :
115 (temp.index(*flip_axis) + (temp.size (*flip_axis) / 2));
116 output.value() = temp.value();
117 }
118 } else {
119 copy (temp, output);
120 }
121 }

130 template <class ComplexImageType>
131 class FFTKernel { MEMALIGN(FFTKernel)
132 public:
133 FFTKernel (const ComplexImageType& voxel, const size_t FFT_axis, const bool inverse_FFT) :
134 vox (voxel),
135 data_in (vox.size (FFT_axis)),
136 data_out (data_in.size()),
137 axis (FFT_axis),
138 inverse (inverse_FFT) { }
139
140 void operator () (const Iterator& pos) {
141 assign_pos_of (pos).to (vox);
142 for (vox.index(axis) = 0; vox.index(axis) < vox.size(axis); ++vox.index(axis))
143 data_in[vox.index(axis)] = cdouble (vox.value());
144 if (inverse)
145 fft.inv (data_out, data_in);
146 else
147 fft.fwd (data_out, data_in);
148 for (vox.index(axis) = 0; vox.index(axis) < vox.size(axis); ++vox.index(axis))
149 vox.value() = typename ComplexImageType::value_type (data_out[vox.index(axis)]);
150 }

164 template <class ImageType>
165 void fft (ImageType&& vox, const size_t axis, const bool inverse = false) {
166 auto axes = Stride::order (vox);
167 for (size_t n = 0; n < axes.size(); ++n)
168 if (axis == axes[n])
169 axes.erase (axes.begin() + n);
170
171 struct Kernel { MEMALIGN(Kernel)
172 Kernel (const ImageType& v, size_t axis, bool inverse) :
173 data_in (v.size (axis)), data_out (data_in.size()), axis (axis), inverse (inverse) { }
174
175 void operator ()(ImageType& v) {
176 for (auto l = Loop (axis, axis+1) (v); l; ++l)
177 data_in[v[axis]] = cdouble (v.value());
178 if (inverse)
179 fft.inv (data_out, data_in);
180 else
181 fft.fwd (data_out, data_in);
182 for (auto l = Loop (axis, axis+1) (v); l; ++l)
183 v.value() = typename std::remove_reference<ImageType>::type::value_type (data_out[v[axis]]);
184 }
185 Eigen::FFT<double> fft;
186 Eigen::Matrix<cdouble, Eigen::Dynamic, 1> data_in, data_out;
187 const size_t axis;
188 const bool inverse;
189 } kernel (vox, axis, inverse);
190
191 ThreadedLoop ("performing in-place FFT", vox, axes)
192 .run (kernel, vox);
193 }
