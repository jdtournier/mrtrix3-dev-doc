106  struct IntPointHasher {
107  size_t operator () (const std::array<int, 3>& v) const {
108  // This hashing function works best if the fixel image dimensions
109  // are bounded above by 2^10 x 2^10 x 2^10 = 1024 x 1024 x 1024
110  return (v[0] + (v[1] << 10) + (v[2] << 20));
111  }
