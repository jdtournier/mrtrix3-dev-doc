31  // Base class to handle case where the factor contributed by the streamline varies along its length
32  // (currently only occurs when the track-wise statistic is Gaussian)
33  class VoxelAddon
34  {
35  public:
36  VoxelAddon() : sum_factors (0.0) { }
37  VoxelAddon (const float v) : sum_factors (v) { }
38  float get_factor() const { return sum_factors; }
39  protected:
40  void operator+= (const float f) const { sum_factors += f; }
41  void operator= (const float f) { sum_factors = f; }
42  void operator= (const VoxelAddon& that) { sum_factors = that.sum_factors; }
43  void normalize (const float l) const { sum_factors /= l; }
44  private:
45  mutable float sum_factors;
46  };
