110  void init (const Stride::List& desired_strides) {
111  Stride::List new_strides = Image::Stride::get_nearest_match (static_cast<ConstHeader&> (*this), desired_strides);
112 
113  if (new_strides == Image::Stride::get (static_cast<ConstHeader&> (*this))) {
114  init();
115  return;
116  }
117 
118  Stride::List old_strides = Stride::get (*this);
119  Stride::set (static_cast<Header&> (*this), new_strides);
120  voxel_type dest_vox (*this);
121 
122  Stride::set (static_cast<Header&> (*this), old_strides);
123  do_load (dest_vox);
124  Stride::set (static_cast<Header&> (*this), new_strides);
125  }

128  void init () {
129  assert (handler_);
130  assert (handler_->nsegments());
131 
132  if (handler_->nsegments() == 1 &&
133  datatype() == DataType::from<value_type>() &&
134  intensity_offset() == 0.0 && intensity_scale() == 1.0) {
135  INFO ("data in \"" + name() + "\" already in required format - mapping as-is");
136  data_ = reinterpret_cast<value_type*> (handler_->segment (0));
137  return;

144  template <class VoxType>
145  void do_load (VoxType& destination) {
146  INFO ("data for image \"" + name() + "\" will be loaded into memory");
147  data_ = new value_type [Image::voxel_count (*this)];
148  Buffer<value_type>& filedata (*this);
149  typename Buffer<value_type>::voxel_type src (filedata);
150  Image::threaded_copy_with_progress_message ("loading data for image \"" + name() + "\"...", src, destination);
151 
152  this->Info::datatype_ = DataType::from<value_type>();
153  handler_ = NULL;
