120  template <class InputImageType, class OutputImageType, typename ValueType = float>
121  void operator() (InputImageType& input, OutputImageType& output)
122  {
123  std::shared_ptr <Image<ValueType> > in (std::make_shared<Image<ValueType> > (Image<ValueType>::scratch (input)));
124  threaded_copy (input, *in);
125  std::shared_ptr <Image<ValueType> > out;
126 
127  std::unique_ptr<ProgressBar> progress;
128  if (message.size()) {
129  size_t axes_to_smooth = 0;
130  for (std::vector<default_type>::const_iterator i = stdev.begin(); i != stdev.end(); ++i)
131  if (*i)
132  ++axes_to_smooth;
133  progress.reset (new ProgressBar (message, axes_to_smooth + 1));
134  }
135 
136  for (size_t dim = 0; dim < 3; dim++) {
137  if (stdev[dim] > 0) {
138  out = std::make_shared<Image<ValueType> > (Image<ValueType>::scratch (input));
139  Adapter::Gaussian1D<Image<ValueType> > gaussian (*in, stdev[dim], dim, extent[dim], zero_boundary);
140  threaded_copy (gaussian, *out, 0, input.ndim(), 2);
141  in = out;
142  if (progress)
143  ++(*progress);
144  }
145  }
146  threaded_copy (*in, output);
147  }
