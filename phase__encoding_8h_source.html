44  template <class MatrixType>
45  void check (const MatrixType& PE)
46  {
47  if (!PE.rows())
48  throw Exception ("No valid phase encoding table found");
49 
50  if (PE.cols() < 3)
51  throw Exception ("Phase-encoding matrix must have at least 3 columns");
52 
53  for (ssize_t row = 0; row != PE.rows(); ++row) {
54  for (ssize_t axis = 0; axis != 3; ++axis) {
55  if (std::round (PE(row, axis)) != PE(row, axis))
56  throw Exception ("Phase-encoding matrix contains non-integral axis designation");
57  }
58  }
59  }

65  void check (const Header& header, const MatrixType& PE)
66  {

70  throw Exception ("Number of volumes in image \"" + header.name() + "\" (" + str(num_volumes) + ") does not match that in phase encoding table (" + str(PE.rows()) + ")");

88  template <class MatrixType>
89  void set_scheme (Header& header, const MatrixType& PE)
90  {
91  auto erase = [&] (const std::string& s) { auto it = header.keyval().find (s); if (it != header.keyval().end()) header.keyval().erase (it); };
92  if (!PE.rows()) {
93  erase ("pe_scheme");
94  erase ("PhaseEncodingDirection");
95  erase ("TotalReadoutTime");
96  return;
97  }
98  check (header, PE);
99  std::string pe_scheme;
100  std::string first_line;
101  bool variation = false;
102  for (ssize_t row = 0; row < PE.rows(); ++row) {
103  std::string line = str(PE(row,0));
104  for (ssize_t col = 1; col < PE.cols(); ++col)
105  line += "," + str(PE(row,col), 3);
106  add_line (pe_scheme, line);
107  if (first_line.empty())
108  first_line = line;
109  else if (line != first_line)
110  variation = true;
111  }
112  if (variation) {
113  header.keyval()["pe_scheme"] = pe_scheme;
114  erase ("PhaseEncodingDirection");
115  erase ("TotalReadoutTime");
116  } else {
117  erase ("pe_scheme");
118  const Eigen::Vector3 dir { PE(0, 0), PE(0, 1), PE(0, 2) };
119  header.keyval()["PhaseEncodingDirection"] = Axes::dir2id (dir);
120  if (PE.cols() >= 4)
121  header.keyval()["TotalReadoutTime"] = str(PE(0, 3), 3);
122  else
123  erase ("TotalReadoutTime");
124  }
125  }

158  template <class MatrixType>
159  void scheme2eddy (const MatrixType& PE, Eigen::MatrixXd& config, Eigen::Array<int, Eigen::Dynamic, 1>& indices)
160  {
161  try {
162  check (PE);
163  } catch (Exception& e) {
164  throw Exception (e, "Cannot convert phase-encoding scheme to eddy format");
165  }
166  if (PE.cols() != 4)
167  throw Exception ("Phase-encoding matrix requires 4 columns to convert to eddy format");
168  config.resize (0, 0);
169  indices = Eigen::Array<int, Eigen::Dynamic, 1>::Constant (PE.rows(), PE.rows());
170  for (ssize_t PE_row = 0; PE_row != PE.rows(); ++PE_row) {
171  for (ssize_t config_row = 0; config_row != config.rows(); ++config_row) {
172  bool dir_match = PE.template block<1,3>(PE_row, 0).isApprox (config.block<1,3>(config_row, 0));
173  bool time_match = abs (PE(PE_row, 3) - config(config_row, 3)) < 1e-3;
174  if (dir_match && time_match) {
175  // FSL-style index file indexes from 1
176  indices[PE_row] = config_row + 1;
177  break;
178  }
179  }
180  if (indices[PE_row] == PE.rows()) {
181  // No corresponding match found in config matrix; create a new entry
182  config.conservativeResize (config.rows()+1, 4);
183  config.row(config.rows()-1) = PE.row(PE_row);
184  indices[PE_row] = config.rows();
185  }
186  }
187  }

195  template <class MatrixType>
196  void save (const MatrixType& PE, const std::string& path)
197  {
198  try {
199  check (PE);
200  } catch (Exception& e) {
201  throw Exception (e, "Cannot export phase-encoding table to file \"" + path + "\"");
202  }
203 
204  File::OFStream out (path);
205  for (ssize_t row = 0; row != PE.rows(); ++row) {
206  // Write phase-encode direction as integers; other information as floating-point
207  out << PE.template block<1, 3>(row, 0).template cast<int>();
208  if (PE.cols() > 3)
209  out << " " << PE.block(row, 3, 1, PE.cols()-3);
210  out << "\n";
211  }
212  }
