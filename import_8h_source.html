118 template <class SubjectDataImport>
119 void CohortDataImport::initialise (const std::string& listpath, const std::string& explicit_from_directory)
120 {
121 // Read the provided text file one at a time
122 // For each file, create an instance of SubjectDataImport
123 // (which must derive from SubjectDataImportBase)
124
125 // - If a directory path is explicitly provided, first try to find
126 // all input files relative to that directory
127 // - If that fails, try to find all files relative to
128 // directory in which subject list text file is stored
129 // - If that fails, try to find all files relative to
130 // current working directory
131 // Only once a directory is selected that contains all inputs listed in the
132 // text file is an attempt made to load all of those files
133 vector<std::string> lines;
134 {
135 std::ifstream ifs (listpath.c_str());
136 if (!ifs)
137 throw Exception ("Unable to open subject file list \"" + listpath + "\"");
138 std::string line;
139 while (getline (ifs, line)) {
140 size_t p = line.find_last_not_of(" \t");
141 if (p != std::string::npos)
142 line.erase (p+1);
143 if (line.size())
144 lines.push_back (line);
145 }
146 }
147
148 vector<std::string> directories { Path::dirname (listpath) };
149 if (directories[0].size() && directories[0] != ".")
150 directories.push_back (".");
151 if (explicit_from_directory.size())
152 directories.insert (directories.begin(), explicit_from_directory);
153
154 Exception e_nosuccess ("Unable to load all input data from file \"" + listpath + "\"");
155 std::string load_from_dir;
156 for (const auto& directory : directories) {
157 try {
158 for (const auto& line : lines) {
159 const std::string full_path = Path::join (directory, line);
160 if (!Path::is_file (full_path))
161 throw Exception ("File \"" + full_path + "\" not found");
162 }
163 load_from_dir = directory;
164 break;
165 } catch (Exception& e) {
166 e_nosuccess.push_back ("If loading relative to directory \"" + directory + "\": ");
167 e_nosuccess.push_back (e);
168 }
169 }
170
171 if (load_from_dir.empty())
172 throw e_nosuccess;
173
174 ProgressBar progress ("Importing data from files listed in \""
175 + Path::basename (listpath)
176 + "\" as found relative to directory \""
177 + load_from_dir + "\"");
178
179 for (const auto& line : lines) {
180 try {
181 std::shared_ptr<SubjectDataImport> subject (new SubjectDataImport (Path::join (load_from_dir, line)));
182 files.emplace_back (subject);
183 } catch (Exception& e) {
184 throw Exception (e, "Input data not successfully loaded: \"" + line + "\"");
185 }
186 ++progress;
187 }
188 }

193 }
194 }
