80  bool set (const float cg, const float sg, const float w, const float c, const float p) {
81  if (std::isnan (cg) || std::isnan (sg) || std::isnan (w) || std::isnan (c) || std::isnan (p)) {
82  cgm = sgm = wm = csf = path = 0.0;
83  return ((is_valid = false));
84  }
85  cgm = (cg < 0.0) ? 0.0 : ((cg > 1.0) ? 1.0 : cg);
86  sgm = (sg < 0.0) ? 0.0 : ((sg > 1.0) ? 1.0 : sg);
87  wm = (w < 0.0) ? 0.0 : ((w > 1.0) ? 1.0 : w );
88  csf = (c < 0.0) ? 0.0 : ((c > 1.0) ? 1.0 : c );
89  path = (p < 0.0) ? 0.0 : ((p > 1.0) ? 1.0 : p );
90  return ((is_valid = ((cgm + sgm + wm + csf + path) >= TISSUE_SUM_THRESHOLD)));
91  }
