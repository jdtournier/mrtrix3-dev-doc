87  bool set (const float cg, const float sg, const float w, const float c, const float p) {
88  if (std::isnan (cg) || std::isnan (sg) || std::isnan (w) || std::isnan (c) || std::isnan (p)) {
89  cgm = sgm = wm = csf = path = 0.0;
90  return ((is_valid = false));
91  }
92  cgm = (cg < 0.0) ? 0.0 : ((cg > 1.0) ? 1.0 : cg);
93  sgm = (sg < 0.0) ? 0.0 : ((sg > 1.0) ? 1.0 : sg);
94  wm = (w < 0.0) ? 0.0 : ((w > 1.0) ? 1.0 : w );
95  csf = (c < 0.0) ? 0.0 : ((c > 1.0) ? 1.0 : c );
96  path = (p < 0.0) ? 0.0 : ((p > 1.0) ? 1.0 : p );
97  return ((is_valid = ((cgm + sgm + wm + csf + path) >= TISSUE_SUM_THRESHOLD)));
98  }
