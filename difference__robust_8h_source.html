116  diff_values = im1_values - im2_values;
117 
118  Eigen::Matrix<default_type, Eigen::Dynamic, 1> residuals, grads;
119  estimator (diff_values.template cast<default_type>(), residuals, grads);

121  Eigen::Matrix<default_type, 1, 3> g;
122  for (ssize_t i = 0; i < volumes; ++i) {

124  gradient.segment<4>(0) += g(0) * jacobian_vec;
125  gradient.segment<4>(4) += g(1) * jacobian_vec;
