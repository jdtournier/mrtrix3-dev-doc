30  FORCE_INLINE Eigen::MatrixXd aPSF_weights_to_FOD_transform (const int num_SH, const Eigen::MatrixXd& directions)
31  {
32  Eigen::MatrixXd aPSF_matrix (num_SH, directions.cols());
33  Math::SH::aPSF<default_type> aPSF_generator (Math::SH::LforN (num_SH));
34  Eigen::Matrix<default_type, Eigen::Dynamic,1> aPSF;
35  for (ssize_t i = 0; i < directions.cols(); ++i) {
36  aPSF_matrix.col(i) = aPSF_generator (aPSF, directions.col(i));
37  }
38  return aPSF_matrix;
39  }

43  template <class FODImageType>
44  class LinearKernel { MEMALIGN(LinearKernel<FODImageType>)
45 
46  public:
47  LinearKernel (const ssize_t n_SH,
48  const transform_type& linear_transform,
49  const Eigen::MatrixXd& directions,
50  const bool modulate) : fod (n_SH)
51  {
52  Eigen::MatrixXd transformed_directions = linear_transform.linear().inverse() * directions;
53 
54  if (modulate) {
55  Eigen::VectorXd modulation_factors = transformed_directions.colwise().norm() / linear_transform.linear().inverse().determinant();
56  transformed_directions.colwise().normalize();
57  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * modulation_factors.asDiagonal()
58  * Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions));
59  } else {
60  transformed_directions.colwise().normalize();
61  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions)
62  * Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions));
63  }
64  }

66  void operator() (FODImageType& in, FODImageType& out)
67  {
68  in.index(3) = 0;
69  if (in.value() > 0.0) { // only reorient voxels that contain a FOD
70  fod = in.row(3);
71  fod = transform * fod;
72  out.row(3) = fod;
73  }
74  }

78  Eigen::VectorXd fod;

119  template <class FODImageType>
120  class NonLinearKernel { MEMALIGN(NonLinearKernel<FODImageType>)
121 
122  public:
123  NonLinearKernel (const ssize_t n_SH, Image<default_type>& warp, const Eigen::MatrixXd& directions, const bool modulate) :
124  n_SH (n_SH),
125  jacobian_adapter (warp),
126  directions (directions),
127  modulate (modulate),
128  FOD_to_aPSF_transform (Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions))),
129  fod (n_SH) {}
130 
131 
132  void operator() (FODImageType& image) {
133  image.index(3) = 0;
134  if (image.value() > 0) { // only reorient voxels that contain a FOD
135  for (size_t dim = 0; dim < 3; ++dim)
136  jacobian_adapter.index(dim) = image.index(dim);
137  Eigen::MatrixXd jacobian = jacobian_adapter.value().inverse().template cast<default_type>();
138  Eigen::MatrixXd transformed_directions = jacobian * directions;
139 
140  if (modulate) {
141  Eigen::MatrixXd modulation_factors = transformed_directions.colwise().norm() / jacobian.determinant();
142  transformed_directions.colwise().normalize();
143 
144  Eigen::MatrixXd temp = aPSF_weights_to_FOD_transform (n_SH, transformed_directions);
145  for (ssize_t i = 0; i < temp.cols(); ++i)
146  temp.col(i) = temp.col(i) * modulation_factors(0,i);
147 
148  transform.noalias() = temp * FOD_to_aPSF_transform;
149  } else {
150  transformed_directions.colwise().normalize();
151  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * FOD_to_aPSF_transform;
152  }
153  fod = image.row(3);
154  fod = transform * fod;
155  image.row(3) = fod;
156  }
157  }
158  protected:
159  const ssize_t n_SH;
160  Adapter::Jacobian<Image<default_type> > jacobian_adapter;
161  const Eigen::MatrixXd& directions;
162  const bool modulate;
163  const Eigen::MatrixXd FOD_to_aPSF_transform;
164  Eigen::MatrixXd transform;
165  Eigen::VectorXd fod;
166  };
