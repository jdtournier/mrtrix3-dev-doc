31  FORCE_INLINE Eigen::MatrixXd aPSF_weights_to_FOD_transform (const int num_SH, const Eigen::MatrixXd& directions)
32  {
33  Eigen::MatrixXd aPSF_matrix (num_SH, directions.cols());
34  Math::SH::aPSF<default_type> aPSF_generator (Math::SH::LforN (num_SH));
35  Eigen::Matrix<default_type, Eigen::Dynamic,1> aPSF;
36  for (ssize_t i = 0; i < directions.cols(); ++i) {
37  aPSF_matrix.col(i) = aPSF_generator (aPSF, directions.col(i));
38  }
39  return aPSF_matrix;
40  }

44  template <class FODImageType>
45  class LinearKernel { MEMALIGN(LinearKernel<FODImageType>)
46 
47  public:
48  LinearKernel (const ssize_t n_SH,
49  const transform_type& linear_transform,
50  const Eigen::MatrixXd& directions,
51  const bool modulate) : fod (n_SH)
52  {
53  Eigen::MatrixXd transformed_directions = linear_transform.linear().inverse() * directions;
54 
55  if (modulate) {
56  Eigen::VectorXd modulation_factors = transformed_directions.colwise().norm() / linear_transform.linear().inverse().determinant();
57  transformed_directions.colwise().normalize();
58  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * modulation_factors.asDiagonal()
59  * Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions));
60  } else {
61  transformed_directions.colwise().normalize();
62  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions)
63  * Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions));
64  }
65  }

67  void operator() (FODImageType& in, FODImageType& out)
68  {
69  in.index(3) = 0;
70  if (in.value() > 0.0) { // only reorient voxels that contain a FOD
71  fod = in.row(3);
72  fod = transform * fod;
73  out.row(3) = fod;
74  }

79  Eigen::VectorXd fod;

120  template <class FODImageType>
121  class NonLinearKernel { MEMALIGN(NonLinearKernel<FODImageType>)
122 
123  public:
124  NonLinearKernel (const ssize_t n_SH, Image<default_type>& warp, const Eigen::MatrixXd& directions, const bool modulate) :
125  n_SH (n_SH),
126  jacobian_adapter (warp),
127  directions (directions),
128  modulate (modulate),
129  FOD_to_aPSF_transform (Math::pinv (aPSF_weights_to_FOD_transform (n_SH, directions))),
130  fod (n_SH) {}
131 
132 
133  void operator() (FODImageType& image) {
134  image.index(3) = 0;
135  if (image.value() > 0) { // only reorient voxels that contain a FOD
136  for (size_t dim = 0; dim < 3; ++dim)
137  jacobian_adapter.index(dim) = image.index(dim);
138  Eigen::MatrixXd jacobian = jacobian_adapter.value().inverse().template cast<default_type>();
139  Eigen::MatrixXd transformed_directions = jacobian * directions;
140 
141  if (modulate) {
142  Eigen::MatrixXd modulation_factors = transformed_directions.colwise().norm() / jacobian.determinant();
143  transformed_directions.colwise().normalize();
144 
145  Eigen::MatrixXd temp = aPSF_weights_to_FOD_transform (n_SH, transformed_directions);
146  for (ssize_t i = 0; i < temp.cols(); ++i)
147  temp.col(i) = temp.col(i) * modulation_factors(0,i);
148 
149  transform.noalias() = temp * FOD_to_aPSF_transform;
150  } else {
151  transformed_directions.colwise().normalize();
152  transform.noalias() = aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * FOD_to_aPSF_transform;
153  }
154  fod = image.row(3);
155  fod = transform * fod;
156  image.row(3) = fod;
157  }
158  }
159  protected:
160  const ssize_t n_SH;
161  Adapter::Jacobian<Image<default_type> > jacobian_adapter;
162  const Eigen::MatrixXd& directions;
163  const bool modulate;
164  const Eigen::MatrixXd FOD_to_aPSF_transform;
165  Eigen::MatrixXd transform;
166  Eigen::VectorXd fod;
167  };
