47  public:
48  LinearKernel (const ssize_t n_SH,
49  const transform_type& linear_transform,
50  const Eigen::MatrixXd& directions,
51  const bool modulate)
52  {
53  Eigen::MatrixXd transformed_directions = linear_transform.linear().inverse() * directions;
54 
55  if (modulate) {
56  Eigen::VectorXd modulation_factors = transformed_directions.colwise().norm() / linear_transform.linear().inverse().determinant();
57  transformed_directions.colwise().normalize();
58  transform = (aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * modulation_factors.asDiagonal()

107  class NonLinearKernel {
108 
109  public:
110  NonLinearKernel (const ssize_t n_SH, WarpImageType& warp, const Eigen::MatrixXd& directions, const bool modulate) :
111  n_SH (n_SH),
112  jacobian_adapter (warp),
113  directions (directions),

117 
118  void operator() (FODImageType& image) {
119  image.index(3) = 0;
120  if (image.value() > 0) { // only reorient voxels that contain a FOD
121  for (size_t dim = 0; dim < 3; ++dim)
122  jacobian_adapter.index(dim) = image.index(dim);
123  Eigen::MatrixXd jacobian = jacobian_adapter.value().inverse().template cast<default_type>();
124  Eigen::MatrixXd transformed_directions = jacobian * directions;
125 
126  if (modulate) {
127  Eigen::MatrixXd modulation_factors = transformed_directions.colwise().norm() / jacobian.determinant();
128  transformed_directions.colwise().normalize();
129 
130  Eigen::MatrixXd temp = aPSF_weights_to_FOD_transform (n_SH, transformed_directions);
131  for (ssize_t i = 0; i < temp.cols(); ++i)
132  temp.col(i) = temp.col(i) * modulation_factors(0,i);
133 
134  transform = temp * FOD_to_aPSF_transform;
135  } else {
136  transformed_directions.colwise().normalize();
137  transform = aPSF_weights_to_FOD_transform (n_SH, transformed_directions) * FOD_to_aPSF_transform;
138  }
139 
140  image.row(3) = transform.cast<typename WarpImageType::value_type>() * image.row(3);
141  }
142  }

144  const ssize_t n_SH;
145  Adapter::Jacobian<WarpImageType> jacobian_adapter;
146  const Eigen::MatrixXd& directions;
147  const bool modulate;
