99  class LinearReorientKernel {
100 
101  public:
102  LinearReorientKernel (FODVoxelType& fod_image_in,
103  FODVoxelType& fod_image_out,
104  const Math::Matrix<float>& directions_transposed,
105  const Math::Matrix<float>& transform) :
106  fod_voxel_in (fod_image_in),
107  fod_voxel_out (fod_image_out),
108  fod_in (fod_image_in.dim(3)),
109  fod_out (fod_image_out.dim(3)) {
110 
111  Image::check_dimensions(fod_image_in, fod_image_out);
112 
113  Math::Matrix<float> forward_transform;
114  Math::LU::inv (forward_transform, transform.sub(0,3,0,3));
115 
116  Math::Matrix<float> transformed_directions;
117  Math::mult (transformed_directions, forward_transform, Math::transpose (directions_transposed));
118 
119  Math::Matrix<float> fod_to_aPSF_weights_transform;
120  precompute_FOD_to_aPSF_weights_transform (fod_image_in.dim(3), directions_transposed, fod_to_aPSF_weights_transform);
121 
122  Math::SH::aPSF<float> aPSF_generator (Math::SH::LforN (fod_image_in.dim(3)));
123  Math::Vector<float> aPSF;
124  Math::Matrix<float> aPSF_matrix (fod_image_in.dim(3), transformed_directions.columns());
125  for (size_t i = 0; i < transformed_directions.columns(); ++i) {
126  Point<float> dir (transformed_directions (0, i), transformed_directions (1, i), transformed_directions (2, i));
127  aPSF_generator (aPSF, dir);
128  for (int j = 0; j < fod_image_in.dim(3); ++j)
129  aPSF_matrix (j, i) = aPSF[j];
130  }
131  Math::mult (reorient_transform, aPSF_matrix, fod_to_aPSF_weights_transform);
132  }
133 
134  void operator() (const Image::Iterator& pos) {
135  Image::voxel_assign (fod_voxel_in, pos, 0, 3);
136  Image::voxel_assign (fod_voxel_out, pos, 0, 3);
137  fod_voxel_in[3] = 0;
138  if (fod_voxel_in.value() > 0) {
139  for (fod_voxel_in[3] = 0; fod_voxel_in[3] < (int)fod_in.size(); ++fod_voxel_in[3])
140  fod_in[fod_voxel_in[3]] = fod_voxel_in.value();
141  Math::mult (fod_out, reorient_transform, fod_in);
142  for (fod_voxel_out[3] = 0; fod_voxel_out[3] < (int)fod_out.size(); ++fod_voxel_out[3])
143  fod_voxel_out.value() = fod_out[fod_voxel_out[3]];
144  }
145  }

147  void precompute_FOD_to_aPSF_weights_transform (const int num_SH,
148  const Math::Matrix<float>& directions,
149  Math::Matrix<float>& fod_to_aPSF_weights_transform) {
150  Math::Matrix<float> aPSF_matrix (num_SH, directions.rows());
151  Math::SH::aPSF<float> aPSF_generator (Math::SH::LforN (num_SH));
152  Math::Vector<float> aPSF;
153  for (size_t i = 0; i < directions.rows(); ++i) {
154  Point<float> dir (directions (i, 0), directions (i, 1), directions (i, 2));
155  aPSF_generator (aPSF, dir);
156  aPSF_matrix.column(i) = aPSF;
157  }
158  Math::pinv (fod_to_aPSF_weights_transform, aPSF_matrix);
159  }
