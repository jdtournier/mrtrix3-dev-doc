91  class WildBootstrap {
92  public:
93  WildBootstrap (const Eigen::MatrixXf& hat_matrix) :
94  H (hat_matrix),
95  uniform_int (0, 1),
96  residuals (H.rows()),
97  log_signal (H.rows()) { }
98 
99  void operator() (float* data) {
100  for (ssize_t i = 0; i < residuals.size(); ++i)
101  log_signal[i] = data[i] > float (0.0) ? -std::log (data[i]) : float (0.0);
102 
103  residuals = H * log_signal;
104 
105  for (ssize_t i = 0; i < residuals.size(); ++i) {
106  residuals[i] = residuals[i] ? (data[i] - std::exp (-residuals[i])) : float(0.0);
107  data[i] += uniform_int (*rng) ? residuals[i] : -residuals[i];
108  }
109  }
