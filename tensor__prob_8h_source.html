143  class WildBootstrap {
144  public:
145  WildBootstrap (const Math::Matrix<value_type>& hat_matrix, Math::RNG& random_number_generator) :
146  H (hat_matrix),
147  rng (random_number_generator),
148  uniform_int (0, 1),
149  residuals (H.rows()),
150  log_signal (H.rows()) { }
151 
152  void operator() (value_type* data) {
153  for (size_t i = 0; i < residuals.size(); ++i)
154  log_signal[i] = data[i] > value_type (0.0) ? -std::log (data[i]) : value_type (0.0);
155 
156  mult (residuals, H, log_signal);
157 
158  for (size_t i = 0; i < residuals.size(); ++i) {
159  residuals[i] = std::exp (-residuals[i]) - data[i];
160  data[i] += uniform_int (rng) ? residuals[i] : -residuals[i];
161  }
162  }
163 
164  private:
165  const Math::Matrix<value_type>& H;
166  Math::RNG& rng;
167  std::uniform_int_distribution<> uniform_int;
168  Math::Vector<value_type> residuals, log_signal;
