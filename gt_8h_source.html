70  Stats(const double T0, const double T1, const uint64_t maxiter)
71  : Text(T1), Tint(T0), EextTot(0.0), EintTot(0.0), n_iter(0), n_max(maxiter),
72  progress("running MH sampler", n_max/ITER_BIGSTEP)
73  {
74  for (int k = 0; k != 5; k++)
75  n_gen[k] = n_acc[k] = 0;
76  alpha = std::pow(T1/T0, double(ITER_BIGSTEP)/double(n_max - n_max/FRAC_BURNIN - n_max/FRAC_PHASEOUT));
77  }

90  bool next() {
91  std::lock_guard<std::mutex> lock (mutex);
92  ++n_iter;
93  if (n_iter % ITER_BIGSTEP == 0) {
94  if ((n_iter >= n_max/FRAC_BURNIN) && (n_iter < n_max - n_max/FRAC_PHASEOUT))
95  Tint *= alpha;
96  progress++;
97  out << *this << std::endl;
98  }
99  return (n_iter < n_max);
100  }
