64  bool operator () (SetVoxelDir& in)
65  {
66  // For each voxel tract tangent, assign to a fixel
67  std::vector<int32_t> tract_fixel_indices;
68  for (SetVoxelDir::const_iterator i = in.begin(); i != in.end(); ++i) {
69  assign_pos_of (*i).to (fixel_indexer);
70  fixel_indexer.index(3) = 0;
71  int32_t first_index = fixel_indexer.value();
72  if (first_index >= 0) {
73  fixel_indexer.index(3) = 1;
74  int32_t last_index = first_index + fixel_indexer.value();
75  int32_t closest_fixel_index = -1;
76  value_type largest_dp = 0.0;
77  Eigen::Vector3f dir (i->get_dir());
78  dir.normalize();
79  for (int32_t j = first_index; j < last_index; ++j) {
80  value_type dp = std::abs (dir.dot (fixel_directions[j]));
81  if (dp > largest_dp) {
82  largest_dp = dp;
83  closest_fixel_index = j;
84  }
85  }
86  if (largest_dp > angular_threshold_dp) {
87  tract_fixel_indices.push_back (closest_fixel_index);
88  fixel_TDI[closest_fixel_index]++;
89  }
90  }
91  }
92

94  for (size_t j = i + 1; j < tract_fixel_indices.size(); j++) {
95  connectivity_matrix[tract_fixel_indices[i]][tract_fixel_indices[j]].value++;
