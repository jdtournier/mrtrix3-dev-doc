69  bool operator () (SetVoxelDir& in)
70  {
71  // For each voxel tract tangent, assign to a fixel
72  std::vector<int32_t> tract_fixel_indices;
73  for (SetVoxelDir::const_iterator i = in.begin(); i != in.end(); ++i) {
74  Image::Nav::set_pos (fixel_indexer, *i);
75  fixel_indexer[3] = 0;
76  int32_t first_index = fixel_indexer.value();
77  if (first_index >= 0) {
78  fixel_indexer[3] = 1;
79  int32_t last_index = first_index + fixel_indexer.value();
80  int32_t closest_fixel_index = -1;
81  value_type largest_dp = 0.0;
82  Point<value_type> dir (i->get_dir());
83  dir.normalise();
84  for (int32_t j = first_index; j < last_index; ++j) {
85  value_type dp = Math::abs (dir.dot (fixel_directions[j]));
86  if (dp > largest_dp) {
87  largest_dp = dp;
88  closest_fixel_index = j;
89  }
90  }
91  if (largest_dp > angular_threshold_dp) {
92  tract_fixel_indices.push_back (closest_fixel_index);
93  fixel_TDI[closest_fixel_index]++;
94  }
95  }
96  }
97 
98  for (size_t i = 0; i < tract_fixel_indices.size(); i++) {
99  for (size_t j = i + 1; j < tract_fixel_indices.size(); j++) {
100  connectivity_matrix[tract_fixel_indices[i]][tract_fixel_indices[j]].value++;
101  connectivity_matrix[tract_fixel_indices[j]][tract_fixel_indices[i]].value++;
102  }
103  }
104 
105 
106  return true;
107  }
