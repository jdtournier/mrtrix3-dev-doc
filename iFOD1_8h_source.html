178  {
179  if (!get_data (source))
180  return EXIT_IMAGE;

182  value_type max_val = 0.0;
183  size_t nan_count = 0;
184  for (size_t i = 0; i < calibrate_list.size(); ++i) {
185  value_type val = FOD (rotate_direction (dir, calibrate_list[i]));
186  if (std::isnan (val))
187  ++nan_count;
188  else if (val > max_val)
189  max_val = val;
190  }
191 
192  if (nan_count == calibrate_list.size())
193  return EXIT_IMAGE;
194 
195  if (max_val <= 0.0)
196  return CALIBRATE_FAIL;
197 
198  max_val *= calibrate_ratio;
199 
200  num_sample_runs++;
201 
202  for (size_t n = 0; n < S.max_trials; n++) {
203  Point<value_type> new_dir = rand_dir (dir);
204  value_type val = FOD (new_dir);
205 
206  if (val > S.threshold) {
207
