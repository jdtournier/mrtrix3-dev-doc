119  struct ThreadFunctor { MEMALIGN(ThreadFunctor)
120  public:

151  void operator() (const Iterator& iter) {
152  auto engine = std::default_random_engine{static_cast<std::default_random_engine::result_type>(rng.get_seed())};
153  auto kern = ThreadKernel<MetricType, ParamType> (metric, params, cost_function, gradient, overlap_count);
154  Iterator iterator (iter);
155  assign_pos_of(iter).to(iterator);
156  auto inner_loop = Random_loop<Iterator, std::default_random_engine>(iterator, engine, inner_axes[1], (float) iterator.size(inner_axes[1]) * density);
157  for (auto j = inner_loop; j; ++j)
158  for (auto k = Loop (inner_axes[0]) (iterator); k; ++k) {
159  kern (iterator);
160  }
161  }
162  protected:
163  vector<size_t> inner_axes;
164  default_type density;
165  MetricType metric;
166  ParamType params;
167  Eigen::VectorXd cost_function;
168  Eigen::VectorXd gradient;
169  Eigen::VectorXd& overall_cost_function;
170  Eigen::VectorXd& overall_gradient;
171  Math::RNG rng;
