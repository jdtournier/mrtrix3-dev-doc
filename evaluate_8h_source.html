126  void operator() (const Iterator& iter) {
127  auto engine = std::default_random_engine{static_cast<std::default_random_engine::result_type>(rng.get_seed())};
128  auto kern = ThreadKernel<MetricType, ParamType> (metric, params, cost_function, gradient);
