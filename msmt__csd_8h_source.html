89  void init()
90  {
91  if (lmax.empty()) {
92  lmax = lmax_response;
93  } else {
94  if (lmax.size() != num_tissues())
95  throw Exception ("Number of lmaxes specified does not match number of tissues");
96  for (const auto i : lmax) {
97  if (i < 0 || i % 2)
98  throw Exception ("Each value of lmax must be a non-negative even integer");
99  }
100  }
101 
102  for (size_t t = 0; t != num_tissues(); ++t) {
103  if (size_t(responses[t].rows()) != num_shells())
104  throw Exception ("number of rows in response function must match number of b-value shells");
105  // Pad response functions out to the requested lmax for this tissue
106  responses[t].conservativeResizeLike (Eigen::MatrixXd::Zero (num_shells(), lmax[t]/2+1));
107  }
108 
110  // Set up the constrained least squares problem //
112 
113  size_t nparams = 0;
114  int maxlmax = 0;
115  for (size_t i = 0; i < num_tissues(); i++) {
116  nparams += Math::SH::NforL (lmax[i]);
117  maxlmax = std::max (maxlmax, lmax[i]);
118  }
119 
120  INFO ("initialising multi-tissue CSD for " + str(num_tissues()) + " tissue types, with " + str (nparams) + " parameters");
121 
122  Eigen::MatrixXd C (grad.rows(), nparams);
123 
124  std::vector<size_t> dwilist;
125  for (size_t i = 0; i != size_t(grad.rows()); i++)
126  dwilist.push_back(i);
127 
128  Eigen::MatrixXd directions = DWI::gen_direction_matrix (grad, dwilist);
129  Eigen::MatrixXd SHT = Math::SH::init_transform (directions, maxlmax);
130  for (ssize_t i = 0; i < SHT.rows(); i++)
131  for (ssize_t j = 0; j < SHT.cols(); j++)
132  if (std::isnan (SHT(i,j)))
133  SHT(i,j) = 0.0;
134 
135  // TODO: is this just computing the Associated Legrendre polynomials...?
136  Eigen::MatrixXd delta(1,2); delta << 0, 0;
137  Eigen::MatrixXd DSH__ = Math::SH::init_transform (delta, maxlmax);
138  Eigen::VectorXd DSH_ = DSH__.row(0);
139  Eigen::VectorXd DSH (maxlmax/2+1);
140  size_t j = 0;
141  for (ssize_t i = 0; i < DSH_.size(); i++)
142  if (DSH_[i] != 0.0) {
143  DSH[j] = DSH_[i];
144  j++;
145  }
146 
147  size_t pbegin = 0;
148  for (size_t tissue_idx = 0; tissue_idx < num_tissues(); ++tissue_idx) {
149  const size_t tissue_lmax = lmax[tissue_idx];
150  const size_t tissue_n = Math::SH::NforL (tissue_lmax);
151  const size_t tissue_nmzero = tissue_lmax/2+1;
152 
153  for (size_t shell_idx = 0; shell_idx < num_shells(); ++shell_idx) {
154  Eigen::VectorXd response_ = responses[tissue_idx].row (shell_idx);
155  response_ = (response_.array() / DSH.head (tissue_nmzero).array()).matrix();
156  Eigen::VectorXd fconv (tissue_n);
157  int li = 0; int mi = 0;
158  for (int l = 0; l <= int(tissue_lmax); l+=2) {
159  for (int m = -l; m <= l; m++) {
160  fconv[mi] = response_[li];
161  mi++;
162  }
163  li++;
164  }
165  std::vector<size_t> vols = shells[shell_idx].get_volumes();
166  for (size_t idx = 0; idx < vols.size(); idx++) {
167  Eigen::VectorXd SHT_(SHT.row (vols[idx]).head (tissue_n));
168  SHT_ = (SHT_.array()*fconv.array()).matrix();
169  C.row (vols[idx]).segment (pbegin, tissue_n) = SHT_;
170  }
171  }
172  pbegin += tissue_n;
173  }
174 
175  std::vector<size_t> m (num_tissues());
176  std::vector<size_t> n (num_tissues());
177  size_t M = 0;
178  size_t N = 0;
179 
180  Eigen::MatrixXd HR_SHT = Math::SH::init_transform (HR_dirs, maxlmax);
181 
182  for (size_t i = 0; i != num_tissues(); i++) {
183  if (lmax[i] > 0)
184  m[i] = HR_dirs.rows();
185  else
186  m[i] = 1;
187  M += m[i];
188  n[i] = Math::SH::NforL (lmax[i]);
189  N += n[i];
190  }
191 
192  Eigen::MatrixXd A (Eigen::MatrixXd::Zero (M, N));
193  size_t b_m = 0; size_t b_n = 0;
194  for (size_t i = 0; i != num_tissues(); i++) {
195  A.block (b_m,b_n,m[i],n[i]) = HR_SHT.block (0,0,m[i],n[i]);
196  b_m += m[i];
197  b_n += n[i];
198  }
199 
200  problem = Math::ICLS::Problem<double> (C, A, 1.0e-10, 1.0e-10);
201 
202  INFO ("Multi-shell, multi-tissue CSD initialised successfully");
203  }

220  void prepare_responses() {
221  for (size_t t = 0; t != num_tissues(); ++t) {
222  Eigen::MatrixXd& r (responses[t]);
223  size_t n = 0;
224  for (size_t row = 0; row < size_t(r.rows()); row++) {
225  for (size_t col = 0; col < size_t(r.cols()); col++) {
226  if (r(row,col))
227  n = std::max (n, col+1);
228  }
229  }
230  // Clip off any empty columns, i.e. degrees containing zero coefficients for all shells
231  r.conservativeResize (r.rows(), n);
232  // Store the lmax for each tissue based on their response functions;
233  // if the user doesn't manually specify lmax, these will determine
234  // the lmax of each tissue ODF output
235  lmax_response.push_back ((r.cols()-1)*2);
236  }
237  }
