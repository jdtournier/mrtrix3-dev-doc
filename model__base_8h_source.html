191  template <class Fixel>
192  void ModelBase<Fixel>::perform_FOD_segmentation (Image<float>& data)
193  {
194  Math::SH::check (data);
195  DWI::FMLS::FODQueueWriter writer (data, proc_mask);
196  DWI::FMLS::Segmenter fmls (dirs, Math::SH::LforN (data.size(3)));
197  fmls.set_dilate_lookup_table (!App::get_options ("no_dilate_lut").size());
198  fmls.set_create_null_lobe (App::get_options ("make_null_lobes").size());
199  Thread::run_queue (writer, Thread::batch (FMLS::SH_coefs()), Thread::multi (fmls), Thread::batch (FMLS::FOD_lobes()), *this);
200  have_null_lobes = fmls.get_create_null_lobe();
201  }

206  template <class Fixel>
207  void ModelBase<Fixel>::scale_FDs_by_GM ()
208  {
209  if (!App::get_options("fd_scale_gm").size())
210  return;
211  if (!act_5tt.valid()) {
212  INFO ("Cannot scale fibre densities according to GM fraction; no ACT image data provided");
213  return;
214  }
215  // Loop through voxels, getting total GM fraction for each, and scale all fixels in each voxel
216  VoxelAccessor v (accessor());
217  FOD_sum = 0.0;
218  for (auto l = Loop(v) (v, act_5tt); l; ++l) {
219  Tractography::ACT::Tissues tissues (act_5tt);
220  const default_type multiplier = 1.0 - tissues.get_cgm() - (0.5 * tissues.get_sgm()); // Heuristic
221  for (typename Fixel_map<Fixel>::Iterator i = begin(v); i; ++i) {
222  i().scale_FOD (multiplier);
223  FOD_sum += i().get_weight() * i().get_FOD();
224  }
225  }
226  }

231  template <class Fixel>
232  void ModelBase<Fixel>::map_streamlines (const std::string& path)
233  {
234  Tractography::Properties properties;
235  Tractography::Reader<> file (path, properties);
236 
237  const track_t count = (properties.find ("count") == properties.end()) ? 0 : to<track_t>(properties["count"]);
238  if (!count)
239  throw Exception ("Cannot map streamlines: track file " + Path::basename(path) + " is empty");
240 
241  Mapping::TrackLoader loader (file, count);
242  Mapping::TrackMapperBase mapper (Fixel_map<Fixel>::header(), dirs);
243  mapper.set_upsample_ratio (Mapping::determine_upsample_ratio (Fixel_map<Fixel>::header(), properties, 0.1));
244  mapper.set_use_precise_mapping (true);
245  Thread::run_queue (
246  loader,
247  Thread::batch (Tractography::Streamline<float>()),
248  Thread::multi (mapper),
249  Thread::batch (Mapping::SetDixel()),
250  *this);
251 
252  INFO ("Proportionality coefficient after streamline mapping is " + str (mu()));
253  }

258  template <class Fixel>
259  bool ModelBase<Fixel>::operator() (const FMLS::FOD_lobes& in)
260  {
261  if (!Fixel_map<Fixel>::operator() (in))
262  return false;
263 
264  VoxelAccessor v (accessor());
265  assign_pos_of (in.vox).to (v);
266 
267  if (v.value()) {
268  assign_pos_of (in.vox).to (proc_mask);
269  const float mask_value = proc_mask.value();
270 
271  for (typename Fixel_map<Fixel>::Iterator i = begin (v); i; ++i) {
272  i().set_weight (mask_value);
273  FOD_sum += i().get_FOD() * mask_value;
274  }
275  }
276  return true;
277  }

282  template <class Fixel>
283  bool ModelBase<Fixel>::operator() (const Mapping::SetDixel& in)
284  {
285  default_type total_contribution = 0.0;
286  for (Mapping::SetDixel::const_iterator i = in.begin(); i != in.end(); ++i) {
287  const size_t fixel_index = Mapping::Fixel_TD_map<Fixel>::dixel2fixel (*i);
288  if (fixel_index) {
289  fixels[fixel_index] += i->get_length();
290  total_contribution += fixels[fixel_index].get_weight() * i->get_length();
291  }
292  }
293  TD_sum += total_contribution;
294  return true;
295  }
