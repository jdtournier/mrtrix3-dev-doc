227  void ModelBase<Fixel>::perform_FOD_segmentation (BufferType& data)
228  {
229  auto data_vox = data.voxel();
230  DWI::FMLS::FODQueueWriter<decltype(data_vox), decltype(proc_mask)> writer (data_vox, proc_mask);
231  DWI::FMLS::Segmenter fmls (dirs, Math::SH::LforN (data.dim(3)));
232  fmls.set_dilate_lookup_table (!App::get_options ("no_dilate_lut").size());
233  fmls.set_create_null_lobe (App::get_options ("make_null_lobes").size());

241  template <class Fixel>
242  void ModelBase<Fixel>::scale_FDs_by_GM ()
243  {
244  if (App::get_options("no_fd_scaling").size())
245  return;
246  if (!act_5tt) {
247  INFO ("Cannot scale fibre densities according to GM fraction; no ACT image data provided");
248  return;
249  }
250  // Loop through voxels, getting total GM fraction for each, and scale all fixels in each voxel
251  VoxelAccessor v (accessor);
252  auto v_anat = act_5tt->voxel();
253  FOD_sum = 0.0;
254  for (auto l = Image::LoopInOrder(v) (v, v_anat); l; ++l) {
255  Tractography::ACT::Tissues tissues (v_anat);
256  const float multiplier = 1.0 - tissues.get_cgm() - (0.5 * tissues.get_sgm()); // Heuristic
257  for (typename Fixel_map<Fixel>::Iterator i = begin(v); i; ++i) {
258  i().scale_FOD (multiplier);
259  FOD_sum += i().get_weight() * i().get_FOD();
260  }
261  }
262  }

268  void ModelBase<Fixel>::map_streamlines (const std::string& path)
269  {
270  Tractography::Properties properties;
271  Tractography::Reader<float> file (path, properties);
272 
273  const track_t count = (properties.find ("count") == properties.end()) ? 0 : to<track_t>(properties["count"]);
274 
275  Mapping::TrackLoader loader (file, count);
276  Mapping::TrackMapperBase mapper (H, dirs);
277  mapper.set_upsample_ratio (Mapping::determine_upsample_ratio (H, properties, 0.1));
278  mapper.set_use_precise_mapping (true);
279  Thread::run_queue (
280  loader,
281  Thread::batch (Tractography::Streamline<float>()),
282  Thread::multi (mapper),
283  Thread::batch (Mapping::SetDixel()),
284  *this);
285 
286  INFO ("Proportionality coefficient after streamline mapping is " + str (mu()));
287  }

292  template <class Fixel>
293  bool ModelBase<Fixel>::operator() (const FMLS::FOD_lobes& in)
294  {
295  if (!Fixel_map<Fixel>::operator() (in))
296  return false;
297  const float mask_value = Image::Nav::get_value_at_pos (proc_mask, in.vox);
298  VoxelAccessor v (accessor);
299  Image::Nav::set_pos (v, in.vox);
300  if (v.value()) {
301  for (typename Fixel_map<Fixel>::Iterator i = begin (v); i; ++i) {
302  i().set_weight (mask_value);
303  FOD_sum += i().get_FOD() * mask_value;
304  }
305  }
306  return true;
307  }

312  template <class Fixel>
313  bool ModelBase<Fixel>::operator() (const Mapping::SetDixel& in)
314  {
315  float total_contribution = 0.0;
316  for (Mapping::SetDixel::const_iterator i = in.begin(); i != in.end(); ++i) {
317  const size_t fixel_index = Mapping::Fixel_TD_map<Fixel>::dixel2fixel (*i);
318  if (fixel_index) {
319  fixels[fixel_index] += i->get_length();
320  total_contribution += fixels[fixel_index].get_weight() * i->get_length();
321  }
322  }
323  TD_sum += total_contribution;
324  return true;
325  }
