114  void init ()
115  {
116  using namespace Math::SH;
117 
118  if (lmax_data <= 0)
119  throw Exception ("data contain too few directions even for lmax = 2");
120 
121  if (lmax_response <= 0)
122  throw Exception ("response function does not contain anisotropic terms");
123 
124  lmax = ( lmax_cmdline ? lmax_cmdline : std::min (lmax_response, DEFAULT_CSD_LMAX) );
125 
126  if (lmax <= 0 || lmax % 2)
127  throw Exception ("lmax must be a positive even integer");
128 
129  assert (response.size());
130  lmax_response = std::min (lmax_response, std::min (lmax_data, lmax));
131  INFO ("calculating even spherical harmonic components up to order " + str (lmax_response) + " for initialisation");
132 
133  if (!init_filter.size())
134  init_filter = Eigen::VectorXd::Ones(3);
135  init_filter.conservativeResizeLike (Eigen::VectorXd::Zero (Math::ZSH::NforL (lmax_response)));
136 
137  auto RH = Math::ZSH::ZSH2RH (response);
138  if (size_t(RH.size()) < Math::ZSH::NforL (lmax))
139  RH.conservativeResizeLike (Eigen::VectorXd::Zero (Math::ZSH::NforL (lmax)));
140 
141  // inverse sdeconv for initialisation:
142  auto fconv = init_transform (DW_dirs, lmax_response);
143  rconv.resize (fconv.cols(), fconv.rows());
144  fconv.diagonal().array() += 1.0e-2;
145  //fconv.save ("fconv.txt");
146  rconv = Math::pinv (fconv);
147  //rconv.save ("rconv.txt");
148  ssize_t l = 0, nl = 1;
149  for (ssize_t row = 0; row < rconv.rows(); ++row) {
150  if (row >= nl) {
151  l++;
152  nl = NforL (2*l);
153  }
154  rconv.row (row).array() *= init_filter[l] / RH[l];
155  }
156 
157  // forward sconv for iteration, using all response function
158  // coefficients up to the requested lmax:
159  INFO ("calculating even spherical harmonic components up to order " + str (lmax) + " for output");
160  fconv = init_transform (DW_dirs, lmax);
161  l = 0;
162  nl = 1;
163  for (ssize_t col = 0; col < fconv.cols(); ++col) {
164  if (col >= nl) {
165  l++;
166  nl = NforL (2*l);
167  }
168  fconv.col (col).array() *= RH[l];
169  }
170 
171  // high-res sampling to apply constraint:
172  HR_trans = init_transform (HR_dirs, lmax);
173  default_type constraint_multiplier = neg_lambda * 50.0 * response[0] / default_type (HR_trans.rows());
174  HR_trans.array() *= constraint_multiplier;
175 
176  // adjust threshold accordingly:
177  threshold *= constraint_multiplier;
178 
179  // precompute as much as possible ahead of Cholesky decomp:
180  assert (fconv.cols() <= HR_trans.cols());
181  M.resize (DW_dirs.rows(), HR_trans.cols());
182  M.leftCols (fconv.cols()) = fconv;
183  M.rightCols (M.cols() - fconv.cols()).setZero();
184  Mt_M.resize (M.cols(), M.cols());
185  Mt_M.triangularView<Eigen::Lower>() = M.transpose() * M;
186 
187 
188  // min-norm constraint:
189  if (norm_lambda) {
190  norm_lambda *= NORM_LAMBDA_MULTIPLIER * Mt_M (0,0);
191 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
192  Mt_M.diagonal().array() += norm_lambda;
193 #else
194  int l = 0;
195  for (size_t i = 0; i < Mt_M.rows(); ++i) {
196  if (Math::SH::index (l,0) == i) {
197  Mt_M(i,i) += norm_lambda;
198  l+=2;
199  }
200  else
201  Mt_M(i,i) += 0.5 * norm_lambda;
202  }
203 #endif
204  }
205 
206  INFO ("constrained spherical deconvolution initialised successfully");
207  }

245  template <class VectorType>
246  void set (const VectorType& DW_signals) {
247  F.head (shared.rconv.rows()) = shared.rconv * DW_signals;
248  F.tail (F.size()-shared.rconv.rows()).setZero();

254  bool iterate() {
255  neg.clear();
256  HR_amps = shared.HR_trans * F;
257  for (ssize_t n = 0; n < HR_amps.size(); n++)
258  if (HR_amps[n] < shared.threshold)
259  neg.push_back (n);
260 
261  if (old_neg == neg)
262  return true;
263 
264  work.triangularView<Eigen::Lower>() = shared.Mt_M.triangularView<Eigen::Lower>();
265 
266  if (neg.size()) {
267  for (size_t i = 0; i < neg.size(); i++)
268  HR_T.row (i) = shared.HR_trans.row (neg[i]);
269  auto HR_T_view = HR_T.topRows (neg.size());
270  work.triangularView<Eigen::Lower>() += HR_T_view.transpose() * HR_T_view;
271  }
272 
273  F.noalias() = llt.compute (work.triangularView<Eigen::Lower>()).solve (Mt_b);
274 
275  old_neg = neg;
276 
277  return false;
278  }
