86  void calc_distance ()
87  {
88  if (!std::isfinite (orientation_z[0]))
89  orientation_z = orientation_x.cross (orientation_y);
90  else {
91  Eigen::Vector3 normal = orientation_x.cross (orientation_y);
92  if (normal.dot (orientation_z) < 0.0)
93  orientation_z = -normal;
94  else
95  orientation_z = normal;
