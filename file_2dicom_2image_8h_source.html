71  bool operator< (const Frame& frame) const {
72  if (series_num != frame.series_num)
73  return series_num < frame.series_num;
74  if (image_type != frame.image_type)
75  return image_type < frame.image_type;
76  if (acq != frame.acq)
77  return acq < frame.acq;
78  if (std::isfinite (distance) && std::isfinite (frame.distance) && distance != frame.distance)
79  return distance < frame.distance;
80  for (size_t n = index.size(); n--;)
81  if (index[n] != frame.index[n])
82  return index[n] < frame.index[n];
83  if (sequence != frame.sequence)
84  return sequence < frame.sequence;
85  if (instance != frame.instance)
86  return instance < frame.instance;
87  return false;
88  }

91  void calc_distance ()
92  {
93  if (!std::isfinite (orientation_z[0]))
94  orientation_z = orientation_x.cross (orientation_y);
95  else {
96  if (!orientation_x.allFinite() || !orientation_y.allFinite())
97  throw Exception ("slice orientation information missing from DICOM header!");
98  Eigen::Vector3 normal = orientation_x.cross (orientation_y);
99  if (normal.dot (orientation_z) < 0.0)
100  orientation_z = -normal;
101  else
102  orientation_z = normal;
103  }
104 
105  if (!position_vector.allFinite())
106  throw Exception ("slice position information missing from DICOM header!");
107 
108  orientation_z.normalize();
109  distance = orientation_z.dot (position_vector);
110  }
