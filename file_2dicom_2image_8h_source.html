56  bool operator< (const Frame& frame) const {
57  if (series_num != frame.series_num)
58  return series_num < frame.series_num;
59  if (acq != frame.acq)
60  return acq < frame.acq;
61  assert (std::isfinite (distance));

76  void calc_distance ()
77  {
78  if (!std::isfinite (orientation_z[0]))
79  orientation_z = orientation_x.cross (orientation_y);
80  else {
81  Eigen::Vector3 normal = orientation_x.cross (orientation_y);
82  if (normal.dot (orientation_z) < 0.0)
83  orientation_z = -normal;
84  else
85  orientation_z = normal;
86  }
87 
88  orientation_z.normalize();
89  distance = orientation_z.dot (position_vector);
90  }
