84  template <class MatrixType>
85  Eigen::Matrix<typename MatrixType::Scalar,Eigen::Dynamic, Eigen::Dynamic> init_transform (const MatrixType& dirs, int lmax)
86  {
87  using namespace Eigen;
88  typedef typename MatrixType::Scalar value_type;
89  if (dirs.cols() != 2)
90  throw Exception ("direction matrix should have 2 columns: [ azimuth elevation ]");
91  Matrix<value_type,Dynamic,Dynamic> SHT (dirs.rows(), NforL (lmax));
92  Matrix<value_type,Dynamic,1,0,64> AL (lmax+1);
93  for (ssize_t i = 0; i < dirs.rows(); i++) {
94  value_type x = std::cos (dirs (i,1));
95  Legendre::Plm_sph (AL, lmax, 0, x);
96  for (int l = 0; l <= lmax; l+=2)
97  SHT (i,index (l,0)) = AL[l];
98  for (int m = 1; m <= lmax; m++) {
99  Legendre::Plm_sph (AL, lmax, m, x);
100  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
101 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
102  SHT(i, index(l, m)) = Math::sqrt2 * AL[l]*std::cos (m*dirs (i,0));
103  SHT(i, index(l,-m)) = Math::sqrt2 * AL[l]*std::sin (m*dirs (i,0));
104 #else
105  SHT(i, index(l, m)) = AL[l]*std::cos (m*dirs (i,0));
106  SHT(i, index(l,-m)) = AL[l]*std::sin (m*dirs (i,0));
107 #endif
108  }
109  }
110  }
111  return SHT;
112  }

204  template <class VectorType>
205  inline typename VectorType::Scalar value (const VectorType& coefs,
206  typename VectorType::Scalar cos_elevation,
207  typename VectorType::Scalar cos_azimuth,
208  typename VectorType::Scalar sin_azimuth,
209  int lmax)
210  {
211  typedef typename VectorType::Scalar value_type;
212  value_type amplitude = 0.0;
213  Eigen::Matrix<value_type,Eigen::Dynamic,1,0,64> AL (lmax+1);
214  Legendre::Plm_sph (AL, lmax, 0, cos_elevation);
215  for (int l = 0; l <= lmax; l+=2)
216  amplitude += AL[l] * coefs[index (l,0)];
217  value_type c0 (1.0), s0 (0.0);
218  for (int m = 1; m <= lmax; m++) {
219  Legendre::Plm_sph (AL, lmax, m, cos_elevation);
220  value_type c = c0 * cos_azimuth - s0 * sin_azimuth; // std::cos(m*azimuth)
221  value_type s = s0 * cos_azimuth + c0 * sin_azimuth; // std::sin(m*azimuth)
222  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
223 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
224  amplitude += AL[l] * Math::sqrt2 * (c * coefs[index (l,m)] + s * coefs[index (l,-m)]);
225 #else
226  amplitude += AL[l] * (c * coefs[index (l,m)] + s * coefs[index (l,-m)]);
227 #endif
228  }
229  c0 = c;
230  s0 = s;
231  }
232  return amplitude;
233  }

269  Legendre::Plm_sph (AL, lmax, m, unit_dir[2]);
270  value_type c = c0 * cp - s0 * sp;
271  value_type s = s0 * cp + c0 * sp;
272  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {

275  delta_vec[index (l,-m)] = AL[l] * Math::sqrt2 * s;
276 #else

278  delta_vec[index (l,-m)] = AL[l] * 2.0 * s;
279 #endif
280  }
281  c0 = c;
282  s0 = s;
283  }

429  inline VectorType& FA2SH (
430  VectorType& sh, default_type FA, default_type ADC, default_type bvalue, int lmax, int precision = 100)
431  {
432  default_type a = FA/sqrt (3.0 - 2.0*FA*FA);
433  default_type ev1 = ADC* (1.0+2.0*a), ev2 = ADC* (1.0-a);

440  default_type el = i*Math::pi / (2.0* (precision-1));
441  sigs[i] = exp (-bvalue* (ev1*std::cos (el) *std::cos (el) + ev2*std::sin (el) *std::sin (el)));

583  template <class VectorType, class UnitVectorType, class ValueType = float>
584  inline typename VectorType::Scalar get_peak (
585  const VectorType& sh,
586  int lmax,
587  UnitVectorType& unit_init_dir,
588  PrecomputedAL<typename VectorType::Scalar>* precomputer = nullptr)
589  {
590  typedef typename VectorType::Scalar value_type;
591  assert (std::isfinite (unit_init_dir[0]));
592  for (int i = 0; i < 50; i++) {
593  value_type az = std::atan2 (unit_init_dir[1], unit_init_dir[0]);
594  value_type el = std::acos (unit_init_dir[2]);
595  value_type amplitude, dSH_del, dSH_daz, d2SH_del2, d2SH_deldaz, d2SH_daz2;
596  derivatives (sh, lmax, el, az, amplitude, dSH_del, dSH_daz, d2SH_del2, d2SH_deldaz, d2SH_daz2, precomputer);
597 
598  value_type del = sqrt (dSH_del*dSH_del + dSH_daz*dSH_daz);
599  value_type daz = dSH_daz/del;
600  del = dSH_del/del;
601 
602  value_type dSH_dt = daz*dSH_daz + del*dSH_del;
603  value_type d2SH_dt2 = daz*daz*d2SH_daz2 + 2.0*daz*del*d2SH_deldaz + del*del*d2SH_del2;
604  value_type dt = d2SH_dt2 ? (-dSH_dt / d2SH_dt2) : 0.0;
605 
606  if (dt < 0.0) dt = -dt;
607  if (dt > MAX_DIR_CHANGE) dt = MAX_DIR_CHANGE;
608 
609  del *= dt;
610  daz *= dt;
611 
612  unit_init_dir[0] += del*std::cos (az) *std::cos (el) - daz*std::sin (az);
613  unit_init_dir[1] += del*std::sin (az) *std::cos (el) + daz*std::cos (az);
614  unit_init_dir[2] -= del*std::sin (el);
615  unit_init_dir.normalize();

633  template <class VectorType>
634  inline void derivatives (
635  const VectorType& sh,
636  const int lmax,
637  const typename VectorType::Scalar elevation,
638  const typename VectorType::Scalar azimuth,
639  typename VectorType::Scalar& amplitude,
640  typename VectorType::Scalar& dSH_del,
641  typename VectorType::Scalar& dSH_daz,
642  typename VectorType::Scalar& d2SH_del2,
643  typename VectorType::Scalar& d2SH_deldaz,
644  typename VectorType::Scalar& d2SH_daz2,
645  PrecomputedAL<typename VectorType::Scalar>* precomputer)
646  {
647  typedef typename VectorType::Scalar value_type;
648  value_type sel = std::sin (elevation);
649  value_type cel = std::cos (elevation);
650  bool atpole = sel < 1e-4;
651 
652  dSH_del = dSH_daz = d2SH_del2 = d2SH_deldaz = d2SH_daz2 = 0.0;
653  VLA_MAX (AL, value_type, NforL_mpos (lmax), 64);
654 
655  if (precomputer) {
656  PrecomputedFraction<value_type> f;
657  precomputer->set (f, elevation);
658  precomputer->get (AL, f);
659  }
660  else {
661  Eigen::Matrix<value_type,Eigen::Dynamic,1,0,64> buf (lmax+1);
662  for (int m = 0; m <= lmax; m++) {
663  Legendre::Plm_sph (buf, lmax, m, cel);
664  for (int l = ( (m&1) ?m+1:m); l <= lmax; l+=2)
665  AL[index_mpos (l,m)] = buf[l];
666  }
667  }
668 
669  amplitude = sh[0] * AL[0];
670  for (int l = 2; l <= (int) lmax; l+=2) {
671  const value_type& v (sh[index (l,0)]);
672  amplitude += v * AL[index_mpos (l,0)];
673  dSH_del += v * sqrt (value_type (l* (l+1))) * AL[index_mpos (l,1)];
674  d2SH_del2 += v * (sqrt (value_type (l* (l+1) * (l-1) * (l+2))) * AL[index_mpos (l,2)] - l* (l+1) * AL[index_mpos (l,0)]) /2.0;
675  }
676 
677  for (int m = 1; m <= lmax; m++) {
678 #ifndef USE_NON_ORTHONORMAL_SH_BASIS
679  value_type caz = Math::sqrt2 * std::cos (m*azimuth);
680  value_type saz = Math::sqrt2 * std::sin (m*azimuth);
681 #else
682  value_type caz = std::cos (m*azimuth);
683  value_type saz = std::sin (m*azimuth);
684 #endif
685  for (int l = ( (m&1) ? m+1 : m); l <= lmax; l+=2) {
686  const value_type& vp (sh[index (l,m)]);
687  const value_type& vm (sh[index (l,-m)]);
688  amplitude += (vp*caz + vm*saz) * AL[index_mpos (l,m)];
689 
690  value_type tmp = sqrt (value_type ( (l+m) * (l-m+1))) * AL[index_mpos (l,m-1)];
691  if (l > m) tmp -= sqrt (value_type ( (l-m) * (l+m+1))) * AL[index_mpos (l,m+1)];
692  tmp /= -2.0;
693  dSH_del += (vp*caz + vm*saz) * tmp;
694 
695  value_type tmp2 = - ( (l+m) * (l-m+1) + (l-m) * (l+m+1)) * AL[index_mpos (l,m)];
696  if (m == 1) tmp2 -= sqrt (value_type ( (l+m) * (l-m+1) * (l+m-1) * (l-m+2))) * AL[index_mpos (l,1)];
697  else tmp2 += sqrt (value_type ( (l+m) * (l-m+1) * (l+m-1) * (l-m+2))) * AL[index_mpos (l,m-2)];
698  if (l > m+1) tmp2 += sqrt (value_type ( (l-m) * (l+m+1) * (l-m-1) * (l+m+2))) * AL[index_mpos (l,m+2)];
699  tmp2 /= 4.0;
700  d2SH_del2 += (vp*caz + vm*saz) * tmp2;
701 
702  if (atpole) dSH_daz += (vm*caz - vp*saz) * tmp;
703  else {
704  d2SH_deldaz += m * (vm*caz - vp*saz) * tmp;
705  dSH_daz += m * (vm*caz - vp*saz) * AL[index_mpos (l,m)];
706  d2SH_daz2 -= (vp*caz + vm*saz) * m*m * AL[index_mpos (l,m)];
707  }
708 
709  }
710  }
711 
712  if (!atpole) {
713  dSH_daz /= sel;
714  d2SH_deldaz /= sel;
715  d2SH_daz2 /= sel*sel;
716  }
717  }
