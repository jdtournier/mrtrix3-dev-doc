192  {

222  template <class InfoType>
223  void sanitise (InfoType& info)
224  {
225  // remove duplicates
226  for (size_t i = 0; i < info.ndim()-1; ++i) {
227  if (!info.stride (i)) continue;
228  for (size_t j = i+1; j < info.ndim(); ++j) {
229  if (!info.stride (j)) continue;
230  if (std::abs (info.stride (i)) == std::abs (info.stride (j))) info.stride (j) = 0;
231  }
232  }
233 
234  size_t max = 0;
235  for (size_t i = 0; i < info.ndim(); ++i)
236  if (size_t (std::abs (info.stride (i))) > max)
237  max = std::abs (info.stride (i));
238 
239  for (size_t i = 0; i < info.ndim(); ++i) {
240  if (info.stride (i)) continue;
241  info.stride (i) = ++max;
242  }
243  }
