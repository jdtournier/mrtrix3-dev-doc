217  template <class Set> void sanitise (Set& set)
218  {
219  for (size_t i = 0; i < set.ndim()-1; ++i) {
220  if (!set.stride (i)) continue;
221  for (size_t j = i+1; j < set.ndim(); ++j) {
222  if (!set.stride (j)) continue;
223  if (abs (set.stride (i)) == abs (set.stride (j))) set.stride (j) = 0;
224  }
225  }
226 
227  size_t max = 0;
228  for (size_t i = 0; i < set.ndim(); ++i)
229  if (size_t (abs (set.stride (i))) > max)
230  max = abs (set.stride (i));
231 
232  for (size_t i = 0; i < set.ndim(); ++i) {
233  if (set.stride (i)) continue;
234  set.stride (i) = ++max;
235  }
236  }

253  {
