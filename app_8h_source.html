240  class ParsedOption { NOMEMALIGN
241  public:
242  ParsedOption (const Option* option, const char* const* arguments) :
243  opt (option), args (arguments)
244  {
245  for (size_t i = 0; i != option->size(); ++i) {
246  if (arguments[i][0] == '-' &&
247  !(((*option)[i].type == ImageIn || (*option)[i].type == ImageOut) && !strcmp(arguments[i], std::string("-").c_str())) &&
248  !((*option)[i].type == Integer || (*option)[i].type == Float || (*option)[i].type == IntSeq || (*option)[i].type == FloatSeq || (*option)[i].type == Various)) {
249  WARN (std::string("Value \"") + arguments[i] + "\" is being used as " +
250  ((option->size() == 1) ? "the expected argument" : ("one of the " + str(option->size()) + " expected arguments")) +
251  " for option \"-" + option->id + "\"; is this what you intended?");
252  }
253  }
254  }

259  const char* const* args;
