257 void error (Exception& e) const {
258 std::string msg ("error parsing token \"");
259 msg += p;
260 if (opt) msg += std::string ("\" for option \"") + opt->id + "\"";
261 else msg += std::string ("\" for argument \"") + arg->id + "\"";
262 throw Exception (e, msg);
263 }

278 class ParsedOption { NOMEMALIGN
279 public:
280 ParsedOption (const Option* option, const char* const* arguments) :
281 opt (option), args (arguments)
282 {
283 for (size_t i = 0; i != option->size(); ++i) {
284 if (arguments[i][0] == '-' &&
285 !(((*option)[i].type == ImageIn || (*option)[i].type == ImageOut) && !strcmp(arguments[i], std::string("-").c_str())) &&
286 !((*option)[i].type == Integer || (*option)[i].type == Float || (*option)[i].type == IntSeq || (*option)[i].type == FloatSeq || (*option)[i].type == Various)) {
287 WARN (std::string("Value \"") + arguments[i] + "\" is being used as " +
288 ((option->size() == 1) ? "the expected argument" : ("one of the " + str(option->size()) + " expected arguments")) +
289 " for option \"-" + option->id + "\"; is this what you intended?");
290 }
291 }
292 }
293
295 const Option* opt;
297 const char* const* args;
298
299 const ParsedArgument operator[] (size_t num) const {
300 assert (num < opt->size());
301 return ParsedArgument (opt, & (*opt) [num], args[num]);
302 }
303
305 bool operator== (const char* match) const {
306 std::string name = lowercase (match);
307 return name == opt->id;
308 }
