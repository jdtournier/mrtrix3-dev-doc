172  Eigen::Vector3f MethodBase::rotate_direction (const Eigen::Vector3f& reference, const Eigen::Vector3f& direction)
173  {
174  float n = std::sqrt (Math::pow2(reference[0]) + Math::pow2(reference[1]));
175  if (n == 0.0) {
176  if (reference[2] < 0.0)
177  return -direction;
178  else
179  return direction;
180  }
181 
182  Eigen::Vector3f m (reference[0]/n, reference[1]/n, 0.0f);
183  Eigen::Vector3f mp (reference[2]*m[0], reference[2]*m[1], -n);
184 
185  float alpha = direction[2];
186  float beta = direction[0]*m[0] + direction[1]*m[1];
187 
188  return {
189  direction[0] + alpha * reference[0] + beta * (mp[0] - m[0]),
190  direction[1] + alpha * reference[1] + beta * (mp[1] - m[1]),
191  direction[2] + alpha * (reference[2]-1.0f) + beta * (mp[2] - m[2])
192  };
