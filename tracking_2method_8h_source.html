95  void truncate_track (GeneratedTrack& tck, const size_t length_to_revert_from, const size_t revert_step)
96  {

106  else

108  tck.resize (length_to_revert_from - revert_step);

141  Eigen::Vector3f MethodBase::random_direction ()
142  {
143  Eigen::Vector3f d;
144  do {
145  d[0] = 2.0 * uniform(*rng) - 1.0;
146  d[1] = 2.0 * uniform(*rng) - 1.0;
147  d[2] = 2.0 * uniform(*rng) - 1.0;
148  } while (d.squaredNorm() > 1.0);
149  d.normalize();

154  Eigen::Vector3f MethodBase::random_direction (float max_angle, float sin_max_angle)
155  {

171  Eigen::Vector3f MethodBase::rotate_direction (const Eigen::Vector3f& reference, const Eigen::Vector3f& direction)
172  {
173  float n = std::sqrt (Math::pow2(reference[0]) + Math::pow2(reference[1]));
174  if (n == 0.0) {
175  if (reference[2] < 0.0)
176  return -direction;
177  else
178  return direction;
179  }
180 
181  Eigen::Vector3f m (reference[0]/n, reference[1]/n, 0.0f);
182  Eigen::Vector3f mp (reference[2]*m[0], reference[2]*m[1], -n);
183 
184  float alpha = direction[2];
185  float beta = direction[0]*m[0] + direction[1]*m[1];
186 
187  return {
188  direction[0] + alpha * reference[0] + beta * (mp[0] - m[0]),
189  direction[1] + alpha * reference[1] + beta * (mp[1] - m[1]),
190  direction[2] + alpha * (reference[2]-1.0f) + beta * (mp[2] - m[2])
191  };
192  }
