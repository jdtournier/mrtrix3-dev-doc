142  Point<value_type> MethodBase::random_direction ()
143  {
144  Point<value_type> d;
145  do {
146  d[0] = 2.0 * uniform_rng() - 1.0;
147  d[1] = 2.0 * uniform_rng() - 1.0;

172  Point<value_type> MethodBase::rotate_direction (const Point<value_type>& reference, const Point<value_type>& direction)
173  {
174  value_type n = std::sqrt (Math::pow2(reference[0]) + Math::pow2(reference[1]));
175  if (n == 0.0)
176  return (reference[2] < 0.0 ? -direction : direction);
177 
178  Point<value_type> m (reference[0]/n, reference[1]/n, 0.0);
179  Point<value_type> mp (reference[2]*m[0], reference[2]*m[1], -n);
180 
181  value_type alpha = direction[2];
182  value_type beta = direction[0]*m[0] + direction[1]*m[1];
183 
184  return (Point<value_type> (
185  direction[0] + alpha * reference[0] + beta * (mp[0] - m[0]),
186  direction[1] + alpha * reference[1] + beta * (mp[1] - m[1]),
187  direction[2] + alpha * (reference[2]-1.0) + beta * (mp[2] - m[2])
188  ));
189  }
