129 // set_scale_factor needs to be the first option that is set as it overwrites the stage vector
130 void set_scale_factor (const vector<default_type>& scalefactor) {
131 stages.resize (scalefactor.size());
132 for (size_t level = 0; level < scalefactor.size(); ++level) {
133 if (scalefactor[level] <= 0 || scalefactor[level] > 1.0)
134 throw Exception ("the linear registration scale factor for each multi-resolution level must be between 0 and 1");
135 stages[level].scale_factor = scalefactor[level];
136 }
137 }

139 // needs to be set before set_stage_iterations is set
140 void set_stage_optimiser_default (const OptimiserAlgoType& type) {
141 for (auto & stage : stages)
142 stage.optimiser_default = type;
143 }
144
145 // needs to be set before set_stage_iterations is set
146 void set_stage_optimiser_first (const OptimiserAlgoType& type) {
147 for (auto & stage : stages)
148 stage.optimiser_first = type;

151 // needs to be set before set_stage_iterations is set
152 void set_stage_optimiser_last (const OptimiserAlgoType& type) {
153 for (auto & stage : stages)
154 stage.optimiser_last = type;
155 }
156
157 void set_stage_iterations (const vector<int>& it) {
158 for (size_t i = 0; i < it.size (); ++i)
159 if (it[i] <= 0)
160 throw Exception ("the number of stage iterations must be positive");
161 if (it.size() == stages.size()) {
162 for (size_t i = 0; i < stages.size (); ++i)
163 stages[i].stage_iterations = it[i];
164 } else if (it.size() == 1) {
165 for (size_t i = 0; i < stages.size (); ++i)
166 stages[i].stage_iterations = it[0];
167 } else
168 throw Exception ("the number of stage iterations must be defined for all stages (1 or " + str(stages.size())+")");
169 for (auto & stage : stages) {
170 stage.optimisers.resize(stage.stage_iterations, stage.optimiser_default);
171 stage.optimisers[0] = stage.optimiser_first;
172 if (stage.stage_iterations > 1)
173 stage.optimisers[stage.stage_iterations - 1] = stage.optimiser_last;
174 }
175 }
176
177 void set_max_iter (const vector<int>& maxiter) {
178 for (size_t i = 0; i < maxiter.size (); ++i)
179 if (maxiter[i] < 0)
180 throw Exception ("the number of iterations must be non-negative");
181 if (maxiter.size() == stages.size()) {
182 for (size_t i = 0; i < stages.size (); ++i)
183 stages[i].gd_max_iter = maxiter[i];
184 } else if (maxiter.size() == 1) {
185 for (size_t i = 0; i < stages.size (); ++i)
186 stages[i].gd_max_iter = maxiter[0];
187 } else
188 throw Exception ("the number of gradient descent iterations must be defined for all stages (1 or " + str(stages.size())+")");
189 }
190
191 // needs to be set before set_lmax
192 void set_directions (Eigen::MatrixXd& dir) {
193 aPSF_directions = dir;
194 do_reorientation = true;
195 }
196
197 void set_lmax (const vector<int>& lmax) {
198 for (size_t i = 0; i < lmax.size (); ++i)
199 if (lmax[i] < 0 || lmax[i] % 2)
200 throw Exception ("the input lmax must be positive and even");
201 if (lmax.size() == stages.size()) {
202 for (size_t i = 0; i < stages.size (); ++i)
203 stages[i].fod_lmax = lmax[i];
204 } else if (lmax.size() == 1) {
205 for (size_t i = 0; i < stages.size (); ++i)
206 stages[i].fod_lmax = lmax[0];
207 } else
208 throw Exception ("the lmax must be defined for all stages (1 or " + str(stages.size())+")");
209 }

216 void set_loop_density (const vector<default_type>& loop_density_){
217 for (size_t d = 0; d < loop_density_.size(); ++d)
218 if (loop_density_[d] < 0.0 or loop_density_[d] > 1.0 )
219 throw Exception ("loop density must be between 0.0 and 1.0");
220 if (loop_density_.size() == stages.size()) {
221 for (size_t i = 0; i < stages.size (); ++i)
222 stages[i].loop_density = loop_density_[i];
223 } else if (loop_density_.size() == 1) {
224 for (size_t i = 0; i < stages.size (); ++i)
225 stages[i].loop_density = loop_density_[0];
226 } else
227 throw Exception ("the lmax must be defined for all stages (1 or " + str(stages.size())+")");
228 }

230 void set_diagnostics_image_prefix (const std::basic_string<char>& diagnostics_image_prefix) {
231 for (size_t level = 0; level < stages.size(); ++level) {
232 auto & stage = stages[level];
233 for (size_t iter = 1; iter <= stage.stage_iterations; ++iter) {
234 std::ostringstream oss;
235 oss << diagnostics_image_prefix << "_stage-" << level + 1 << "_iter-" << iter << ".mif";
236 if (Path::exists(oss.str()) && !App::overwrite_files)
237 throw Exception ("diagnostics image file \"" + oss.str() + "\" already exists (use -force option to force overwrite)");
238 stage.diagnostics_images.push_back(oss.str());
239 }
240 }

319 template <class MetricType, class TransformType, class Im1ImageType, class Im2ImageType, class Im1MaskType, class Im2MaskType>
320 void run_masked (
321 MetricType& metric,
322 TransformType& transform,
323 Im1ImageType& im1_image,
324 Im2ImageType& im2_image,
325 Im1MaskType& im1_mask,
326 Im2MaskType& im2_mask) {
327
328 if (do_reorientation)
329 for (auto & s : stages)
330 if (s.fod_lmax < 0)
331 throw Exception ("the lmax needs to be defined for each registration stage");
332
333 if (init_translation_type == Transform::Init::mass)
334 Transform::Init::initialise_using_image_mass (im1_image, im2_image, im1_mask, im2_mask, transform, init, contrasts);
335 else if (init_translation_type == Transform::Init::geometric)
336 Transform::Init::initialise_using_image_centres (im1_image, im2_image, im1_mask, im2_mask, transform, init);
337 else if (init_translation_type == Transform::Init::set_centre_mass) // doesn't change translation or linear matrix
338 Transform::Init::set_centre_via_mass (im1_image, im2_image, im1_mask, im2_mask, transform, init, contrasts);
339 else if (init_translation_type == Transform::Init::set_centre_geometric) // doesn't change translation or linear matrix
340 Transform::Init::set_centre_via_image_centres (im1_image, im2_image, im1_mask, im2_mask, transform, init);
341
342 if (init_rotation_type == Transform::Init::moments)
343 Transform::Init::initialise_using_image_moments (im1_image, im2_image, im1_mask, im2_mask, transform, init, contrasts);
344 else if (init_rotation_type == Transform::Init::rot_search)
345 Transform::Init::initialise_using_rotation_search (im1_image, im2_image, im1_mask, im2_mask, transform, init, contrasts);
346
347
348 INFO ("Transformation before registration:");
349 INFO (transform.info());
350
351 INFO ("Linear registration stage parameters:");
352 for (auto & stage : stages) {
353 INFO(stage.info());
354 }
355
356 // TODO global search
357 // if (global_search) {
358 // GlobalSearch::GlobalSearch transformation_search;
359 // if (log_stream) {
360 // transformation_search.set_log_stream(log_stream);
361 // }
362 // // std::ofstream outputFile( "/tmp/log.txt" );
363 // // transformation_search.set_log_stream(outputFile.rdbuf());
364 // // transformation_search.set_log_stream(std::cerr.rdbuf());
365 // transformation_search.run_masked (metric, transform, im1_image, im2_image, im1_mask, im2_mask, init);
366 // // transform.debug();
367 // }
368
369 using MidwayImageType = Header;
370 using ProcessedImageType = Im1ImageType;
371 using ProcessedMaskType = Image<bool>;
372
373#ifdef REGISTRATION_CUBIC_INTERP
374 // using Im1ImageInterpolatorType = Interp::SplineInterp<Im1ImageType, Math::UniformBSpline<typename Im1ImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
375 // using Im2ImageInterpolatorType = Interp::SplineInterp<Im2ImageType, Math::UniformBSpline<typename Im2ImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
376 // using ProcessedImageInterpolatorType = Interp::SplineInterp<ProcessedImageType, Math::UniformBSpline<typename ProcessedImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
377#else
378 using Im1ImageInterpolatorType = Interp::LinearInterp<Im1ImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
379 using Im2ImageInterpolatorType = Interp::LinearInterp<Im2ImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
380 using ProcessedImageInterpolatorType = Interp::LinearInterp<ProcessedImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
381#endif
382
383 using ParamType = Metric::Params<TransformType,
384 Im1ImageType,
385 Im2ImageType,
386 MidwayImageType,
387 Im1MaskType,
388 Im2MaskType,
389 Im1ImageInterpolatorType,
390 Im2ImageInterpolatorType,
391 Interp::Linear<Im1MaskType>,
392 Interp::Linear<Im2MaskType>,
393 ProcessedImageType,
394 ProcessedImageInterpolatorType,
395 ProcessedMaskType,
396 Interp::Nearest<ProcessedMaskType>>;
397
398 Eigen::Matrix<typename TransformType::ParameterType, Eigen::Dynamic, 1> optimiser_weights = transform.get_optimiser_weights();
399
400 // calculate midway (affine average) space which will be constant for each resolution level
401 midway_image_header = compute_minimum_average_header (im1_image, im2_image, transform.get_transform_half_inverse(), transform.get_transform_half());
402
403 for (size_t istage = 0; istage < stages.size(); istage++) {
404 auto& stage = stages[istage];
405
406 CONSOLE ("linear stage " + str(istage + 1) + "/"+str(stages.size()) + ", " + stage.info(do_reorientation));
407 // define or adjust tissue contrast lmax, nvols for this stage
408 stage_contrasts = contrasts;
409 if (stage_contrasts.size()) {
410 for (auto & mc : stage_contrasts) {
411 mc.lower_lmax (stage.fod_lmax);
412 }
413 } else {
414 MultiContrastSetting mc (im1_image.ndim()<4? 1:im1_image.size(3), do_reorientation, stage.fod_lmax);
415 stage_contrasts.push_back(mc);
416 }
417
418 DEBUG ("before downsampling:");
419 for (const auto & mc : stage_contrasts)
420 DEBUG (str(mc));
421
422 INFO ("smoothing image 1");
423 auto im1_smoothed = Registration::multi_resolution_lmax (im1_image, stage.scale_factor, do_reorientation, stage_contrasts);
424 INFO ("smoothing image 2");
425 auto im2_smoothed = Registration::multi_resolution_lmax (im2_image, stage.scale_factor, do_reorientation, stage_contrasts, &stage_contrasts);
426
427 DEBUG ("after downsampling:");
428 for (const auto & mc : stage_contrasts)
429 INFO (str(mc));
430
431
432 Filter::Resize midway_resize_filter (midway_image_header);
433 midway_resize_filter.set_scale_factor (stage.scale_factor);
434 Header midway_resized (midway_resize_filter);
435
436 ParamType parameters (transform, im1_smoothed, im2_smoothed, midway_resized, im1_mask, im2_mask);
437 parameters.loop_density = stage.loop_density;
438 if (contrasts.size())
439 parameters.set_mc_settings (stage_contrasts);
440
441
442 // if (robust_estimate)
443 // INFO ("using robust estimate");
444 // parameters.robust_estimate = robust_estimate; // TODO
445
446 // set control point coordinates inside +-1/3 of the midway_image size
447 {
448 Eigen::Vector3 ext (midway_image_header.spacing(0) / 6.0,
449 midway_image_header.spacing(1) / 6.0,
450 midway_image_header.spacing(2) / 6.0);
451 for (size_t i = 0; i<3; ++i)
452 ext(i) *= midway_image_header.size(i) - 0.5;
453 parameters.set_control_points_extent(ext);
454 }
455 DEBUG ("neighbourhood kernel extent: " + str(kernel_extent));
456 parameters.set_extent (kernel_extent);
457 Eigen::Vector3d spacing (
458 midway_image_header.spacing(0),
459 midway_image_header.spacing(1),
460 midway_image_header.spacing(2));
461 Eigen::Vector3d coherence(spacing);
462 Eigen::Vector3d stop(spacing);
463 //CONF option: RegCoherenceLen
464 //CONF default: 3.0
465 //CONF Linear registration: estimated spatial coherence length in voxels.
466 default_type reg_coherence_len = File::Config::get_float ("RegCoherenceLen", 3.0); // = 3 stdev blur
467 coherence *= reg_coherence_len * 1.0 / (2.0 * stage.scale_factor);
468 //CONF option: RegStopLen
469 //CONF default: 0.0001
470 //CONF Linear registration: smallest gradient descent step measured in fraction of a voxel at which to stop registration.
471 default_type reg_stop_len = File::Config::get_float ("RegStopLen", 0.0001);
472 stop.array() *= reg_stop_len;
473 DEBUG ("coherence length: " + str(coherence));
474 DEBUG ("stop length: " + str(stop));
475 transform.get_gradient_descent_updator()->set_control_points (parameters.control_points, coherence, stop, spacing);
476
477 // convergence check using slope of smoothed parameter trajectories
478 Eigen::VectorXd slope_threshold = Eigen::VectorXd::Ones (12);
479 //CONF option: RegGdConvergenceThresh
480 //CONF default: 5e-3
481 //CONF Linear registration: threshold for convergence check using the smoothed control point trajectories
482 //CONF measured in fraction of a voxel.
483 slope_threshold.fill (spacing.mean() * File::Config::get_float ("RegGdConvergenceThresh", 5e-3f));
484 DEBUG ("convergence slope threshold: " + str(slope_threshold[0]));
485 //CONF option: RegGdConvergenceDataSmooth
486 //CONF default: 0.8
487 //CONF Linear registration: control point trajectory smoothing value used in convergence check
488 //CONF parameter range: [0...1].
489 const default_type alpha (MR::File::Config::get_float ("RegGdConvergenceDataSmooth", 0.8));
490 if ( (alpha < 0.0f ) || (alpha > 1.0f) )
491 throw Exception ("config file option RegGdConvergenceDataSmooth has to be in the range: [0...1]");
492 //CONF option: RegGdConvergenceSlopeSmooth
493 //CONF default: 0.1
494 //CONF Linear registration: control point trajectory slope smoothing value used in convergence check
495 //CONF parameter range: [0...1].
496 const default_type beta (MR::File::Config::get_float ("RegGdConvergenceSlopeSmooth", 0.1));
497 if ( (beta < 0.0f ) || (beta > 1.0f) )
498 throw Exception ("config file option RegGdConvergenceSlopeSmooth has to be in the range: [0...1]");
499 size_t buffer_len (MR::File::Config::get_float ("RegGdConvergenceBufferLen", 4));
500 //CONF option: RegGdConvergenceMinIter
501 //CONF default: 10
502 //CONF Linear registration: minimum number of iterations until convergence check is activated.
503 size_t min_iter (MR::File::Config::get_float ("RegGdConvergenceMinIter", 10));
504 transform.get_gradient_descent_updator()->set_convergence_check (slope_threshold, alpha, beta, buffer_len, min_iter);
505
506 Metric::Evaluate<MetricType, ParamType> evaluate (metric, parameters);
507 if (do_reorientation && stage.fod_lmax > 0)
508 evaluate.set_directions (aPSF_directions);
509
510 INFO ("registration stage running...");
511 for (auto stage_iter = 1U; stage_iter <= stage.stage_iterations; ++stage_iter) {
512 if (stage.gd_max_iter > 0 and stage.optimisers[stage_iter - 1] == OptimiserAlgoType::bbgd) {
513 Math::GradientDescentBB<Metric::Evaluate<MetricType, ParamType>, typename TransformType::UpdateType>
514 optim (evaluate, *transform.get_gradient_descent_updator());
515 optim.be_verbose (analyse_descent);
516 optim.precondition (optimiser_weights);
517 optim.run (stage.gd_max_iter, grad_tolerance, analyse_descent ? std::cout.rdbuf() : log_stream);
518 parameters.optimiser_update (optim, evaluate.overlap());
519 INFO (" iteration: "+str(stage_iter)+"/"+str(stage.stage_iterations)+" GD iterations: "+
520 str(optim.function_evaluations())+" cost: "+str(optim.value())+" overlap: "+str(evaluate.overlap()));
521 } else if (stage.gd_max_iter > 0) {
522 Math::GradientDescent<Metric::Evaluate<MetricType, ParamType>, typename TransformType::UpdateType>
523 optim (evaluate, *transform.get_gradient_descent_updator());
524 optim.be_verbose (analyse_descent);
525 optim.precondition (optimiser_weights);
526 optim.run (stage.gd_max_iter, grad_tolerance, analyse_descent ? std::cout.rdbuf() : log_stream);
527 parameters.optimiser_update (optim, evaluate.overlap());
528 INFO (" iteration: "+str(stage_iter)+"/"+str(stage.stage_iterations)+" GD iterations: "+
529 str(optim.function_evaluations())+" cost: "+str(optim.value())+" overlap: "+str(evaluate.overlap()));
530 }
531
532 if (log_stream) {
533 std::ostream log_os(log_stream);
534 log_os << "\n\n"; // two empty lines for gnuplot's index recognition
535 }
536 // debug cost function gradient
537 // auto params = optim.state();
538 // VAR(optim.function_evaluations());
539 // Math::check_function_gradient (evaluate, params, 0.0001, true, optimiser_weights);
540 if (stage.diagnostics_images.size()) {
541 CONSOLE(" creating diagnostics image: " + stage.diagnostics_images[stage_iter - 1]);
542 parameters.make_diagnostics_image (stage.diagnostics_images[stage_iter - 1], File::Config::get_bool ("RegLinregDiagnosticsImageMasked", false));
543 }
544 }
545 // update midway (affine average) space using the current transformations
546 midway_image_header = compute_minimum_average_header (im1_image, im2_image, parameters.transformation.get_transform_half_inverse(), parameters.transformation.get_transform_half());
547 }
548 INFO("linear registration done");
549 INFO (transform.info());
550 }
