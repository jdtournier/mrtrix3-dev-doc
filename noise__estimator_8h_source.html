43  typedef ValueType value_type;
44 
45  NoiseEstimatorFunctor (InputVoxelType& dwi, OutputVoxelType& noise, const Math::Matrix<value_type>& SH2amp_mapping, int axis) :
46  dwi (dwi),
47  noise (noise),
48  axis (axis) {
49 
50  Math::Matrix<value_type> iSH = Math::pinv (SH2amp_mapping);
51  Math::mult (H, SH2amp_mapping, iSH);
52 
53  S.allocate (H.columns(), dwi.dim(axis));
54  R.allocate (S);
55 
56  leverage.allocate (H.rows());
57  for (size_t n = 0; n < leverage.size(); ++n)
58  leverage[n] = H(n,n) < 1.0 ? 1.0 / std::sqrt (1.0 - H(n,n)) : 1.0;
59 
60  }
61 
62  void operator () (const Image::Iterator& pos) {
63  Image::voxel_assign (dwi, pos);
64  for (dwi[axis] = 0; dwi[axis] < dwi.dim(axis); ++dwi[axis])
65  for (dwi[3] = 0; dwi[3] < dwi.dim(3); ++dwi[3])
66  S(dwi[3],dwi[axis]) = dwi.value();
67 
68  Math::mult (R, CblasLeft, value_type(0.0), value_type(1.0), CblasUpper, H, S);
69  R -= S;
70 
71  Image::voxel_assign (noise, pos);
72  for (noise[axis] = 0; noise[axis] < noise.dim(axis); ++noise[axis]) {
73  R.column(noise[axis]) *= leverage;
74  noise.value() = scale_estimator (R.column (noise[axis]));
75  }
76  }
