88  class Random_sparse_loop {
89  public:
90  Random_sparse_loop (ImageType& in,
91  const size_t& axis = 0,
92  const size_t& number_iterations = std::numeric_limits<ssize_t>::max(),
93  const bool repeat = false,
94  const ssize_t& min_index = 0,
95  const ssize_t& max_index = std::numeric_limits<ssize_t>::max()):
96  image (in),
97  repeat_ (repeat),
98  status (true),
99  ax (axis),
100  cnt (0),
101  min_idx (min_index)
102  {
103  if (max_index < image.size(ax))
104  range = max_index - min_idx + 1;
105  else
106  range = image.size(ax) - min_idx;
107  if (number_iterations < range)
108  max_cnt = number_iterations;
109  else
110  max_cnt = range;
111  assert (range >= 1);
112  assert (max_cnt >= 1);
113  if (repeat_)
114  set_next_index_with_repeat();
115  else
116  set_next_index_no_repeat();
117  }
118 
119  void set_next_index_no_repeat() {
120  while (cnt < max_cnt){
121  index = rand() % range + min_idx;
122  if (!idx_done.count(index)){
123  idx_done.insert(index);
124  break;
125  }
126  }
127  ++cnt;
128  image.index(ax) = index;
129  assert(idx_done.size() < range);
130  }
131 
132  void set_next_index_with_repeat() {
133  index = rand() % range + min_idx;
134  ++cnt;
135  image.index(ax) = index;
136  }

138  void operator++ ()
139  {

141  set_next_index_with_repeat();

144  if (cnt > max_cnt) {
145  status = false;
146  }
147  }
