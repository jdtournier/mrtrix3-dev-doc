43  template <class HeaderType>
44  ParticleGrid(const HeaderType& image)
45  {
46  DEBUG("Initialise particle grid.");
47  dims[0] = Math::ceil<size_t>( image.size(0) * image.spacing(0) / (2.0*Particle::L) );
48  dims[1] = Math::ceil<size_t>( image.size(1) * image.spacing(1) / (2.0*Particle::L) );
49  dims[2] = Math::ceil<size_t>( image.size(2) * image.spacing(2) / (2.0*Particle::L) );
50  grid.resize(dims[0]*dims[1]*dims[2]);
51 
52  // Initialise scanner-to-grid transform
53  Eigen::DiagonalMatrix<default_type, 3> newspacing (2.0*Particle::L, 2.0*Particle::L, 2.0*Particle::L);
54  Eigen::Vector3 shift (image.spacing(0)/2.0 - Particle::L,
55  image.spacing(1)/2.0 - Particle::L,
56  image.spacing(2)/2.0 - Particle::L);
57  T_s2g = image.transform() * newspacing;
58  T_s2g = T_s2g.inverse().translate(shift);
59  }
