35  // Compute the *regularised* incomplete beta function using
36  // the incomplete beta function as provided in Eigen 3.3 and above
37  template <typename ArgADerived, typename ArgBDerived, typename ArgXDerived>
38  inline const Eigen::Array<typename ArgXDerived::Scalar, Eigen::Dynamic, Eigen::Dynamic>
39  betaincreg (const Eigen::ArrayBase<ArgADerived>& a, const Eigen::ArrayBase<ArgBDerived>& b, const Eigen::ArrayBase<ArgXDerived>& x)
40  {
41  Eigen::Array<typename ArgXDerived::Scalar, Eigen::Dynamic, Eigen::Dynamic> ones (x);
42  ones.fill (typename ArgXDerived::Scalar (1));
43  return (Eigen::betainc (a, b, x) / Eigen::betainc (a, b, ones)).eval();
44  };
