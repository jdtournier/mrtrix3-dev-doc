189 template <class value_type, typename Cont>
190 class MapWriter : public MapWriterBase<Cont>
191 {
192 
193  typedef typename Image::Buffer<value_type> image_type;
194  typedef typename Image::Buffer<value_type>::voxel_type image_voxel_type;
195  typedef typename Image::BufferScratch<value_type> buffer_type;
196  typedef typename Image::BufferScratch<value_type>::voxel_type buffer_voxel_type;
197 
198  public:
199  MapWriter (Image::Header& header, const std::string& name, const bool direct_dump = false, const vox_stat_t voxel_statistic = V_SUM) :
200  MapWriterBase<Cont> (header, name, direct_dump, voxel_statistic),
201  buffer (header, "buffer"),
202  v_buffer (buffer)
203  {
204  Image::Loop loop;
205  if (voxel_statistic == V_MIN) {
206  for (loop.start (v_buffer); loop.ok(); loop.next (v_buffer))
207  v_buffer.value() = std::numeric_limits<value_type>::max();
208  } else if (voxel_statistic == V_MAX) {
209  for (loop.start (v_buffer); loop.ok(); loop.next (v_buffer))
210  v_buffer.value() = -std::numeric_limits<value_type>::max();
211  } else {
212  for (loop.start (v_buffer); loop.ok(); loop.next (v_buffer))
213  v_buffer.value() = value_type (0);
214  }
215  }
216 
217  MapWriter (const MapWriter& that) :
218  MapWriterBase<Cont> (that),
219  buffer (MapWriterBase<Cont>::H, ""),
220  v_buffer (buffer)
221  {
222  throw Exception ("Do not instantiate copy constructor for MapWriter");
223  }
224 
225  ~MapWriter ()
226  {
227 
228  Image::Loop loop;
229  switch (MapWriterBase<Cont>::voxel_statistic) {
230 
231  case V_SUM: break;
232 
233  case V_MIN:
234  for (loop.start (v_buffer); loop.ok(); loop.next (v_buffer)) {
235  if (v_buffer.value() == std::numeric_limits<value_type>::max())
236  v_buffer.value() = value_type(0);
237  }
238  break;
239 
240  case V_MEAN:
241  for (loop.start (v_buffer, *MapWriterBase<Cont>::v_counts); loop.ok(); loop.next (v_buffer, *MapWriterBase<Cont>::v_counts)) {
242  if ((*MapWriterBase<Cont>::v_counts).value())
243  v_buffer.value() /= float(MapWriterBase<Cont>::v_counts->value());
244  }
245  break;
246 
247  case V_MAX:
248  for (loop.start (v_buffer); loop.ok(); loop.next (v_buffer)) {
249  if (v_buffer.value() == -std::numeric_limits<value_type>::max())
250  v_buffer.value() = value_type(0);
251  }
252  break;
253 
254  default:
255  throw Exception ("Unknown / unhandled voxel statistic in ~MapWriter()");
256 
257  }
258 
259  if (MapWriterBase<Cont>::direct_dump) {
260  fprintf (stderr, "%s: writing image to file... ", App::NAME.c_str());
261  buffer.dump_to_file (MapWriterBase<Cont>::output_image_name, MapWriterBase<Cont>::H);
262  fprintf (stderr, "done.\n");
263  } else {
264  image_type out (MapWriterBase<Cont>::output_image_name, MapWriterBase<Cont>::H);
265  image_voxel_type v_out (out);
266  Image::LoopInOrder loopinorder (v_out, "writing image to file...", 0, 3);
267  for (loopinorder.start (v_out, v_buffer); loopinorder.ok(); loopinorder.next (v_out, v_buffer))
268  v_out.value() = v_buffer.value();
269  }
270  }
271 
272 
273  bool operator() (const Cont& in)
274  {
275  for (typename Cont::const_iterator i = in.begin(); i != in.end(); ++i) {
276  Image::Nav::set_pos (v_buffer, *i);
277  const value_type factor = get_factor (in, i);
278  switch (MapWriterBase<Cont>::voxel_statistic) {
279  case V_SUM: v_buffer.value() += in.weight * factor; break;
280  case V_MIN: v_buffer.value() = MIN(v_buffer.value(), factor); break;
281  case V_MAX: v_buffer.value() = MAX(v_buffer.value(), factor); break;
282  case V_MEAN:
283  // Only increment counts[] if it is necessary to do so given the chosen statistic
284  v_buffer.value() += in.weight * factor;
285  Image::Nav::set_pos (*MapWriterBase<Cont>::v_counts, *i);
286  (*MapWriterBase<Cont>::v_counts).value() += in.weight;
287  break;
288  default:
289  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::execute()");
290  }
291  }
292  return true;
293  }

360  set_value (value);

391  bool operator() (const Cont& in)
392  {
393  for (typename Cont::const_iterator i = in.begin(); i != in.end(); ++i) {
394  Image::Nav::set_pos (v_buffer, *i);
395  const float factor = get_factor (in, i);
396  Point<float> scaled_colour (i->get_colour());
397  scaled_colour *= factor;
398  const Point<float> current_value = get_value();
399  switch (MapWriterBase<Cont>::voxel_statistic) {
400  case V_SUM:
401  set_value (current_value + (scaled_colour * in.weight));
402  break;
403  case V_MIN:
404  if (scaled_colour.norm2() < current_value.norm2())
405  set_value (scaled_colour);
406  break;
407  case V_MEAN:
408  set_value (current_value + (scaled_colour * in.weight));
409  Image::Nav::set_pos (*MapWriterBase<Cont>::v_counts, *i);
410  (*MapWriterBase<Cont>::v_counts).value() += in.weight;
411  break;
412  case V_MAX:
413  if (scaled_colour.norm2() > current_value.norm2())
414  set_value (scaled_colour);
415  break;
416  default:
417  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::execute()");
418  }
419  }
420  return true;
421  }
