173  (type == DEC && voxel_statistic == V_SUM))

176  if (type == DEC || type == TOD) {

198  switch (voxel_statistic) {
199 
200  case V_SUM:
201  if (type == DEC) {
202  for (auto l = loop (v_buffer, *v_counts); l; ++l) {
203  if (v_counts->value()) {
204  Point<value_type> value (get_dec());
205  value *= v_counts->value() / value.norm();
206  set_dec (value);
207  }
208  }
209  }
210  break;
211 
212  case V_MIN:
213  for (auto l = loop (v_buffer); l; ++l ) {
214  if (v_buffer.value() == std::numeric_limits<value_type>::max())
215  v_buffer.value() = value_type(0);
216  }
217  break;

219  case V_MEAN:
220  if (type == GREYSCALE) {

222  if (v_counts->value())

235  if (v_counts->value()) {

363 template <typename value_type>
364 template <class Cont>
365 void MapWriter<value_type>::receive_greyscale (const Cont& in)
366 {
367  assert (MapWriterBase::type == GREYSCALE);
368  for (typename Cont::const_iterator i = in.begin(); i != in.end(); ++i) {
369  Image::Nav::set_pos (v_buffer, *i);
370  const float factor = get_factor (*i, in);
371  const float weight = in.weight * i->get_length();
372  switch (voxel_statistic) {
373  case V_SUM: v_buffer.value() += weight * factor; break;
374  case V_MIN: v_buffer.value() = MIN(v_buffer.value(), factor); break;
375  case V_MAX: v_buffer.value() = MAX(v_buffer.value(), factor); break;
376  case V_MEAN:
377  v_buffer.value() += weight * factor;
378  assert (v_counts);
379  Image::Nav::set_pos (*v_counts, *i);
380  (*v_counts).value() += weight;
381  break;
382  default:
383  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_greyscale()");
384  }
385  }

390 template <typename value_type>
391 template <class Cont>
392 void MapWriter<value_type>::receive_dec (const Cont& in)
393 {
394  assert (type == DEC);
395  for (typename Cont::const_iterator i = in.begin(); i != in.end(); ++i) {
396  Image::Nav::set_pos (v_buffer, *i);
397  const float factor = get_factor (*i, in);
398  const float weight = in.weight * i->get_length();
399  Point<value_type> scaled_colour (i->get_colour());
400  scaled_colour *= factor;
401  const Point<value_type> current_value = get_dec();
402  switch (voxel_statistic) {
403  case V_SUM:
404  set_dec (current_value + (scaled_colour * weight));
405  assert (v_counts);
406  Image::Nav::set_pos (*v_counts, *i);
407  (*v_counts).value() += weight;
408  break;
409  case V_MIN:
410  if (scaled_colour.norm2() < current_value.norm2())
411  set_dec (scaled_colour);
412  break;
413  case V_MEAN:
414  set_dec (current_value + (scaled_colour * weight));
415  Image::Nav::set_pos (*v_counts, *i);
416  (*v_counts).value() += weight;
417  break;
418  case V_MAX:
419  if (scaled_colour.norm2() > current_value.norm2())
420  set_dec (scaled_colour);
421  break;
422  default:
423  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_dec()");
424  }

459 template <typename value_type>
460 template <class Cont>
461 void MapWriter<value_type>::receive_tod (const Cont& in)
462 {
463  assert (type == TOD);
464  Math::Vector<float> sh_coefs;
465  for (typename Cont::const_iterator i = in.begin(); i != in.end(); ++i) {
466  Image::Nav::set_pos (v_buffer, *i, 0, 3);
467  const float factor = get_factor (*i, in);
468  const float weight = in.weight * i->get_length();
469  get_tod (sh_coefs);
470  if (v_counts)
471  Image::Nav::set_pos (*v_counts, *i, 0, 3);
472  switch (voxel_statistic) {
473  case V_SUM:
474  for (size_t index = 0; index != sh_coefs.size(); ++index)
475  sh_coefs[index] += (i->get_tod()[index] * weight * factor);
476  set_tod (sh_coefs);
477  break;
478  // For TOD, need to store min/max factors - counts buffer is hijacked to do this
479  case V_MIN:
480  assert (v_counts);
481  if (factor < (*v_counts).value()) {
482  (*v_counts).value() = factor;
483  Math::Vector<float> tod (i->get_tod());
484  tod *= factor;
485  set_tod (tod);
486  }
487  break;
488  case V_MAX:
489  assert (v_counts);
490  if (factor > (*v_counts).value()) {
491  (*v_counts).value() = factor;
492  Math::Vector<float> tod (i->get_tod());
493  tod *= factor;
494  set_tod (tod);
495  }
496  break;
497  case V_MEAN:
498  assert (v_counts);
499  for (size_t index = 0; index != sh_coefs.size(); ++index)
500  sh_coefs[index] += (i->get_tod()[index] * weight * factor);
501  set_tod (sh_coefs);
502  (*v_counts).value() += weight;
503  break;
504  default:
505  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_tod()");
