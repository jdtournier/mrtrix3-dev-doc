340  template <class Cont>
341  void MapWriter<value_type>::receive_greyscale (const Cont& in)
342  {
343  assert (MapWriterBase::type == GREYSCALE);
344  for (const auto& i : in) {
345  assign_pos_of (i).to (buffer);
346  const float factor = get_factor (i, in);
347  const float weight = in.weight * i.get_length();
348  switch (voxel_statistic) {
349  case V_SUM: buffer.value() += weight * factor; break;
350  case V_MIN: buffer.value() = std::min (float (buffer.value()), factor); break;
351  case V_MAX: buffer.value() = std::max (float (buffer.value()), factor); break;
352  case V_MEAN:
353  buffer.value() += weight * factor;
354  assert (counts);
355  assign_pos_of (i).to (*counts);
356  counts->value() += weight;
357  break;
358  default:
359  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_greyscale()");
360  }
361  }

366  template <typename value_type>
367  template <class Cont>
368  void MapWriter<value_type>::receive_dec (const Cont& in)
369  {
370  assert (type == DEC);
371  for (const auto& i : in) {
372  assign_pos_of (i).to (buffer);
373  const float factor = get_factor (i, in);
374  const float weight = in.weight * i.get_length();
375  auto scaled_colour = i.get_colour();
376  scaled_colour *= factor;
377  const auto current_value = get_dec();
378  switch (voxel_statistic) {
379  case V_SUM:
380  set_dec (current_value + (scaled_colour * weight));
381  assert (counts);
382  assign_pos_of (i).to (*counts);
383  counts->value() += weight;
384  break;
385  case V_MIN:
386  if (scaled_colour.squaredNorm() < current_value.squaredNorm())
387  set_dec (scaled_colour);
388  break;
389  case V_MEAN:
390  set_dec (current_value + (scaled_colour * weight));
391  assign_pos_of (i).to (*counts);
392  counts->value() += weight;
393  break;
394  case V_MAX:
395  if (scaled_colour.squaredNorm() > current_value.squaredNorm())
396  set_dec (scaled_colour);
397  break;
398  default:
399  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_dec()");
400  }
401  }

435  template <typename value_type>
436  template <class Cont>
437  void MapWriter<value_type>::receive_tod (const Cont& in)
438  {
439  assert (type == TOD);
440  Eigen::VectorXf sh_coefs;
441  for (const auto& i : in) {
442  assign_pos_of (i, 0, 3).to (buffer);
443  const float factor = get_factor (i, in);
444  const float weight = in.weight * i.get_length();
445  get_tod (sh_coefs);
446  if (counts)
447  assign_pos_of (i, 0, 3).to (*counts);
448  switch (voxel_statistic) {
449  case V_SUM:
450  for (ssize_t index = 0; index != sh_coefs.size(); ++index)
451  sh_coefs[index] += i.get_tod()[index] * weight * factor;
452  set_tod (sh_coefs);
453  break;
454  // For TOD, need to store min/max factors - counts buffer is hijacked to do this
455  case V_MIN:
456  assert (counts);
457  if (factor < counts->value()) {
458  counts->value() = factor;
459  auto tod = i.get_tod();
460  tod *= factor;
461  set_tod (tod);
462  }
463  break;
464  case V_MAX:
465  assert (counts);
466  if (factor > counts->value()) {
467  counts->value() = factor;
468  auto tod = i.get_tod();
469  tod *= factor;
470  set_tod (tod);
471  }
472  break;
473  case V_MEAN:
474  assert (counts);
475  for (ssize_t index = 0; index != sh_coefs.size(); ++index)
476  sh_coefs[index] += i.get_tod()[index] * weight * factor;
477  set_tod (sh_coefs);
478  counts->value() += weight;
479  break;
480  default:
481  throw Exception ("Unknown / unhandled voxel statistic in MapWriter::receive_tod()");
482  }
483  }
