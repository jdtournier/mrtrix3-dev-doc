46  template <class MatrixType>
47  Eigen::MatrixXd& normalise_grad (MatrixType& grad)
48  {
49  if (grad.cols() < 3)
50  throw Exception ("invalid diffusion gradient table dimensions");
51  for (ssize_t i = 0; i < grad.rows(); i++) {
52  auto norm = grad.row(i).template head<3>().norm();
53  if (norm)
54  grad.row(i).template head<3>().array() /= norm;
55  }
56  return grad;
57  }

63  template <class MatrixType, class IndexVectorType>
64  inline Eigen::MatrixXd gen_direction_matrix (
65  const MatrixType& grad,
66  const IndexVectorType& dwi)
67  {
68  Eigen::MatrixXd dirs (dwi.size(),2);
69  for (size_t i = 0; i < dwi.size(); i++) {
70  dirs (i,0) = std::atan2 (grad (dwi[i],1), grad (dwi[i],0));
71  auto z = grad (dwi[i],2) / grad.row (dwi[i]).template head<3>().norm();
72  if (z >= 1.0)
73  dirs(i,1) = 0.0;
74  else if (z <= -1.0)
75  dirs (i,1) = Math::pi;
76  else
77  dirs (i,1) = std::acos (z);
78  }
79  return dirs;
80  }

86  template <class MatrixType>
87  default_type condition_number_for_lmax (const MatrixType& dirs, int lmax)
88  {
89  Eigen::MatrixXd g;
90  if (dirs.cols() == 2) // spherical coordinates:
91  g = dirs;
92  else // Cartesian to spherical:
93  g = Math::Sphere::cartesian2spherical (dirs).leftCols(2);
94 
95  auto v = Eigen::JacobiSVD<Eigen::MatrixXd> (Math::SH::init_transform (g, lmax)).singularValues();
96  return v[0] / v[v.size()-1];
97  }

139  template <class MatrixType>
140  void set_DW_scheme (Header& header, const MatrixType& G)
141  {
142  if (!G.rows()) {
143  header.keyval().erase ("dw_scheme");
144  return;
145  }
146  std::string dw_scheme;
147  for (ssize_t row = 0; row < G.rows(); ++row) {
148  std::string line;
149  for (ssize_t col = 0; col < G.cols(); ++col) {
150  line += str(G(row,col), 10);
151  if (col < G.cols() - 1) line += ",";

153  add_line (dw_scheme, line);
154  }
155  if (dw_scheme.size())
156  header.keyval()["dw_scheme"] = dw_scheme;
157  else
158  WARN ("attempt to add empty DW scheme to header - ignored");
159  }

259  template <class MatrixType>
260  Eigen::MatrixXd compute_SH2amp_mapping (
261  const MatrixType& directions,
262  bool lmax_from_command_line = true,
263  int default_lmax = 8)
264  {
265  int lmax = -1;
266  int lmax_from_ndir = Math::SH::LforN (directions.rows());
267  bool lmax_set_from_commandline = false;
268  if (lmax_from_command_line) {
269  auto opt = App::get_options ("lmax");
270  if (opt.size()) {
271  lmax_set_from_commandline = true;
272  lmax = to<int> (opt[0][0]);
273  if (lmax % 2)
274  throw Exception ("lmax must be an even number");
275  if (lmax < 0)
276  throw Exception ("lmax must be a non-negative number");
277  if (lmax > lmax_from_ndir) {
278  WARN ("not enough directions for lmax = " + str(lmax) + " - dropping down to " + str(lmax_from_ndir));
279  lmax = lmax_from_ndir;
280  }
281  }
282  }
283 
284  if (lmax < 0) {
285  lmax = lmax_from_ndir;
286  if (lmax > default_lmax)
287  lmax = default_lmax;
288  }
289 
290  INFO ("computing SH transform using lmax = " + str (lmax));
291 
292  int lmax_prev = lmax;
293  Eigen::MatrixXd mapping;
294  do {
295  mapping = Math::SH::init_transform (directions, lmax);
296  auto v = Eigen::JacobiSVD<Eigen::MatrixXd> (mapping).singularValues();
297  auto cond = v[0] / v[v.size()-1];
298  if (cond < 10.0)
299  break;
300  WARN ("directions are poorly distributed for lmax = " + str(lmax) + " (condition number = " + str (cond) + ")");
301  if (cond < 100.0 || lmax_set_from_commandline)
302  break;
303  lmax -= 2;
304  } while (lmax >= 0);
305 
306  if (lmax_prev != lmax)
307  WARN ("reducing lmax to " + str(lmax) + " to improve conditioning");
308 
309  return mapping;
310  }
