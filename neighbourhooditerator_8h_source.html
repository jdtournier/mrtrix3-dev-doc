49  NeighbourhoodIterator() = delete;
50  template <class IteratorType>
51  NeighbourhoodIterator (const IteratorType& iter, const std::vector<size_t>& extent) :
52  dim (iter.ndim()),
53  offset (iter.ndim()),
54  // pos (iter.ndim()),
55  pos_orig (iter.ndim()),
56  ext (container_cast<decltype(ext)>(extent)),
57  has_next_(false) {
58  assert (iter.ndim() == extent.size());
59  pos.resize(iter.ndim());
60  for (size_t i = 0; i < iter.ndim(); ++i){
61  ext[i] = (ext[i]-1) / 2;
62  offset[i] = iter.index(i);
63  // set pos to lower bound
64  pos[i] = (offset[i] - ext[i] < 0) ? 0 : offset[i] - ext[i];
65  pos_orig[i] = pos[i];
66  // upper bound:
67  auto high = (offset[i] + ext[i] >= iter.size(i)) ? iter.size(i) - 1 : offset[i] + ext[i];
68  dim[i] = high - pos[i] + 1;
69  }
70  }

85  bool loop () {
86  if (not this->has_next_){
87  this->has_next_ = true;
88  for (auto axis = dim.size(); axis-- != 0; )
89  this->reset(axis);
90  return true;
91  }
92  for (auto axis = dim.size(); axis-- != 0; ) { // TODO loop in stride order
93  ++pos[axis];
94  if (pos[axis] != (pos_orig[axis] + dim[axis]))
95  return true;
96  this->reset(axis);
97  }
98  this->has_next_ = false;
99  return false;
