161  bool within_bounds (const Point<float>& p) {
162  // Bounds testing is different for sinc interpolation than others
163  // Not only due to the width of the kernel, but also the mirroring of the image data beyond the FoV
164  return (round (p[0]) > -dim(0) + kernel_width) && (round (p[0]) < (2 * dim(0)) - kernel_width)
165  && (round (p[1]) > -dim(1) + kernel_width) && (round (p[1]) < (2 * dim(1)) - kernel_width)
