195 template <class Cont>
196 void TrackMapperTWI<Cont>::load_factors (const std::vector< Point<float> >& tck)
197 {
198 
199  if (contrast != CURVATURE)
200  throw Exception ("Function TrackMapperTWI::load_factors() only works with curvature contrast");
201 
202  std::vector< Point<float> > tangents;
203  tangents.reserve (tck.size());
204 
205  // Would like to be able to manipulate the length over which the tangent calculation is affected
206  // However don't want to just take a pair of distant points and get the tangent that way; would rather
207  // find a way to 'smooth' the curvature in a non-scalar fashion i.e. inverted curvature cancels
208  // Ideally would like to get a curvature measurement & azimuth at each point; these can be averaged
209  // using a Gaussian kernel
210  // But how to define azimuth & make it consistent between points?
211  // Average principal normal vectors using a gaussian kernel, re-determine the curvature
212 
213  for (size_t i = 0; i != tck.size(); ++i) {
214  Point<float> this_tangent;
215  if (i == 0)
216  this_tangent = ((tck[1] - tck[0] ).normalise());
217  else if (i == tck.size() - 1)
218  this_tangent = ((tck[i] - tck[i-1]).normalise());
219  else
220  this_tangent = ((tck[i+1] - tck[i-1]).normalise());
221  if (this_tangent.valid())
222  tangents.push_back (this_tangent);
223  else
224  tangents.push_back (Point<float> (0.0, 0.0, 0.0));
225  }
226 
227  // For those tangents which are invalid, fill with valid factors from neighbours
228  for (size_t i = 0; i != tangents.size(); ++i) {
229  if (tangents[i] == Point<float> (0.0, 0.0, 0.0)) {
230 
231  if (i == 0) {
232  size_t j;
233  for (j = 1; (j < tck.size() - 1) && (tangents[j] != Point<float> (0.0, 0.0, 0.0)); ++j);
234  tangents[i] = tangents[j];
235  } else if (i == tangents.size() - 1) {
236  size_t k;
237  for (k = i - 1; k && (tangents[k] != Point<float> (0.0, 0.0, 0.0)); --k);
238  tangents[i] = tangents[k];
239  } else {
240  size_t j, k;
241  for (j = 1; (j < tck.size() - 1) && (tangents[j] != Point<float> (0.0, 0.0, 0.0)); ++j);
242  for (k = i - 1; k && (tangents[k] != Point<float> (0.0, 0.0, 0.0)); --k);
243  tangents[i] = (tangents[j] + tangents[k]).normalise();
244  }
245 
246  }
247  }
248 
249  // Smooth both the tangent vectors and the principal normal vectors according to a Gaussuan kernel
250  // Remember: tangent vectors are unit length, but for principal normal vectors length must be preserved!
251 
252  std::vector< Point<float> > smoothed_tangents;
253  smoothed_tangents.reserve (tangents.size());
254 
255  static const float gaussian_theta = CURVATURE_TRACK_SMOOTHING_FWHM / (2.0 * sqrt (2.0 * log (2.0)));
256  static const float gaussian_denominator = 2.0 * gaussian_theta * gaussian_theta;
257 
258  for (size_t i = 0; i != tck.size(); ++i) {
259 
260  Point<float> this_tangent (0.0, 0.0, 0.0);
261  std::pair<float, Point<float> > this_normal (std::make_pair (0.0, Point<float>(0.0, 0.0, 0.0)));
262 
263  for (size_t j = 0; j != tck.size(); ++j) {
264  const float distance = step_size * Math::abs ((int)i - (int)j);
265  const float this_weight = exp (-distance * distance / gaussian_denominator);
266  this_tangent += (tangents[j] * this_weight);
267  }
268 
269  smoothed_tangents.push_back (this_tangent.normalise());
270 
271  }
272 
273  for (size_t i = 0; i != tck.size(); ++i) {
274 
275  // Risk of acos() returning NAN if the dot product is greater than 1.0
276  float this_value;
277  if (i == 0)
278  this_value = acos (smoothed_tangents[ 1 ].dot (smoothed_tangents[ 0 ])) / step_size;
279  else if (i == tck.size() - 1)
280  this_value = acos (smoothed_tangents[ i ].dot (smoothed_tangents[i-1])) / step_size;
281  else
282  this_value = acos (smoothed_tangents[i+1].dot (smoothed_tangents[i-1])) * 0.5 / step_size;
283 
284  if (isnan (this_value))
285  factors.push_back (0.0);
286  else
287  factors.push_back (this_value);
288 
289  }
290 
291 }

297 template <class Cont>
298 void TrackMapperTWI<Cont>::gaussian_smooth_factors ()
299 {
300 
301  std::vector<float> unsmoothed (factors);
302 
303  for (size_t i = 0; i != unsmoothed.size(); ++i) {
304 
305  double sum = 0.0, norm = 0.0;
306 
307  if (std::isfinite (unsmoothed[i])) {
308  sum = unsmoothed[i];
309  norm = 1.0; // Gaussian is unnormalised -> e^0 = 1
310  }
311 
312  float distance = 0.0;
313  for (size_t j = i; j--; ) { // Decrement AFTER null test, so loop_ runs with j = 0
314  distance += step_size;
315  if (std::isfinite (unsmoothed[j])) {
316  const float this_weight = exp (-distance * distance / gaussian_denominator);
317  norm += this_weight;
318  sum += this_weight * unsmoothed[j];
319  }
320  }
321  distance = 0.0;
322  for (size_t j = i + 1; j < unsmoothed.size(); ++j) {
323  distance += step_size;
324  if (std::isfinite (unsmoothed[j])) {
325  const float this_weight = exp (-distance * distance / gaussian_denominator);
326  norm += this_weight;
327  sum += this_weight * unsmoothed[j];
328  }
329  }
330 
331  if (norm)
332  factors[i] = (sum / norm);
333  else
334  factors[i] = 0.0;
335 
336  }

345 template <class Cont>
346 void TrackMapperTWI<Cont>::set_factor (const std::vector< Point<float> >& tck, Cont& out)
347 {
348 
349  size_t count = 0;
350 
351  switch (contrast) {
352 
353  case TDI: out.factor = 1.0; break;
354  case PRECISE_TDI: out.factor = 1.0; break;
355  case ENDPOINT: out.factor = 1.0; break;
356  case LENGTH: out.factor = (step_size * (tck.size() - 1)); break;
357  case INVLENGTH: out.factor = (step_size / (tck.size() - 1)); break;
358 
359  case SCALAR_MAP:
360  case SCALAR_MAP_COUNT:
361  case FOD_AMP:
362  case CURVATURE:
363 
364  factors.clear();
365  factors.reserve (tck.size());
366  load_factors (tck); // This should call the overloaded virtual function for TrackMapperImage
367 
368  switch (track_statistic) {
369 
370  case T_SUM:
371  out.factor = 0.0;
372  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
373  if (std::isfinite (*i))
374  out.factor += *i;
375  }
376  break;
377 
378  case T_MIN:
379  out.factor = INFINITY;
380  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
381  if (std::isfinite (*i))
382  out.factor = MIN (out.factor, *i);
383  }
384  break;
385 
386  case T_MEAN:
387  out.factor = 0.0;
388  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
389  if (std::isfinite (*i)) {
390  out.factor += *i;
391  ++count;
392  }
393  }
394  out.factor = (count ? (out.factor / float(count)) : 0.0);
395  break;
396 
397  case T_MAX:
398  out.factor = -INFINITY;
399  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
400  if (std::isfinite (*i))
401  out.factor = MAX (out.factor, *i);
402  }
403  break;
404 
405  case T_MEDIAN:
406  if (factors.empty()) {
407  out.factor = 0.0;
408  } else {
409  nth_element (factors.begin(), factors.begin() + (factors.size() / 2), factors.end());
410  out.factor = *(factors.begin() + (factors.size() / 2));
411  }
412  break;
413 
414  case T_MEAN_NONZERO:
415  out.factor = 0.0;
416  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
417  if (std::isfinite (*i) && *i) {
418  out.factor += *i;
419  ++count;
420  }
421  }
422  out.factor = (count ? (out.factor / float(count)) : 0.0);
423  break;
424 
425  case GAUSSIAN:
426  gaussian_smooth_factors();
427  out.factor = 0.0;
428  for (std::vector<float>::const_iterator i = factors.begin(); i != factors.end(); ++i) {
429  if (*i) {
430  out.factor = 1.0;
431  break;
432  }
433  }
434  break;
435 
436  case ENDS_MIN:
437  assert (factors.size() == 2);
438  out.factor = (Math::abs(factors[0]) < Math::abs(factors[1])) ? factors[0] : factors[1];
439  break;
440 
441  case ENDS_MEAN:
442  assert (factors.size() == 2);
443  out.factor = 0.5 * (factors[0] + factors[1]);
444  break;
445 
446  case ENDS_MAX:
447  assert (factors.size() == 2);
448  out.factor = (Math::abs(factors[0]) > Math::abs(factors[1])) ? factors[0] : factors[1];
449  break;
450 
451  case ENDS_PROD:
452  assert (factors.size() == 2);
453  if ((factors[0] < 0.0 && factors[1] < 0.0) || (factors[0] > 0.0 && factors[1] > 0.0))
454  out.factor = factors[0] * factors[1];
455  else
456  out.factor = 0.0;
457  break;
458 
459  case ENDS_CORR:
460  assert (factors.size() == 1);
461  out.factor = factors.front();
462  break;
463 
464  default:
465  throw Exception ("FIXME: Undefined / unsupported track statistic in TrackMapperTWI::get_factor()");
466 
467  }
468  break;
469 
470  default:
471  throw Exception ("FIXME: Undefined / unsupported contrast mechanism in TrackMapperTWI::get_factor()");
472 
473  }
474 
475  if (contrast == SCALAR_MAP_COUNT)
476  out.factor = (out.factor ? 1.0 : 0.0);
477 
478  if (!std::isfinite (out.factor))
479  out.factor = 0.0;
480 
481 }

549 template <class Cont>
550 void TrackMapperTWIImage<Cont>::load_factors (const std::vector< Point<float> >& tck)
551 {
552 
553  switch (TrackMapperTWI<Cont>::contrast) {
554 
555  case SCALAR_MAP:
556  case SCALAR_MAP_COUNT:
557 
558  if (TrackMapperTWI<Cont>::track_statistic == ENDS_MIN || TrackMapperTWI<Cont>::track_statistic == ENDS_MEAN || TrackMapperTWI<Cont>::track_statistic == ENDS_MAX || TrackMapperTWI<Cont>::track_statistic == ENDS_PROD) { // Only the track endpoints contribute
559 
560  for (size_t tck_end_index = 0; tck_end_index != 2; ++tck_end_index) {
561  const Point<float> endpoint = get_last_point_in_fov (tck, tck_end_index);
562  if (endpoint.valid())
563  TrackMapperTWI<Cont>::factors.push_back (interp.value());
564  else
565  TrackMapperTWI<Cont>::factors.push_back (NAN);
566  }
567 
568  } else if (TrackMapperTWI<Cont>::track_statistic == ENDS_CORR) {
569 
570  TrackMapperTWI<Cont>::factors.assign (1, 0.0);
571  input_voxel_type start (voxel), end (voxel);
572  const Point<float> p_start (get_last_point_in_fov (tck, false));
573  if (!p_start) return;
574  const Point<int> v_start (int(Math::round (p_start[0])), int(Math::round (p_start[1])), int(Math::round (p_start[2])));
575  Image::Nav::set_pos (start, v_start);
576  const Point<float> p_end (get_last_point_in_fov (tck, true));
577  if (!p_end) return;
578  const Point<int> v_end (int(Math::round (p_end[0])), int(Math::round (p_end[1])), int(Math::round (p_end[2])));
579  Image::Nav::set_pos (end, v_end);
580 
581  double start_sum = 0.0, end_sum = 0.0;
582  for (start[3] = end[3] = 0; start[3] != start.dim (3); ++start[3], ++end[3]) {
583  start_sum += start.value();
584  end_sum += end .value();
585  }
586  const float start_mean = start_sum / double (start.dim (3));
587  const float end_mean = end_sum / double (end .dim (3));
588 
589  double product = 0.0, start_sum_variance = 0.0, end_sum_variance = 0.0;
590  for (start[3] = end[3] = 0; start[3] != start.dim (3); ++start[3], ++end[3]) {
591  product += ((start.value() - start_mean) * (end.value() - end_mean));
592  start_sum_variance += Math::pow2 (start.value() - start_mean);
593  end_sum_variance += Math::pow2 (end .value() - end_mean);
594  }
595  const float product_expectation = product / double (start.dim(3));
596  const float start_stdev = Math::sqrt (start_sum_variance / double(start.dim(3) - 1));
597  const float end_stdev = Math::sqrt (end_sum_variance / double(end .dim(3) - 1));
598 
599  TrackMapperTWI<Cont>::factors[0] = product_expectation / (start_stdev * end_stdev);
600 
601  } else { // The entire length of the track contributes
602 
603  for (std::vector< Point<float> >::const_iterator i = tck.begin(); i != tck.end(); ++i) {
604  if (!interp.scanner (*i))
605  TrackMapperTWI<Cont>::factors.push_back (interp.value());
606  else
607  TrackMapperTWI<Cont>::factors.push_back (NAN);
608  }
609 
610  }
611  break;
612 
613  case FOD_AMP:
614  for (size_t i = 0; i != tck.size(); ++i) {
615  const Point<float>& p = tck[i];
616  if (!interp.scanner (p)) {
617  // Get the interpolated spherical harmonics at this point
618  for (interp[3] = 0; interp[3] != interp.dim(3); ++interp[3])
619  sh_coeffs[interp[3]] = interp.value();
620  const Point<float> dir = (tck[(i == tck.size()-1) ? i : (i+1)] - tck[i ? (i-1) : 0]).normalise();
621  TrackMapperTWI<Cont>::factors.push_back (precomputer.value (sh_coeffs, dir));
622  } else {
623  TrackMapperTWI<Cont>::factors.push_back (NAN);
624  }
625  }
626  break;
