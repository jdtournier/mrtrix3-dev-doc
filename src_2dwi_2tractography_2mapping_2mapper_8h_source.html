186 void TrackMapperBase::voxelise (const Streamline<>& tck, Cont& output) const
187 {
188 
189  Streamline<>::const_iterator prev = tck.begin();
190  const Streamline<>::const_iterator last = tck.end() - 1;
191 
192  Point<int> vox;
193  for (Streamline<>::const_iterator i = tck.begin(); i != last; ++i) {
194  vox = round (transform.scanner2voxel (*i));
195  if (check (vox, info)) {
196  const Point<float> dir ((*(i+1) - *prev).normalise());
197  add_to_set (output, vox, dir, 1.0f);
198  }
199  prev = i;
200  }
201 
202  vox = round (transform.scanner2voxel (*last));
203  if (check (vox, info)) {
204  const Point<float> dir ((*last - *prev).normalise());
205  add_to_set (output, vox, dir, 1.0f);
206  }
207 
208  for (typename Cont::iterator i = output.begin(); i != output.end(); ++i)
209  i->normalise();
210 
211 }

218 template <class Cont>
219 void TrackMapperBase::voxelise_precise (const Streamline<>& tck, Cont& out) const
220 {
221  typedef Point<float> PointF;
222 
223  static const float accuracy = Math::pow2 (0.005 * minvalue (info.vox (0), info.vox (1), info.vox (2)));
224 
225  if (tck.size() < 2)
226  return;
227 
228  Math::Hermite<float> hermite (0.1);
229 
230  const PointF tck_proj_front = (tck[ 0 ] * 2.0) - tck[ 1 ];
231  const PointF tck_proj_back = (tck[tck.size()-1] * 2.0) - tck[tck.size()-2];
232 
233  unsigned int p = 0;
234  PointF p_voxel_exit = tck.front();
235  float mu = 0.0;
236  bool end_track = false;
237  Point<int> next_voxel (round (transform.scanner2voxel (tck.front())));
238 
239  do {
240 
241  const PointF p_voxel_entry (p_voxel_exit);
242  PointF p_prev (p_voxel_entry);
243  float length = 0.0;
244  const Point<int> this_voxel = next_voxel;
245 
246  while ((p != tck.size()) && ((next_voxel = round (transform.scanner2voxel (tck[p]))) == this_voxel)) {
247  length += dist (p_prev, tck[p]);
248  p_prev = tck[p];
249  ++p;
250  mu = 0.0;
251  }
252 
253  if (p == tck.size()) {
254  p_voxel_exit = tck.back();
255  end_track = true;
256  } else {
257 
258  float mu_min = mu;
259  float mu_max = 1.0;
260 
261  const PointF* p_one = (p == 1) ? &tck_proj_front : &tck[p - 2];
262  const PointF* p_four = (p == tck.size() - 1) ? &tck_proj_back : &tck[p + 1];
263 
264  PointF p_min = p_prev;
265  PointF p_max = tck[p];
266 
267  while (dist2 (p_min, p_max) > accuracy) {
268 
269  mu = 0.5 * (mu_min + mu_max);
270  hermite.set (mu);
271  const PointF p_mu = hermite.value (*p_one, tck[p - 1], tck[p], *p_four);
272  const Point<int> mu_voxel = round (transform.scanner2voxel (p_mu));
273 
274  if (mu_voxel == this_voxel) {
275  mu_min = mu;
276  p_min = p_mu;
277  } else {
278  mu_max = mu;
279  p_max = p_mu;
280  next_voxel = mu_voxel;
281  }
282 
283  }
284  p_voxel_exit = p_max;
285 
286  }
287 
288  length += dist (p_prev, p_voxel_exit);
289  PointF traversal_vector ((p_voxel_exit - p_voxel_entry).normalise());
290  if (traversal_vector.valid() && check (this_voxel, info))
291  add_to_set (out, this_voxel, traversal_vector, length);
292 
293  } while (!end_track);
294 
295 }
