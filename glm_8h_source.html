41  inline void SVD_invert (Math::Matrix<double>& I, const Math::Matrix<double>& M, double precision = 1.0e-10)
42  {
43  size_t N = std::min (M.rows(), M.columns());
44  Math::Matrix<double> U (M), V (N,N);
45  Math::Vector<double> S (N), work (N);
46  gsl_linalg_SV_decomp (U.gsl(), V.gsl(), S.gsl(), work.gsl());
47  for (size_t n = 0; n < N; ++n) {
48  double sv = S[n] < precision ? 0.0 : 1.0/S[n];
49  for (size_t r = 0; r < N; ++r)
50  V(r,n) *= sv;
51  }
52  Math::mult (I, 1.0, CblasNoTrans, V, CblasTrans, U);
53  }

74  template <typename ValueType>
75  inline Math::Matrix<ValueType> scale_contrasts (const Math::Matrix<ValueType>& contrasts, const Math::Matrix<ValueType>& design, size_t degrees_of_freedom)
76  {
77  Math::Matrix<ValueType> XtX;
78  Math::mult (XtX, ValueType(1.0), CblasTrans, design, CblasNoTrans, design);
79  {
80  Math::Matrix<double> d_XtX = XtX;
81  Math::Matrix<double> d_pinv_XtX;
82  SVD_invert (d_pinv_XtX, d_XtX);
83  XtX = d_pinv_XtX;
84  }
85 
86  // make sure contrast is a column vector:
87  Math::Matrix<ValueType> scaled_contrasts (contrasts);
88  if (scaled_contrasts.columns() > 1 && scaled_contrasts.rows() > 1)
89  throw Exception ("too many columns in contrast matrix: this implementation currently only supports univariate GLM");
90  if (scaled_contrasts.rows() > 1)
91  scaled_contrasts = Math::transpose (scaled_contrasts);
92  scaled_contrasts.resize (scaled_contrasts.rows(), design.columns());
93 
94  for (size_t n = 0; n < contrasts.rows(); ++n) {
95  Math::Vector<ValueType> pinv_XtX_c;
96  Math::mult (pinv_XtX_c, XtX, contrasts.row(n));
97  scaled_contrasts.row(n) *= std::sqrt (ValueType(degrees_of_freedom) / Math::dot (contrasts.row(n), pinv_XtX_c));
98  }
99 
100  return scaled_contrasts;
101  }

173  template <typename ValueType>
174  inline void std_effect_size (const Math::Matrix<ValueType>& measurements,
175  const Math::Matrix<ValueType>& design,
176  const Math::Matrix<ValueType>& contrast,
177  Math::Matrix<ValueType>& cohens_d) {
178  Math::Matrix<ValueType> betas;
179  GLM::solve_betas (measurements, design, betas);
180  Math::Matrix<ValueType> residuals;
181  Math::mult (residuals, ValueType(-1.0), CblasNoTrans, betas, CblasTrans, design);
182  residuals += measurements;
183  residuals.pow (2.0);
184  Math::Matrix<ValueType> one_over_dof (measurements.columns(), 1);
185  one_over_dof = 1.0 / ValueType(design.rows()-rank(design));
186  Math::Matrix<ValueType> stdev;
187  Math::mult (stdev, residuals, one_over_dof);
188  stdev.sqrt();
189  Math::mult (cohens_d, ValueType(1.0), CblasNoTrans, betas, CblasTrans, contrast);
190  cohens_d /= stdev;
191  }

263  scaled_contrasts, betas, residuals);
