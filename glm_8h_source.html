35  template <typename ValueType>
36  inline Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic> scale_contrasts (const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& contrasts,
37  const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& design,
38  size_t degrees_of_freedom)
39  {
40  assert (contrasts.cols() == design.cols());
41  Eigen::Matrix<default_type,Eigen::Dynamic, Eigen::Dynamic> XtX = (design.transpose() * design).template cast<default_type>();
42  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic> pinv_XtX = ((XtX.transpose() * XtX).fullPivLu().solve (XtX.transpose())).template cast<ValueType>();
43  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic> scaled_contrasts (contrasts);
44 
45  for (size_t n = 0; n < size_t(contrasts.rows()); ++n) {
46  Eigen::Matrix<ValueType, Eigen::Dynamic, 1> pinv_XtX_c = pinv_XtX * contrasts.row(n).transpose();
47  scaled_contrasts.row(n) *= std::sqrt (ValueType (degrees_of_freedom) / contrasts.row(n).dot (pinv_XtX_c));
48  }
49  return scaled_contrasts;
50  }

61  template <typename ValueType>
62  inline void ttest (
63  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& tvalues,
64  const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& design,
65  const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& pinv_design,
66  const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& measurements,
67  const Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& scaled_contrasts,
68  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& betas,
69  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic>& residuals)
70  {
71  betas.noalias() = measurements * pinv_design;
72  residuals.noalias() = measurements - betas * design;
73  tvalues.noalias() = betas * scaled_contrasts;
74  for (size_t n = 0; n < size_t(tvalues.rows()); ++n)
75  tvalues.row(n).array() /= residuals.row(n).norm();
76  }

165  void operator() (const std::vector<size_t>& perm_labelling, std::vector<float>& stats,
166  float& max_stat, float& min_stat) const
167  {
168  stats.resize (y.rows(), 0.0);
169  Eigen::MatrixXf tvalues, betas, residuals, SX, pinvSX;
170 
171  SX.resize (X.rows(), X.cols());
172  pinvSX.resize (pinvX.rows(), pinvX.cols());
173  for (ssize_t i = 0; i < X.rows(); ++i) {
174  SX.row(i) = X.row (perm_labelling[i]);
175  pinvSX.col(i) = pinvX.col (perm_labelling[i]);
176  }
177 
178  pinvSX.transposeInPlace();
179  SX.transposeInPlace();
180  for (ssize_t i = 0; i < y.rows(); i += GLM_BATCH_SIZE) {
181  Eigen::MatrixXf tmp = y.block (i, 0, std::min (GLM_BATCH_SIZE, (int)(y.rows()-i)), y.cols());
182  GLM::ttest (tvalues, SX, pinvSX, tmp, scaled_contrasts, betas, residuals);
183  for (ssize_t n = 0; n < tvalues.rows(); ++n) {
184  float val = tvalues(n,0);
185  if (std::isfinite (val)) {
186  if (val > max_stat)
187  max_stat = val;
188  if (val < min_stat)
189  min_stat = val;
190  } else {
191  val = float(0.0);
192  }
193  stats[i+n] = val;
194  }
195  }
196  }
197

199  size_t num_elements () const { return y.rows(); }
