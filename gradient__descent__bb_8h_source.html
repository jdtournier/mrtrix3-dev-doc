35  class LinearUpdateBB { NOMEMALIGN
36  public:
37  template <typename ValueType>
38  inline bool operator() (Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& newx, const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& x,
39  const Eigen::Matrix<ValueType, Eigen::Dynamic, 1>& g, ValueType step_size) {
40  assert (newx.size() == x.size());
41  assert (g.size() == x.size());
42  newx = x - step_size * g;
43  return !newx.isApprox(x);
44  }
45  };

172  bool iterate (std::ostream& log_os) {
173  assert (std::isfinite (normg));
174  if (!update_func (x3, x2, g2, dt))
175  return false;

177  f = evaluate_func (x3, g3, verbose);
178  x2.swap(x3);
179  x1.swap(x3);
