25 // These constants set how many times a tracking algorithm should attempt to propagate
26 // from a given seed point, based on the mechanism used to provide the seed point
27 //
28 // Update 12/03/2017: By default a greater number of attempts will be made to
29 // find an appropriate direction in which to initiate tracking from all
30 // seeding mechanisms
31 //
32 // Mechanisms that provide random seed locations
33 #define MAX_TRACKING_SEED_ATTEMPTS_RANDOM 1000
34 //
35 // Dynamic seeding also provides the mean direction of the fixel, so only a small number of
36 // attempts should be required to find a direction above the FOD amplitude threshold;
37 // this will however depend on this threshold as well as the angular threshold
38 #define MAX_TRACKING_SEED_ATTEMPTS_DYNAMIC 1000
39 //
40 // GM-WM interface seeding incurs a decent overhead when generating the seed points;
41 // therefore want to make maximal use of each seed point generated, bearing in mind that
42 // the FOD amplitudes may be small there.
43 #define MAX_TRACKING_SEED_ATTEMPTS_GMWMI 1000
44 //
45 // Mechanisms that provide a fixed number of seed points; hence the maximum effort should
46 // be made to find an appropriate tracking direction from every seed point provided
47 #define MAX_TRACKING_SEED_ATTEMPTS_FIXED 1000

64  template <class ImageType>
65  uint32_t get_count (ImageType& data)
66  {
67  std::atomic<uint32_t> count (0);
68  ThreadedLoop (data).run ([&] (ImageType& v) { if (v.value()) count.fetch_add (1, std::memory_order_relaxed); }, data);
69  return count;
70  }

73  template <class ImageType>
74  float get_volume (ImageType& data)
75  {
76  std::atomic<default_type> volume (0.0);
77  ThreadedLoop (data).run (
78  [&] (decltype(data)& v) {
79  const typename ImageType::value_type value = v.value();
80  if (value) {
81  default_type current = volume.load (std::memory_order_relaxed);
82  default_type target;
83  do {
84  target = current + value;
85  } while (!volume.compare_exchange_weak (current, target, std::memory_order_relaxed));
86  }
87  }, data);
88  return volume;
89  }
