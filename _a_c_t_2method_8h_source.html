60 term_t check_structural (const Eigen::Vector3f& pos)
61 {
62 if (!fetch_tissue_data (pos))
63 return EXIT_IMAGE;
64
65 if (tissues().is_csf())
66 return (sgm_depth ? EXIT_SGM : ENTER_CSF);
67
68 if (tissues().is_gm()) {
69 if (tissues().get_cgm() >= tissues().get_sgm())
70 return ENTER_CGM;
71 ++sgm_depth;
72 } else if (sgm_depth) {
73 if (seed_in_sgm && !sgm_seed_to_wm) {
74 sgm_seed_to_wm = true;
75 sgm_depth = 0;
76 return CONTINUE;
77 }
78 return EXIT_SGM;
79 }
80
81 return CONTINUE;
82 }

85 bool check_seed (const Eigen::Vector3f& pos)
86 {
87 sgm_depth = 0;
88
89 if (!fetch_tissue_data (pos))
90 return false;
91
92 if (tissues().is_sgm()) {
93 seed_in_sgm = true;
94 sgm_seed_to_wm = false;
95 return true;
96 }
97
98 seed_in_sgm = false;
99
100 if ((tissues().is_csf()) || !tissues().get_wm() || ((tissues().get_gm() - tissues().get_wm()) >= GMWMI_ACCURACY))
101 return false;
102
103 return true;
104 }

107 bool seed_is_unidirectional (const Eigen::Vector3f& pos, Eigen::Vector3f& dir)
108 {
109 // Tissue values should have already been acquired for the seed point when this function is run
110 if (tissues().is_sgm()) return false;
111 if ((tissues().get_wm() >= tissues().get_gm()) || (tissues().get_sgm() >= tissues().get_cgm()))
112 return false;
113
114 const Tissues tissues_at_pos (tissues());
115
116 const Eigen::Vector3f pos_plus (pos + (dir * GMWMI_NORMAL_PERTURBATION));
117 fetch_tissue_data (pos_plus);
118 const Tissues tissues_plus (tissues());
119
120 const Eigen::Vector3f pos_minus (pos - (dir * GMWMI_NORMAL_PERTURBATION));
121 fetch_tissue_data (pos_minus);
122 const Tissues& tissues_minus (tissues());
123
124 const float gradient = (tissues_plus.get_gm() - tissues_plus.get_wm()) - (tissues_minus.get_gm() - tissues_minus.get_wm());
125 if (gradient > 0.0)
126 dir = -dir;
127
128 // Restore the tissue values to those at the seed point
129 tissue_values = tissues_at_pos;
130 return true;
131 }

135 {
136 if (!act_image.scanner (pos)) {
137 tissue_values.reset();
138 return false;
139 }
140 return tissue_values.set (act_image);
141 }
