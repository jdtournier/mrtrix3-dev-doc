107  bool seed_is_unidirectional (const Eigen::Vector3f& pos, Eigen::Vector3f& dir)
108  {
109  // Tissue values should have already been acquired for the seed point when this function is run
110  if (tissues().is_sgm()) return false;
111  if ((tissues().get_wm() >= tissues().get_gm()) || (tissues().get_sgm() >= tissues().get_cgm()))
112  return false;
113 
114  const Tissues tissues_at_pos (tissues());
115 
116  const Eigen::Vector3f pos_plus (pos + (dir * GMWMI_NORMAL_PERTURBATION));
117  fetch_tissue_data (pos_plus);
118  const Tissues tissues_plus (tissues());
119 
120  const Eigen::Vector3f pos_minus (pos - (dir * GMWMI_NORMAL_PERTURBATION));
121  fetch_tissue_data (pos_minus);
122  const Tissues& tissues_minus (tissues());
123 
124  const float gradient = (tissues_plus.get_gm() - tissues_plus.get_wm()) - (tissues_minus.get_gm() - tissues_minus.get_wm());
125  if (gradient > 0.0)
126  dir = -dir;
127 
128  // Restore the tissue values to those at the seed point
129  tissue_values = tissues_at_pos;
130  return true;
