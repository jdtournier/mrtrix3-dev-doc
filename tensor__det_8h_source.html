129 void get_EV ()
130 {
131 M(0,0) = dt[0];
132 M(1,1) = dt[1];
133 M(2,2) = dt[2];
134 M(1,0) = M(0,1) = dt[3];
135 M(2,0) = M(0,2) = dt[4];
136 M(2,1) = M(1,2) = dt[5];

138 eig.computeDirect (M);
139 dir = eig.eigenvectors().col(2);
140 }

143 bool do_init()
144 {
145 dwi2tensor (dt, S.binv, values);
146
147 if (tensor2FA (dt) < S.init_threshold)
148 return false;
149
150 get_EV();
151
152 return true;
153 }
154
155
156 term_t do_next()
157 {
158
159 dwi2tensor (dt, S.binv, values);
160
161 if (tensor2FA (dt) < S.threshold)
162 return MODEL;
163
164 Eigen::Vector3f prev_dir = dir;
165
166 get_EV();
167
168 float dot = prev_dir.dot (dir);
169 if (abs (dot) < S.cos_max_angle_1o)
170 return HIGH_CURVATURE;
171
172 if (dot < 0.0)
173 dir = -dir;
174
175 pos += dir * S.step_size;
176
177 return CONTINUE;
178
179 }
