126  {

130 
131  std::unique_ptr<ProgressBar> progress;
132  if (message.size()) {

134  for (vector<default_type>::const_iterator i = stdev.begin(); i != stdev.end(); ++i)
135  if (*i)
136  ++axes_to_smooth;
137  progress.reset (new ProgressBar (message, axes_to_smooth + 1));
138  }
139 
140  for (size_t dim = 0; dim < 3; dim++) {
141  if (stdev[dim] > 0) {
142  DEBUG ("creating scratch image for smoothing image along dimension " + str(dim));
143  out = make_shared<Image<ValueType> > (Image<ValueType>::scratch (input));
144  Adapter::Gaussian1D<Image<ValueType> > gaussian (*in, stdev[dim], dim, extent[dim], zero_boundary);
145  threaded_copy (gaussian, *out, 0, input.ndim(), 2);

191  template <class ImageType>
192  class SmoothFunctor1D { MEMALIGN (SmoothFunctor1D)
193  public:
194  SmoothFunctor1D (ImageType& image,
195  default_type stdev_in = 1.0,
196  size_t axis_in = 0,
197  size_t extent = 0,
198  bool zero_boundary_in = false):
199  stdev (stdev_in),
200  axis (axis_in),
201  zero_boundary (zero_boundary_in),
202  spacing (image.spacing(axis_in)),
203  buffer_size (image.size(axis_in)) {
204  buffer.resize(buffer_size);
205  if (!extent)
206  radius = std::ceil(2 * stdev / spacing);
207  else if (extent == 1)
208  radius = 0;
209  else
210  radius = (extent - 1) / 2;
211  compute_kernel();
212  }
213 
214  using value_type = typename ImageType::value_type;
215 
216  void compute_kernel() {
217  if ((radius < 1) || stdev <= 0.0)
218  return;
219  kernel.resize (2 * radius + 1);
220  default_type norm_factor = 0.0;
221  for (ssize_t c = 0; c < kernel.size(); ++c) {
222  kernel[c] = exp(-((c-radius) * (c-radius) * spacing * spacing) / (2 * stdev * stdev));
223  norm_factor += kernel[c];
224  }
225  for (ssize_t c = 0; c < kernel.size(); c++) {
226  kernel[c] /= norm_factor;
227  }
228  }
229 
230  // SmoothFunctor1D operator():
231  // the inner loop axis has to be the dimension the smoothing is applied to and
232  // the loop has to start with image.index (smooth_axis) == 0
233  void operator () (ImageType& image) {
234  if (!kernel.size())
235  return;
236 
237  const ssize_t pos = image.index (axis);
238 
239  // fill buffer for current image line if necessary
240  if (pos == 0) {
241  for (ssize_t k = 0; k < buffer_size; ++k) {
242  image.index(axis) = k;
243  buffer(k) = image.value();
244  }
245  image.index (axis) = pos;
246  }
247 
248  if (zero_boundary)
249  if (pos == 0 || pos == image.size(axis) - 1) {
250  image.value() = 0.0;
251  return;
252  }
253 
254  const ssize_t from = (pos < radius) ? 0 : pos - radius;
255  const ssize_t to = (pos + radius) >= image.size(axis) ? image.size(axis) - 1 : pos + radius;
256 
257  ssize_t c = (pos < radius) ? radius - pos : 0;
258  ssize_t kernel_size = to - from + 1;
259 
260  value_type result = kernel.segment(c, kernel_size).dot(buffer.segment(from, kernel_size));
261 
262  if (!std::isfinite(result)) {
263  result = 0.0;
264  value_type av_weights = 0.0;
265  for (ssize_t k = from; k <= to; ++k, ++c) {
266  value_type neighbour_value = buffer(k);
267  if (std::isfinite (neighbour_value)) {
268  av_weights += kernel[c];
269  result += neighbour_value * kernel[c];
270  }
271  }
272  result /= av_weights;
273  } else if (kernel_size != kernel.size()) {
274  result /= kernel.segment(c, kernel_size).sum();
275  }
276  image.value() = result;
277  }
