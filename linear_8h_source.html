126  // set_scale_factor needs to be the first option that is set as it overwrites the stage vector
127  void set_scale_factor (const vector<default_type>& scalefactor) {
128  stages.resize (scalefactor.size());
129  for (size_t level = 0; level < scalefactor.size(); ++level) {
130  if (scalefactor[level] <= 0 || scalefactor[level] > 1.0)
131  throw Exception ("the linear registration scale factor for each multi-resolution level must be between 0 and 1");
132  stages[level].scale_factor = scalefactor[level];
133  }
134  }

154  void set_stage_iterations (const vector<int>& it) {
155  for (size_t i = 0; i < it.size (); ++i)
156  if (it[i] <= 0)
157  throw Exception ("the number of stage iterations must be positive");
158  if (it.size() == stages.size()) {
159  for (size_t i = 0; i < stages.size (); ++i)
160  stages[i].stage_iterations = it[i];
161  } else if (it.size() == 1) {
162  for (size_t i = 0; i < stages.size (); ++i)
163  stages[i].stage_iterations = it[0];
164  } else
165  throw Exception ("the number of stage iterations must be defined for all stages (1 or " + str(stages.size())+")");
166  for (auto & stage : stages) {
167  stage.optimisers.resize(stage.stage_iterations, stage.optimiser_default);
168  stage.optimisers[0] = stage.optimiser_first;
169  if (stage.stage_iterations > 1)
170  stage.optimisers[stage.stage_iterations - 1] = stage.optimiser_last;
171  }
172  }
173 
174  void set_max_iter (const vector<int>& maxiter) {
175  for (size_t i = 0; i < maxiter.size (); ++i)
176  if (maxiter[i] < 0)
177  throw Exception ("the number of iterations must be non-negative");
178  if (maxiter.size() == stages.size()) {
179  for (size_t i = 0; i < stages.size (); ++i)
180  stages[i].gd_max_iter = maxiter[i];
181  } else if (maxiter.size() == 1) {
182  for (size_t i = 0; i < stages.size (); ++i)
183  stages[i].gd_max_iter = maxiter[0];
184  } else
185  throw Exception ("the number of gradient descent iterations must be defined for all stages (1 or " + str(stages.size())+")");
186  }

194  void set_lmax (const vector<int>& lmax) {
195  for (size_t i = 0; i < lmax.size (); ++i)
196  if (lmax[i] < 0 || lmax[i] % 2)
197  throw Exception ("the input lmax must be positive and even");
198  if (lmax.size() == stages.size()) {
199  for (size_t i = 0; i < stages.size (); ++i)
200  stages[i].fod_lmax = lmax[i];
201  } else if (lmax.size() == 1) {
202  for (size_t i = 0; i < stages.size (); ++i)
203  stages[i].fod_lmax = lmax[0];
204  } else
205  throw Exception ("the lmax must be defined for all stages (1 or " + str(stages.size())+")");
206  }

208  void set_loop_density (const vector<default_type>& loop_density_){
209  for (size_t d = 0; d < loop_density_.size(); ++d)
210  if (loop_density_[d] < 0.0 or loop_density_[d] > 1.0 )
211  throw Exception ("loop density must be between 0.0 and 1.0");
212  if (loop_density_.size() == stages.size()) {
213  for (size_t i = 0; i < stages.size (); ++i)
214  stages[i].loop_density = loop_density_[i];
215  } else if (loop_density_.size() == 1) {
216  for (size_t i = 0; i < stages.size (); ++i)
217  stages[i].loop_density = loop_density_[0];
218  } else
219  throw Exception ("the lmax must be defined for all stages (1 or " + str(stages.size())+")");
220  }

231  }
232  }

305  template <class MetricType, class TransformType, class Im1ImageType, class Im2ImageType, class Im1MaskType, class Im2MaskType>
306  void run_masked (
307  MetricType& metric,
308  TransformType& transform,
309  Im1ImageType& im1_image,
310  Im2ImageType& im2_image,
311  Im1MaskType& im1_mask,
312  Im2MaskType& im2_mask) {
313 
314  if (do_reorientation)
315  for (auto & s : stages)
316  if (s.fod_lmax < 0)
317  throw Exception ("the lmax needs to be defined for each registration stage");
318 
319  if (init_translation_type == Transform::Init::mass)
320  Transform::Init::initialise_using_image_mass (im1_image, im2_image, im1_mask, im2_mask, transform, init);
321  else if (init_translation_type == Transform::Init::geometric)
322  Transform::Init::initialise_using_image_centres (im1_image, im2_image, im1_mask, im2_mask, transform, init);
323  else if (init_translation_type == Transform::Init::set_centre_mass) // doesn't change translation or linear matrix
324  Transform::Init::set_centre_via_mass (im1_image, im2_image, im1_mask, im2_mask, transform, init);
325  else if (init_translation_type == Transform::Init::set_centre_geometric) // doesn't change translation or linear matrix
326  Transform::Init::set_centre_via_image_centres (im1_image, im2_image, im1_mask, im2_mask, transform, init);
327 
328  if (init_rotation_type == Transform::Init::moments)
329  Transform::Init::initialise_using_image_moments (im1_image, im2_image, im1_mask, im2_mask, transform, init);
330  else if (init_rotation_type == Transform::Init::rot_search) {
331  Transform::Init::initialise_using_rotation_search (
332  im1_image, im2_image, im1_mask, im2_mask, transform, init);
333  }
334 
335  INFO ("Transformation before registration:");
336  INFO (transform.info());
337 
338  INFO("Linear registration stage parameters:");
339  for (auto & stage : stages) {
340  INFO(stage.info());
341  }
342 
343  // TODO global search
344  // if (global_search) {
345  // GlobalSearch::GlobalSearch transformation_search;
346  // if (log_stream) {
347  // transformation_search.set_log_stream(log_stream);
348  // }
349  // // std::ofstream outputFile( "/tmp/log.txt" );
350  // // transformation_search.set_log_stream(outputFile.rdbuf());
351  // // transformation_search.set_log_stream(std::cerr.rdbuf());
352  // transformation_search.run_masked (metric, transform, im1_image, im2_image, im1_mask, im2_mask, init);
353  // // transform.debug();
354  // }
355 
356  using MidwayImageType = Header;
357  using ProcessedImageType = Im1ImageType;
358  using ProcessedMaskType = Image<bool>;
359 
360 #ifdef REGISTRATION_CUBIC_INTERP
361  // using Im1ImageInterpolatorType = Interp::SplineInterp<Im1ImageType, Math::UniformBSpline<typename Im1ImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
362  // using Im2ImageInterpolatorType = Interp::SplineInterp<Im2ImageType, Math::UniformBSpline<typename Im2ImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
363  // using ProcessedImageInterpolatorType = Interp::SplineInterp<ProcessedImageType, Math::UniformBSpline<typename ProcessedImageType::value_type>, Math::SplineProcessingType::ValueAndDerivative>;
364 #else
365  using Im1ImageInterpolatorType = Interp::LinearInterp<Im1ImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
366  using Im2ImageInterpolatorType = Interp::LinearInterp<Im2ImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
367  using ProcessedImageInterpolatorType = Interp::LinearInterp<ProcessedImageType, Interp::LinearInterpProcessingType::ValueAndDerivative>;
368 #endif
369 
370  using ParamType = Metric::Params<TransformType,
371  Im1ImageType,
372  Im2ImageType,
373  MidwayImageType,
374  Im1MaskType,
375  Im2MaskType,
376  Im1ImageInterpolatorType,
377  Im2ImageInterpolatorType,
378  Interp::Linear<Im1MaskType>,
379  Interp::Linear<Im2MaskType>,
380  ProcessedImageType,
381  ProcessedImageInterpolatorType,
382  ProcessedMaskType,
383  Interp::Nearest<ProcessedMaskType>>;
384 
385  Eigen::Matrix<typename TransformType::ParameterType, Eigen::Dynamic, 1> optimiser_weights = transform.get_optimiser_weights();
386  const Eigen::Matrix<default_type, 4, 1> midspace_padding = Eigen::Matrix<default_type, 4, 1>(1.0, 1.0, 1.0, 1.0);
387  const int midspace_voxel_subsampling = 1;
388 
389  // calculate midway (affine average) space which will be constant for each resolution level
390  midway_image_header = compute_minimum_average_header (im1_image, im2_image, transform, midspace_voxel_subsampling, midspace_padding);
391 
392  for (size_t istage = 0; istage < stages.size(); istage++) {
393  auto& stage = stages[istage];
394 
395  CONSOLE ("linear stage " + str(istage + 1) + "/"+str(stages.size()) + ", " + stage.info(do_reorientation));
396 
397  INFO ("smoothing image 1");
398  auto im1_smoothed = Registration::multi_resolution_lmax (im1_image, stage.scale_factor, do_reorientation, stage.fod_lmax);
399  INFO ("smoothing image 2");
400  auto im2_smoothed = Registration::multi_resolution_lmax (im2_image, stage.scale_factor, do_reorientation, stage.fod_lmax);
401 
402  Filter::Resize midway_resize_filter (midway_image_header);
403  midway_resize_filter.set_scale_factor (stage.scale_factor);
404  Header midway_resized (midway_resize_filter);
405 
406  ParamType parameters (transform, im1_smoothed, im2_smoothed, midway_resized, im1_mask, im2_mask);
407  parameters.loop_density = stage.loop_density;
408  // if (robust_estimate)
409  // INFO ("using robust estimate");
410  // parameters.robust_estimate = robust_estimate; // TODO
411 
412  // set control point coordinates inside +-1/3 of the midway_image size
413  {
414  Eigen::Vector3 ext (midway_image_header.spacing(0) / 6.0,
415  midway_image_header.spacing(1) / 6.0,
416  midway_image_header.spacing(2) / 6.0);
417  for (size_t i = 0; i<3; ++i)
418  ext(i) *= midway_image_header.size(i) - 0.5;
419  parameters.set_control_points_extent(ext);
420  }
421  DEBUG ("neighbourhood kernel extent: " + str(kernel_extent));
422  parameters.set_extent (kernel_extent);
423  Eigen::Vector3d spacing (
424  midway_image_header.spacing(0),
425  midway_image_header.spacing(1),
426  midway_image_header.spacing(2));
427  Eigen::Vector3d coherence(spacing);
428  Eigen::Vector3d stop(spacing);
429  //CONF option: reg_coherence_len
430  //CONF default: 3.0
431  //CONF Linear registration: estimated spatial coherence length in voxels
432  default_type reg_coherence_len = File::Config::get_float ("reg_coherence_len", 3.0); // = 3 stdev blur
433  coherence *= reg_coherence_len * 1.0 / (2.0 * stage.scale_factor);
434  //CONF option: reg_stop_len
435  //CONF default: 0.0001
436  //CONF Linear registration: smallest gradient descent step measured in fraction of a voxel at which to stop registration
437  default_type reg_stop_len = File::Config::get_float ("reg_stop_len", 0.0001);
438  stop.array() *= reg_stop_len;
439  DEBUG ("coherence length: " + str(coherence));
440  DEBUG ("stop length: " + str(stop));
441  transform.get_gradient_descent_updator()->set_control_points (parameters.control_points, coherence, stop, spacing);
442 
443  // convergence check using slope of smoothed parameter trajectories
444  Eigen::VectorXd slope_threshold = Eigen::VectorXd::Ones (12);
445  //CONF option: reg_gd_convergence_thresh
446  //CONF default: 5e-3
447  //CONF Linear registration: threshold for convergence check using the smoothed control point trajectories
448  //CONF measured in fraction of a voxel
449  slope_threshold.fill (spacing.mean() * File::Config::get_float ("reg_gd_convergence_thresh", 5e-3f));
450  DEBUG ("convergence slope threshold: " + str(slope_threshold[0]));
451  //CONF option: reg_gd_convergence_data_smooth
452  //CONF default: 0.8
453  //CONF Linear registration: control point trajectory smoothing value used in convergence check
454  //CONF parameter range: [0...1]
455  const default_type alpha (MR::File::Config::get_float ("reg_gd_convergence_data_smooth", 0.8));
456  if ( (alpha < 0.0f ) || (alpha > 1.0f) )
457  throw Exception ("config file option reg_gd_convergence_data_smooth has to be in the range: [0...1]");
458  //CONF option: reg_gd_convergence_slope_smooth
459  //CONF default: 0.1
460  //CONF Linear registration: control point trajectory slope smoothing value used in convergence check
461  //CONF parameter range: [0...1]
462  const default_type beta (MR::File::Config::get_float ("reg_gd_convergence_slope_smooth", 0.1));
463  if ( (beta < 0.0f ) || (beta > 1.0f) )
464  throw Exception ("config file option reg_gd_convergence_slope_smooth has to be in the range: [0...1]");
465  size_t buffer_len (MR::File::Config::get_float ("reg_gd_convergence_buffer_len", 4));
466  //CONF option: reg_gd_convergence_min_iter
467  //CONF default: 10
468  //CONF Linear registration: minimum number of iterations until convergence check is activated
469  size_t min_iter (MR::File::Config::get_float ("reg_gd_convergence_min_iter", 10));
470  transform.get_gradient_descent_updator()->set_convergence_check (slope_threshold, alpha, beta, buffer_len, min_iter);
471 
472  Metric::Evaluate<MetricType, ParamType> evaluate (metric, parameters);
473  if (do_reorientation && stage.fod_lmax > 0)
474  evaluate.set_directions (aPSF_directions);
475 
476  INFO ("registration stage running...");
477  for (auto stage_iter = 1U; stage_iter <= stage.stage_iterations; ++stage_iter) {
478  if (stage.gd_max_iter > 0 and stage.optimisers[stage_iter - 1] == OptimiserAlgoType::bbgd) {
479  Math::GradientDescentBB<Metric::Evaluate<MetricType, ParamType>, typename TransformType::UpdateType>
480  optim (evaluate, *transform.get_gradient_descent_updator());
481  optim.be_verbose (analyse_descent);
482  optim.precondition (optimiser_weights);
483  optim.run (stage.gd_max_iter, grad_tolerance, analyse_descent ? std::cout.rdbuf() : log_stream);
484  parameters.optimiser_update (optim, evaluate.overlap());
485  INFO (" iteration: "+str(stage_iter)+"/"+str(stage.stage_iterations)+" GD iterations: "+
486  str(optim.function_evaluations())+" cost: "+str(optim.value())+" overlap: "+str(evaluate.overlap()));
487  } else if (stage.gd_max_iter > 0) {
488  Math::GradientDescent<Metric::Evaluate<MetricType, ParamType>, typename TransformType::UpdateType>
489  optim (evaluate, *transform.get_gradient_descent_updator());
490  optim.be_verbose (analyse_descent);
491  optim.precondition (optimiser_weights);
492  optim.run (stage.gd_max_iter, grad_tolerance, analyse_descent ? std::cout.rdbuf() : log_stream);
493  parameters.optimiser_update (optim, evaluate.overlap());
494  INFO (" iteration: "+str(stage_iter)+"/"+str(stage.stage_iterations)+" GD iterations: "+
495  str(optim.function_evaluations())+" cost: "+str(optim.value())+" overlap: "+str(evaluate.overlap()));
496  }
497 
498  if (log_stream) {
499  std::ostream log_os(log_stream);
500  log_os << "\n\n"; // two empty lines for gnuplot's index recognition
501  }
502  // debug cost function gradient
503  // auto params = optim.state();
504  // VAR(optim.function_evaluations());
505  // Math::check_function_gradient (evaluate, params, 0.0001, true, optimiser_weights);
506  if (stage.diagnostics_images.size()) {
507  CONSOLE(" creating diagnostics image: " + stage.diagnostics_images[stage_iter - 1]);
508  parameters.make_diagnostics_image (stage.diagnostics_images[stage_iter - 1], File::Config::get_bool ("reg_linreg_diagnostics_image_masked", false));
509  }
