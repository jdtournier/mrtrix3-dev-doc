113 class Extract :
114 public Base<Extract<ImageType>,ImageType>
115 { MEMALIGN (Extract<ImageType>)
116 public:
117
118 using base_type = Base<Extract<ImageType>, ImageType>;
119 using value_type = typename ImageType::value_type;
120

122 using base_type::spacing;

125
126 Extract (const ImageType& original, const vector<vector<int>>& indices) :
127 base_type (original),
128 current_pos (ndim()),
129 indices (indices),
130 trans (original.transform()) {
131 reset();
132 trans.translation() = trans * Eigen::Vector3d (
133 indices[0][0] * spacing (0),
134 indices[1][0] * spacing (1),
135 indices[2][0] * spacing (2)
136 );
137
138 for (auto& idx : this->indices) {
139 sizes.push_back (idx.size());
140 idx.push_back (idx.back());
141 }
142 }
143
144
145 ssize_t size (size_t axis) const { return sizes[axis]; }
146
147 const transform_type& transform () const { return trans; }
148
149 void reset () {
150 for (size_t n = 0; n < ndim(); ++n) {
151 current_pos[n] = 0;
152 parent().index(n) = indices[n][0];
153 }
154 }
155
156 ssize_t get_index (size_t axis) const { return current_pos[axis]; }
157 void move_index (size_t axis, ssize_t increment) {
158 ssize_t prev = current_pos[axis];
159 current_pos[axis] += increment;
160 parent().index (axis) += indices[axis][current_pos[axis]] - indices[axis][prev];
161 }
162
163 private:
164 vector<size_t> current_pos;
165 vector<vector<int> > indices;
166 vector<ssize_t> sizes;
167 transform_type trans;
