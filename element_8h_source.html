40 class Date { NOMEMALIGN
41 public:
42 Date (const std::string& entry) :
43 year(0), month(0), day(0) {
44 if (entry.size() >= 8) {
45 year = to<uint32_t> (entry.substr (0, 4));

47 day = to<uint32_t> (entry.substr (6, 2));

56 class Time { NOMEMALIGN
57 public:
58 Time (const std::string& entry) : Time() {
59 if (entry.size() < 6)
60 throw Exception ("field \"" + entry + "\" is too short to be interpreted as a time");
61 hour = to<uint32_t> (entry.substr (0, 2));
62 minute = to<uint32_t> (entry.substr (2, 2));
63 second = to<uint32_t> (entry.substr (4, 2));
64 fraction = entry.size() > 6 ? to<default_type> (entry.substr (6)) : 0.0;
65 }
66 Time (default_type i)
67 {
68 if (i < 0.0)
69 throw Exception ("Error converting negative floating-point number to a time");
70 hour = std::floor (i / 3600.0); i -= hour * 3600.0;
71 if (hour >= 24)
72 throw Exception ("Error converting floating-point number to a time: Beyond 24 hours");
73 minute = std::floor (i / 60.0); i -= minute * 60;
74 second = std::floor (i);
75 fraction = i - second;
76 }
77 Time () : hour (0), minute (0), second (0), fraction (0.0) { }
78 operator default_type() const { return (hour*3600.0 + minute*60 + second + fraction); }
79 Time operator- (const Time& t) const { return Time (default_type (*this) - default_type (t)); }
80 uint32_t hour, minute, second;
81 default_type fraction;
82 friend std::ostream& operator<< (std::ostream& stream, const Time& item);
83 };
