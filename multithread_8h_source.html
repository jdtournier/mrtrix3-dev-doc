132  bool operator() (const Mapped_track& in) {
133  assert (in.get_first_node() < data.rows());
134  assert (in.get_second_node() < data.rows());
135  assert (in.get_first_node() <= in.get_second_node());
136  data (in.get_first_node(), in.get_second_node()) += in.get_factor() * in.get_weight();
137  counts (in.get_first_node(), in.get_second_node()) += in.get_weight();
138  return true;
139  }

142  void scale_by_streamline_count() {
143  for (node_t i = 0; i != counts.rows(); ++i) {
144  for (node_t j = i; j != counts.columns(); ++j) {
145  if (counts (i, j)) {
146  data (i, j) /= counts (i, j);
147  counts (i, j) = 1;
148  }
149  }
150  }
151  }

154  void error_check (const std::set<node_t>& missing_nodes) {
155  std::vector<uint32_t> node_counts (num_nodes(), 0);
156  for (node_t i = 0; i != counts.rows() - 1; ++i) {
157  for (node_t j = i; j != counts.columns() - 1; ++j) {
158  node_counts[i] += counts (i, j);
159  node_counts[j] += counts (i, j);
160  }
161  }
162  std::vector<node_t> empty_nodes;
163  for (size_t i = 0; i != node_counts.size(); ++i) {
164  if (!node_counts[i] && missing_nodes.find (i) == missing_nodes.end())
165  empty_nodes.push_back (i);
166  }
167  if (empty_nodes.size()) {
168  WARN ("The following nodes do not have any streamlines assigned:");
169  std::string list = str(empty_nodes.front());
170  for (size_t i = 1; i != empty_nodes.size(); ++i)
171  list += ", " + str(empty_nodes[i]);
172  WARN (list);
173  WARN ("(This may indicate a poor registration)");
174  }
175  }
176 
177 
178  void remove_unassigned() {
179  for (node_t i = 0; i != data.rows() - 1; ++i) {
180  for (node_t j = i; j != data.columns() - 1; ++j) {
181  data (i, j) = data (i+1, j+1);
182  counts (i, j) = counts (i+1, j+1);
183  }
184  }
185  data .resize (data .rows() - 1, data .columns() - 1);
186  counts.resize (counts.rows() - 1, counts.columns() - 1);
187  }
188 
189 
190  void zero_diagonal() {
191  for (node_t i = 0; i != data.rows(); ++i)
192  data (i, i) = counts (i, i) = 0.0;
193  }
