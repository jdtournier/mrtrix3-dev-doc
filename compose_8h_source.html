73  ComposeDispKernel (Image<default_type>& disp_input1, Image<default_type>& disp_input2, default_type step) :
74  disp1_transform (disp_input1), disp2_interp (disp_input2), step (step) {}

78  void operator() (Image<default_type>& disp_input1, Image<default_type>& disp_output) {
79  Eigen::Vector3 voxel ((default_type)disp_input1.index(0), (default_type)disp_input1.index(1), (default_type)disp_input1.index(2));
80  Eigen::Vector3 voxel_position = disp1_transform.voxel2scanner * voxel;
81  Eigen::Vector3 original_position = voxel_position + disp_input1.row(3);
82  disp2_interp.scanner (original_position);
83  if (!disp2_interp) {
84  disp_output.row(3) = disp_input1.row(3);
85  } else {
86  Eigen::Vector3 displacement (disp2_interp.row(3).array() * step);
87  Eigen::Vector3 new_position = displacement + original_position;
88  disp_output.row(3) = new_position - voxel_position;
89  }
90  }
91 
92  protected:
93  MR::Transform disp1_transform;
94  Interp::Linear<Image<default_type> > disp2_interp;
95  default_type step;

99  template <class DeformationField1Type, class DeformationField2Type>
100  class ComposeHalfwayKernel {
101  public:
102  ComposeHalfwayKernel (const transform_type& linear1, DeformationField1Type& deform1,
103  DeformationField2Type& deform2, const transform_type& linear2) :
104  linear1 (linear1), deform1_interp (deform1), deform2_interp (deform2), linear2 (linear2) {
105  out_of_bounds.setOnes();
106  out_of_bounds *= NaN;
107  }
108 
109  EIGEN_MAKE_ALIGNED_OPERATOR_NEW // avoid memory alignment errors in Eigen3;
110 
111  void operator() (Image<default_type>& deform) {
112  Eigen::Vector3 voxel ((default_type)deform.index(0), (default_type)deform.index(1), (default_type)deform.index(2));
113  Eigen::Vector3 position = linear1 * voxel;
114  deform1_interp.scanner (position);
115  if (!deform1_interp) {
116  deform.row(3) = out_of_bounds;
117  } else {
118  Eigen::Vector3 position2 = deform1_interp.row(3);
119  deform2_interp.scanner (position2);
120  if (!deform2_interp) {
121  deform.row(3) = out_of_bounds;
122  } else {
123  Eigen::Vector3 position3 = deform2_interp.row(3);
124  deform.row(3) = linear2 * position3;
125  }
126  }
127  }
128 
129  protected:
130  const transform_type linear1;
131  Interp::Linear<DeformationField2Type> deform1_interp;
132  Interp::Linear<DeformationField2Type> deform2_interp;
133  const transform_type linear2;
134  Eigen::Vector3 out_of_bounds;
135  };

161  // Compose two displacement fields and output a displacement field using scaling and squaring. The input and output can be the same image.
162  FORCE_INLINE void update_displacement_scaling_and_squaring (Image<default_type>& input, Image<default_type>& update, Image<default_type>& output, const default_type step = 1.0)
163  {
164  check_dimensions (input, output, 0, 3);
165 
166  default_type max_norm = 0.0;
167  auto max_norm_func = [&max_norm](Image<default_type>& update) {
168  default_type norm = update.row(3).norm();
169  if (norm > max_norm)
170  max_norm = norm;
171  };
172  ThreadedLoop (update).run (max_norm_func, update);
173  default_type min_vox_size = static_cast<default_type> (std::min (input.spacing(0), std::min (input.spacing(1), input.spacing(2))));
174 
175  // if the maximum update is larger than half a voxel, perform scaling and squaring to ensure the displacement field remains diffeomorphic
176 
177  default_type scale_factor = 1.0;
178  if (max_norm * step < min_vox_size / 2.0) {
179  update_displacement (input, update, output, step);
180  } else {
181  scale_factor = std::pow (2, std::ceil (std::log ((max_norm * step) / (min_vox_size / 2.0)) / std::log (2.0)));
182 
183  std::shared_ptr<Image<default_type>> scaled_update = std::make_shared<Image<default_type> >(Image<default_type>::scratch (update));
184  std::shared_ptr<Image<default_type>> composed = std::make_shared<Image<default_type> >(Image<default_type>::scratch (update));
185 
186  // Scaling
187  default_type scaled_step = step / scale_factor; // apply the step size and scale factor at once
188  ThreadedLoop (update).run (
189  [&scaled_step](Image<default_type>& update, Image<default_type>& scaled_update) {
190  scaled_update.row(3) = update.row(3) * scaled_step;
191  }, update, *scaled_update);
192 
193 // CONSOLE ("composing " + str(std::log2 (scale_factor)) + "times");
194 
195  // Squaring
196  for (size_t i = 0; i < std::log2 (scale_factor); ++i) {
197  update_displacement (*scaled_update, *scaled_update, *composed);
198  std::swap (scaled_update, composed);
199  }

201 // Adapter::Jacobian<Image<default_type> > jacobian (*scaled_update);
202 // Header header (*scaled_update);
203 // header.ndim() = 3;
204 // bool is_neg = false;

206 // Eigen::MatrixXd ident = Eigen::MatrixXd::Identity (3,3);
207 // for (auto i = Loop (0,3) (jacobian, jacobian_det); i; ++i) {
208 // auto jac_matrix = ident + jacobian.value();

212 // }
