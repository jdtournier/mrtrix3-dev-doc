47  template <class MatrixType>
48  Eigen::MatrixXd& normalise_grad (MatrixType& grad)
49  {
50  if (grad.cols() < 3)
51  throw Exception ("invalid diffusion gradient table dimensions (" + str(grad.rows()) + " x " + str(grad.cols()) + ")");
52  for (ssize_t i = 0; i < grad.rows(); i++) {
53  auto norm = grad.row(i).template head<3>().norm();
54  if (norm)
55  grad.row(i).template head<3>().array() /= norm;
56  }
57  return grad;
58  }

64  template <class MatrixType, class IndexVectorType>
65  inline Eigen::MatrixXd gen_direction_matrix (
66  const MatrixType& grad,
67  const IndexVectorType& dwi)
68  {
69  Eigen::MatrixXd dirs (dwi.size(),2);
70  for (size_t i = 0; i < dwi.size(); i++) {
71  dirs (i,0) = std::atan2 (grad (dwi[i],1), grad (dwi[i],0));
72  auto z = grad (dwi[i],2) / grad.row (dwi[i]).template head<3>().norm();
73  if (z >= 1.0)
74  dirs(i,1) = 0.0;
75  else if (z <= -1.0)
76  dirs (i,1) = Math::pi;
77  else
78  dirs (i,1) = std::acos (z);
79  }
80  return dirs;
81  }

87  template <class MatrixType>
88  default_type condition_number_for_lmax (const MatrixType& dirs, int lmax)
89  {
90  Eigen::MatrixXd g;
91  if (dirs.cols() == 2) // spherical coordinates:
92  g = dirs;
93  else // Cartesian to spherical:
94  g = Math::Sphere::cartesian2spherical (dirs).leftCols(2);
95 
96  auto v = Eigen::JacobiSVD<Eigen::MatrixXd> (Math::SH::init_transform (g, lmax)).singularValues();
97  return v[0] / v[v.size()-1];
98  }

140  template <class MatrixType>
141  void set_DW_scheme (Header& header, const MatrixType& G)
142  {
143  if (!G.rows()) {
144  auto it = header.keyval().find ("dw_scheme");
145  if (it != header.keyval().end())
146  header.keyval().erase (it);

157  }

268  template <class MatrixType>
269  Eigen::MatrixXd compute_SH2amp_mapping (
270  const MatrixType& directions,
271  bool lmax_from_command_line = true,
272  int default_lmax = 8)
273  {
274  int lmax = -1;
275  int lmax_from_ndir = Math::SH::LforN (directions.rows());
276  bool lmax_set_from_commandline = false;
277  if (lmax_from_command_line) {
278  auto opt = App::get_options ("lmax");
279  if (opt.size()) {
280  lmax_set_from_commandline = true;
281  lmax = to<int> (opt[0][0]);
282  if (lmax % 2)
283  throw Exception ("lmax must be an even number");
284  if (lmax < 0)
285  throw Exception ("lmax must be a non-negative number");
286  if (lmax > lmax_from_ndir) {
287  WARN ("not enough directions for lmax = " + str(lmax) + " - dropping down to " + str(lmax_from_ndir));
288  lmax = lmax_from_ndir;
289  }
290  }
291  }
292 
293  if (lmax < 0) {
294  lmax = lmax_from_ndir;
295  if (lmax > default_lmax)
296  lmax = default_lmax;
297  }
298 
299  INFO ("computing SH transform using lmax = " + str (lmax));
300 
301  int lmax_prev = lmax;
302  Eigen::MatrixXd mapping;
303  do {
304  mapping = Math::SH::init_transform (directions, lmax);
305  auto v = Eigen::JacobiSVD<Eigen::MatrixXd> (mapping).singularValues();
306  auto cond = v[0] / v[v.size()-1];
307  if (cond < 10.0)
308  break;
309  WARN ("directions are poorly distributed for lmax = " + str(lmax) + " (condition number = " + str (cond) + ")");
310  if (cond < 100.0 || lmax_set_from_commandline)
311  break;
312  lmax -= 2;
313  } while (lmax >= 0);
314 
315  if (lmax_prev != lmax)
316  WARN ("reducing lmax to " + str(lmax) + " to improve conditioning");
317 
318  return mapping;
319  }
