52  Math::Matrix<ValueType>& normalise_grad (Math::Matrix<ValueType>& grad)
53  {
54  if (grad.columns() < 3)
55  throw Exception ("invalid gradient matrix dimensions");
56  for (size_t i = 0; i < grad.rows(); i++) {
57  ValueType norm = Math::norm (grad.row (i).sub (0,3));
58  if (norm)
59  grad.row (i).sub (0,3) /= norm;
60  }
61  return grad;
62  }

68  template <typename ValueType>
69  inline Math::Matrix<ValueType> gen_direction_matrix (
70  const Math::Matrix<ValueType>& grad,
71  const std::vector<size_t>& dwi)
72  {
73  Math::Matrix<ValueType> dirs (dwi.size(),2);
74  for (size_t i = 0; i < dwi.size(); i++) {
75  dirs (i,0) = std::atan2 (grad (dwi[i],1), grad (dwi[i],0));
76  ValueType z = grad (dwi[i],2) / Math::norm (grad.row (dwi[i]).sub (0,3));
77  if (z >= 1.0)
78  dirs(i,1) = 0.0;
79  else if (z <= -1.0)
80  dirs (i,1) = Math::pi;
81  else
82  dirs (i,1) = std::acos (z);
83  }
84  return dirs;
85  }

110  template <typename ValueType>
111  void load_bvecs_bvals (Math::Matrix<ValueType>& grad, const Image::Header& header, const std::string& bvecs_path, const std::string& bvals_path)
112  {
113  Math::Matrix<ValueType> bvals, bvecs;
114  bvals.load (bvals_path);
115  bvecs.load (bvecs_path);
116 
117  if (bvals.rows() != 1) throw Exception ("bvals file must contain 1 row only");
118  if (bvecs.rows() != 3) throw Exception ("bvecs file must contain exactly 3 rows");
119 
120  if (bvals.columns() != bvecs.columns() || bvals.columns() != size_t(header.dim (3)))
121  throw Exception ("bvals and bvecs files must have same number of diffusion directions as DW-image");
122 
123  // account for the fact that bvecs are specified wrt original image axes,
124  // which may have been re-ordered and/or inverted by MRtrix to match the
125  // expected anatomical frame of reference:
126  std::vector<size_t> order = Image::Stride::order (header, 0, 3);
127  Math::Matrix<ValueType> G (bvecs.columns(), 3);
128  for (size_t n = 0; n < G.rows(); ++n) {
129  G(n,order[0]) = header.stride(order[0]) > 0 ? bvecs(0,n) : -bvecs(0,n);
130  G(n,order[1]) = header.stride(order[1]) > 0 ? bvecs(1,n) : -bvecs(1,n);
131  G(n,order[2]) = header.stride(order[2]) > 0 ? bvecs(2,n) : -bvecs(2,n);
132  }
133 
134  // rotate gradients into scanner coordinate system:
135  grad.allocate (G.rows(), 4);
136  Math::Matrix<ValueType> grad_G = grad.sub (0, grad.rows(), 0, 3);
137  Math::Matrix<ValueType> rotation = header.transform().sub (0,3,0,3);
138  Math::mult (grad_G, ValueType(0.0), ValueType(1.0), CblasNoTrans, G, CblasTrans, rotation);
139 
140  grad.column(3) = bvals.row(0);
141  }

180  template <typename ValueType>
181  Math::Matrix<ValueType> get_DW_scheme (const Image::Header& header)
182  {
183  DEBUG ("searching for suitable gradient encoding...");
184  using namespace App;
185  Math::Matrix<ValueType> grad;
186 
187  try {
188  const Options opt_mrtrix = get_options ("grad");
189  if (opt_mrtrix.size())
190  grad.load (opt_mrtrix[0][0]);
191  const Options opt_fsl = get_options ("fslgrad");
192  if (opt_fsl.size()) {
193  if (opt_mrtrix.size())
194  throw Exception ("Please provide diffusion encoding using either -grad or -fslgrad option (not both)");
195  load_bvecs_bvals (grad, header, opt_fsl[0][0], opt_fsl[0][1]);
196  }
197  if (!opt_mrtrix.size() && !opt_fsl.size() && header.DW_scheme().is_set())
198  grad = header.DW_scheme();
199  }
200  catch (Exception& E) {
201  E.display (3);
202  throw Exception ("error importing diffusion encoding for image \"" + header.name() + "\"");
203  }
204 
205  if (!grad.rows())
206  return grad;
207 
208  if (grad.columns() < 4)
209  throw Exception ("unexpected diffusion encoding matrix dimensions");
210 
211  INFO ("found " + str (grad.rows()) + "x" + str (grad.columns()) + " diffusion-weighted encoding");
212 
213  return grad;
214  }

281  template <typename ValueType>
282  inline Math::Matrix<ValueType> compute_SH2amp_mapping (
283  const Math::Matrix<ValueType>& directions,
284  bool lmax_from_command_line = true,
285  int default_lmax = 8,
286  ValueType max_cond = 2.0)
287  {

290  bool lmax_set_from_commandline = false;

293  if (opt.size()) {
294  lmax_set_from_commandline = true;
295  lmax = to<int> (opt[0][0]);
296  if (lmax > lmax_from_ndir) {
297  WARN ("not enough directions for lmax = " + str(lmax) + " - dropping down to " + str(lmax_from_ndir));
298  lmax = lmax_from_ndir;
299  }
300  }
301  }
302 
303  if (lmax < 0) {
304  lmax = lmax_from_ndir;
305  if (lmax > default_lmax)
306  lmax = default_lmax;
307  }
308 
309  INFO ("computing SH transform using lmax = " + str (lmax));
310 
311  int lmax_prev = lmax;
312  Math::Matrix<ValueType> mapping;
313  do {
314  Math::SH::init_transform (mapping, directions, lmax);
315  double cond = Math::cond (mapping);
316  if (cond < 10.0)
317  break;
318  WARN ("directions are poorly distributed for lmax = " + str(lmax) + " (condition number = " + str (cond) + ")");
319  if (cond < 100.0 || lmax_set_from_commandline)
320  break;
321  lmax -= 2;
322  } while (lmax >= 0);
323 
324  if (lmax_prev != lmax)
325  WARN ("reducing lmax to " + str(lmax) + " to improve conditioning");
326 
327  return mapping;
328  }
