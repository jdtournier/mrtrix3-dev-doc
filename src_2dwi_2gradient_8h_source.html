46  template <class MatrixType>
47  Eigen::MatrixXd& normalise_grad (MatrixType& grad)
48  {
49  if (grad.cols() < 3)
50  throw Exception ("invalid diffusion gradient table dimensions");
51  for (ssize_t i = 0; i < grad.rows(); i++) {
52  auto norm = grad.row(i).template head<3>().norm();
53  if (norm)
54  grad.row(i).template head<3>().array() /= norm;
55  }
56  return grad;
57  }

63  template <class MatrixType, class IndexVectorType>
64  inline Eigen::MatrixXd gen_direction_matrix (
65  const MatrixType& grad,
66  const IndexVectorType& dwi)
67  {
68  Eigen::MatrixXd dirs (dwi.size(),2);
69  for (size_t i = 0; i < dwi.size(); i++) {
70  dirs (i,0) = std::atan2 (grad (dwi[i],1), grad (dwi[i],0));
71  auto z = grad (dwi[i],2) / grad.row (dwi[i]).template head<3>().norm();
72  if (z >= 1.0)
73  dirs(i,1) = 0.0;
74  else if (z <= -1.0)
75  dirs (i,1) = Math::pi;
76  else
77  dirs (i,1) = std::acos (z);
78  }
79  return dirs;
80  }

87  default_type condition_number_for_lmax (const MatrixType& dirs, int lmax)
88  {
89  Eigen::MatrixXd g;
90  if (dirs.cols() == 2) // spherical coordinates:

221  template <class MatrixType>
222  Eigen::MatrixXd compute_SH2amp_mapping (
223  const MatrixType& directions,
224  bool lmax_from_command_line = true,
225  int default_lmax = 8)
226  {
227  int lmax = -1;
228  int lmax_from_ndir = Math::SH::LforN (directions.rows());
229  bool lmax_set_from_commandline = false;
230  if (lmax_from_command_line) {
231  auto opt = App::get_options ("lmax");
232  if (opt.size()) {
233  lmax_set_from_commandline = true;
234  lmax = to<int> (opt[0][0]);
235  if (lmax % 2)
236  throw Exception ("lmax must be an even number");
237  if (lmax < 0)
238  throw Exception ("lmax must be a non-negative number");
239  if (lmax > lmax_from_ndir) {
240  WARN ("not enough directions for lmax = " + str(lmax) + " - dropping down to " + str(lmax_from_ndir));
241  lmax = lmax_from_ndir;
242  }
243  }
244  }
245 
246  if (lmax < 0) {
247  lmax = lmax_from_ndir;
248  if (lmax > default_lmax)
249  lmax = default_lmax;
250  }
251 
252  INFO ("computing SH transform using lmax = " + str (lmax));
253 
254  int lmax_prev = lmax;
255  Eigen::MatrixXd mapping;
256  do {
257  mapping = Math::SH::init_transform (directions, lmax);
258  auto v = Eigen::JacobiSVD<Eigen::MatrixXd> (mapping).singularValues();
259  auto cond = v[0] / v[v.size()-1];
260  if (cond < 10.0)
261  break;
262  WARN ("directions are poorly distributed for lmax = " + str(lmax) + " (condition number = " + str (cond) + ")");
263  if (cond < 100.0 || lmax_set_from_commandline)
264  break;
265  lmax -= 2;
266  } while (lmax >= 0);
267 
268  if (lmax_prev != lmax)
269  WARN ("reducing lmax to " + str(lmax) + " to improve conditioning");
270 
271  return mapping;
272  }
