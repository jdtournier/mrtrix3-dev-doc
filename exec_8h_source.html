120  bool operator() (GeneratedTrack& item) {
121  if (!seed_track (item))
122  return false;
123  if (track_excluded) {
124  item.set_status (GeneratedTrack::status_t::SEED_REJECTED);

128  gen_track (item);

130  item.clear();
131  item.set_status (GeneratedTrack::status_t::TRACK_REJECTED);
132  } else if (S.downsampler.get_ratio() > 1 || (S.is_act() && S.act().crop_at_gmwmi())) {
133  S.downsampler (item);
134  check_downsampled_length (item);
135  } else {
136  item.set_status (GeneratedTrack::status_t::ACCEPTED);

150  term_t iterate ()
151  {
152  const term_t method_term = (S.rk4 ? next_rk4() : method.next());
153 
154  if (method_term)
155  return (S.is_act() && method.act().sgm_depth) ? TERM_IN_SGM : method_term;
156 
157  if (S.is_act()) {
158  const term_t structural_term = method.act().check_structural (method.pos);
159  if (structural_term)
160  return structural_term;
161  }
162 
163  if (S.properties.mask.size() && !S.properties.mask.contains (method.pos))
164  return EXIT_MASK;
165 
166  if (S.properties.exclude.contains (method.pos))
167  return ENTER_EXCLUDE;
168 
169  // If backtracking is not enabled, add streamline to include regions as it is generated
170  // If it is enabled, this check can only be performed after the streamline is completed
171  if (!(S.is_act() && S.act().backtrack()))
172  include_visitation (method.pos);
173 
174  if (S.stop_on_all_include && bool(include_visitation))
175  return TRAVERSE_ALL_INCLUDE;
176 
177  return CONTINUE;

182  bool seed_track (GeneratedTrack& tck)
183  {
184  tck.clear();
185  track_excluded = false;
186  include_visitation.reset();
187  method.dir = { NaN, NaN, NaN };
188 
189  if (S.properties.seeds.is_finite()) {
190 
191  if (!S.properties.seeds.get_seed (method.pos, method.dir))
192  return false;
193  if (!method.check_seed() || !method.init()) {
194  track_excluded = true;
195  tck.set_status (GeneratedTrack::status_t::SEED_REJECTED);
196  }
197  return true;
198

200 
201  for (size_t num_attempts = 0; num_attempts != MAX_NUM_SEED_ATTEMPTS; ++num_attempts) {
202  if (S.properties.seeds.get_seed (method.pos, method.dir)) {
203  if (!(method.check_seed() && method.init())) {
204  track_excluded = true;
205  tck.set_status (GeneratedTrack::status_t::SEED_REJECTED);
206  }
207  return true;

245  void gen_track_unidir (GeneratedTrack& tck)
246  {
247  term_t termination = CONTINUE;
248 
249  if (S.is_act() && S.act().backtrack()) {
250 
251  size_t revert_step = 1;
252  size_t max_size_at_backtrack = tck.size();
253  unsigned int revert_count = 0;
254 
255  do {
256  termination = iterate();
257  if (term_add_to_tck[termination])
258  tck.push_back (method.pos);
259  if (termination) {
260  apply_priors (termination);
261  if (track_excluded && termination != ENTER_EXCLUDE) {
262  if (tck.size() > max_size_at_backtrack) {
263  max_size_at_backtrack = tck.size();
264  revert_step = 1;
265  revert_count = 1;
266  } else {
267  if (revert_count++ == ACT_BACKTRACK_ATTEMPTS) {
268  revert_count = 1;
269  ++revert_step;
270  }
271  }
272  method.truncate_track (tck, max_size_at_backtrack, revert_step);
273  if (method.pos.allFinite()) {
274  track_excluded = false;
275  termination = CONTINUE;
276  }
277  }
278  } else if (tck.size() >= S.max_num_points_preds) {
279  termination = LENGTH_EXCEED;
280  }
281  } while (!termination);
282 
283  } else {
284 
285  do {
286  termination = iterate();
287  if (term_add_to_tck[termination])
288  tck.push_back (method.pos);
289  if (!termination && tck.size() >= S.max_num_points_preds)
290  termination = LENGTH_EXCEED;
291  } while (!termination);
292 
293  }
294 
295  apply_priors (termination);
296 
297  if (termination == EXIT_SGM) {
298  truncate_exit_sgm (tck);
299  method.pos = tck.back();
300  }
301 
302  if (track_excluded) {
303  switch (termination) {
304  case CALIBRATOR: case ENTER_CSF: case MODEL: case HIGH_CURVATURE:
305  S.add_rejection (ACT_POOR_TERMINATION);
306  break;
307  case LENGTH_EXCEED:
308  S.add_rejection (TRACK_TOO_LONG);
309  break;
310  case ENTER_EXCLUDE:
311  S.add_rejection (ENTER_EXCLUDE_REGION);
312  break;
313  default:
314  throw Exception ("\nFIXME: Unidirectional track excluded but termination is good!\n");
315  }
316  }
317 
318  if (S.is_act() && (termination == ENTER_CGM) && S.act().crop_at_gmwmi())
319  S.act().crop_at_gmwmi (tck);
320 
321 #ifdef DEBUG_TERMINATIONS
322  S.add_termination (termination, method.pos);
323 #else
324  S.add_termination (termination);
325 #endif
326 
327  }

331  void apply_priors (term_t& termination)
332  {
333 
334  if (S.is_act()) {
335 
336  switch (termination) {
337 
338  case CONTINUE:
339  throw Exception ("\nFIXME: undefined termination of track in apply_priors()\n");
340 
341  case ENTER_CGM: case EXIT_IMAGE: case EXIT_MASK: case EXIT_SGM: case TERM_IN_SGM: case TRAVERSE_ALL_INCLUDE:
342  break;
343 
344  case ENTER_CSF: case LENGTH_EXCEED: case ENTER_EXCLUDE:
345  track_excluded = true;
346  break;
347 
348  case CALIBRATOR: case MODEL: case HIGH_CURVATURE:
349  if (method.act().sgm_depth)
350  termination = TERM_IN_SGM;
351  else if (!method.act().in_pathology())
352  track_excluded = true;
353  break;
354 
355  }
356 
357  } else {
358 
359  switch (termination) {
360 
361  case CONTINUE:
362  throw Exception ("\nFIXME: undefined termination of track in apply_priors()\n");
363 
364  case ENTER_CGM: case ENTER_CSF: case EXIT_SGM: case TERM_IN_SGM:
365  throw Exception ("\nFIXME: Have received ACT-based termination for non-ACT tracking in apply_priors()\n");
366 
367  case EXIT_IMAGE: case EXIT_MASK: case LENGTH_EXCEED: case CALIBRATOR: case MODEL: case HIGH_CURVATURE: case TRAVERSE_ALL_INCLUDE:
368  break;
369 
370  case ENTER_EXCLUDE:
371  track_excluded = true;
372  break;

382  bool track_rejected (const GeneratedTrack& tck)
383  {
384 
385  if (track_excluded)
386  return true;
387 
388  // seedtest algorithm uses min_num_points_preds = 1; should be 2 or more for all other algorithms
389  if (tck.size() == 1 && S.min_num_points_preds > 1) {
390  S.add_rejection (NO_PROPAGATION_FROM_SEED);
391  return true;
392  }
393 
394  if (tck.size() < S.min_num_points_preds) {
395  S.add_rejection (TRACK_TOO_SHORT);
396  return true;
397  }
398 
399  if (S.is_act()) {
400 
401  if (!satisfy_wm_requirement (tck)) {
402  S.add_rejection (ACT_FAILED_WM_REQUIREMENT);
403  return true;
404  }
405 
406  if (S.act().backtrack()) {
407  for (const auto& i : tck)
408  include_visitation (i);
409  }
410 
411  }
412 
413  if (!bool(include_visitation)) {
414  S.add_rejection (MISSED_INCLUDE_REGION);
415  return true;
416  }
417 
418  return false;

422  bool satisfy_wm_requirement (const vector<Eigen::Vector3f>& tck)
423  {
424  // If using the Seed_test algorithm (indicated by max_num_points == 2), don't want to execute this check
425  if (S.max_num_points_preds == 2)
426  return true;
427  // If the seed was in SGM, need to confirm that one side of the track actually made it to WM
428  if (method.act().seed_in_sgm && !method.act().sgm_seed_to_wm)
429  return false;
430  // Used these in the ACT paper, but wasn't entirely happy with the method; can change these #defines to re-enable
431  // ACT instead now defaults to a 2-voxel minimum length
432  if (!ACT_WM_INT_REQ && !ACT_WM_ABS_REQ)
433  return true;
434  float integral = 0.0, max_value = 0.0;
435  for (const auto& i : tck) {
436  if (method.act().fetch_tissue_data (i)) {
437  const float wm = method.act().tissues().get_wm();
438  max_value = std::max (max_value, wm);
439  if (((integral += (Math::pow2 (wm) * S.internal_step_size())) > ACT_WM_INT_REQ) && (max_value > ACT_WM_ABS_REQ))
440  return true;
441  }
442  }
443  return false;
444  }

448  void truncate_exit_sgm (vector<Eigen::Vector3f>& tck)
449  {
450  Interpolator<Image<float>>::type source (S.source);
451 
452  const size_t sgm_start = tck.size() - method.act().sgm_depth;
453  assert (sgm_start >= 0 && sgm_start < tck.size());
454  size_t best_termination = tck.size() - 1;
455  float min_value = std::numeric_limits<float>::infinity();
456  for (size_t i = sgm_start; i != tck.size(); ++i) {
457  method.pos = tck[i];
458  method.get_data (source);
459  method.dir = (tck[i] - tck[i-1]).normalized();
460  const float this_value = method.get_metric();
461  if (this_value < min_value) {
462  min_value = this_value;
463  best_termination = i;
464  }
465  }
466  tck.erase (tck.begin() + best_termination + 1, tck.end());
467  }

471  void check_downsampled_length (GeneratedTrack& tck)
472  {
473  // Don't quantify the precise streamline length if we don't have to
474  // (i.e. we know for sure that even in the presence of downsampling,
475  // we're not going to break either of these two criteria)
476  if (tck.size() > S.min_num_points_postds && tck.size() < S.max_num_points_postds) {
477  tck.set_status (GeneratedTrack::status_t::ACCEPTED);
478  return;
479  }
480  const float length = Tractography::length (tck);
481  if (length < S.min_dist) {
482  tck.clear();
483  tck.set_status (GeneratedTrack::status_t::TRACK_REJECTED);
484  S.add_rejection (TRACK_TOO_SHORT);
485  } else if (length > S.max_dist) {
486  if (S.is_act()) {
487  tck.clear();
488  tck.set_status (GeneratedTrack::status_t::TRACK_REJECTED);
489  S.add_rejection (TRACK_TOO_LONG);
490  } else {
491  truncate_maxlength (tck);
492  tck.set_status (GeneratedTrack::status_t::ACCEPTED);
493  }
494  } else {
495  tck.set_status (GeneratedTrack::status_t::ACCEPTED);
496  }
497  }

501  void truncate_maxlength (GeneratedTrack& tck)
502  {
503  // Chop the track short so that the length is precisely the maximum length
504  // If the truncation would result in removing the seed point, truncate all
505  // the way back to the seed point, reverse the streamline, and then
506  // truncate off the end (which used to be the start)
507  float length_sum = 0.0f;
508  size_t index;
509  for (index = 1; index != tck.size(); ++index) {
510  const float seg_length = (tck[index] - tck[index-1]).norm();
511  if (length_sum + seg_length > S.max_dist)
512  break;
513  length_sum += seg_length;
514  }
515 
516  // If we don't exceed the maximum length, this function should never have been called!
517  assert (index != tck.size());
518  // But nevertheless; if we happen to somehow get here, let's just allow processing to continue
519  if (index == tck.size())
520  return;
521 
522  // Would truncation at this vertex (plus including a new small segment at the end) result in
523  // discarding the seed point? If so:
524  // - Truncate so that the seed point is the last point on the streamline
525  // - Reverse the order of the vertices
526  // - Re-run this function recursively in order to truncate from the opposite end of the streamline
527  if (tck.get_seed_index() >= index) {
528  tck.resize (tck.get_seed_index()+1);
529  tck.reverse();
530  truncate_maxlength (tck);
531  return;
532  }
533 
534  // We want to determine a new vertex in between "index" and the prior vertex, which
535  // is of the appropriate distance away from "index" in order to make the streamline
536  // precisely the maximum length
537  tck.resize (index+1);
538  tck[index] = tck[index-1] + ((tck[index] - tck[index-1]).normalized() * (S.max_dist - length_sum));
539  }

543  term_t next_rk4()
544  {
545  term_t termination = CONTINUE;
546  const Eigen::Vector3f init_pos (method.pos);
547  const Eigen::Vector3f init_dir (method.dir);
548  if ((termination = method.next()))
549  return termination;
550  const Eigen::Vector3f dir_rk1 (method.dir);
551  method.pos = init_pos + (dir_rk1 * (0.5 * S.step_size));
552  method.dir = init_dir;
553  if ((termination = method.next()))
554  return termination;
555  const Eigen::Vector3f dir_rk2 (method.dir);
556  method.pos = init_pos + (dir_rk2 * (0.5 * S.step_size));
557  method.dir = init_dir;
558  if ((termination = method.next()))
559  return termination;
560  const Eigen::Vector3f dir_rk3 (method.dir);
561  method.pos = init_pos + (dir_rk3 * S.step_size);
562  method.dir = (dir_rk2 + dir_rk3).normalized();
563  if ((termination = method.next()))
564  return termination;
565  const Eigen::Vector3f dir_rk4 (method.dir);
566  method.dir = (dir_rk1 + (dir_rk2 * 2.0) + (dir_rk3 * 2.0) + dir_rk4).normalized();
567  method.pos = init_pos + (method.dir * S.step_size);
568  const Eigen::Vector3f final_pos (method.pos);
569  const Eigen::Vector3f final_dir (method.dir);
570  if ((termination = method.next()))
571  return termination;
572  if (dir_rk1.dot (method.dir) < S.cos_max_angle_ho)
573  return HIGH_CURVATURE;
574  method.pos = final_pos;
575  method.dir = final_dir;
576  return CONTINUE;
577  }
