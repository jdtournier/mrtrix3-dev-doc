171  bool gen_track (GeneratedTrack& tck)
172  {
173  tck.clear();
174  track_excluded = false;
175  track_included.assign (track_included.size(), false);
176  method.dir.invalidate();
177 
178  bool unidirectional = S.unidirectional;
179 
180  if (S.properties.seeds.is_finite()) {
181 
182  if (!S.properties.seeds.get_seed (method.pos, method.dir))
183  return false;
184  if (!method.check_seed() || !method.init()) {
185  track_excluded = true;
186  return true;
187  }
188 
189  } else {
190 
191  for (size_t num_attempts = 0; num_attempts != MAX_NUM_SEED_ATTEMPTS; ++num_attempts) {
192  if (S.properties.seeds.get_seed (method.pos, method.dir) && method.check_seed() && method.init())
193  break;
194  }
195  if (!method.pos.valid()) {
196  FAIL ("Failed to find suitable seed point after " + str (MAX_NUM_SEED_ATTEMPTS) + " attempts - aborting");
197  return false;
198  }
199 
200  }
201 
202  if (S.is_act() && !unidirectional)
203  unidirectional = method.act().seed_is_unidirectional (method.pos, method.dir);
204 
205  S.properties.include.contains (method.pos, track_included);
206

213  tck.reverse();
214  method.pos = tck.back();
215  method.dir = -seed_dir;

227  void gen_track_unidir (GeneratedTrack& tck)
228  {
229 
230  if (S.is_act())

238

404  bool satisfy_wm_requirement (const std::vector< Point<float> >& tck)
405  {
406  // If using the Seed_test algorithm (indicated by max_num_points == 2), don't want to execute this check
407  if (S.max_num_points == 2)
408  return true;
409  // Used these in the ACT paper, but wasn't entirely happy with the method; can change these #defines to re-enable
410  // ACT instead now defaults to a 2-voxel minimum length
411  if (!ACT_WM_INT_REQ && !ACT_WM_ABS_REQ)
412  return true;
413  float integral = 0.0, max_value = 0.0;
414  for (std::vector< Point<float> >::const_iterator i = tck.begin(); i != tck.end(); ++i) {
415  if (method.act().fetch_tissue_data (*i)) {
416  const float wm = method.act().tissues().get_wm();
417  max_value = MAX (max_value, wm);
418  if (((integral += (Math::pow2 (wm) * S.internal_step_size())) > ACT_WM_INT_REQ) && (max_value > ACT_WM_ABS_REQ))
419  return true;
420  }
421  }
422  return false;

452  term_t next_rk4()
453  {
454  term_t termination = CONTINUE;
455  const Point<value_type> init_pos (method.pos);
456  const Point<value_type> init_dir (method.dir);
457  if ((termination = method.next()))
458  return termination;

461  method.dir = init_dir;
462  if ((termination = method.next()))
463  return termination;
464  const Point<value_type> dir_rk2 (method.dir);
465  method.pos = init_pos + (dir_rk2 * (0.5 * S.step_size));
466  method.dir = init_dir;
467  if ((termination = method.next()))
468  return termination;
469  const Point<value_type> dir_rk3 (method.dir);
470  method.pos = init_pos + (dir_rk3 * S.step_size);
471  method.dir = (dir_rk2 + dir_rk3).normalise();
472  if ((termination = method.next()))
473  return termination;
474  const Point<value_type> dir_rk4 (method.dir);
475  method.dir = (dir_rk1 + (dir_rk2 * 2.0) + (dir_rk3 * 2.0) + dir_rk4).normalise();

477  const Point<value_type> final_pos (method.pos);
478  const Point<value_type> final_dir (method.dir);
479  if ((termination = method.next()))
480  return termination;
481  if (dir_rk1.dot (method.dir) < S.cos_max_angle_rk4)
482  return HIGH_CURVATURE;
483  method.pos = final_pos;
484  method.dir = final_dir;
