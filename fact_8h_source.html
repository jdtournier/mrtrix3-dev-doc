106

128 float do_next (Eigen::Vector3f& d) const
129 {
130
131 int idx = -1;
132 float max_abs_dot = 0.0, max_dot = 0.0, max_norm = 0.0;
133
134 for (size_t n = 0; n < S.num_vec; ++n) {
135 Eigen::Vector3f v (values[3*n], values[3*n+1], values[3*n+2]);
136 float norm = v.norm();
137 float dot = v.dot(d) / norm;
138 float abs_dot = abs (dot);
139 if (abs_dot < S.dot_threshold) continue;
140 if (max_abs_dot < abs_dot) {
141 max_abs_dot = abs_dot;
142 max_dot = dot;
143 max_norm = norm;
144 idx = n;
145 }
146 }
147
148 if (idx < 0) return (0.0);
149
150 d = { values[3*idx], values[3*idx+1], values[3*idx+2] };
151 d.normalize();
152 if (max_dot < 0.0)
153 d = -d;
154
155 return max_norm;
156 }
157
158 };
