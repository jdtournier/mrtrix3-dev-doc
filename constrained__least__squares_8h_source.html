74  const matrix_type& constraint_matrix,
75  value_type solution_min_norm_regularisation = 0.0,
76  value_type constraint_min_norm_regularisation = 0.0,
77  size_t max_iterations = 0,
78  value_type tolerance = 0.0) :
79  H (problem_matrix),
80  chol_HtH (H.cols(), H.cols()),
81  lambda_min_norm (constraint_min_norm_regularisation),
82  tol (tolerance),
83  max_niter (max_iterations ? max_iterations : 10*problem_matrix.cols()) {
84 
85  if (H.cols() != constraint_matrix.cols())
86  throw Exception ("FIXME: dimensions of problem and constraint matrices do not match (ICLS)");
87 
88  if (solution_min_norm_regularisation < 0.0)
89  throw Exception ("FIXME: solution norm regularisation is negative (ICLS)");
90 
91  if (lambda_min_norm < 0.0)
92  throw Exception ("FIXME: constraint norm regularisation is negative (ICLS)");
93 
94  if (tolerance < 0.0)
95  throw Exception ("FIXME: tolerance is negative (ICLS)");
96 
97  // form quadratic problem matrix H'*H:
98  chol_HtH.setZero();
99  chol_HtH.template triangularView<Eigen::Lower>() = H.transpose() * H;
100  // add minimum norm constraint:
101  chol_HtH.diagonal().array() += solution_min_norm_regularisation * chol_HtH.diagonal().maxCoeff();
102  // get Cholesky decomposition:
103  chol_HtH.template triangularView<Eigen::Lower>() = chol_HtH.template selfadjointView<Eigen::Lower>().llt().matrixL();
104 
105  // form (transpose of) matrix projecting b onto preconditioned
106  // quadratic problem chol_HtH\H:
107  b2d.noalias() = chol_HtH.template triangularView<Eigen::Lower>().transpose().template solve<Eigen::OnTheRight> (H);
108 
109  // project constraint onto preconditioned quadratic domain,
110  // and normalise each row to help preconditioning (the norm of
111  // each row is irrelevant to the constraint itself):
112  B.noalias() = chol_HtH.template triangularView<Eigen::Lower>().transpose().template solve<Eigen::OnTheRight> (constraint_matrix);
113  for (ssize_t n = 0; n < B.rows(); ++n)
114  B.row(n).normalize();
115  }

137  Solver (const Problem<value_type>& problem) :
138  P (problem),
139  BtB (P.chol_HtH.rows(), P.chol_HtH.cols()),
140  B (P.B.rows(), P.B.cols()),
141  y_u (BtB.rows()),
142  c (P.B.rows()),
143  c_u (P.B.rows()),
144  lambda (c.size()),
145  lambda_prev (c.size()),
146  l (lambda.size()),
147  active (lambda.size(), false) { }
148 
149  size_t operator() (vector_type& x, const vector_type& b)
150  {
151 #ifdef MRTRIX_ICLS_DEBUG
152  std::ofstream l_stream ("l.txt");
153  std::ofstream n_stream ("n.txt");
154 #endif
155  // compute unconstrained solution:
156  y_u = P.b2d.transpose() * b;
157  // compute constraint violations for unconstrained solution:
158  c_u = P.B * y_u;
159 
160  // set all Lagrangian multipliers to zero:
161  lambda.setZero();
162  lambda_prev.setZero();
163  // set active set empty:
164  std::fill (active.begin(), active.end(), false);
165 
166  // initial estimate of constraint values:
167  c = c_u;
168  // initial estimate of solution:
169  x = y_u;
170 
171  size_t min_c_index;
172  size_t niter = 0;
173 
174  while (c.minCoeff (&min_c_index) < -P.tol) {
175  bool active_set_changed = !active[min_c_index];
176  active[min_c_index] = true;
177 
178  while (1) {
179  // form submatrix of active constraints:
180  size_t num_active = 0;
181  for (size_t n = 0; n < active.size(); ++n) {
182  if (active[n]) {
183  B.row (num_active) = P.B.row (n);
184  l[num_active] = -c_u[n];
185  ++num_active;
186  }
187  }
188  auto B_active = B.topRows (num_active);
189  auto l_active = l.head (num_active);
190 
191  BtB.resize (num_active, num_active);
192  // solve for l in B*B'l = -c_u by Cholesky decomposition:
193  BtB = B_active * B_active.transpose();
194  BtB.diagonal().array() += P.lambda_min_norm;
195  BtB.template selfadjointView<Eigen::Lower>().llt().solveInPlace (l_active);
196 
197  // update lambda values in full vector
198  // and identify worst offender if any lambda < 0
199  // by projection from previous onto feasible
200  // subset (i.e. l>=0):
201  value_type s_min = std::numeric_limits<value_type>::infinity();
202  size_t s_min_index = 0;
203  size_t a = 0;
204  for (size_t n = 0; n < active.size(); ++n) {
205  if (active[n]) {
206  if (l_active[a] < 0.0) {
207  value_type s = lambda_prev[n] / (lambda_prev[n] - l_active[a]);
208  if (s < s_min) {
209  s_min = s;
210  s_min_index = n;
211  }
212  }
213  lambda[n] = l_active[a];
214  ++a;
215  }
216  else
217  lambda[n] = 0.0;
218  }
219 
220  // if no lambda < 0, proceed:
221  if (!std::isfinite (s_min)) {
222  // update solution vector:
223  x = y_u + B_active.transpose() * l_active;
224  break;
225  }
226 #ifdef MRTRIX_ICLS_DEBUG
227  l_stream << lambda << "\n";
228 #endif
229 
230  // remove worst offending lambda from active set,
231  // and re-estimate remaining lambdas:
232  if (active[s_min_index])
233  active_set_changed = true;
234  active[s_min_index] = false;
235  }
236 
237  // store feasible subset of lambdas:
238  lambda_prev = lambda;
239 
240 
241 #ifdef MRTRIX_ICLS_DEBUG
242  l_stream << lambda << "\n";
243  for (const auto& a : active)
244  n_stream << a << " ";
245  n_stream << "\n";
246 #endif
247 
248  ++niter;
249  if (!active_set_changed || niter > P.max_niter)
250  break;
251 
252  // compute constraint values at updated solution:
253  c = P.B * x;
254  }
255 
256  // project back to unconditioned domain:
257  P.chol_HtH.template triangularView<Eigen::Lower>().transpose().solveInPlace (x);
258  return niter;
259  }
