116  NormFixel (const BaseType& i) :
117  BaseType (i),
118  norm_multiplier (connectivity_value_type (1)) { }

120  BaseType (std::move (i)),
121  norm_multiplier (connectivity_value_type (1)) { }
122  void normalise() {
123  norm_multiplier = connectivity_value_type (0);
124  for (const auto& c : *this)
125  norm_multiplier += c.value();
126  norm_multiplier = norm_multiplier ? (connectivity_value_type (1) / norm_multiplier) : connectivity_value_type(0);
127  }

129  norm_multiplier = sum ? (connectivity_value_type(1) / sum) : connectivity_value_type(0);

161  // New code for handling load/save of fixel-fixel connectivity matrix
162  // Use something akin to the fixel directory format, with a sub-directory
163  // within an existing fixel directory containing the following data:
164  // - index.mif: Similar to the index image in the fixel directory format,
165  // but has dimensions Nx1x1x2, where:
166  // - N is the number of fixels in the fixel template
167  // - The first volume contains the number of connected fixels for that fixel
168  // - The second volume contains the offset of the first connected fixel for that fixel
169  // - connectivity.mif: Floating-point image of dimension Cx1x1, where C is the total
170  // number of fixel-fixel connections stored in the entire matrix. Each value should be
171  // between 0.0 and 1.0, corresponding to the fraction of streamlines passing through
172  // the fixel that additionally pass through some other fixel.
173  // - fixel.mif: Unsigned integer image of dimension Cx1x1, where C is the total number of
174  // fixel-fixel connections stored in the entire matrix. Each value indexes the
175  // fixel to which the fixel-fixel connection refers.
176  //
177  // In order to avoid duplication of memory usage, the writing function should
178  // perform, for each fixel in turn:
179  // - Normalisation of the matrix
180  // - Writing to the three images
181  // - Erasing the memory used for that matrix in the initial building
182

184  const connectivity_value_type threshold,
