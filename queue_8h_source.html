631  Mutex::Lock lock (mutex);
632  while (full() && reader_count) more_space.wait();
633  if (!reader_count) return (false);
634  *back = item;
635  back = inc (back);
636  more_data.signal();
637  if (item_stack.empty()) {
638  item = new T;
639  items.push_back (item);
640  }
641  else {
642  item = item_stack.top();
643  item_stack.pop();
644  }
645  return (true);
646  }

649  Mutex::Lock lock (mutex);
650  if (item)
651  item_stack.push (item);
652  item = NULL;
653  while (empty() && writer_count) more_data.wait();
654  if (empty() && !writer_count) return (false);
655  item = *front;
656  front = inc (front);
