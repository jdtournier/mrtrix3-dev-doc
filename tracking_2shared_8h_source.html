229  void set_step_size (float stepsize)
230  {
231  step_size = stepsize * vox();
232  properties.set (step_size, "step_size");
233  INFO ("step size = " + str (step_size) + " mm");
234 
235  if (downsampler.get_ratio() > 1)
236  properties["output_step_size"] = str (step_size * downsampler.get_ratio());
237 
238  float max_dist = 100.0 * vox();
239  properties.set (max_dist, "max_dist");
240  max_num_points = std::round (max_dist/step_size) + 1;
241 
242  float min_dist = is_act() ? (2.0 * vox()) : (5.0 * vox());
243  properties.set (min_dist, "min_dist");
244  min_num_points = std::max (2, int(std::round (min_dist/step_size) + 1));
245 
246  max_angle = 90.0 * step_size / vox();
247  properties.set (max_angle, "max_angle");
248  INFO ("maximum deviation angle = " + str (max_angle) + " deg");
249  max_angle *= Math::pi / 180.0;
250  cos_max_angle = std::cos (max_angle);
251 
252  if (rk4) {
253  max_angle_rk4 = max_angle;
254  cos_max_angle_rk4 = cos_max_angle;
255  max_angle = Math::pi;
