102  const size_t from, to;
103  const std::tuple<ImageType&...> vox;
104  const ssize_t size0;
105  bool ok;
106  FORCE_INLINE Run (const size_t axis_from, const size_t axis_to, const std::tuple<ImageType&...>& vox) :
107  from (axis_from), to (axis_to ? axis_to : std::get<0>(vox).ndim()), vox (vox), size0 (std::get<0>(vox).size(from)), ok (true) {
108  for (size_t n = from; n < to; ++n)
109  apply (set_pos (n, 0), vox);
110  }
111  FORCE_INLINE operator bool() const { return ok; }
112  FORCE_INLINE void operator++() {
113  apply (inc_pos (from), vox);
114  if (std::get<0>(vox).index(from) < size0)
115  return;
116 
117  apply (set_pos (from, 0), vox);
118  size_t axis = from+1;
119  while (axis < to) {
120  apply (inc_pos (axis), vox);
121  if (std::get<0>(vox).index(axis) < std::get<0>(vox).size(axis))
122  return;
123  apply (set_pos (axis, 0), vox);
124  ++axis;
125  }
126  ok = false;
127  }
128  FORCE_INLINE void operator++(int) { operator++(); }
129  };

347  /* \section restrictedorderloop Looping over a specific range of axes
348  * It is also possible to explicitly specify the range of axes to be looped
349  * over. In the following example, the program will loop over each 3D
350  * volume in the ImageType in turn using the Loop class, and use the
351  * LoopInOrder class to iterate over the axes of each volume to ensure
352  * efficient memory bandwidth use when each volume is being processed.
353  * \code
354  * // define inner loop to iterate over axes 0 to 2
355  * LoopInOrder inner (vox, 0, 3);
356  *
357  * // outer loop iterates over axes 3 and above:
358  * for (auto i = Loop(3).run (vox); i; ++i) {
359  * float sum = 0.0;
360  * for (auto j = inner.run (vox); j; ++j) {
361  * sum += vox.value();
362  * print ("total = " + str (sum) + "\n");
363  * }
364  * \endcode
365  *
366  * \section arbitraryorderloop Arbitrary order loop
367  * It is also possible to specify the looping order explictly, as in the
368  * following example:
369  * \code
370  * float value = 0.0;
371  * std::vector<size_t> order = { 1, 0, 2 };
372  *
373  * LoopInOrder loop (vox, order);
374  * for (auto i = loop.run (vox); i; ++i)
375  * value += std::exp (-vox.value());
376  * \endcode
377  * This will iterate over the axes in the same order as the first example
378  * above, irrespective of the strides of the ImageType.
379  *
380  * \section multiorderloop Looping over multiple ImageType objects:
381  *
382  * As with the Loop class, it is possible to loop over more than one
383  * ImageType of the same dimensions, by passing any additional ImageType
384  * objects to be looped over to the run() member function. For example,
385  * this code snippet will copy the contents of the ImageType \a src into a
386  * ImageType \a dest (assumed to have the same dimensions as \a src),
387  * with the looping order optimised for the \a src ImageType:
388  * \code
389  * LoopInOrder loop (src);
390  * for (auto i = loop.run(src, dest); i; ++i)
391  * dest.value() = src.value();
392  * \endcode
393  */
394  /* \section progressloopinroder Displaying progress status
395  * As in the Loop class, the LoopInOrder object can also display its
396  * progress as it proceeds, using the appropriate constructor. In the
397  * following example, the program will display its progress as it averages
398  * a ImageType:
399  * \code
400  * float sum = 0.0;
401  *
402  * LoopInOrder loop (vox, "averaging");
403  * for (auto i = loop.run (vox); i; ++i)
404  * sum += vox.value();
405  *
406  * float average = sum / float (Image::voxel_count (vox));
407  * print ("average = " + str (average) + "\n");
408  * \endcode
409  * The output would look something like this:
410  * \code
411  * myprogram: [100%] averaging
412  * average = 23.42
413  * \endcode
414  */
