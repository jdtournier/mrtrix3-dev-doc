229  struct LoopAlongAxisRange {
230  const size_t from, to;
231 
232  template <class... ImageType>
233  struct Run {
234  const size_t from, to;
235  const std::tuple<ImageType&...> vox;
236  const ssize_t size0;
237  bool ok;
238  FORCE_INLINE Run (const size_t axis_from, const size_t axis_to, const std::tuple<ImageType&...>& vox) :
239  from (axis_from), to (axis_to ? axis_to : std::get<0>(vox).ndim()), vox (vox), size0 (std::get<0>(vox).size(from)), ok (true) {
240  for (size_t n = from; n < to; ++n)
241  apply (set_pos (n, 0), vox);
242  }
243  FORCE_INLINE operator bool() const { return ok; }
244  FORCE_INLINE void operator++() {
245  apply (inc_pos (from), vox);
246  if (std::get<0>(vox).index(from) < size0)
247  return;
248 
249  apply (set_pos (from, 0), vox);
250  size_t axis = from+1;
251  while (axis < to) {
252  apply (inc_pos (axis), vox);
253  if (std::get<0>(vox).index(axis) < std::get<0>(vox).size(axis))
254  return;
255  apply (set_pos (axis, 0), vox);
256  ++axis;
257  }
258  ok = false;
259  }
260  FORCE_INLINE void operator++(int) { operator++(); }
261  };
262 
263  template <class... ImageType>
264  FORCE_INLINE Run<ImageType...> operator() (ImageType&... images) const { return { from, to, std::tie (images...) }; }
265  };
