192  template <typename value_type>
193  inline void Versor<value_type>::from_matrix (const Math::Matrix<value_type>& matrix)
194  {
195  x[0] = 1.0 + matrix(0,0) + matrix(1,1) + matrix(2,2);
196  x[0] = x[0] > 0.0 ? 0.5 * std::sqrt (x[0]) : 0.0;
197  if (std::abs (x[0]) < 0.1) {
198  x[1] = 1.0 + matrix(0,0) - matrix(1,1) - matrix(2,2);
199  x[1] = x[1] > 0.0 ? 0.5 * std::sqrt (x[1]) : 0.0;
200  if (std::abs (x[1]) < 0.1) {
201  x[2] = 1.0 - matrix(0,0) + matrix(1,1) - matrix(2,2);
202  x[2] = x[2] > 0.0 ? 0.5 * std::sqrt (x[2]) : 0.0;
203  if (std::abs (x[2]) < 0.1) {
204  x[3] = 0.5 * std::sqrt (1.0 - matrix(0,0) - matrix(1,1) + matrix(2,2));
205  x[0] = (matrix(1,0) - matrix(0,1)) / (4.0 * x[3]);
206  x[1] = (matrix(0,2) + matrix(2,0)) / (4.0 * x[3]);
207  x[2] = (matrix(2,1) + matrix(1,2)) / (4.0 * x[3]);
208  }
209  else {
210  x[0] = (matrix(0,2) - matrix(2,0)) / (4.0 * x[2]);
211  x[1] = (matrix(1,0) + matrix(0,1)) / (4.0 * x[2]);
212  x[3] = (matrix(2,1) + matrix(1,2)) / (4.0 * x[2]);
213  }
214  }
215  else {
216  x[0] = (matrix(2,1) - matrix(1,2)) / (4.0 * x[1]);
217  x[2] = (matrix(1,0) + matrix(0,1)) / (4.0 * x[1]);
218  x[3] = (matrix(0,2) + matrix(2,0)) / (4.0 * x[1]);
219  }
220  }
221  else {
222  x[1] = (matrix(2,1) - matrix(1,2)) / (4.0 * x[0]);
223  x[2] = (matrix(0,2) - matrix(2,0)) / (4.0 * x[0]);
224  x[3] = (matrix(1,0) - matrix(0,1)) / (4.0 * x[0]);
225  }
226 
227  normalise();
228  }

248  template <typename value_type>
249  inline void Versor<value_type>::slerp (Versor& i, float t, const Versor& y) const
250  {
251  double cosHalfTheta = x[0] * y[0] + x[1] * y[1] + x[2] * y[2] + x[3] * y[3];
252  if (std::abs(cosHalfTheta) >= 1.0) {
253  i[0] = x[0]; i[1] = x[1]; i[2] = x[2]; i[3] = x[3];
254  return;
255  }
256 
257  if (cosHalfTheta < 0) {
258  y[0] = -y[0]; y[1] = -y[1]; y[2] = -y[2]; y[3] = -y[3];
259  cosHalfTheta = -cosHalfTheta;
260  }
261 
262  double halfTheta = std::acos (cosHalfTheta);
263  double sinHalfTheta = std::sqrt(1.0 - cosHalfTheta*cosHalfTheta);
264  if (std::abs(sinHalfTheta) < 0.001) {
265  i[0] = (x[0] * 0.5 + y[0] * 0.5);
266  i[1] = (x[1] * 0.5 + y[1] * 0.5);
267  i[2] = (x[2] * 0.5 + y[2] * 0.5);
268  i[3] = (x[3] * 0.5 + y[3] * 0.5);
269  return;
270  }
271  double ratioA = std::sin((1 - t) * halfTheta) / sinHalfTheta;
272  double ratioB = std::sin(t * halfTheta) / sinHalfTheta;
273 
274  i[0] = (x[0] * ratioA + y[0] * ratioB);
275  i[1] = (x[1] * ratioA + y[1] * ratioB);
276  i[2] = (x[2] * ratioA + y[2] * ratioB);
277  i[3] = (x[3] * ratioA + y[3] * ratioB);
278  return;
279  }
