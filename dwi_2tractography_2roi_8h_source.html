56  class ROI { MEMALIGN(ROI)

58  ROI (const Eigen::Vector3f& sphere_pos, float sphere_radius) :
59  pos (sphere_pos), radius (sphere_radius), radius2 (Math::pow2 (radius)) { }
60 
61  ROI (const std::string& spec) :
62  radius (NaN), radius2 (NaN)
63  {
64  try {
65  auto F = parse_floats (spec);
66  if (F.size() != 4)
67  throw 1;
68  pos[0] = F[0];
69  pos[1] = F[1];
70  pos[2] = F[2];
71  radius = F[3];
72  radius2 = Math::pow2 (radius);
73  }
74  catch (...) {
75  DEBUG ("could not parse spherical ROI specification \"" + spec + "\" - assuming mask image");
76  mask.reset (new Mask (spec));
77  }
78  }
79 
80  std::string shape () const { return (mask ? "image" : "sphere"); }
81 
82  std::string parameters () const {
83  return mask ? mask->name() : str(pos[0]) + "," + str(pos[1]) + "," + str(pos[2]) + "," + str(radius);
84  }
85 
86  bool contains (const Eigen::Vector3f& p) const
87  {
88 
89  if (mask) {
90  Eigen::Vector3f v = *(mask->scanner2voxel) * p;
91  Mask temp (*mask); // Required for thread-safety
92  temp.index(0) = std::round (v[0]);
93  temp.index(1) = std::round (v[1]);
94  temp.index(2) = std::round (v[2]);
95  if (is_out_of_bounds (temp))
96  return false;
97  return temp.value();
98  }
99 
100  return (pos-p).squaredNorm() <= radius2;
101 
102  }
