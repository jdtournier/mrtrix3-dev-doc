77  ROI (const std::string& spec) :

81  std::vector<float> F (parse_floats (spec));
82  if (F.size() != 4) throw 1;
83  pos.set (F[0], F[1], F[2]);
84  radius = F[3];
85  radius2 = Math::pow2 (radius);
86  }

88  DEBUG ("could not parse spherical ROI specification \"" + spec + "\" - assuming mask image");
89  mask = get_mask (spec);
90  }
91  }

99  bool contains (const Point<>& p) const
100  {
101 
102  if (mask) {
103  auto vox = mask->voxel();
104  Point<> v = mask->transform.scanner2voxel (p);
105  vox[0] = std::round (v[0]);
106  vox[1] = std::round (v[1]);
107  vox[2] = std::round (v[2]);
108  if (!Image::Nav::within_bounds (vox))

111  }
