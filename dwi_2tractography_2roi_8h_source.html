59  class ROI {
60  public:
61  ROI (const Eigen::Vector3f& sphere_pos, float sphere_radius) :
62  pos (sphere_pos), radius (sphere_radius), radius2 (Math::pow2 (radius)) { }
63 
64  ROI (const std::string& spec) :
65  radius (NaN), radius2 (NaN)
66  {
67  try {
68  auto F = parse_floats (spec);
69  if (F.size() != 4)
70  throw 1;
71  pos[0] = F[0];
72  pos[1] = F[1];
73  pos[2] = F[2];
74  radius = F[3];
75  radius2 = Math::pow2 (radius);
76  }
77  catch (...) {
78  DEBUG ("could not parse spherical ROI specification \"" + spec + "\" - assuming mask image");
79  mask.reset (new Mask (spec));
80  }
81  }
82 
83  std::string shape () const { return (mask ? "image" : "sphere"); }
84 
85  std::string parameters () const {
86  return mask ? mask->name() : str(pos[0]) + "," + str(pos[1]) + "," + str(pos[2]) + "," + str(radius);
87  }
88 
89  bool contains (const Eigen::Vector3f& p) const
90  {
91 
92  if (mask) {
93  Eigen::Vector3f v = *(mask->scanner2voxel) * p;
94  Mask temp (*mask); // Required for thread-safety
95  temp.index(0) = std::round (v[0]);
96  temp.index(1) = std::round (v[1]);
97  temp.index(2) = std::round (v[2]);
98  if (is_out_of_bounds (temp))
99  return false;
100  return temp.value();
101  }
102 
103  return (pos-p).squaredNorm() <= radius2;
104 
105  }
