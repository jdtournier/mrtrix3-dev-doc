62  ROI (const std::string& spec) :
63  radius (NaN), radius2 (NaN)
64  {
65  try {
66  auto F = parse_floats (spec);
67  if (F.size() != 4)
68  throw 1;
69  pos[0] = F[0];
70  pos[1] = F[1];
71  pos[2] = F[2];
72  radius = F[3];
73  radius2 = Math::pow2 (radius);
74  }
75  catch (...) {
76  DEBUG ("could not parse spherical ROI specification \"" + spec + "\" - assuming mask image");
77  mask.reset (new Mask (spec));
78  }
79  }
80 
81  std::string shape () const { return (mask ? "image" : "sphere"); }
82 
83  std::string parameters () const {
84  return mask ? mask->name() : str(pos[0]) + "," + str(pos[1]) + "," + str(pos[2]) + "," + str(radius);
85  }

87  bool contains (const Eigen::Vector3f& p) const
88  {
89 
90  if (mask) {
91  Eigen::Vector3f v = *(mask->scanner2voxel) * p;
92  Mask temp (*mask); // Required for thread-safety
93  temp.index(0) = std::round (v[0]);
94  temp.index(1) = std::round (v[1]);
95  temp.index(2) = std::round (v[2]);
96  if (is_out_of_bounds (temp))
97  return false;
98  return temp.value();
99  }
100 
101  return (pos-p).squaredNorm() <= radius2;
