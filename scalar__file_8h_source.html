75  bool operator() (vector<value_type>& tck_scalar)
76  {
77  tck_scalar.clear();
78 
79  if (!in.is_open())
80  return false;
81  do {
82  value_type val = get_next_scalar();
83  if (std::isinf (val)) {
84  in.close();
85  return false;
86  }
87  if (in.eof()) {
88  in.close();
