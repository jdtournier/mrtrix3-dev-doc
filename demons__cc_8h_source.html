74 
75  if (im1_meansubtracted.index(0) == 0 || im1_meansubtracted.index(0) == im1_meansubtracted.size(0) - 1 ||
76  im1_meansubtracted.index(1) == 0 || im1_meansubtracted.index(1) == im1_meansubtracted.size(1) - 1 ||

78  im1_update.row(3) = 0.0;
79  im2_update.row(3) = 0.0;
80  return;
81  }
82 
83  typename Im1MaskType::value_type im1_mask_value = 1.0;
84  if (im1_mask.valid()) {
85  assign_pos_of (im1_meansubtracted, 0, 3).to (im1_mask);
86  im1_mask_value = im1_mask.value();
87  if (im1_mask_value < 0.1) {
88  im1_update.row(3) = 0.0;
89  im2_update.row(3) = 0.0;
90  return;
91  }
92  }
93 
94  typename Im2MaskType::value_type im2_mask_value = 1.0;
95  if (im2_mask.valid()) {
96  assign_pos_of (im2_meansubtracted, 0, 3).to (im2_mask);
97  im2_mask_value = im2_mask.value();
98  if (im2_mask_value < 0.1) {
99  im1_update.row(3) = 0.0;
100  im2_update.row(3) = 0.0;
101  return;
102  }
103  }
104 
105  default_type sfm = A.value();
106  default_type smm = B.value();
107  default_type sff = C.value();
108  default_type asq = sfm * sfm;
109 
110  default_type denom = smm * sff;
111  if (std::isnan(sfm) || std::isnan(denom) || MR::abs (denom) < denominator_threshold) {
112  // if (std::abs (asq) > robustness_parameter)
113  // thread_voxel_count++; // TODO to count or not to count?
114  im1_update.row(3) = 0.0;
115  im2_update.row(3) = 0.0;
116  return;
117  }
118 
119  default_type lcc = asq / denom;
120  thread_cost -= lcc;
121  thread_voxel_count++;
122 
123  assign_pos_of (im1_meansubtracted, 0, 3).to (im1_gradient, im2_gradient);
124 
125  default_type i1 = im1_meansubtracted.value();
126  default_type i2 = im2_meansubtracted.value();
127 
128  // Avants eq. 6 and 7
129  Eigen::Matrix<typename Im1ImageType::value_type, 3, 1> grad = 2.0 * sfm / (sff * smm) * (
130  (i2 - sfm / smm * i1 ) * im1_gradient.value() - (i1 - sfm / sff * i2 ) * im2_gradient.value());
131  // TODO: add det(jacobian(Phi)))
132 
133  im1_update.row(3) = grad * 40.0; // TODO: normalise the update?
134  im2_update.row(3) = -grad * 40.0;
135  }
