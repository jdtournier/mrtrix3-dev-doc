95  void add (const dir_t bin, const default_type value)
96  {
97  assert ((value <= 0.0 && neg) || (value >= 0.0 && !neg));
98  mask[bin] = true;
99  values[bin] = value;
100  const Eigen::Vector3f& dir = mask.get_dirs()[bin];
101  const float multiplier = (peak_dir.dot (dir)) > 0.0 ? 1.0 : -1.0;
102  mean_dir += dir * multiplier * value;
103  integral += std::abs (value);
104  }
105 
106  void revise_peak (const Eigen::Vector3f& real_peak, const float value)
107  {
108  assert (!neg);
109  peak_dir = real_peak;
110  peak_value = value;
111  }

121  void finalise()
122  {
123  // 2pi == solid angle of halfsphere in steradians
124  integral *= 2.0 * Math::pi / float(mask.size());
125  // This is calculated as the lobe is built, just needs to be set to unit length
126  mean_dir.normalize();
127  }
128 
129  void merge (const FOD_lobe& that)
130  {
131  assert (neg == that.neg);
132  mask |= that.mask;
133  for (size_t i = 0; i != mask.size(); ++i)
134  values[i] += that.values[i];
135  if (that.peak_value > peak_value) {
136  peak_dir_bin = that.peak_dir_bin;
137  peak_value = that.peak_value;
138  peak_dir = that.peak_dir;
139  }
140  const float multiplier = (mean_dir.dot (that.mean_dir)) > 0.0 ? 1.0 : -1.0;
141  mean_dir += that.mean_dir * that.integral * multiplier;
142  integral += that.integral;
143  }

186  class FODQueueWriter
187  {

189  typedef Image<float> FODImageType;
190  typedef Image<float> MaskImageType;
191 
192  public:
193  FODQueueWriter (const FODImageType& fod_image, const MaskImageType& mask_image = MaskImageType()) :
194  fod (fod_image),
195  mask (mask_image),
196  loop (Loop("segmenting FODs", 0, 3) (fod)) { }
197 
198  bool operator() (SH_coefs& out)
199  {
200  if (!loop)
201  return false;
202  if (mask.valid()) {
203  do {
204  assign_pos_of (fod, 0, 3).to (mask);
205  if (!mask.value())
206  ++loop;
207  } while (loop && !mask.value());
208  }
209  assign_pos_of (fod).to (out.vox);
210  out.resize (fod.size (3));
211  for (auto l = Loop (3) (fod); l; ++l)
212  out[fod.index(3)] = fod.value();
213  ++loop;
214  return true;
215  }
216 
217  private:
218  FODImageType fod;
219  MaskImageType mask;
220  decltype(Loop("text", 0, 3) (fod)) loop;
