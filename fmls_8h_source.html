93  mask (i),

101  void add (const dir_t bin, const float value)
102  {
103  assert ((value <= 0.0 && neg) || (value > 0.0 && !neg));
104  mask[bin] = true;
105  values[bin] = value;
106  const Point<float>& dir = mask.get_dirs()[bin];
107  const float multiplier = (peak_dir.dot (dir)) > 0.0 ? 1.0 : -1.0;
108  mean_dir += dir * multiplier * value;
109  integral += Math::abs (value);
110  }
111 
112  void revise_peak (const Point<float>& real_peak, const float value)
113  {
114  assert (!neg);
115  peak_dir = real_peak;
116  peak_value = value;
117  }
118 
119 #ifdef FMLS_OPTIMISE_MEAN_DIR
120  void revise_mean_dir (const Point<float>& real_mean)
121  {
122  assert (!neg);
123  mean_dir = real_mean;
124  }
125 #endif
126 
127  void finalise()
128  {
129  // 4pi == solid angle of sphere in steradians
130  integral *= 4.0 * M_PI / float(mask.size());
131  // This is calculated as the lobe is built, just needs to be set to unit length
132  mean_dir.normalise();
133  }
134 
135  void merge (const FOD_lobe& that)
136  {
137  assert (neg == that.neg);
138  mask |= that.mask;
139  for (size_t i = 0; i != mask.size(); ++i)
140  values[i] += that.values[i];
141  if (that.peak_value > peak_value) {
142  peak_dir_bin = that.peak_dir_bin;
143  peak_value = that.peak_value;
144  peak_dir = that.peak_dir;
145  }
146  const float multiplier = (mean_dir.dot (that.mean_dir)) > 0.0 ? 1.0 : -1.0;
147  mean_dir += that.mean_dir * that.integral * multiplier;
148  integral += that.integral;
149  }

221 
222  bool operator () (SH_coefs& out)
223  {
224  if (!loop.ok())
225  return false;
226  if (mask_vox_ptr) {
227  do {
228  Image::voxel_assign (*mask_vox_ptr, fod_vox, 0, 3);
229  if (!mask_vox_ptr->value())
230  loop.next (fod_vox);
231  } while (loop.ok() && !mask_vox_ptr->value());
232  }
233  out.vox[0] = fod_vox[0]; out.vox[1] = fod_vox[1]; out.vox[2] = fod_vox[2];
234  out.allocate (fod_vox.dim (3));
235  for (fod_vox[3] = 0; fod_vox[3] != fod_vox.dim (3); ++fod_vox[3])
236  out[fod_vox[3]] = fod_vox.value();
237  loop.next (fod_vox);
238  return true;
239  }
