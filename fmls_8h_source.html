67  class FOD_lobe {
68 
69  public:
70  FOD_lobe (const DWI::Directions::Set& dirs, const dir_t seed, const default_type value, const default_type weight) :
71  mask (dirs),
72  values (dirs.size(), 0.0),
73  max_peak_value (std::abs (value)),
74  peak_dirs (1, dirs.get_dir (seed)),
75  mean_dir (peak_dirs.front() * value * weight),
76  integral (std::abs (value * weight)),
77  neg (value <= 0.0)
78  {
79  mask[seed] = true;
80  values[seed] = value;
81  }
82 
83  // This is used for creating a `null lobe' i.e. an FOD lobe with zero size, containing all directions not
84  // assigned to any other lobe in the voxel
85  FOD_lobe (const DWI::Directions::Mask& i) :
86  mask (i),
87  values (i.size(), 0.0),
88  max_peak_value (0.0),
89  integral (0.0),
90  neg (false) { }
91 
92 
93  void add (const dir_t bin, const default_type value, const default_type weight)
94  {
95  assert ((value <= 0.0 && neg) || (value >= 0.0 && !neg));
96  mask[bin] = true;
97  values[bin] = value;
98  const Eigen::Vector3f& dir = mask.get_dirs()[bin];
99  const float multiplier = (mean_dir.dot (dir)) > 0.0 ? 1.0 : -1.0;
100  mean_dir += dir * multiplier * value * weight;
101  integral += std::abs (value * weight);
102  }
103 
104  void revise_peak (const size_t index, const Eigen::Vector3f& real_peak, const float value)
105  {
106  assert (!neg);
107  assert (index < num_peaks());
108  peak_dirs[index] = real_peak;
109  if (!index)
110  max_peak_value = value;
111  }
112 
113 #ifdef FMLS_OPTIMISE_MEAN_DIR
114  void revise_mean_dir (const Eigen::Vector3f& real_mean)
115  {
116  assert (!neg);
117  mean_dir = real_mean;
118  }
119 #endif
120 
121  void finalise()
122  {
123  // 2pi == solid angle of halfsphere in steradians
124  //integral *= 2.0 * Math::pi / float(mask.size());
125  // No longer needed: Segmenter::weights should deal with this
126  // This is calculated as the lobe is built, just needs to be set to unit length
127  mean_dir.normalize();
128  }
129 
130  void merge (const FOD_lobe& that)
131  {
132  assert (neg == that.neg);
133  mask |= that.mask;
134  for (size_t i = 0; i != mask.size(); ++i)
135  values[i] += that.values[i];
136  if (that.max_peak_value > max_peak_value) {
137  max_peak_value = that.max_peak_value;
138  peak_dirs.insert (peak_dirs.begin(), that.peak_dirs.begin(), that.peak_dirs.end());
139  } else {
140  peak_dirs.insert (peak_dirs.end(), that.peak_dirs.begin(), that.peak_dirs.end());
141  }
142  const float multiplier = (mean_dir.dot (that.mean_dir)) > 0.0 ? 1.0 : -1.0;
143  mean_dir += that.mean_dir * that.integral * multiplier;
144  integral += that.integral;
145  }

147  const DWI::Directions::Mask& get_mask() const { return mask; }
148  const std::vector<float>& get_values() const { return values; }
149  float get_max_peak_value() const { return max_peak_value; }
150  size_t num_peaks() const { return peak_dirs.size(); }
151  const Eigen::Vector3f& get_peak_dir (const size_t i) const { assert (i < num_peaks()); return peak_dirs[i]; }
152  const Eigen::Vector3f& get_mean_dir() const { return mean_dir; }
153  float get_integral() const { return integral; }
154  bool is_negative() const { return neg; }

159  std::vector<float> values;

162  Eigen::Vector3f mean_dir;
163  float integral;

187  class FODQueueWriter
188  {
189 
190  typedef Image<float> FODImageType;
191  typedef Image<float> MaskImageType;
192 
193  public:
194  FODQueueWriter (const FODImageType& fod_image, const MaskImageType& mask_image = MaskImageType()) :
195  fod (fod_image),
196  mask (mask_image),
197  loop (Loop("segmenting FODs", 0, 3) (fod)) { }
198 
199  bool operator() (SH_coefs& out)
200  {
201  if (!loop)
202  return false;
203  if (mask.valid()) {
204  do {
205  assign_pos_of (fod, 0, 3).to (mask);
206  if (!mask.value())
207  ++loop;
208  } while (loop && !mask.value());
209  }
210  assign_pos_of (fod).to (out.vox);
211  out.resize (fod.size (3));
212  for (auto l = Loop (3) (fod); l; ++l)
213  out[fod.index(3)] = fod.value();
214  ++loop;
215  return true;
216  }
217 
218  private:
219  FODImageType fod;
220  MaskImageType mask;
221  decltype(Loop("text", 0, 3) (fod)) loop;
222 
223  };
