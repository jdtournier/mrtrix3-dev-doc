122  template <class VectorType>
123  bool voxel (const VectorType& pos) {
124  Eigen::Vector3 f = Base<ImageType>::intravoxel_offset (pos);
125  if (Base<ImageType>::out_of_bounds)
126  return false;
127  P = pos;
128  for(size_t i = 0; i < 3; ++i)
129  H[i].set (f[i]);
130 
131  // Precompute weights
132  size_t i(0);
133  for (ssize_t z = 0; z < 4; ++z) {
134  for (ssize_t y = 0; y < 4; ++y) {
135  value_type partial_weight = H[1].weights[y] * H[2].weights[z];
136  for (ssize_t x = 0; x < 4; ++x)
137  weights_vec[i++] = H[0].weights[x] * partial_weight;
138  }
139  }
140 
141  return true;
