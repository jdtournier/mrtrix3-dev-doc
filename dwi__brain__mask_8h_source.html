81  // Loop over each shell, including b=0, in turn

84  const DWI::Shell shell (shells[s]);

89  value_type mean = 0;
90  for (std::vector<size_t>::const_iterator v = shell.get_volumes().begin(); v != shell.get_volumes().end(); ++v) {
91  input.index(3) = *v;
92  mean += (input.value() < 0) ? 0 : input.value();
93  }
94  shell_image.value() = mean / value_type(shell.count());
95  }
96  if (progress)
97  ++(*progress);
98

101

103  threshold_filter (shell_image, shell_mask_voxel);
104  if (progress)

106 
107  // Add this mask to the master
108  for (auto l = Loop (0, 3)(mask_image, shell_mask_voxel); l; ++l) {
109  if (shell_mask_voxel.value())

112  if (progress)

115  }
