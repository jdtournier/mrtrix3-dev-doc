139  float get_cumulative_prob (const uint64_t track_count)
140  {
141  while (updating.test_and_set (std::memory_order_acquire));
142  float cumulative_prob = old_prob;
143  if (track_count > track_count_at_last_update) {
144  cumulative_prob = ((track_count_at_last_update * old_prob) + ((track_count - track_count_at_last_update) * applied_prob)) / float(track_count);
145  old_prob = cumulative_prob;
146  track_count_at_last_update = track_count;
147  }
148  return cumulative_prob;
149  }
