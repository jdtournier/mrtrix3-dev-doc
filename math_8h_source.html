119  template <class ValueType = default_type>
120  std::vector<std::vector<ValueType>> load_matrix_2D_vector (const std::string& filename)
121  {
122  std::ifstream stream (filename, std::ios_base::in | std::ios_base::binary);
123  std::vector<std::vector<ValueType>> V;
124  std::string sbuf;
125 
126  while (getline (stream, sbuf)) {
127  sbuf = strip (sbuf.substr (0, sbuf.find_first_of ('#')));
128  if (sbuf.empty())
129  continue;
130 
131  V.push_back (std::vector<ValueType>());
132 
133  const auto elements = MR::split (sbuf, " ,;\t", true);
134  try {
135  for (const auto& entry : elements)
136  V.back().push_back (to<ValueType> (entry));
137  } catch (...) {
138  throw Exception ("File \"" + filename + "\" contains non-numerical data");
139  }
140 
141  if (V.size() > 1)
142  if (V.back().size() != V[0].size())
143  throw Exception ("uneven rows in matrix file \"" + filename + "\"");
144  }
145  if (stream.bad())
146  throw Exception (strerror (errno));
147 
148  if (!V.size())
149  throw Exception ("no data in matrix file \"" + filename + "\"");
150 
151  return V;
152  }

177  inline transform_type load_transform (const std::string& filename)
178  {
179  DEBUG ("loading transform file \"" + filename + "\"...");
180 
181  std::vector<std::vector<default_type>> V;
182  try {
183  V = load_matrix_2D_vector<> (filename);
184  } catch (Exception& e) {
185  throw e;
186  }
187 
188  if (V.empty())
189  throw Exception ("transform in file " + filename + " is empty");
190 
191  if (V[0].size() != 4)
192  throw Exception ("transform in file " + filename + " is invalid: does not contain 4 columns.");
193 
194  if (V.size() != 3 && V.size() != 4)
195  throw Exception ("transform in file " + filename + " is invalid: must contain either 3 or 4 rows.");
196 
197  transform_type M;
198 
199  for (ssize_t i = 0; i < 3; i++)
200  for (ssize_t j = 0; j < 4; j++)
201  M(i,j) = V[i][j];
202 
203  return M;
204  }
