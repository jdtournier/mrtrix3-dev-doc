141  template <class ValueType = default_type>
142  vector<vector<ValueType>> load_matrix_2D_vector (const std::string& filename)
143  {
144  std::ifstream stream (filename, std::ios_base::in | std::ios_base::binary);
145  vector<vector<ValueType>> V;
146  std::string sbuf;
147 
148  while (getline (stream, sbuf)) {
149  sbuf = strip (sbuf.substr (0, sbuf.find_first_of ('#')));
150  if (sbuf.empty())
151  continue;
152 
153  V.push_back (vector<ValueType>());
154 
155  const auto elements = MR::split (sbuf, " ,;\t", true);
156  try {
157  for (const auto& entry : elements)
158  V.back().push_back (to<ValueType> (entry));
159  } catch (...) {
160  throw Exception ("File \"" + filename + "\" contains non-numerical data");
161  }
162 
163  if (V.size() > 1)
164  if (V.back().size() != V[0].size())
165  throw Exception ("uneven rows in matrix file \"" + filename + "\"");
166  }
167  if (stream.bad())
168  throw Exception (strerror (errno));
169 
170  if (!V.size())
171  throw Exception ("no data in matrix file \"" + filename + "\"");
172 
173  return V;
174  }

199  template <class ValueType = default_type>
200  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic> parse_matrix (const std::string& spec)
201  {
202  Eigen::Matrix<ValueType, Eigen::Dynamic, Eigen::Dynamic> M;
203  const auto lines = split_lines (spec);
204  for (size_t row = 0; row < lines.size(); ++row) {
205  const auto values = parse_floats (lines[row]);
206  if (M.cols() == 0)
207  M.resize (lines.size(), values.size());
208  else if (M.cols() != ssize_t (values.size()))
209  throw Exception ("error converting string to matrix - uneven number of entries per row");
210  for (size_t col = 0; col < values.size(); ++col)
211  M(row, col) = values[col];
212  }
213  return M;
