174  template <class Fixel>
175  void Model<Fixel>::map_streamlines (const std::string& path)
176  {
177  Tractography::Properties properties;
178  Tractography::Reader<float> file (path, properties);
179 
180  if (properties.find ("count") == properties.end())
181  throw Exception ("Input .tck file does not specify number of streamlines (run tckfixcount on your .tck file!)");
182  const track_t count = to<track_t>(properties["count"]);
183 
184  contributions.assign (count, NULL);
185 
186  // Determine appropriate upsampling ratio for mapping
187  // In this particular context want the calculation of length to be precise - 1/10th voxel size at worst
188  float step_size = 0.0;
189  if (properties.find ("output_step_size") != properties.end())
190  step_size = to<float> (properties["output_step_size"]);
191  else
192  step_size = to<float> (properties["step_size"]);
193  if (!step_size || !std::isfinite (step_size))
194  throw Exception ("Cannot perform appropriate streamline mapping without knowledge of track step size!");
195  const float upsample_ratio = Math::ceil<size_t> (step_size / (minvalue (H.vox(0), H.vox(1), H.vox(2)) * 0.1));
196 
197  Mapping::TrackLoader loader (file, count);
198  Mapping::TrackMapperDixel mapper (H, upsample_ratio, true, dirs);
199  MappedTrackReceiver receiver (*this);
200  Thread::run_queue (
201  loader,
202  Thread::batch (Tractography::Streamline<float>()),
203  Thread::multi (mapper),
204  Thread::batch (Mapping::SetDixel()),
205  Thread::multi (receiver));
206 
207  if (!contributions.back()) {
208  track_t num_tracks = 0, max_index = 0;
209  for (track_t i = 0; i != contributions.size(); ++i) {
210  if (contributions[i]) {
211  ++num_tracks;
212  max_index = std::max (max_index, i);
213  }
214  WARN ("Only " + str (num_tracks) + " tracks read from input track file; expected " + str (contributions.size()));
215  WARN ("(suggest running command tckfixcount on file " + path + ")");
216  contributions.resize (max_index + 1);
217  }
218  }
219 
220  tck_file_path = path;
221 
222  INFO ("Proportionality coefficient after streamline mapping is " + str (mu()));
223  }

229  template <class Fixel>
230  void Model<Fixel>::remove_excluded_fixels ()
231  {
232 
233  const bool remove_untracked_fixels = App::get_options ("remove_untracked").size();
234  App::Options opt = App::get_options ("min_fod_integral");
235  const float min_FOD_integral = opt.size() ? float(opt[0][0]) : 0.0;
236 
237  if (!remove_untracked_fixels && !min_FOD_integral)
238  return;
239 
240  std::vector<size_t> fixel_index_mapping (fixels.size(), 0);
241  typename Fixel_map<Fixel>::VoxelAccessor v (accessor);
242  Image::LoopInOrder loop (v);
243 
244  std::vector<Fixel> new_fixels;
245  new_fixels.push_back (Fixel());
246  FOD_sum = 0.0;
247 
248  for (loop.start (v); loop.ok(); loop.next (v)) {
249  if (v.value()) {
250 
251  size_t new_start_index = new_fixels.size();
252 
253  for (typename Fixel_map<Fixel>::ConstIterator i = begin(v); i; ++i) {
254  if ((!remove_untracked_fixels || i().get_TD()) && (i().get_FOD() > min_FOD_integral)) {
255  fixel_index_mapping [size_t (i)] = new_fixels.size();
256  new_fixels.push_back (i());
257  FOD_sum += i().get_weight() * i().get_FOD();
258  }
259  }
260 
261  delete v.value();
262 
263  if (new_fixels.size() == new_start_index)
264  v.value() = NULL;
265  else
266  v.value() = new MapVoxel (new_start_index, new_fixels.size() - new_start_index);
267 
268  }
269  }
270 
271  INFO (str (fixels.size() - new_fixels.size()) + " out of " + str(fixels.size()) + " fixels removed from reconstruction (" + str(new_fixels.size()) + ") remaining)");
272 
273  fixels.swap (new_fixels);
274 
275  TrackIndexRangeWriter writer (SIFT_TRACK_INDEX_BUFFER_SIZE, num_tracks(), "Removing excluded fixels...");
276  FixelRemapper remapper (*this, fixel_index_mapping);
277  Thread::run_queue (writer, TrackIndexRange(), Thread::multi (remapper));
278 
279  TD_sum = 0.0;
280  for (typename std::vector<Fixel>::const_iterator i = fixels.begin(); i != fixels.end(); ++i)
281  TD_sum += i->get_weight() * i->get_TD();
282 
283  INFO ("After fixel removal, the proportionality coefficient is " + str(mu()));
284 
285  }

294  void Model<Fixel>::check_TD()
295  {
296  VAR (TD_sum);
297  double sum_from_fixels = 0.0, sum_from_fixels_weighted = 0.0;
298  for (typename std::vector<Fixel>::const_iterator i = fixels.begin(); i != fixels.end(); ++i) {
299  sum_from_fixels += i->get_TD();
300  sum_from_fixels_weighted += i->get_TD() * i->get_weight();
301  }
302  VAR (sum_from_fixels);
303  VAR (sum_from_fixels_weighted);
304  double sum_from_tracks = 0.0;
305  for (std::vector<TrackContribution*>::const_iterator i = contributions.begin(); i != contributions.end(); ++i) {
306  if (*i)
307  sum_from_tracks += (*i)->get_total_contribution();
308  }
309  VAR (sum_from_tracks);
310  }

317  void Model<Fixel>::output_non_contributing_streamlines (const std::string& output_path) const
318  {
319  Tractography::Properties p;
320  Tractography::Reader<float> reader (tck_file_path, p);
321  Tractography::Writer<float> writer (output_path, p);
322  Tractography::Streamline<float> tck, null_tck;
323  ProgressBar progress ("Writing non-contributing streamlines output file...", contributions.size());
324  track_t tck_counter = 0;
325  while (reader (tck) && tck_counter < contributions.size()) {
326  if (contributions[tck_counter] && !contributions[tck_counter++]->get_total_contribution())
327  writer (tck);
328  else
329  writer (null_tck);
330  ++progress;
331  }
332  reader.close();
333  }

353  template <class Fixel>
354  bool Model<Fixel>::MappedTrackReceiver::operator() (const Mapping::SetDixel& in)
355  {
356 
357  if (in.index >= master.contributions.size())
358  throw Exception ("Received mapped streamline beyond the expected number of streamlines (run tckfixcount on your .tck file!)");
359  if (master.contributions[in.index])
360  throw Exception ("FIXME: Same streamline has been mapped multiple times! (?)");
361 
362  std::vector<Track_fixel_contribution> masked_contributions;
363  double total_contribution = 0.0, total_length = 0.0;
364 
365  for (Mapping::SetDixel::const_iterator i = in.begin(); i != in.end(); ++i) {
366  total_length += i->get_value();
367  const size_t fixel_index = master.dixel2fixel (*i);
368  if (fixel_index && (i->get_value() > Track_fixel_contribution::min())) {
369  total_contribution += i->get_value() * master.fixels[fixel_index].get_weight();
370  bool incremented = false;
371  for (std::vector<Track_fixel_contribution>::iterator c = masked_contributions.begin(); !incremented && c != masked_contributions.end(); ++c) {
372  if ((c->get_fixel_index() == fixel_index) && c->add (i->get_value()))
373  incremented = true;
374  }
375  if (!incremented)
376  masked_contributions.push_back (Track_fixel_contribution (fixel_index, i->get_value()));
377  }
378  }
379 
380  master.contributions[in.index] = new TrackContribution (masked_contributions, total_contribution, total_length);
381 
382  TD_sum += total_contribution;
383  for (std::vector<Track_fixel_contribution>::const_iterator i = masked_contributions.begin(); i != masked_contributions.end(); ++i)
384  fixel_TDs [i->get_fixel_index()] += i->get_value();
385 
386  return true;
387 
388  }

394  template <class Fixel>
395  bool Model<Fixel>::FixelRemapper::operator() (const TrackIndexRange& in)
396  {
397  for (track_t track_index = in.first; track_index != in.second; ++track_index) {
398  if (master.contributions[track_index]) {
399  TrackContribution& this_cont (*master.contributions[track_index]);
400  std::vector<Track_fixel_contribution> new_cont;
401  double total_contribution = 0.0;
402  for (size_t i = 0; i != this_cont.dim(); ++i) {
403  const size_t new_index = remapper[this_cont[i].get_fixel_index()];
404  if (new_index) {
405  new_cont.push_back (Track_fixel_contribution (new_index, this_cont[i].get_value()));
406  total_contribution += this_cont[i].get_value() * master[new_index].get_weight();
407  }
408  }
409  TrackContribution* new_contribution = new TrackContribution (new_cont, total_contribution, this_cont.get_total_length());
410  delete master.contributions[track_index];
411  master.contributions[track_index] = new_contribution;
412  }
413  }
414  return true;
415  }
