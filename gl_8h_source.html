115  class Texture : CheckContext { NOMEMALIGN
116  public:
117  Texture () : id (0), tex_type (0) { }
118  ~Texture () { clear(); }
119  Texture (const Texture&) : id (0), tex_type (0) { }
120  Texture (Texture&& t) : id (t.id), tex_type (t.tex_type) { t.id = t.tex_type = 0; }
121  Texture& operator= (Texture&& t) { clear(); id = t.id; tex_type = t.tex_type; t.id = t.tex_type = 0; return *this; }
122  void cache_copy(const Texture& t) { id = t.id; tex_type = t.tex_type; }
123  operator GLuint () const { return id; }
124  void gen (GLenum target, GLint interp_type = gl::LINEAR) {
125  if (!id) {
126  grab_context();
127  tex_type = target;
128  gl::GenTextures (1, &id);
129  GL_DEBUG ("created OpenGL texture ID " + str(id));
130  bind();
131  gl::TexParameteri (tex_type, gl::TEXTURE_BASE_LEVEL, 0);
132  gl::TexParameteri (tex_type, gl::TEXTURE_MAX_LEVEL, 0);
133  gl::TexParameteri (tex_type, gl::TEXTURE_MAG_FILTER, interp_type);
134  gl::TexParameteri (tex_type, gl::TEXTURE_MIN_FILTER, interp_type);
135  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_S, gl::CLAMP_TO_EDGE);
136  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_T, gl::CLAMP_TO_EDGE);
137  if (tex_type == gl::TEXTURE_3D)
138  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_R, gl::CLAMP_TO_EDGE);
139  }
140  }
141  GLenum type () const { return tex_type; }
142  void clear () {
143  if (id) {
144  check_context();
145  GL_DEBUG ("deleting OpenGL texture ID " + str(id));
146  gl::DeleteTextures (1, &id);
147  }
148  id = 0;
149  tex_type = 0;
150  }
151  void bind () const {
152  assert (id);
153  check_context();
154  GL_DEBUG ("binding OpenGL texture ID " + str(id));
155  gl::BindTexture (tex_type, id);
156  }
157  void set_interp (GLint type) const {
158  bind();
159  gl::TexParameteri (tex_type, gl::TEXTURE_MAG_FILTER, type);
160  gl::TexParameteri (tex_type, gl::TEXTURE_MIN_FILTER, type);
161  }
162  void set_interp_on (bool interpolate) const { set_interp (interpolate ? gl::LINEAR : gl::NEAREST); }
163  protected:
164  GLuint id;
165  GLenum tex_type;
166  };
