111  class Texture : CheckContext {
112  public:
113  Texture () : id (0) { }
114  ~Texture () { clear(); }
115  Texture (const Texture&) : id (0) { }
116  Texture (Texture&& t) : id (t.id) { t.id = 0; }
117  Texture& operator= (Texture&& t) { clear(); id = t.id; t.id = 0; return *this; }
118  operator GLuint () const { return id; }
119  void gen (GLenum target, GLint interp_type = gl::LINEAR) {
120  if (!id) {
121  grab_context();
122  tex_type = target;
123  gl::GenTextures (1, &id);
124  GL_DEBUG ("created OpenGL texture ID " + str(id));
125  bind();
126  gl::TexParameteri (tex_type, gl::TEXTURE_BASE_LEVEL, 0);
127  gl::TexParameteri (tex_type, gl::TEXTURE_MAX_LEVEL, 0);
128  gl::TexParameteri (tex_type, gl::TEXTURE_MAG_FILTER, interp_type);
129  gl::TexParameteri (tex_type, gl::TEXTURE_MIN_FILTER, interp_type);
130  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_S, gl::CLAMP_TO_EDGE);
131  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_T, gl::CLAMP_TO_EDGE);
132  if (tex_type == gl::TEXTURE_3D)
133  gl::TexParameteri (tex_type, gl::TEXTURE_WRAP_R, gl::CLAMP_TO_EDGE);
134  }
135  }
136  GLenum type () const { return tex_type; }
137  void clear () {
138  if (id) {
139  check_context();
140  GL_DEBUG ("deleting OpenGL texture ID " + str(id));
141  gl::DeleteTextures (1, &id);
142  }
143  id = 0;
144  }
145  void bind () const {
146  assert (id);
147  check_context();
148  GL_DEBUG ("binding OpenGL texture ID " + str(id));
149  gl::BindTexture (tex_type, id);
150  }
151  void set_interp (GLint type) const {
152  bind();
153  gl::TexParameteri (tex_type, gl::TEXTURE_MAG_FILTER, type);
154  gl::TexParameteri (tex_type, gl::TEXTURE_MIN_FILTER, type);
155  }
156  void set_interp_on (bool interpolate) const { set_interp (interpolate ? gl::LINEAR : gl::NEAREST); }
157  protected:
158  GLuint id;
159  GLenum tex_type;
160  };
