118  class Connectivity {
119  public:
120  Connectivity (const std::vector<std::map<int32_t, connectivity> >& connectivity_map,
121  value_type dh, value_type E, value_type H) :
122  connectivity_map (connectivity_map), dh (dh), E (E), H (H) { }
123 
124  value_type operator() (const value_type max_stat, const std::vector<value_type>& stats,
125  std::vector<value_type>* get_tfce_stats)
126  {
127  tfce_stats.resize (stats.size());
128  std::fill (tfce_stats.begin(), tfce_stats.end(), 0.0);
129  value_type max_tfce_stat = 0.0;
130  for (size_t fixel = 0; fixel < connectivity_map.size(); ++fixel) {
131  std::map<int32_t, connectivity>::const_iterator connected_fixel;
132  for (value_type h = this->dh; h < stats[fixel]; h += this->dh) {
133  value_type extent = 0.0;
134  for (connected_fixel = connectivity_map[fixel].begin(); connected_fixel != connectivity_map[fixel].end(); ++connected_fixel)
135  if (stats[connected_fixel->first] > h)
136  extent += connected_fixel->second.value;
137  tfce_stats[fixel] += Math::pow (extent, E) * Math::pow (h, H);
138  }
139  if (tfce_stats[fixel] > max_tfce_stat)
140  max_tfce_stat = tfce_stats[fixel];
141  }
142 
143  if (get_tfce_stats)
144  *get_tfce_stats = tfce_stats;
145 
146  return max_tfce_stat;
147  }
148

150  const std::vector<std::map<int32_t, connectivity> >& connectivity_map;
151  std::vector<value_type> tfce_stats;
