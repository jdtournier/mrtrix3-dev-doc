63  bool include_default)
64  {

84  inline void statistic2pvalue (const Math::Vector<value_type>& perm_dist,
85  const std::vector<value_type>& stats,
86  std::vector<value_type>& pvalues)
87  {

89  for (size_t i = 0; i < perm_dist.size(); i++)
90  permutations[i] = perm_dist[i];
91  std::sort (permutations.begin(), permutations.end());
92  pvalues.resize (stats.size());
93  for (size_t i = 0; i < stats.size(); ++i) {
94  if (stats[i] > 0.0) {
95  value_type pvalue = 1.0;
96  for (size_t j = 0; j < permutations.size(); ++j) {
97  if (stats[i] < permutations[j]) {
98  pvalue = value_type (j) / value_type (permutations.size());
99  break;
100  }
101  }
