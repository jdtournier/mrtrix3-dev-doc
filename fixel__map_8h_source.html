192  template <class Fixel>
193  bool Fixel_map<Fixel>::operator() (const FMLS::FOD_lobes& in)
194  {
195  if (in.empty())
196  return true;
197  auto v = accessor();
198  assign_pos_of (in.vox).to (v);
199  if (is_out_of_bounds (v))
200  return false;
201  if (v.value())
202  throw Exception ("FIXME: FOD_map has received multiple segmentations for the same voxel!");
203  v.value() = new MapVoxel (in, fixels.size());
204  for (const auto& i : in)
205  fixels.push_back (Fixel (i));
206  return true;
207  }
