207  template <class Fixel>
208  bool Fixel_map<Fixel>::operator() (const FMLS::FOD_lobes& in)
209  {
210  if (in.empty())
211  return true;
212  if (!Image::Nav::within_bounds (data, in.vox))
213  return false;
214  VoxelAccessor v (accessor);
215  Image::Nav::set_pos (v, in.vox);
216  if (v.value())
217  throw Exception ("FIXME: FOD_map has received multiple segmentations for the same voxel!");
218  v.value() = new MapVoxel (in, fixels.size());
219  for (FMLS::FOD_lobes::const_iterator i = in.begin(); i != in.end(); ++i)
220  fixels.push_back (Fixel (*i));
221  return true;
222  }
