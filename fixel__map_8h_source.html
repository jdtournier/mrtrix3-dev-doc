184  template <class Fixel>
185  bool Fixel_map<Fixel>::operator() (const FMLS::FOD_lobes& in)
186  {
187  if (in.empty())
188  return true;
189  auto v = accessor();
190  assign_pos_of (in.vox).to (v);
191  if (is_out_of_bounds (v))
192  return false;
193  if (v.value())
194  throw Exception ("FIXME: FOD_map has received multiple segmentations for the same voxel!");
195  v.value() = new MapVoxel (in, fixels.size());
196  for (const auto& i : in)
197  fixels.push_back (Fixel (i));
198  return true;
199  }
