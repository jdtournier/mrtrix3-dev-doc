79  properties.set (max_trials, "max_trials");

81  properties.set (fod_power, "fod_power");
82  bool precomputed = true;
83  properties.set (precomputed, "sh_precomputed");
84  if (precomputed)
85  precomputer.init (lmax);
86 
87  // num_samples is number of samples excluding first point
88  --num_samples;
89  INFO ("iFOD2 generating " + str(num_samples) + " vertices per " + str (step_size) + " mm step");
90 
91  // iFOD2 by default downsamples after track propagation back to the desired 'step size'
92  // i.e. the sub-step detail is removed from the output
93  size_t downsample_factor = num_samples;
94  properties.set (downsample_factor, "downsample_factor");
95  downsampler.set_ratio (downsample_factor);
96 
97  // For iFOD2, "step_size" represents the length of the chord represented
98  // using "num_samples" vertices rather than just one; the following two
99  // variables need to be calculated accordingly:
100  // - The arc angle subtended by two sequential vertices on a circle of minimal radius
101  // (prior to downsampling)
102  const float angle_minradius_preds = 2.0 * std::asin (step_size / (2.0 * min_radius)) / float(num_samples);
103  // - The maximal possible distance between vertices after downsampling
104  const float max_step_postds = downsample_factor * step_size / float(num_samples);
105  set_num_points (angle_minradius_preds, max_step_postds);
106  }

231  term_t next () override
232  {
233 
234  if (++sample_idx < S.num_samples) {
235  pos = positions[sample_idx];
236  dir = tangents [sample_idx];
237  return CONTINUE;
238  }
239 
240  Eigen::Vector3f next_pos, next_dir;
241 
242  float max_val = 0.0;
243  for (size_t i = 0; i < calibrate_list.size(); ++i) {
244  get_path (calib_positions, calib_tangents, rotate_direction (dir, calibrate_list[i]));
245  float val = path_prob (calib_positions, calib_tangents);
246  if (std::isnan (val))
247  return EXIT_IMAGE;
248  else if (val > max_val)
249  max_val = val;
250  }
251 
252  if (max_val <= 0.0)
253  return CALIBRATOR;
254 
255  max_val *= calibrate_ratio;
256 
257  num_sample_runs++;
258 
259  for (size_t n = 0; n < S.max_trials; n++) {
260  float val = rand_path_prob ();
261 
262  if (val > max_val) {
263  DEBUG ("max_val exceeded!!! (val = " + str(val) + ", max_val = " + str (max_val) + ")");
264  ++num_truncations;
265  if (val/max_val > max_truncation)
266  max_truncation = val/max_val;
267  }
268 
269  if (uniform(rng) < val/max_val) {
270  mean_sample_num += n;
271  half_log_prob0 = last_half_log_probN;
272  pos = positions[0];
273  dir = tangents [0];
274  sample_idx = 0;

298  void truncate_track (GeneratedTrack& tck, const size_t length_to_revert_from, const size_t revert_step) override
299  {
300  // OK, if we know length_to_revert_from, we can reconstruct what sample_idx was at that point
301  size_t sample_idx_at_full_length = (length_to_revert_from - tck.get_seed_index()) % S.num_samples;
302  // Unfortunately can't distinguish between sample_idx = 0 and sample_idx = S.num_samples
303  // However the former would result in zero truncation with revert_step = 1...
304  if (!sample_idx_at_full_length)
305  sample_idx_at_full_length = S.num_samples;
306  const size_t points_to_remove = sample_idx_at_full_length + ((revert_step - 1) * S.num_samples);
307  if (tck.get_seed_index() + points_to_remove >= tck.size()) {
308  tck.clear();
309  pos = { NaN, NaN, NaN };
310  dir = { NaN, NaN, NaN };
311  return;
312  }
313  const size_t new_size = length_to_revert_from - points_to_remove;
314  if (tck.size() == 2 || new_size == 1)
315  dir = (tck[1] - tck[0]).normalized();
316  else if (new_size != tck.size())
317  dir = (tck[new_size] - tck[new_size - 2]).normalized();
318  tck.resize (new_size);
319 
320  // Need to get the path probability contribution from the FOD at this point
321  pos = tck.back();
322  get_data (source);
323  half_log_prob0 = 0.5 * std::log (FOD (dir));
324 
325  // Make sure that arc is re-calculated when next() is called
326  sample_idx = S.num_samples;
327 
328  // Need to update sgm_depth appropriately, remembering that it is tracked by exec
329  if (S.is_act())
330  act().sgm_depth = (act().sgm_depth > points_to_remove) ? act().sgm_depth - points_to_remove : 0;
331  }

379  float path_prob (vector<Eigen::Vector3f>& positions, vector<Eigen::Vector3f>& tangents)
380  {
381 
382  // Early exit for ACT when path is not sensible
383  if (S.is_act()) {
384  if (!act().fetch_tissue_data (positions[S.num_samples - 1]))
385  return (NaN);
386  if (act().tissues().get_csf() >= 0.5)
387  return 0.0;
388  }
389 
390  float log_prob = half_log_prob0;
391  for (size_t i = 0; i < S.num_samples; ++i) {
392 
393  float fod_amp = FOD (positions[i], tangents[i]);
394  if (std::isnan (fod_amp))
395  return NaN;
396  if (fod_amp < S.threshold)
397  return 0.0;
398  fod_amp = std::log (fod_amp);
399  if (i < S.num_samples-1) {
400  log_prob += fod_amp;
401  } else {
402  last_half_log_probN = 0.5*fod_amp;
403  log_prob += last_half_log_probN;
404  }
405  }
406 
407  return std::exp (S.fod_power * log_prob);
408  }

412  void get_path (vector<Eigen::Vector3f>& positions, vector<Eigen::Vector3f>& tangents, const Eigen::Vector3f& end_dir) const
413  {
414  float cos_theta = end_dir.dot (dir);
415  cos_theta = std::min (cos_theta, float(1.0));
416  float theta = std::acos (cos_theta);
417 
418  if (theta) {
419 
420  Eigen::Vector3f curv = end_dir - cos_theta * dir;
421  curv.normalize();
422  float R = S.step_size / theta;
423 
424  for (size_t i = 0; i < S.num_samples-1; ++i) {
425  float a = (theta * (i+1)) / S.num_samples;
426  float cos_a = std::cos (a);
427  float sin_a = std::sin (a);
428  positions[i] = pos + R * (sin_a * dir + (float(1.0) - cos_a) * curv);
429  tangents[i] = cos_a * dir + sin_a * curv;
430  }
431  positions[S.num_samples-1] = pos + R * (std::sin (theta) * dir + (float(1.0)-cos_theta) * curv);
432  tangents[S.num_samples-1] = end_dir;
433 
434  } else { // straight on:
435 
436  for (size_t i = 0; i < S.num_samples; ++i) {
437  float f = (i+1) * (S.step_size / S.num_samples);
438  positions[i] = pos + f * dir;
439  tangents[i] = dir;
440  }
441 
442  }
443  }
