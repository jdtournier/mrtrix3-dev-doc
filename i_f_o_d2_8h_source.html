293  void truncate_track (GeneratedTrack& tck, const size_t length_to_revert_from, const size_t revert_step)
294  {
295  // OK, if we know length_to_revert_from, we can reconstruct what sample_idx was at that point
296  size_t sample_idx_at_full_length = (length_to_revert_from - tck.get_seed_index()) % S.num_samples;
297  // Unfortunately can't distinguish between sample_idx = 0 and sample_idx = S.num_samples
298  // However the former would result in zero truncation with revert_step = 1...
299  if (!sample_idx_at_full_length)
300  sample_idx_at_full_length = S.num_samples;
301  const size_t points_to_remove = sample_idx_at_full_length + ((revert_step - 1) * S.num_samples);
302  if (tck.get_seed_index() + points_to_remove >= tck.size()) {
303  tck.clear();
304  pos = { NaN, NaN, NaN };
305  dir = { NaN, NaN, NaN };
306  return;
307  }
308  const size_t new_size = length_to_revert_from - points_to_remove;
309  if (tck.size() == 2 || new_size == 1)
310  dir = (tck[1] - tck[0]).normalized();
311  else if (new_size != tck.size())
312  dir = (tck[new_size] - tck[new_size - 2]).normalized();
313  tck.resize (new_size);
314 
315  // Need to get the path probability contribution from the FOD at this point
316  pos = tck.back();
317  get_data (source);
318  half_log_prob0 = 0.5 * std::log (FOD (dir));
319 
320  // Make sure that arc is re-calculated when next() is called
321  sample_idx = S.num_samples;
322 
323  // Need to update sgm_depth appropriately, remembering that it is tracked by exec
324  act().sgm_depth = (act().sgm_depth > points_to_remove) ? act().sgm_depth - points_to_remove : 0;
325  }

373  float path_prob (std::vector<Eigen::Vector3f>& positions, std::vector<Eigen::Vector3f>& tangents)
374  {
375 
376  // Early exit for ACT when path is not sensible
377  if (S.is_act()) {
378  if (!act().fetch_tissue_data (positions[S.num_samples - 1]))
379  return (NaN);
380  if (act().tissues().get_csf() >= 0.5)
381  return 0.0;
382  }
383 
384  float log_prob = half_log_prob0;
385  for (size_t i = 0; i < S.num_samples; ++i) {
386 
387  float fod_amp = FOD (positions[i], tangents[i]);
388  if (std::isnan (fod_amp))
389  return NaN;
390  if (fod_amp < S.threshold)
391  return 0.0;
392  fod_amp = std::log (fod_amp);
393  if (i < S.num_samples-1) {
394  log_prob += fod_amp;
395  } else {
396  last_half_log_probN = 0.5*fod_amp;
397  log_prob += last_half_log_probN;
398  }
399  }
400 
401  return std::exp (S.fod_power * log_prob);
402  }

406  void get_path (std::vector<Eigen::Vector3f>& positions, std::vector<Eigen::Vector3f>& tangents, const Eigen::Vector3f& end_dir) const
407  {
408  float cos_theta = end_dir.dot (dir);
409  cos_theta = std::min (cos_theta, float(1.0));
410  float theta = std::acos (cos_theta);
411 
412  if (theta) {
413 
414  Eigen::Vector3f curv = end_dir - cos_theta * dir;
415  curv.normalize();
416  float R = S.step_size / theta;
417 
418  for (size_t i = 0; i < S.num_samples-1; ++i) {
419  float a = (theta * (i+1)) / S.num_samples;
420  float cos_a = std::cos (a);
421  float sin_a = std::sin (a);
422  positions[i] = pos + R * (sin_a * dir + (float(1.0) - cos_a) * curv);
423  tangents[i] = cos_a * dir + sin_a * curv;
424  }
425  positions[S.num_samples-1] = pos + R * (std::sin (theta) * dir + (float(1.0)-cos_theta) * curv);
426  tangents[S.num_samples-1] = end_dir;
427 
428  } else { // straight on:
429 
430  for (size_t i = 0; i < S.num_samples; ++i) {
431  float f = (i+1) * (S.step_size / S.num_samples);
432  positions[i] = pos + f * dir;
433  tangents[i] = dir;
434  }
435 
436  }
437  }
