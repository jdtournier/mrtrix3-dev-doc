193  // TOD class: tore the SH coefficients in the voxel class so that aPSF generation can be multi-threaded
194  // Provide a normalize() function to remove any length dependence, and have unary contribution per streamline
195  class VoxelTOD : public Voxel
196  {
197 
198  public:
199  VoxelTOD () :
200  Voxel (),
201  sh_coefs () { }
202 
203  VoxelTOD (const Eigen::Vector3i& V) :
204  Voxel (V),
205  sh_coefs () { }
206 
207  VoxelTOD (const Eigen::Vector3i& V, const Eigen::VectorXf& t) :
208  Voxel (V),
209  sh_coefs (t) { }
210 
211  VoxelTOD (const Eigen::Vector3i& V, const Eigen::VectorXf& t, const float l) :
212  Voxel (V, l),
213  sh_coefs (t) { }
214 
215  VoxelTOD& operator= (const VoxelTOD& V) { Voxel::operator= (V); sh_coefs = V.sh_coefs; return (*this); }
216  VoxelTOD& operator= (const Eigen::Vector3i& V) { Voxel::operator= (V); sh_coefs.resize(0,0); return (*this); }
217 
218  // For sorting / inserting, want to identify the same voxel, even if the TOD is different
219  bool operator== (const VoxelTOD& V) const { return Voxel::operator== (V); }
220  bool operator< (const VoxelTOD& V) const { return Voxel::operator< (V); }
221 
222  void normalize() const
223  {
224  const float multiplier = 1.0f / get_length();
225  sh_coefs *= multiplier;
226  Voxel::normalize();
227  }
228  void set_tod (const Eigen::VectorXf& i) { sh_coefs = i; }
229  void add (const Eigen::VectorXf& i, const float l) const
230  {
231  assert (i.size() == sh_coefs.size());
232  for (ssize_t index = 0; index != sh_coefs.size(); ++index)
233  sh_coefs[index] += l * i[index];
234  Voxel::operator+= (l);
235  }
236  void operator+= (const Eigen::VectorXf& i) const
237  {
238  assert (i.size() == sh_coefs.size());
239  sh_coefs += i;
240  Voxel::operator+= (1.0f);
241  }
242  const Eigen::VectorXf& get_tod() const { return sh_coefs; }
243 
244  private:
245  mutable Eigen::VectorXf sh_coefs;
246 
247  };
